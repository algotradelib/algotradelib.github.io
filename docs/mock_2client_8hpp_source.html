281 template <typename T, typename Ask>
282 Result<T> read(Call call, std::optional<T>& last, Ask ask) {
283 auto fault = take(call);
284 if (fault && fault->effect_ == Effect::FAIL) {
285 return Result<T>(std::move(fault->error_));
286 }
287 if (fault && last) {
288 auto repeated = *last;
289 repeated.received_at_ = market_->now();
290 return Result<T>(std::move(repeated));
291 }
292 auto answer = ask();
293 if (answer) {
294 last = answer.value();
295 }
296 return answer;
297 }
298
300 std::optional<Fault> take(Call call) {
301 for (auto it = faults_.begin(); it != faults_.end(); ++it) {
302 if (it->call_ == call) {
303 auto fault = std::move(*it);
304 faults_.erase(it);
305 return fault;
306 }
307 }
308 return std::nullopt;
309 }
