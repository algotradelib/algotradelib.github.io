64std::int64_t parse_account_sequence(const char* value) {
65 const std::string text(value);
66 std::size_t parsed = 0;
67 const auto sequence = std::stoll(text, &parsed);
68 if (parsed != text.size() || sequence <= 0) {
69 throw std::invalid_argument("account sequence must be a positive integer");
70 }
71 return sequence;
72}

76int main(int argc, char* argv[]) {
77 try {
78 algotradelib::tossinvest::TossConfig config;
79 config.from_env();
80
81 ssl::context tls(ssl::context::tls_client);
82 tls.set_default_verify_paths();
83 tls.set_verify_mode(ssl::verify_peer);
84
85 algotradelib::tossinvest::TossBlockingClient client(tls, std::move(config));
86 auto accounts_result = client.accounts();
87 if (!accounts_result) {
88 print_error("account discovery", accounts_result.error());
89 return 1;
90 }
91
92 const auto& accounts = accounts_result.value();
93 std::cout << "accounts: " << accounts.size() << '\n';
94 for (const auto& account : accounts) {
95 std::cout << account.sequence_ << " " << mask_account_number(account.account_number_)
96 << " " << account.account_type_ << '\n';
97 }
98 if (accounts.empty()) {
99 return 0;
100 }
101
102 const std::int64_t selected_sequence =
103 argc > 1 ? parse_account_sequence(argv[1]) : accounts.front().sequence_;
104 const auto selected = std::find_if(
105 accounts.begin(),
106 accounts.end(),
107 [selected_sequence](const algotradelib::tossinvest::TossAccount& account) {
108 return account.sequence_ == selected_sequence;
109 }
110 );
111 if (selected == accounts.end()) {
112 throw std::invalid_argument("selected account sequence was not returned by discovery");
113 }
114
115 std::cout << "selected account: " << selected_sequence << '\n';
116 auto holdings_result = client.holdings(selected_sequence);
117 if (!holdings_result) {
118 print_error("holdings request", holdings_result.error());
119 return 1;
120 }
121
122 print_holdings(holdings_result.value());
123 return 0;
124 } catch (const std::exception& exception) {
125 std::cerr << "example failed: " << exception.what() << '\n';
126 return 1;
127 }
