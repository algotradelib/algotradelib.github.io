110live_program::Settings settings_for(const atl::InstrumentId& instrument) {
111 live_program::Settings settings;
112 settings.universe_ = {instrument};
113
114 settings.limits_.max_order_notional_ = atl::Decimal64::parse("500000");
115 settings.limits_.max_position_quantity_ = atl::Decimal64::parse("1");
116 settings.limits_.max_orders_per_window_ = 2;
117
118 // Daily bars, `SessionSource`'s default. The loop reads the order list and holdings before it feeds
119 // a bar. At start the strategy is shown the history read, but only the newest closed bar's decision
120 // can be sent: at once if the market is open; otherwise an opening order waits for the first `OPEN`
121 // iteration (which needs a calendar), and an exit goes out at once and is kept if the broker refuses
122 // it. After that a day's bar is fed once the day has ended, at the first poll after midnight KST,
123 // and the same rule applies ("kept for the next session" in running-a-strategy-live.md). Polled
124 // every minute so that poll comes promptly; a minute-bar strategy sets `bar_interval_`.
125 settings.schedule_.bars_ = std::chrono::minutes(1);
126 settings.schedule_.holdings_ = std::chrono::seconds(30);
127 settings.schedule_.working_orders_ = std::chrono::seconds(30);
128 // A band is a per-trading-day value; refresh it well inside a session.
129 settings.schedule_.price_limits_ = std::chrono::minutes(10);
130 // The calendar keeps its six-hour default.
131
132 // Every age stated. A holdings snapshot older than two minutes -- four missed polls -- stops
133 // sizing; a band older than twenty -- two missed polls -- stops clamping.
134 settings.maximum_portfolio_age_ = std::chrono::minutes(2);
135 settings.maximum_price_limits_age_ = std::chrono::minutes(20);
136 return settings;
137}

222int run_kis(ssl::context& tls, const Arguments& arguments) {
223 namespace kis = algotradelib::kis;
224
225 kis::KisConfig config;
226 kis::KisAccount account;
227 kis_credentials::load(arguments.credentials_, config, &account, std::cout);
228 const bool demo = config.is_demo_;
229
230 const atl::InstrumentId instrument{atl::Country::KR, atl::Venue::COMBINED, arguments.symbol_};
231 auto settings = settings_for(instrument);
232 name_journal(settings, arguments, "kis", demo, account.account_number_ + '-' + account.product_code_);
233 announce(arguments, settings, demo, "KIS_DEMO");
234
235 atl::BlockingSession<kis::KisBlockingClient> session(
236 std::make_shared<kis::KisBlockingClient>(tls, std::move(config), kis_credentials::paced_options()),
237 std::move(account)
238 );
239
240 // Market data from a session with no account, so it cannot trade: swapped by mistake, every order
241 // and account read through it is refused before any I/O.
242 if (arguments.market_data_ == "toss") {
243 if (!demo) {
244 // Toss publishes the next session's band after the close, before KIS does, and when each rolls
245 // over is not settled: a KIS demo run only.
246 throw std::runtime_error("--market-data toss is for a KIS demo run only");
247 }
248 if (!arguments.credentials_) {
249 throw std::runtime_error("--market-data toss reads tossinvest/real.json; give --credentials DIR");
250 }
251 const auto path = *arguments.credentials_ / "tossinvest" / "real.json";
252 algotradelib::tossinvest::TossConfig toss_config;
253 toss_config.from_json(path.string());
254 atl::BlockingSession<algotradelib::tossinvest::TossBlockingClient> market_data(
255 std::make_shared<algotradelib::tossinvest::TossBlockingClient>(tls, std::move(toss_config)), 0
256 );
257 std::cout << "market data: toss " << path.string() << " (no account: cannot trade)\n";
258 return run(session, market_data, instrument, settings, arguments.live_);
259 }
260 if (!demo && arguments.market_data_ == "kis") {
261 throw std::runtime_error("--market-data kis is for a demo run; a live run reads its own market data");
262 }
263 if (demo) {
264 auto market_config = kis_credentials::market_data_config(arguments.credentials_);
265 if (market_config) {
266 atl::BlockingSession<kis::KisBlockingClient> market_data(
267 std::make_shared<kis::KisBlockingClient>(
268 tls, std::move(*market_config), kis_credentials::paced_options()
269 ),
270 kis::KisAccount{}
271 );
272 std::cout << "market data: kis " << (*arguments.credentials_ / "kis" / "real.json").string()
273 << " (no account: cannot trade)\n";
274 return run(session, market_data, instrument, settings, arguments.live_);
275 }
276 if (arguments.market_data_ == "kis") {
277 throw std::runtime_error(
278 arguments.credentials_ ? "--market-data kis reads kis/real.json, and there is none" :
279 "--market-data kis reads kis/real.json; give --credentials DIR"
280 );
281 }
282 std::cout << "market data: the demo account's own (no kis/real.json); a demo host refuses\n"
283 << " reference data, so this run stops there\n";
284 }
285 return run(session, instrument, settings, arguments.live_);
286}

288int run_kiwoom(ssl::context& tls, const Arguments& arguments) {
289 namespace kiwoom = algotradelib::kiwoom;
290
291 kiwoom::KiwoomConfig config;
292 kiwoom_credentials::load(arguments.credentials_, arguments.kiwoom_account_, config, std::cout);
293 // Kiwoom's balance inquiry names one venue and does not say whether that filters the positions or
294 // only values them, so the adapter refuses holdings until you state both. Until then the loop has
295 // no portfolio and holds: this program runs and places nothing. Measure what your account returns
296 // (run `example-kiwoom-account`, then set its `holdings_venue_` to NXT, rebuild and run it again),
297 // then assert what you measured. The sellable scope is the third thing to state: without it the
298 // guard refuses every sell, a flatten's closing sell included.
299 config.holdings_venue_ = atl::Venue::KRX;
300 // config.holdings_position_scope_ = kiwoom::KiwoomPositionScope::WHOLE_HOLDING;
301 // config.holdings_sellable_scope_ = kiwoom::KiwoomSellableScope::WHOLE_HOLDING;
302 const bool demo = config.is_demo_;
303
304 atl::BlockingSession<kiwoom::KiwoomBlockingClient> session(
305 std::make_shared<kiwoom::KiwoomBlockingClient>(tls, std::move(config), kiwoom_credentials::paced_options()),
306 kiwoom::KiwoomAccount{}
307 );
308 // Kiwoom takes no account number on a request -- the application key is the account -- so the
309 // journal is named after the account the key drives, as Kiwoom reports it.
310 const auto accounts = session.accounts();
311 if (!accounts || accounts.value().empty()) {
312 std::cerr << "could not read which account this Kiwoom key drives"
313 << (accounts ? std::string() : ": " + accounts.error().message_) << '\n';
314 return static_cast<int>(live_program::Exit::ACCOUNT_UNRESOLVED);
315 }
316
317 const atl::InstrumentId instrument{atl::Country::KR, atl::Venue::COMBINED, arguments.symbol_};
318 auto settings = settings_for(instrument);
319 // A Kiwoom order must name its venue (`dmst_stex_tp` has no default), and a strategy's orders name
320 // none, so the binding states where they go. KRX here; `Venue::NXT` for Nextrade, or
321 // `Venue::COMBINED` for Kiwoom's SOR, which its order endpoints document -- whether a demo account
322 // accepts SOR, and whether the listing's `sor_yn` flag reads as documented, has not been observed,
323 // so KRX is the route with the fewest open questions.
324 settings.default_route_ = atl::Venue::KRX;
325 name_journal(settings, arguments, "kiwoom", demo, accounts.value().front().display_number_);
326 announce(arguments, settings, demo, "KIWOOM_DEMO");
327 // Two reasons nothing is placed on Kiwoom yet, said up front so nobody waits for an order that
328 // cannot come. Both are statements you can make:
329 // - Holdings refuse until the scopes above are asserted, so the loop has no portfolio and holds.
330 // - Kiwoom publishes no trading calendar, so the market phase stays unknown, and the default
331 // limits open a position only while the market is open.
332 std::cout << "note : on Kiwoom this program places nothing yet -- holdings need a stated\n"
333 << " scope, and the market phase is unknown without a calendar\n";
334 return run(session, instrument, settings, arguments.live_);
335}

337int run_toss(ssl::context& tls, const Arguments& arguments) {
338 namespace toss = algotradelib::tossinvest;
339
340 toss::TossConfig config;
341 config.from_env();
342 std::cout << "credentials: environment variables (Toss reads no credentials directory)\n";
343 auto client = std::make_shared<toss::TossBlockingClient>(tls, std::move(config));
344
345 // Toss selects an account by sequence number and publishes the list. With one account that is the
346 // one; with several, the program will not guess which should trade, because a sequence is a request
347 // selector rather than an identity.
348 const int unresolved = static_cast<int>(live_program::Exit::ACCOUNT_UNRESOLVED);
349 auto accounts = atl::list_accounts(*client);
350 if (!accounts || accounts.value().empty()) {
351 std::cerr << "could not list the accounts these Toss credentials reach"
352 << (accounts ? std::string() : ": " + accounts.error().message_) << '\n';
353 return unresolved;
354 }
355 if (!arguments.toss_account_ && accounts.value().size() > 1) {
356 std::cerr << "these Toss credentials reach " << accounts.value().size()
357 << " accounts; choose one with --toss-account:\n";
358 for (const auto& account : accounts.value()) {
359 std::cerr << " --toss-account " << account.selector_ << " (" << account.type_ << ")\n";
360 }
361 return unresolved;
362 }
363 auto chosen = accounts.value().front();
364 if (arguments.toss_account_) {
365 bool found = false;
366 for (const auto& account : accounts.value()) {
367 if (account.selector_ == *arguments.toss_account_) {
368 chosen = account;
369 found = true;
370 }
371 }
372 if (!found) {
373 std::cerr << "no Toss account has sequence " << *arguments.toss_account_ << '\n';
374 return unresolved;
375 }
376 }
377 if (chosen.display_number_.empty()) {
378 std::cerr << "Toss reported no account number for sequence " << chosen.selector_
379 << ", so the journal could not be named after the account\n";
380 return unresolved;
381 }
382
383 const atl::InstrumentId instrument{atl::Country::KR, atl::Venue::COMBINED, arguments.symbol_};
384 auto settings = settings_for(instrument);
385 // Named after the account number, not the sequence: two credential sets can both have a sequence 1,
386 // and a journal that one account's run could continue would read its orders as the other's.
387 name_journal(settings, arguments, "toss", false, chosen.display_number_);
388 announce(arguments, settings, false, nullptr);
389 std::cout << "note : Toss has no paper-trading environment; --live trades real money\n";
390
391 atl::BlockingSession<toss::TossBlockingClient> session(std::move(client), chosen.selector_);
392 return run(session, instrument, settings, arguments.live_);
393}

396std::optional<Arguments> parse(int argc, char** argv) {
397 Arguments arguments;
398 const int given = argc;
399 try {
400 arguments.credentials_ = kis_credentials::take_directory(argc, argv);
401 arguments.credentials_from_flag_ = argc != given;
402 arguments.kiwoom_account_ = kiwoom_credentials::take_label(argc, argv);
403 } catch (const std::runtime_error& error) {
404 std::cerr << "example-live-program: " << error.what() << '\n';
405 return std::nullopt;
406 }
407 std::vector<std::string> positional;
408 for (int i = 1; i < argc; ++i) {
409 const std::string argument = argv[i];
410 const bool has_value = i + 1 < argc;
411 if (argument == "--live") {
412 arguments.live_ = true;
413 } else if (argument == "--journal" && has_value) {
414 arguments.journal_ = argv[++i];
415 } else if (argument == "--control" && has_value) {
416 arguments.control_ = argv[++i];
417 } else if (argument == "--toss-account" && has_value) {
418 arguments.toss_account_ = std::strtoll(argv[++i], nullptr, 10);
419 } else if (argument == "--market-data" && has_value) {
420 arguments.market_data_ = argv[++i];
421 if (*arguments.market_data_ != "kis" && *arguments.market_data_ != "toss") {
422 return std::nullopt;
423 }
424 } else if (argument.rfind("--", 0) == 0) {
425 return std::nullopt;
426 } else {
427 positional.push_back(argument);
428 }
429 }
430 if (positional.size() != 2) {
431 return std::nullopt;
432 }
433 arguments.broker_ = positional[0];
434 arguments.symbol_ = positional[1];
435 if (arguments.broker_ != "kis" && arguments.broker_ != "kiwoom" && arguments.broker_ != "toss") {
436 return std::nullopt;
437 }
438 // A setting the chosen broker would ignore is refused: ignored, it would trade whichever account
439 // the environment names. Market data is KIS's; Toss reads only its environment variables; the
440 // label selects a Kiwoom file.
441 if (arguments.market_data_ && arguments.broker_ != "kis") {
442 return std::nullopt;
443 }
444 if (arguments.credentials_from_flag_ && arguments.broker_ == "toss") {
445 return std::nullopt;
446 }
447 if (arguments.kiwoom_account_ && arguments.broker_ != "kiwoom") {
448 return std::nullopt;
449 }
450 return arguments;
451}

462int main(int argc, char** argv) {
463 const auto arguments = parse(argc, argv);
464 if (!arguments) {
465 std::cerr
466 << "usage: example-live-program <kis|kiwoom|toss> <symbol> [--live] [--journal PATH]\n"
467 << " [--control PATH] [--toss-account SEQUENCE]\n"
468 << " [--credentials DIR] [--market-data kis|toss]\n"
469 << " [--kiwoom-account LABEL]\n\n"
470 << "Without --live it observes: nothing can be sent. KIS and Kiwoom are demo unless\n"
471 << "KIS_DEMO=0 / KIWOOM_DEMO=0; Toss has no demo, so toss --live trades real money.\n";
472 return 64;
473 }
474
475 try {
476 ssl::context tls(ssl::context::tls_client);
477 tls.set_default_verify_paths();
478 tls.set_verify_mode(ssl::verify_peer);
479
480 if (arguments->broker_ == "kis") {
481 return run_kis(tls, *arguments);
482 }
483 if (arguments->broker_ == "kiwoom") {
484 return run_kiwoom(tls, *arguments);
485 }
486 return run_toss(tls, *arguments);
487 } catch (const std::exception& exception) {
488 std::cerr << "example-live-program: " << exception.what() << '\n';
489 return 70;
490 }
