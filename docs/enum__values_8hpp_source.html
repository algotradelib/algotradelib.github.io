25// A scoped enum (`enum class`) always has a fixed underlying type, so every value of that type is a value of
26// the enum and the conversions below are defined. An unscoped one converts to an integer implicitly.
27template <typename Enum, bool = std::is_enum<Enum>::value>
28struct is_scoped_enum : std::false_type {};
29
30template <typename Enum>
31struct is_scoped_enum<Enum, true>
32: std::integral_constant<bool, !std::is_convertible<Enum, std::underlying_type_t<Enum>>::value> {};

34template <typename Enum, typename IsEnumerator>
35constexpr bool lists_every_value(const Enum* values, std::size_t count, IsEnumerator is_enumerator) noexcept {
36 using Underlying = std::underlying_type_t<Enum>;
37 for (std::size_t index = 0; index < count; ++index) {
38 const auto value = static_cast<Underlying>(values[index]);
39 if constexpr (std::is_signed<Underlying>::value) {
40 if (value < 0) {
41 return false;
42 }
43 }
44 if (static_cast<std::size_t>(value) >= count || !is_enumerator(values[index])) {
45 return false;
46 }
47 for (std::size_t other = 0; other < index; ++other) {
48 if (values[other] == values[index]) {
49 return false;
50 }
51 }
52 }
53 // `count` distinct values below `count` that fill the underlying type leave nothing beyond them, and
54 // converting `count` would wrap onto 0 (an `enum class : bool` with both values listed).
55 if (count > static_cast<std::size_t>(std::numeric_limits<Underlying>::max())) {
56 return true;
57 }
58 return !is_enumerator(static_cast<Enum>(static_cast<Underlying>(count)));
59}
