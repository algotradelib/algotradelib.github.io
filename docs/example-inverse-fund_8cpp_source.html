1// A bearish view on an index, held as an ordinary long position in an inverse fund.
2//
3// No adapter this library ships can sell what the account does not hold: KIS, Kiwoom and Toss take cash
4// orders. A target below zero there is refused by the guard before it is sent
5// (`RiskLimits::enforce_sellable_quantity_`, on by default). A replay can still simulate a short by
6// turning that check off where the replay is configured; that models no borrow, and nothing live.
7//
8// What a cash account can do is **buy a fund that moves against an index**. This example does exactly
9// that and nothing cleverer: the strategy reads one instrument for its signal and states a target for
10// another, the fund, by the fund's own identity. Nothing is translated, aliased or substituted behind
11// the strategy's back: the order, the fill, the holding and the exit are the fund's.
12//
13// What this is not, so the output is not read as more than it is:
14//
15// - It is not a short of the index, and it cannot stand in for a short of one stock. The fund targets
16// the opposite of its benchmark's **daily** return and resets every day, so over weeks its value
17// parts from what a short opened on day one would be worth, in either direction.
18// - It spends cash to buy fund shares. A short credits cash and owes shares, with a borrow fee, a
19// recall risk and margin. None of that is here.
20// - It carries the fund's own costs and tracking difference, and its benchmark is a futures index.
21// - Whether an account may buy a given fund (some need an agreement or a deposit) is the broker's to
22// say. This replay proves nothing about that.
23//
24// No credentials and no network. The two series below are a sample: the daily closes of 069500 and the
25// daily opens and closes of 114800 on the same forty trading days, 2026-08-11 to 2026-10-12, as a
26// public chart service reported them, unadjusted. They are here to make the example run, not as data
27// to rely on, and the replay stamps them one calendar day apart from an arbitrary start, not with
28// their real dates: replace them with your own history. A replay of this strategy must use the
29// fund's real bars: negating the index's returns to invent the fund's would hide exactly the
30// difference this file is warning about.
31//
32// Run it as:
33//
34// example-inverse-fund

76struct InverseFundOnDowntrend {
78 InstrumentId signal_;
80 InstrumentId fund_;
82 std::size_t lookback_{5};
84 Decimal64 quantity_{Decimal64::parse("1000")};
85
86 std::vector<Decimal64> closes_;
87
88 template <class Ctx>
89 void on_bar(Ctx& ctx, const strategy::BarEvent& bar) {
90 // The fund's own bars are fed too, because its fills and its marks come from them. They carry
91 // no signal.
92 if (!(bar.instrument_ == signal_)) {
93 return;
94 }
95 closes_.push_back(bar.bar_.close_);
96 if (closes_.size() < lookback_) {
97 return;
98 }
99 // `close * n < sum` instead of `close < sum / n`: exact, and no scale to choose.
100 std::optional<Decimal64> sum = Decimal64::parse("0");
101 for (std::size_t at = closes_.size() - lookback_; sum && at < closes_.size(); ++at) {
102 sum = algotradelib::decimal_add(*sum, closes_[at]);
103 }
104 const auto scaled =
105 algotradelib::decimal_multiply(closes_.back(), Decimal64::parse(std::to_string(lookback_)));
106 if (!sum || !scaled) {
107 return;
108 }
109 const bool bearish = algotradelib::decimal_less(*scaled, *sum);
110
111 // A target for the fund, by the fund's name: never a negative target for the index.
112 strategy::TargetPosition target;
113 target.instrument_ = fund_;
114 target.target_quantity_ = bearish ? quantity_ : Decimal64::parse("0");
115 target.tag_ = "inverse-fund";
116 ctx.submit(target);
117 }
118};

121Instrument reference(const InstrumentId& id, const char* name) {
122 Instrument instrument;
123 instrument.id_ = id;
124 instrument.name_ = name;
125 instrument.type_ = SecurityType::ETF;
126 instrument.currency_ = Currency{"KRW"};
127 instrument.lot_size_ = Decimal64::parse("1");
128 // This example sends market orders and states no price. A fund has its own tick grid, which is
129 // not the equity ladder: take it from reference data before pricing an order for one.
130 instrument.price_rule_ = PriceRule::unknown("an exchange-traded fund's tick grid is not stated here");
131 return instrument;
132}

159Bar daily(std::chrono::system_clock::time_point at, int open, int close) {
160 Bar bar;
161 bar.timestamp_ = at;
162 const auto opened = Decimal64::parse(std::to_string(open));
163 const auto closed = Decimal64::parse(std::to_string(close));
164 bar.open_ = opened;
165 bar.close_ = closed;
166 bar.high_ = open > close ? opened : closed;
167 bar.low_ = open > close ? closed : opened;
168 bar.volume_ = Decimal64::parse("1000000");
169 return bar;
170}

177int main() {
178 backtest::ReplayConfig config;
179 config.cash_ = Decimal64::parse("10000000");
180 config.instruments_ = {
181 reference(index_fund(), "an index fund (signal only)"),
182 reference(inverse_fund(), "an inverse fund (traded)"),
183 };
184 config.series_capacity_ = 64;
185 // Any phase, so the point stays the strategy: daily bars end outside the session.
186 config.limits_.permitted_phases_ = std::nullopt;
187 // The sellable check is left on, as it is by default. Every sell below is an exit of fund shares
188 // the replay holds, so nothing here needs it off; a strategy that wanted a true short would.
189
190 InverseFundOnDowntrend strategy_instance;
191 strategy_instance.signal_ = index_fund();
192 strategy_instance.fund_ = inverse_fund();
193
194 backtest::Backtester<InverseFundOnDowntrend> engine(std::move(strategy_instance), config);
195 const auto start = std::chrono::system_clock::time_point{} + std::chrono::hours(24 * 20000);
196 for (std::size_t at = 0; at < sample().size(); ++at) {
197 const auto when = start + std::chrono::hours(24 * static_cast<int>(at));
198 const auto& day = sample()[at];
199 // The fund's bar first: an order decided on yesterday's signal fills at this open. Then the
200 // signal's bar, on whose close the next decision is made.
201 engine.feed(strategy::make_bar_event(
202 inverse_fund(), BarInterval::ONE_DAY, BarStamp::CLOSE, daily(when, day.fund_open_, day.fund_close_)));
203 engine.feed(strategy::make_bar_event(
204 index_fund(), BarInterval::ONE_DAY, BarStamp::CLOSE, daily(when, day.index_close_, day.index_close_)));
205 }
206 engine.finish();
207
208 const auto& result = engine.result();
209 std::cout << "inverse fund 114800 traded on a signal from index fund 069500, " << sample().size()
210 << " sample days -- no credentials, no network\n\n";
211 for (const auto& execution : result.executions_) {
212 const auto& fill = execution.fill_;
213 const auto day = std::chrono::duration_cast<std::chrono::hours>(fill.at_ - start).count() / 24;
214 std::cout << " day " << day << " " << (fill.side_ == OrderSide::BUY ? "BUY " : "SELL") << " "
215 << fill.quantity_.str() << " fund shares at " << fill.price_.str() << "\n";
216 }
217 std::cout << "\n orders filled: " << result.executions_.size() << ", refused by the guard: "
218 << result.refused_.size() << ", not buildable: " << result.rejected_.size() << "\n";
219 if (!result.equity_.empty()) {
220 std::cout << " equity: " << config.cash_.str() << " -> " << result.equity_.back().equity_.str() << "\n";
221 }
222 std::cout
223 << "\nEvery order above is for the fund, and every sell is an exit of shares bought earlier: the\n"
224 << "account is never short, and the default sellable check never has anything to refuse.\n"
225 << "The fund follows the opposite of the index's move one day at a time. Held across many days its\n"
226 << "value is not what a short opened at the start would be worth, and it says nothing about any\n"
227 << "single stock. Forty sample days rank nothing.\n";
228 return 0;
229}
