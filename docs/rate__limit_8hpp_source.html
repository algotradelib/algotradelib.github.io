138inline constexpr int usable_burst(const RateLimitBudget& budget, RateLimitGroup group) noexcept {
139 switch (group) {
140 // An order, and the token it needs, may use the reserve.
141 case RateLimitGroup::TRADING:
142 case RateLimitGroup::AUTH:
143 return budget.burst_;
144 case RateLimitGroup::ACCOUNT:
145 case RateLimitGroup::MARKET_DATA:
146 case RateLimitGroup::REFERENCE:
147 break;
148 }
149 const int reserved = budget.burst_ - budget.trading_reserve_;
150 return reserved > 0 ? reserved : 1;
151}

163template <typename Clock = std::chrono::steady_clock>
164class BasicRateLimitBucket {
165public:
167 using ClockType = Clock;
168
170 using TimePoint = typename Clock::time_point;
171
173 using Duration = std::chrono::nanoseconds;
174
181 BasicRateLimitBucket(int burst, int per_second) { resize(burst, per_second); }
182
188 explicit BasicRateLimitBucket(const RateLimitBudget& budget)
189 : BasicRateLimitBucket(budget.burst_, budget.per_second_) {}
190
197 void resize(int burst, int per_second) {
198 if (burst < 1) {
199 throw std::invalid_argument("rate limit burst must be at least one request");
200 }
201 if (per_second < 1) {
202 throw std::invalid_argument("rate limit rate must be at least one request per second");
203 }
204 burst_ = burst;
205 per_second_ = per_second;
206 interval_ = Duration(std::chrono::nanoseconds(std::chrono::seconds(1)).count() / per_second);
207 }
208
215 Duration delay_until(TimePoint now, int allowance) const {
216 if (!started_) {
217 return Duration::zero();
218 }
219 const auto earliest = schedule_ - tolerance(allowance);
220 if (now >= earliest) {
221 return Duration::zero();
222 }
223 return std::chrono::duration_cast<Duration>(earliest - now);
224 }
225
232 bool ready(TimePoint now, int allowance) const {
233 return delay_until(now, allowance) == Duration::zero();
234 }
235
243 bool try_acquire(TimePoint now, int allowance) {
244 if (!ready(now, allowance)) {
245 return false;
246 }
247 if (!started_) {
248 schedule_ = now + interval_;
249 started_ = true;
250 return true;
251 }
252 schedule_ = (schedule_ > now ? schedule_ : now) + interval_;
253 return true;
254 }
255
265 void cool_down(TimePoint now, Duration duration) {
266 if (duration < Duration::zero()) {
267 duration = Duration::zero();
268 }
269 const auto held = now + duration + tolerance(burst_);
270 if (!started_ || held > schedule_) {
271 schedule_ = held;
272 started_ = true;
273 }
274 }
275
284 void observe_limit(TimePoint now, int burst, int per_second, Duration refill) {
285 resize(burst, per_second);
286 cool_down(now, refill > Duration::zero() ? refill : interval_);
287 }
288
295 int available(TimePoint now, int allowance) const {
296 if (!started_) {
297 return allowance;
298 }
299 const auto earliest = schedule_ - tolerance(allowance);
300 if (now < earliest) {
301 return 0;
302 }
303 const auto spare = std::chrono::duration_cast<Duration>(now - earliest);
304 const auto slots = spare / interval_ + 1;
305 return slots > allowance ? allowance : static_cast<int>(slots);
306 }
307
312 int burst() const noexcept { return burst_; }
313
318 int per_second() const noexcept { return per_second_; }
319
324 Duration interval() const noexcept { return interval_; }
325
326private:
327 // Returns how far ahead of now the schedule may run for `allowance`.
328 Duration tolerance(int allowance) const noexcept {
329 const int slots = allowance < 1 ? 1 : (allowance > burst_ ? burst_ : allowance);
330 return interval_ * (slots - 1);
331 }
332
333 int burst_{1};
334 int per_second_{1};
335 Duration interval_{std::chrono::seconds(1)};
336 TimePoint schedule_{};
337 bool started_{false};
338};

397class RateLimitController {
398public:
400 using Duration = RateLimitBucket::Duration;
401
407 explicit RateLimitController(RateLimitBudget budget = {})
408 : budget_(budget), bucket_(budget) {
409 stats_.burst_ = budget.burst_;
410 stats_.per_second_ = budget.per_second_;
411 }
412
418 Duration delay_until(RateLimitGroup group) const {
419 return bucket_.delay_until(RateLimitBucket::ClockType::now(), usable_burst(budget_, group));
420 }
421
427 bool acquire(RateLimitGroup group) {
428 const auto now = RateLimitBucket::ClockType::now();
429 if (!bucket_.try_acquire(now, usable_burst(budget_, group))) {
430 return false;
431 }
432 ++stats_.group(group).admitted_;
433 stats_.available_ = bucket_.available(now, usable_burst(budget_, group));
434 return true;
435 }
436
442 void record_delay(RateLimitGroup group, Duration waited) {
443 auto& counters = stats_.group(group);
444 ++counters.delayed_;
445 counters.waited_ += waited;
446 }
447
452 void record_shed(RateLimitGroup group) { ++stats_.group(group).shed_; }
453
462 void hold(Duration duration) {
463 ++stats_.broker_rejections_;
464 bucket_.cool_down(RateLimitBucket::ClockType::now(), duration);
465 }
466
477 void shrink(int burst, int per_second, Duration retry) {
478 // The bucket validates first, so a refused figure leaves the reported budget the one in force.
479 bucket_.observe_limit(RateLimitBucket::ClockType::now(), burst, per_second, retry);
480 ++stats_.broker_rejections_;
481 apply_budget(burst, per_second);
482 }
483
488 const RateLimitBudget& budget() const noexcept { return budget_; }
489
494 const RateLimitStats& stats() const noexcept { return stats_; }
495
496private:
497 // Records a new allowance, keeping the reserve smaller than the burst.
498 void apply_budget(int burst, int per_second) {
499 budget_.burst_ = burst;
500 budget_.per_second_ = per_second;
501 if (budget_.trading_reserve_ >= burst) {
502 budget_.trading_reserve_ = burst > 1 ? burst - 1 : 0;
503 }
504 stats_.burst_ = burst;
505 stats_.per_second_ = per_second;
506 }
507
508 RateLimitBudget budget_;
509 RateLimitBucket bucket_;
510 RateLimitStats stats_{};
511};
