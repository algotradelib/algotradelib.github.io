141inline constexpr int usable_burst(const RateLimitBudget& budget, RateLimitGroup group) noexcept {
142 switch (group) {
143 // An order, and the token it needs, may use the reserve.
144 case RateLimitGroup::TRADING:
145 case RateLimitGroup::AUTH:
146 return budget.burst_;
147 case RateLimitGroup::ACCOUNT:
148 case RateLimitGroup::MARKET_DATA:
149 case RateLimitGroup::REFERENCE:
150 break;
151 }
152 const int reserved = budget.burst_ - budget.trading_reserve_;
153 return reserved > 0 ? reserved : 1;
154}

167class BasicRateLimitBucket {
168public:
170 using ClockType = Clock;
171
173 using TimePoint = typename Clock::time_point;
174
176 using Duration = std::chrono::nanoseconds;
177
184 BasicRateLimitBucket(int burst, int per_second) { resize(burst, per_second); }
185
191 explicit BasicRateLimitBucket(const RateLimitBudget& budget)
192 : BasicRateLimitBucket(budget.burst_, budget.per_second_) {}
193
200 void resize(int burst, int per_second) {
201 if (burst < 1) {
202 throw std::invalid_argument("rate limit burst must be at least one request");
203 }
204 if (per_second < 1) {
205 throw std::invalid_argument("rate limit rate must be at least one request per second");
206 }
207 burst_ = burst;
208 per_second_ = per_second;
209 interval_ = Duration(std::chrono::nanoseconds(std::chrono::seconds(1)).count() / per_second);
210 }
211
218 Duration delay_until(TimePoint now, int allowance) const {
219 if (!started_) {
220 return Duration::zero();
221 }
222 const auto earliest = schedule_ - tolerance(allowance);
223 if (now >= earliest) {
224 return Duration::zero();
225 }
226 return std::chrono::duration_cast<Duration>(earliest - now);
227 }
228
235 bool ready(TimePoint now, int allowance) const {
236 return delay_until(now, allowance) == Duration::zero();
237 }
238
246 bool try_acquire(TimePoint now, int allowance) {
247 if (!ready(now, allowance)) {
248 return false;
249 }
250 if (!started_) {
251 schedule_ = now + interval_;
252 started_ = true;
253 return true;
254 }
255 schedule_ = (schedule_ > now ? schedule_ : now) + interval_;
256 return true;
257 }
258
268 void cool_down(TimePoint now, Duration duration) {
269 if (duration < Duration::zero()) {
270 duration = Duration::zero();
271 }
272 const auto held = now + duration + tolerance(burst_);
273 if (!started_ || held > schedule_) {
274 schedule_ = held;
275 started_ = true;
276 }
277 }
278
287 void observe_limit(TimePoint now, int burst, int per_second, Duration refill) {
288 resize(burst, per_second);
289 cool_down(now, refill > Duration::zero() ? refill : interval_);
290 }
291
298 int available(TimePoint now, int allowance) const {
299 if (!started_) {
300 return allowance;
301 }
302 const auto earliest = schedule_ - tolerance(allowance);
303 if (now < earliest) {
304 return 0;
305 }
306 const auto spare = std::chrono::duration_cast<Duration>(now - earliest);
307 const auto slots = spare / interval_ + 1;
308 return slots > allowance ? allowance : static_cast<int>(slots);
309 }
310
315 int burst() const noexcept { return burst_; }
316
321 int per_second() const noexcept { return per_second_; }
322
327 Duration interval() const noexcept { return interval_; }
328
329private:
330 // Returns how far ahead of now the schedule may run for `allowance`.
331 Duration tolerance(int allowance) const noexcept {
332 const int slots = allowance < 1 ? 1 : (allowance > burst_ ? burst_ : allowance);
333 return interval_ * (slots - 1);
334 }
335
336 int burst_{1};
337 int per_second_{1};
338 Duration interval_{std::chrono::seconds(1)};
339 TimePoint schedule_{};
340 bool started_{false};
341};

400class RateLimitController {
401public:
403 using Duration = RateLimitBucket::Duration;
404
410 explicit RateLimitController(RateLimitBudget budget = {})
411 : budget_(budget), bucket_(budget) {
412 stats_.burst_ = budget.burst_;
413 stats_.per_second_ = budget.per_second_;
414 }
415
421 Duration delay_until(RateLimitGroup group) const {
422 return bucket_.delay_until(RateLimitBucket::ClockType::now(), usable_burst(budget_, group));
423 }
424
430 bool acquire(RateLimitGroup group) {
431 const auto now = RateLimitBucket::ClockType::now();
432 if (!bucket_.try_acquire(now, usable_burst(budget_, group))) {
433 return false;
434 }
435 ++stats_.group(group).admitted_;
436 stats_.available_ = bucket_.available(now, usable_burst(budget_, group));
437 return true;
438 }
439
445 void record_delay(RateLimitGroup group, Duration waited) {
446 auto& counters = stats_.group(group);
447 ++counters.delayed_;
448 counters.waited_ += waited;
449 }
450
455 void record_shed(RateLimitGroup group) { ++stats_.group(group).shed_; }
456
465 void hold(Duration duration) {
466 ++stats_.broker_rejections_;
467 bucket_.cool_down(RateLimitBucket::ClockType::now(), duration);
468 }
469
480 void shrink(int burst, int per_second, Duration retry) {
481 // The bucket validates first, so a refused figure leaves the reported budget the one in force.
482 bucket_.observe_limit(RateLimitBucket::ClockType::now(), burst, per_second, retry);
483 ++stats_.broker_rejections_;
484 apply_budget(burst, per_second);
485 }
486
491 const RateLimitBudget& budget() const noexcept { return budget_; }
492
497 const RateLimitStats& stats() const noexcept { return stats_; }
498
499private:
500 // Records a new allowance, keeping the reserve smaller than the burst.
501 void apply_budget(int burst, int per_second) {
502 budget_.burst_ = burst;
503 budget_.per_second_ = per_second;
504 if (budget_.trading_reserve_ >= burst) {
505 budget_.trading_reserve_ = burst > 1 ? burst - 1 : 0;
506 }
507 stats_.burst_ = burst;
508 stats_.per_second_ = per_second;
509 }
510
511 RateLimitBudget budget_;
512 RateLimitBucket bucket_;
513 RateLimitStats stats_{};
514};
