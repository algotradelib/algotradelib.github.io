140class BasicRateLimitBucket {
141public:
143 using ClockType = Clock;
144
146 using TimePoint = typename Clock::time_point;
147
149 using Duration = std::chrono::nanoseconds;
150
157 BasicRateLimitBucket(int burst, int per_second) { resize(burst, per_second); }
158
164 explicit BasicRateLimitBucket(const RateLimitBudget& budget)
165 : BasicRateLimitBucket(budget.burst_, budget.per_second_) {}
166
173 void resize(int burst, int per_second) {
174 if (burst < 1) {
175 throw std::invalid_argument("rate limit burst must be at least one request");
176 }
177 if (per_second < 1) {
178 throw std::invalid_argument("rate limit rate must be at least one request per second");
179 }
180 burst_ = burst;
181 per_second_ = per_second;
182 interval_ = Duration(std::chrono::nanoseconds(std::chrono::seconds(1)).count() / per_second);
183 }
184
191 Duration delay_until(TimePoint now, int allowance) const {
192 if (!started_) {
193 return Duration::zero();
194 }
195 const auto earliest = schedule_ - tolerance(allowance);
196 if (now >= earliest) {
197 return Duration::zero();
198 }
199 return std::chrono::duration_cast<Duration>(earliest - now);
200 }
201
208 bool ready(TimePoint now, int allowance) const {
209 return delay_until(now, allowance) == Duration::zero();
210 }
211
219 bool try_acquire(TimePoint now, int allowance) {
220 if (!ready(now, allowance)) {
221 return false;
222 }
223 if (!started_) {
224 schedule_ = now + interval_;
225 started_ = true;
226 return true;
227 }
228 schedule_ = (schedule_ > now ? schedule_ : now) + interval_;
229 return true;
230 }
231
241 void cool_down(TimePoint now, Duration duration) {
242 if (duration < Duration::zero()) {
243 duration = Duration::zero();
244 }
245 const auto held = now + duration + tolerance(burst_);
246 if (!started_ || held > schedule_) {
247 schedule_ = held;
248 started_ = true;
249 }
250 }
251
260 void observe_limit(TimePoint now, int burst, int per_second, Duration refill) {
261 resize(burst, per_second);
262 cool_down(now, refill > Duration::zero() ? refill : interval_);
263 }
264
271 int available(TimePoint now, int allowance) const {
272 if (!started_) {
273 return allowance;
274 }
275 const auto earliest = schedule_ - tolerance(allowance);
276 if (now < earliest) {
277 return 0;
278 }
279 const auto spare = std::chrono::duration_cast<Duration>(now - earliest);
280 const auto slots = spare / interval_ + 1;
281 return slots > allowance ? allowance : static_cast<int>(slots);
282 }
283
288 int burst() const noexcept { return burst_; }
289
294 int per_second() const noexcept { return per_second_; }
295
300 Duration interval() const noexcept { return interval_; }
301
302private:
303 // Returns how far ahead of now the schedule may run for `allowance`.
304 Duration tolerance(int allowance) const noexcept {
305 const int slots = allowance < 1 ? 1 : (allowance > burst_ ? burst_ : allowance);
306 return interval_ * (slots - 1);
307 }
308
309 int burst_{1};
310 int per_second_{1};
311 Duration interval_{std::chrono::seconds(1)};
312 TimePoint schedule_{};
313 bool started_{false};
314};

373class RateLimitController {
374public:
376 using Duration = RateLimitBucket::Duration;
377
383 explicit RateLimitController(RateLimitBudget budget = {})
384 : budget_(budget), bucket_(budget) {
385 stats_.burst_ = budget.burst_;
386 stats_.per_second_ = budget.per_second_;
387 }
388
394 Duration delay_until(RateLimitGroup group) const {
395 return bucket_.delay_until(RateLimitBucket::ClockType::now(), usable_burst(budget_, group));
396 }
397
403 bool acquire(RateLimitGroup group) {
404 const auto now = RateLimitBucket::ClockType::now();
405 if (!bucket_.try_acquire(now, usable_burst(budget_, group))) {
406 return false;
407 }
408 ++stats_.group(group).admitted_;
409 stats_.available_ = bucket_.available(now, usable_burst(budget_, group));
410 return true;
411 }
412
418 void record_delay(RateLimitGroup group, Duration waited) {
419 auto& counters = stats_.group(group);
420 ++counters.delayed_;
421 counters.waited_ += waited;
422 }
423
428 void record_shed(RateLimitGroup group) { ++stats_.group(group).shed_; }
429
438 void hold(Duration duration) {
439 ++stats_.broker_rejections_;
440 bucket_.cool_down(RateLimitBucket::ClockType::now(), duration);
441 }
442
453 void shrink(int burst, int per_second, Duration retry) {
454 // The bucket validates first, so a refused figure leaves the reported budget the one in force.
455 bucket_.observe_limit(RateLimitBucket::ClockType::now(), burst, per_second, retry);
456 ++stats_.broker_rejections_;
457 apply_budget(burst, per_second);
458 }
459
464 const RateLimitBudget& budget() const noexcept { return budget_; }
465
470 const RateLimitStats& stats() const noexcept { return stats_; }
471
472private:
473 // Records a new allowance, keeping the reserve smaller than the burst.
474 void apply_budget(int burst, int per_second) {
475 budget_.burst_ = burst;
476 budget_.per_second_ = per_second;
477 if (budget_.trading_reserve_ >= burst) {
478 budget_.trading_reserve_ = burst > 1 ? burst - 1 : 0;
479 }
480 stats_.burst_ = burst;
481 stats_.per_second_ = per_second;
482 }
483
484 RateLimitBudget budget_;
485 RateLimitBucket bucket_;
486 RateLimitStats stats_{};
487};
