146inline constexpr int usable_burst(const RateLimitBudget& budget, RateLimitGroup group) noexcept {
147 switch (group) {
148 // An order, and the token it needs, may use the reserve.
149 case RateLimitGroup::TRADING:
150 case RateLimitGroup::AUTH:
151 return budget.burst_;
152 case RateLimitGroup::ACCOUNT:
153 case RateLimitGroup::MARKET_DATA:
154 case RateLimitGroup::REFERENCE:
155 break;
156 }
157 const int reserved = budget.burst_ - budget.trading_reserve_;
158 return reserved > 0 ? reserved : 1;
159}

172class BasicRateLimitBucket {
173public:
175 using ClockType = Clock;
176
178 using TimePoint = typename Clock::time_point;
179
181 using Duration = std::chrono::nanoseconds;
182
189 BasicRateLimitBucket(int burst, int per_second) {
190 resize(burst, per_second);
191 }
192
198 explicit BasicRateLimitBucket(const RateLimitBudget& budget)
199 : BasicRateLimitBucket(budget.burst_, budget.per_second_) {}
200
207 void resize(int burst, int per_second) {
208 if (burst < 1) {
209 throw std::invalid_argument("rate limit burst must be at least one request");
210 }
211 if (per_second < 1) {
212 throw std::invalid_argument("rate limit rate must be at least one request per second");
213 }
214 burst_ = burst;
215 per_second_ = per_second;
216 interval_ = Duration(std::chrono::nanoseconds(std::chrono::seconds(1)).count() / per_second);
217 }
218
225 Duration delay_until(TimePoint now, int allowance) const {
226 if (!started_) {
227 return Duration::zero();
228 }
229 const auto earliest = schedule_ - tolerance(allowance);
230 if (now >= earliest) {
231 return Duration::zero();
232 }
233 return std::chrono::duration_cast<Duration>(earliest - now);
234 }
235
242 bool ready(TimePoint now, int allowance) const {
243 return delay_until(now, allowance) == Duration::zero();
244 }
245
253 bool try_acquire(TimePoint now, int allowance) {
254 if (!ready(now, allowance)) {
255 return false;
256 }
257 if (!started_) {
258 schedule_ = now + interval_;
259 started_ = true;
260 return true;
261 }
262 schedule_ = (schedule_ > now ? schedule_ : now) + interval_;
263 return true;
264 }
265
275 void cool_down(TimePoint now, Duration duration) {
276 if (duration < Duration::zero()) {
277 duration = Duration::zero();
278 }
279 const auto held = now + duration + tolerance(burst_);
280 if (!started_ || held > schedule_) {
281 schedule_ = held;
282 started_ = true;
283 }
284 }
285
294 void observe_limit(TimePoint now, int burst, int per_second, Duration refill) {
295 resize(burst, per_second);
296 cool_down(now, refill > Duration::zero() ? refill : interval_);
297 }
298
305 int available(TimePoint now, int allowance) const {
306 if (!started_) {
307 return allowance;
308 }
309 const auto earliest = schedule_ - tolerance(allowance);
310 if (now < earliest) {
311 return 0;
312 }
313 const auto spare = std::chrono::duration_cast<Duration>(now - earliest);
314 const auto slots = spare / interval_ + 1;
315 return slots > allowance ? allowance : static_cast<int>(slots);
316 }
317
322 int burst() const noexcept {
323 return burst_;
324 }
325
330 int per_second() const noexcept {
331 return per_second_;
332 }
333
338 Duration interval() const noexcept {
339 return interval_;
340 }
341
342private:
343 // Returns how far ahead of now the schedule may run for `allowance`.
344 Duration tolerance(int allowance) const noexcept {
345 const int slots = allowance < 1 ? 1 : (allowance > burst_ ? burst_ : allowance);
346 return interval_ * (slots - 1);
347 }
348
349 int burst_{1};
350 int per_second_{1};
351 Duration interval_{std::chrono::seconds(1)};
352 TimePoint schedule_{};
353 bool started_{false};
354};

413class RateLimitController {
414public:
416 using Duration = RateLimitBucket::Duration;
417
423 explicit RateLimitController(RateLimitBudget budget = {}) : budget_(budget), bucket_(budget) {
424 stats_.burst_ = budget.burst_;
425 stats_.per_second_ = budget.per_second_;
426 }
427
433 Duration delay_until(RateLimitGroup group) const {
434 return bucket_.delay_until(RateLimitBucket::ClockType::now(), usable_burst(budget_, group));
435 }
436
442 bool acquire(RateLimitGroup group) {
443 const auto now = RateLimitBucket::ClockType::now();
444 if (!bucket_.try_acquire(now, usable_burst(budget_, group))) {
445 return false;
446 }
447 ++stats_.group(group).admitted_;
448 stats_.available_ = bucket_.available(now, usable_burst(budget_, group));
449 return true;
450 }
451
457 void record_delay(RateLimitGroup group, Duration waited) {
458 auto& counters = stats_.group(group);
459 ++counters.delayed_;
460 counters.waited_ += waited;
461 }
462
467 void record_shed(RateLimitGroup group) {
468 ++stats_.group(group).shed_;
469 }
470
479 void hold(Duration duration) {
480 ++stats_.broker_rejections_;
481 bucket_.cool_down(RateLimitBucket::ClockType::now(), duration);
482 }
483
494 void shrink(int burst, int per_second, Duration retry) {
495 // The bucket validates first, so a refused figure leaves the reported budget the one in force.
496 bucket_.observe_limit(RateLimitBucket::ClockType::now(), burst, per_second, retry);
497 ++stats_.broker_rejections_;
498 apply_budget(burst, per_second);
499 }
500
505 const RateLimitBudget& budget() const noexcept {
506 return budget_;
507 }
508
513 const RateLimitStats& stats() const noexcept {
514 return stats_;
515 }
516
517private:
518 // Records a new allowance, keeping the reserve smaller than the burst.
519 void apply_budget(int burst, int per_second) {
520 budget_.burst_ = burst;
521 budget_.per_second_ = per_second;
522 if (budget_.trading_reserve_ >= burst) {
523 budget_.trading_reserve_ = burst > 1 ? burst - 1 : 0;
524 }
525 stats_.burst_ = burst;
526 stats_.per_second_ = per_second;
527 }
528
529 RateLimitBudget budget_;
530 RateLimitBucket bucket_;
531 RateLimitStats stats_{};
532};
