46 std::optional<Decimal64> on(const Decimal64& quantity, const Decimal64& price) const {
47 const auto value = decimal_multiply(quantity, price);
48 if (!value) {
49 return std::nullopt;
50 }
51 auto charged = decimal_multiply(*value, rate_);
52 if (!charged) {
53 return std::nullopt;
54 }
55 // Truncated to whole currency units, as a broker charges.
56 const auto whole = decimal_divide(*charged, Decimal64::parse("1"), 0);
57 if (!whole) {
58 return std::nullopt;
59 }
60 return decimal_less(*whole, minimum_) ? minimum_ : *whole;
61 }

94inline Result<CommissionModel> commission_model_from(
95 const CommissionSchedule& schedule,
96 Country country,
97 const Date& date
98) {
99 const auto selection = schedule.rate_for(country, date);
100 if (!selection.rate_) {
101 BrokerError error;
102 error.category_ = ErrorCategory::INVALID_ARGUMENT;
103 // Nothing was asked of anybody: this reads a schedule already in hand.
104 error.reached_broker_ = false;
105 // Distinct codes: a missing row wants a rate stated; an unreadable window wants a refetch.
106 if (selection.bounds_unreadable_) {
107 error.broker_code_ = "rate_bounds_unreadable";
108 error.message_ = "the published schedule holds a commission rate for that market whose date "
109 "bounds could not be read, so whether it applies on that date is unknown";
110 return Result<CommissionModel>(std::move(error));
111 }
112 error.broker_code_ = "no_rate_for_date";
113 error.message_ = "the published schedule states no commission rate for that market and date";
114 return Result<CommissionModel>(std::move(error));
115 }
116 const auto& rate = selection.rate_;
117 auto model = CommissionModel::of(rate->rate_);
118 // A caller's stated schedule stays the caller's: the report says where the rate it charged came from.
119 model.from_broker_ = commission_origin_is_broker(schedule.origin_);
120 return Result<CommissionModel>(std::move(model));
121}

270 Result<CommissionCharge> charge(
271 const Decimal64& quantity,
272 const Decimal64& price,
273 const Date& date,
274 Country country
275 ) const {
276 if (!schedule_) {
277 const auto charged = flat_.on(quantity, price);
278 if (!charged) {
279 return Result<CommissionCharge>(refusal(
280 "commission_not_representable",
281 "the commission on that execution does not fit an exact decimal"
282 ));
283 }
284 CommissionCharge result;
285 result.broker_ = *charged;
286 // A flat model is one inclusive figure: no institution split.
287 result.applied_.country_ = country;
288 result.applied_.rate_ = flat_.rate_;
289 result.applied_.minimum_ = flat_.minimum_;
290 result.applied_.origin_ = flat_.from_broker_ ?
291 CommissionOrigin::BROKER_PUBLISHED :
292 CommissionOrigin::CALLER_STATED;
293 return Result<CommissionCharge>(std::move(result));
294 }
295
296 // A market other than the dating market is refused if any of its rates is dated: the replay's
297 // date is another market's day, so which dated rate applies cannot be decided.
298 if (country != dating_market_ && schedule_->any_dated_rate_for(country)) {
299 BrokerError error;
300 error.category_ = ErrorCategory::INVALID_ARGUMENT;
301 error.reached_broker_ = false;
302 error.broker_code_ = "commission_date_not_market_local";
303 error.message_ = "this market's rates have dated bounds, and the replay's date " +
304 named(date) + " is that of another market's trading day, so which rate applies cannot be "
305 "decided here; price it in a replay whose session is that market's, or state rates with "
306 "no dated bounds";
307 return Result<CommissionCharge>(std::move(error));
308 }
309
310 const auto selection = schedule_->rate_for(country, asserted_, date);
311 const auto& rate = selection.rate_;
312 if (!rate) {
313 BrokerError error;
314 error.category_ = ErrorCategory::INVALID_ARGUMENT;
315 error.reached_broker_ = false;
316 // Name the date, and distinguish an unreadable window (refetch) from a missing rate (state one).
317 if (selection.bounds_unreadable_) {
318 error.broker_code_ = "rate_bounds_unreadable";
319 error.message_ = "the schedule holds a commission rate for " + described(asserted_) +
320 " in this instrument's market whose date bounds could not be read, so whether it applies "
321 "on " + named(date) + " is unknown; refetch the schedule or state the rates";
322 return Result<CommissionCharge>(std::move(error));
323 }
324 error.broker_code_ = "no_rate_for_date";
325 error.message_ = "the schedule states no commission rate for " + named(date) +
326 " at venue " + described(asserted_) + " in the market this instrument is listed in";
327 return Result<CommissionCharge>(std::move(error));
328 }
329
330 // No minimum on a scheduled rate: no schedule states one. A caller with a minimum uses `stated`.
331 const auto broker = component(quantity, price, rate->rate_, Decimal64::parse("0"));
332 if (!broker) {
333 return Result<CommissionCharge>(refusal(
334 "commission_not_representable",
335 "the commission on that execution does not fit an exact decimal"
336 ));
337 }
338 CommissionCharge result;
339 result.broker_ = *broker;
340 if (rate->institution_rate_) {
341 // Its own component, with no minimum (the minimum is the broker's).
342 const auto institution =
343 component(quantity, price, *rate->institution_rate_, Decimal64::parse("0"));
344 if (!institution) {
345 return Result<CommissionCharge>(refusal(
346 "commission_not_representable",
347 "the institution fee on that execution does not fit an exact decimal"
348 ));
349 }
350 result.institution_ = *institution;
351 }
352 result.applied_.country_ = country;
353 result.applied_.venue_ = rate->venue_;
354 result.applied_.rate_ = rate->rate_;
355 result.applied_.institution_rate_ = rate->institution_rate_;
356 result.applied_.from_ = rate->from_;
357 result.applied_.until_ = rate->until_;
358 result.applied_.origin_ = schedule_->origin_;
359 return Result<CommissionCharge>(std::move(result));
360 }

425struct SlippageModel {
427 Decimal64 basis_points_{Decimal64::parse("0")};
429 int ticks_{0};
430
444 std::optional<std::pair<Decimal64, Decimal64>> apply(
445 const Decimal64& price,
446 OrderSide side,
447 const PriceRule& rule
448 ) const {
449 auto moved = price;
450
451 if (basis_points_.coefficient() != 0) {
452 const auto fraction = decimal_divide(basis_points_, Decimal64::parse("10000"), 8);
453 if (!fraction) {
454 return std::nullopt;
455 }
456 const auto amount = decimal_multiply(price, *fraction);
457 if (!amount) {
458 return std::nullopt;
459 }
460 const auto shifted = side == OrderSide::BUY ?
461 decimal_add(moved, *amount) :
462 decimal_subtract(moved, *amount);
463 if (!shifted) {
464 return std::nullopt;
465 }
466 moved = *shifted;
467 }
468
469 if (ticks_ > 0) {
470 // One step at a time, each with the tick where it starts: the first tick of a band is not the
471 // last tick of the band below.
472 const auto zero = Decimal64::parse("0");
473 auto walked = price;
474 for (int step = 0; step < ticks_; ++step) {
475 const auto tick = side == OrderSide::BUY ?
476 rule.tick_at(walked) :
477 tick_below(rule, walked);
478 if (!tick) {
479 // An unknown grid, or the bottom of a known one, contributes nothing rather than a guessed step.
480 break;
481 }
482 const auto next = side == OrderSide::BUY ?
483 decimal_add(walked, *tick) :
484 decimal_subtract(walked, *tick);
485 if (!next) {
486 return std::nullopt;
487 }
488 if (!decimal_less(zero, *next)) {
489 // No price at or below zero.
490 break;
491 }
492 walked = *next;
493 }
494 const auto steps = side == OrderSide::BUY ?
495 decimal_subtract(walked, price) :
496 decimal_subtract(price, walked);
497 if (!steps) {
498 return std::nullopt;
499 }
500 const auto shifted = side == OrderSide::BUY ?
501 decimal_add(moved, *steps) :
502 decimal_subtract(moved, *steps);
503 if (!shifted) {
504 return std::nullopt;
505 }
506 moved = *shifted;
507 }
508
509 const auto given_up = side == OrderSide::BUY ?
510 decimal_subtract(moved, price) :
511 decimal_subtract(price, moved);
512 if (!given_up) {
513 return std::nullopt;
514 }
515 return std::pair<Decimal64, Decimal64>{moved, *given_up};
516 }
517
522 static SlippageModel none() { return SlippageModel{}; }
523
529 static SlippageModel of_basis_points(Decimal64 points) {
530 SlippageModel model;
531 model.basis_points_ = std::move(points);
532 return model;
533 }
534
540 static SlippageModel of_ticks(int count) {
541 SlippageModel model;
542 model.ticks_ = count;
543 return model;
544 }
545
546private:
547 // The tick of the band below a price: the step down to the next lower grid price, for a price on a
548 // grid like KRX's. None at or below the first band's start, or without bands.
549 static std::optional<Decimal64> tick_below(const PriceRule& rule, const Decimal64& price) {
550 std::optional<Decimal64> tick;
551 for (const auto& band : rule.bands_) {
552 if (!decimal_less(band.from_, price)) {
553 break;
554 }
555 tick = band.tick_;
556 }
557 return tick;
558 }
