89template <typename Out, typename Handler>
90void post_refusal(Handler&& handler) {
91 auto executor = net::get_associated_executor(handler, net::system_executor());
92 auto complete = [handler = std::forward<Handler>(handler)]() mutable {
93 std::move(handler)(Result<Out>(refusal()));
94 };
95 if constexpr (detail::postable<decltype(executor)>::value) {
96 net::post(executor, std::move(complete));
97 } else {
98 net::post(net::system_executor(), net::bind_executor(executor, std::move(complete)));
99 }
100}
