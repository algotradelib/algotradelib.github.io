143template <typename Session, typename Runner>
144bool load_reference_data(Session& session, Runner& bound, const Settings& settings, std::ostream& out) {
145 for (const auto& instrument : settings.universe_) {
146 auto reference = session.instrument(instrument);
147 if (!reference) {
148 out << "could not read reference data for " << instrument.symbol_ << ": "
149 << reference.error().message_ << '\n';
150 return false;
151 }
152 bound.context().set_instrument(std::move(reference.value()));
153 }
154 return true;
155}

162inline void print_recovery(const runner::RecoveryReport& recovery, std::ostream& out) {
163 if (recovery.clean()) {
164 out << "recovery: clean -- the journal owes nothing\n";
165 return;
166 }
167 out << "recovery: NOT clean\n";
168 for (const auto& instrument : recovery.marked_) {
169 out << " withheld until the broker's lists resolve it: " << instrument.symbol_ << '\n';
170 }
171 if (recovery.truncated_) {
172 out << " the last line was torn: the previous process died while writing it\n";
173 }
174 if (recovery.unreadable_lines_ > 0) {
175 out << " " << recovery.unreadable_lines_ << " line(s) this library did not write\n";
176 }
177 if (recovery.unreadable_file_) {
178 out << " the journal exists and could not be read\n";
179 }
180 if (recovery.unattributable_) {
181 out << " an unresolved instruction names no instrument\n";
182 }
183 if (!recovery.detail_.empty()) {
184 out << " detail: " << recovery.detail_ << '\n';
185 }

193inline void print_how_to_stop(const Settings& settings, std::ostream& out) {
194 out << "to stop: Ctrl-C or SIGTERM (quiesce), or\n"
195 << " echo quiesce > " << settings.control_file_ << " # stop placing, leave orders resting\n"
196 << " echo withdraw > " << settings.control_file_ << " # cancel the account's working orders\n"
197 << " echo flatten > " << settings.control_file_ << " # withdraw, then close the account's positions\n"
198 << "the file is not deleted: a word left in it is carried out at the next start; remove it\n"
199 << "before trading again\n";
200 if (settings.closing_wait_ > std::chrono::nanoseconds::zero()) {
201 out << "a withdraw or a flatten may first wait up to "
202 << std::chrono::duration_cast<std::chrono::seconds>(settings.closing_wait_).count()
203 << " s for the order list and holdings to account for this\n"
204 << "program's own orders. ";
205 }
206 out << "A supervisor's grace period must cover that stop: the wait, the calls in\n"
207 << "flight (a paginated read, a paced request) and the closing orders themselves, with margin; a\n"
208 << "SIGTERM does not turn a pending flatten into a quiesce\n";
209}

216inline void print_loop(const runner::LoopReport& report, std::ostream& out) {
217 out << "\niterations : " << report.iterations_ << '\n'
218 << "order list read : " << (report.order_list_read_ ? "yes" : "no") << '\n'
219 << "polls failed : bars " << report.failed(runner::PollKind::BARS) << '/'
220 << report.attempted(runner::PollKind::BARS) << ", holdings "
221 << report.failed(runner::PollKind::HOLDINGS) << '/' << report.attempted(runner::PollKind::HOLDINGS)
222 << ", orders " << report.failed(runner::PollKind::WORKING_ORDERS) << '/'
223 << report.attempted(runner::PollKind::WORKING_ORDERS) << '\n'
224 << "blocked : " << report.blocked_ << " iteration(s)\n"
225 << "stopped by : "
226 << (report.stopped_by_ ? runner::stop_intent_name(*report.stopped_by_) : std::string_view("nothing"))
227 << '\n'
228 << "final gate : "
229 << (report.final_gate_ ? strategy::halt_reason_name(*report.final_gate_) : std::string_view("none"))
230 << '\n'
231 << "market phase : " << strategy::market_phase_name(report.final_phase_) << '\n'
232 // Every way an order the strategy wanted did not go out, each with its own count, so an empty
233 // placement count explains itself.
234 << "plans withheld : " << report.plans_withheld_ << " (account unread, too old or older than the order listing, or a hold while it would cancel)\n"
235 << "not buildable : " << report.orders_unbuildable_ << '\n'
236 << "withheld : " << report.orders_reconcile_withheld_ << " (unresolved earlier orders)\n"
237 << "refused by guard: " << report.orders_guard_refused_ << '\n'
238 << "refused locally : " << report.orders_binding_refused_ << " (by the adapter, before sending)\n"
239 << "refused by broker: " << report.orders_broker_refused_ << '\n'
240 << "unknown outcome : " << report.orders_indeterminate_ << " (read the broker's order list)\n"
241 // Cancellations that did not go through, which no line above counts. Accepted ones are not here.
242 << "cancels : " << report.cancels_rejected_ << " rejected (by the broker or the adapter), "
243 << report.cancels_indeterminate_ << " of unknown outcome, " << report.cancels_refused_
244 << " refused by a hold, " << report.cancels_unresolved_ << " for an order not in the picture\n"
245 << "unconfirmed : " << report.orders_unconfirmed_timed_out_
246 << " (sent, and not confirmed in time: its instrument waits for holdings or its history, or for you if "
247 "it was never named or the runner overlaps orders)\n";
248 if (report.last_unbuildable_) {
249 out << "last unbuildable: " << strategy::reject_reason_name(report.last_unbuildable_->reason_) << " for "
250 << report.last_unbuildable_->instrument_.symbol_
251 << (report.last_unbuildable_->tag_.empty() ? "" : " (" + report.last_unbuildable_->tag_ + ")") << '\n';
252 }
253 if (report.last_broker_refusal_) {
254 out << "last refusal : " << report.last_broker_refusal_->code_ << ' '
255 << report.last_broker_refusal_->message_ << '\n';
256 }
257 if (report.abandoned_) {
258 out << "ABANDONED : after repeated failures polling "
259 << (report.abandoned_kind_ ? runner::poll_kind_name(*report.abandoned_kind_) : std::string_view("?"))
260 << '\n';
261 }
262 for (const auto& note : report.notes_) {
263 out << "note : " << note << '\n';
264 }

324template <typename Strategy, typename Session, typename MarketData, typename Clock = runner::SteadyClock>
325Exit trade(
326 Session& session,
327 MarketData& market_data,
328 Strategy strategy_instance,
329 const Settings& settings,
330 runner::StopSource& stop,
331 std::ostream& out,
332 Clock clock = {}
333) {
334 runner::FileJournal journal(settings.journal_path_, settings.journal_scope_);
335 if (!journal.open()) {
336 out << "journal refused: " << journal.rejection() << '\n'
337 << "a runner that cannot record what it sends does not send\n";
338 return Exit::JOURNAL_REFUSED;
339 }
340
341 auto bound = live::bind_journaled(
342 session, journal, std::move(strategy_instance), settings.limits_, live::default_series_capacity,
343 settings.default_route_
344 );
345
346 const auto recovery = runner::recover_runner(bound, journal);
347 print_recovery(recovery, out);
348 if (const auto failure = bound.local_failure()) {
349 // Damage that names no single instrument halts everything, and restarting will find the same
350 // line. There is deliberately no flag to trade past it: a flag that is needed on every restart
351 // ends up in a unit file. The way out is by hand, and it keeps the evidence.
352 out << "HALTED: " << *failure << '\n'
353 << "nothing was started. This program's only way on gives up the whole journal. That is safe\n"
354 << "only once no order the journal may track can execute any more:\n"
355 << " 1. read the broker's order list, order history and holdings for this account yourself.\n"
356 << " A damaged journal may not name every order it tracks: take every order sent on this\n"
357 << " account since the journal began, on every day and every page of the order history, and\n"
358 << " every order still working;\n"
359 << " 2. each must have ended: filled in full, or cancelled with its history row saying so. A\n"
360 << " cancellation only requested, an order list that no longer shows it, and the close of the\n"
361 << " session do not show that an order has ended;\n"
362 << " 3. holdings must show the final fills of each;\n"
363 << " 4. only then move the journal aside, keeping it: mv " << settings.journal_path_ << ' '
364 << settings.journal_path_ << ".damaged-$(date +%Y%m%d)\n"
365 << "the next start then begins a new journal and sizes from the holdings and working orders the\n"
366 << "broker reports. If steps 2 and 3 cannot be shown for every order, leave the journal where it\n"
367 << "is. Do NOT move it aside while an order it may track is left working: a new journal does not\n"
368 << "know the order, and when it fills before holdings show it, the order is sent again.\n";
369 return Exit::RECOVERY_HALTED;
370 }
371
372 if (!load_reference_data(market_data, bound, settings, out)) {
373 return Exit::REFERENCE_UNREADABLE;
374 }
375 // Said before trading as well as in the report at the end, so it is read before it matters.
376 if constexpr (decltype(bound)::on_fill_never_fires) {
377 out << "note: this strategy defines on_fill, which never fires on a live runner; anything it\n"
378 << " does there (a protective stop, say) will not happen. See writing-a-strategy.md.\n";
379 }
380
381 live::SessionSource<Session, runner::FileJournal, MarketData> source(
382 session, market_data, settings.universe_
383 );
384 source.set_journal(journal);
385 source.set_bar_query(settings.bar_interval_, settings.bar_count_);
386
387 runner::PollingLoop<decltype(bound), decltype(source), Clock> loop(
388 bound, source, settings.schedule_, std::move(clock)
389 );
390 configure(loop, settings);
391
392 print_how_to_stop(settings, out);
393 const auto report = loop.run(stop, settings.max_iterations_);
394 print_report(report, out);
395
396 // Checked before abandonment: sticky, set by a failed journal write as much as by recovery, and
397 // the more urgent of the two -- what the runner was sending at that moment is unknown. A run can
398 // end both ways, and the report above still says ABANDONED.
399 if (const auto failure = bound.local_failure()) {
400 out << "LOCAL FAILURE : " << *failure << '\n'
401 << " read the broker's order list before starting again\n";
402 return Exit::LOCAL_FAILURE;
403 }
404 if (report.abandoned_) {
405 return Exit::LOOP_ABANDONED;
406 }
407 if (stop_closes(report) && !report.closing_complete()) {
408 return Exit::CLOSING_INCOMPLETE;
409 }
410 if (report.owed_stop_) {
411 out << "OWED ORDER : the run stopped placing by itself (see the note above). A supervisor must not\n"
412 << " restart it: the order is restored from the journal and the run stops again.\n"
413 << "this journal still owes:\n";
414 for (const auto& owed : bound.obligations()) {
415 out << " " << owed.instrument_.symbol_ << ' ' << (owed.side_ == atl::OrderSide::BUY ? "buy " : "sell ")
416 << owed.quantity_.str() << ", sent " << atl::format_iso8601_utc(owed.sent_at_) << ", "
417 << (owed.state_ == runner::ObligationState::OPERATOR ? "held for an operator" :
418 owed.state_ == runner::ObligationState::UNNAMED ? "no order found" :
419 "order " + owed.order_id_ + ", not yet in holdings")
420 << '\n';
421 }
422 out << "this program has no command to release one order, and its only way on gives up the whole\n"
423 << "journal. That is safe only once none of the orders above can execute any more:\n"
424 << " 1. read the broker's order list, order history and holdings for every order above;\n"
425 << " 2. each must have ended: filled in full, or cancelled with its history row saying so. A\n"
426 << " cancellation only requested, an order list that no longer shows it, and the close of the\n"
427 << " session do not show that an order has ended;\n"
428 << " 3. holdings must show the final fills of each;\n"
429 << " 4. only then move the journal aside, keeping it: mv " << settings.journal_path_ << ' '
430 << settings.journal_path_ << ".owed-$(date +%Y%m%d)\n"
431 << "the next start then begins a new journal and sizes from the account as it is.\n"
432 << "Do NOT move the journal aside while an order it tracks is left working: a new journal does not\n"
433 << "know the order, and when it fills before holdings show it, the order is sent again. A program\n"
434 << "of your own releases that instrument's obligations instead (Runner::release_obligations; see\n"
435 << "set_owed_order_limit) and keeps the rest of the journal.\n";
436 return Exit::OWED_ORDER;
437 }
438 return Exit::CLEAN;
439}

491template <typename Strategy, typename Session, typename MarketData, typename Clock = runner::SteadyClock>
492Exit observe(
493 Session& session,
494 MarketData& market_data,
495 Strategy strategy_instance,
496 const Settings& settings,
497 runner::StopSource& stop,
498 std::ostream& out,
499 Clock clock = {}
500) {
501 auto bound = live::bind_observer(session, std::move(strategy_instance), settings.limits_);
502 if (!load_reference_data(market_data, bound, settings, out)) {
503 return Exit::REFERENCE_UNREADABLE;
504 }
505
506 live::SessionSource<Session, runner::MemoryJournal, MarketData> source(
507 session, market_data, settings.universe_
508 );
509 source.set_bar_query(settings.bar_interval_, settings.bar_count_);
510 runner::PollingLoop<decltype(bound), decltype(source), Clock> loop(
511 bound, source, settings.schedule_, std::move(clock)
512 );
513 configure(loop, settings);
514
515 out << "observing: nothing can be sent (pass --live to trade)\n";
516 print_how_to_stop(settings, out);
517 const auto report = loop.run(stop, settings.max_iterations_);
518 print_loop(report, out);
519 // `closing_complete()` is deliberately not printed: an observer's executor accepts every
520 // instruction without sending it, so "every closing instruction was taken" would be true and would
521 // say nothing about the broker.
522 out << "closing : simulated -- the observer sent nothing, so nothing at the broker was\n"
523 << " placed or withdrawn by this run\n";
524 return report.abandoned_ ? Exit::LOOP_ABANDONED : Exit::CLEAN;
525}
