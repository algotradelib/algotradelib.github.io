435 ChargedCosts since(const ChargedCosts& before) const {
436 ChargedCosts delta;
437 std::size_t refused = 0;
438 const auto difference = [&refused](
439 const std::vector<StatedTotal>& now,
440 const std::vector<StatedTotal>& then
441 ) {
442 std::vector<StatedTotal> out;
443 for (const auto& held : now) {
444 auto earlier = Decimal64::parse("0");
445 std::size_t before_statements = 0;
446 for (const auto& was : then) {
447 if (was.amount_.currency_ == held.amount_.currency_) {
448 earlier = was.amount_.amount_;
449 before_statements = was.statements_;
450 break;
451 }
452 }
453 // Whether anything was stated is the statement count, not the money.
454 if (held.statements_ == before_statements) {
455 continue;
456 }
457 const auto moved = decimal_subtract(held.amount_.amount_, earlier);
458 if (!moved) {
459 // Counted, not skipped: something was stated and its amount cannot be represented.
460 ++refused;
461 continue;
462 }
463 out.push_back(
464 StatedTotal{Money{*moved, held.amount_.currency_}, held.statements_ - before_statements}
465 );
466 }
467 return out;
468 };
469 delta.commission_ = difference(commission_, before.commission_);
470 delta.tax_ = difference(tax_, before.tax_);
471 delta.orders_stating_cost_ = orders_stating_cost_ - before.orders_stating_cost_;
472 delta.orders_stating_none_ = orders_stating_none_ - before.orders_stating_none_;
473 delta.working_list_disappearances_ =
474 working_list_disappearances_ - before.working_list_disappearances_;
475 delta.conflicting_restatements_ = conflicting_restatements_ - before.conflicting_restatements_;
476 delta.unrepresentable_sums_ =
477 (unrepresentable_sums_ - before.unrepresentable_sums_) + refused;
478 return delta;
479 }

1241 LoopReport run(StopSource& stop, std::size_t max_iterations = 0) {
1242 LoopReport report;
1243 // Carried in rather than left false: the field says whether the order list has been read at
1244 // least once, and the loop remembers that across runs. A second `run` on the same loop had been
1245 // reporting it as unread, which reads as a startup gate that never opened.
1246 report.order_list_read_ = order_list_read_;
1247 // **Differenced around the run, not accumulated from the passes.** An order can leave the
1248 // process from inside a source's `poll`, which calls `Runner::feed` itself and discards the
1249 // report it gets back, so a per-pass figure would only ever see what a stop did. The runner's
1250 // counter is monotonic for exactly this reason, and reading it at both ends is how
1251 // `RiskGuard::total_sent` is already used around a callback. Taken per `run` so a second run on
1252 // the same loop reports its own sends rather than the first one's as well.
1253 const auto outside_phase_before = runner_->sends_outside_permitted_phase();
1254 // Differenced for the reason above: both are monotone on the runner, which outlives this call.
1255 const auto bands_withheld_before = runner_->price_bands_withheld();
1256 const auto unclamped_rejected_before = runner_->unclamped_orders_rejected();
1257 const auto unclamped_refused_before = runner_->unclamped_orders_refused();
1258 // **The polling path's own figures, taken before any stop is honoured.** `honour` runs the stop
1259 // handler and the closing pass through the same `Runner::settle`, so a snapshot taken at the end
1260 // of `run` would count a refused flatten here *and* in `closing_refused_` -- one failure reported
1261 // twice, under a contract that says this family describes the polling path.
1262 const auto outcomes_before = instruction_outcomes();
1263 auto outcomes_after = outcomes_before;

1509 // Runs one iteration: poll what is due, then let the runner act if the gate is open.
1510 IterationReport step(StopSource& stop, LoopReport& report) {
1511 IterationReport iteration;
1512 iteration.at_ = clock_.now();
1513
1514 // Read before polling, so an operator's stop is honoured without first spending requests on a
1515 // broker the loop is about to walk away from.
1516 check_control_file(stop, report);
1517 // A stop discards what was retained for a better account read: nothing decided before it is to be
1518 // replayed after it.
1519 if (stop.requested()) {
1520 runner_->clear_retained_targets();
1521 }
1522
1523 // **Installed before anything can send, and announced only after the polls.** The two halves are
1524 // separate on purpose and neither is optional.
1525 //
1526 // A source's `poll` calls `Runner::feed`, which settles, so an order can leave the process from
1527 // inside a poll -- the same fact the gate below is computed early for. If the phase were installed
1528 // after the polls, the first bar arriving after the close would be settled while the context still
1529 // said `OPEN`, and an opening order would pass a phase limit that exists to refuse it. Once per
1530 // session, on the close.
1531 //
1532 // Announcing the change is the opposite case: `on_phase` may place orders, and it should see this
1533 // iteration's polls rather than the previous one's -- fresh prices, a fresh portfolio, and a gate
1534 // re-evaluated against them. That is what the replay's step does (`backtest/engine.hpp`, "The price moves with the portfolio"), with
1535 // the same argument: a handler deciding on a mark from one moment beside a price from another is
1536 // not looking at a fresher picture, it is looking at an incoherent one.
1537 install_phase(iteration);
1538
1539 // **Before anything this loop calls can feed, because several of them can.** The runner measures
1540 // a snapshot's age against the later of this and its own decision instant, and its decision
1541 // instant is a bar's period end -- which on a daily bar is yesterday's close. Set here, a plan
1542 // settled from inside `read_order_list` or a poll is aged against the clock rather than against
1543 // the event that triggered it.
1544 runner_->set_observed_at(iteration.at_);
1545 // Stated here too, and for the same reason: the runner enforces the absence rule at the moment
1546 // an instruction is decided, which is the only place that is inside every callback this loop
1547 // makes. A run that opted out says so, and the runner then places as it did before.
1548 runner_->require_known_portfolio(!allow_absent_portfolio_);
1549
1550 // **Established before any poll runs, because a poll is what feeds the runner.** A source's
1551 // `poll` calls `Runner::feed` itself -- `live::SessionSource::poll_bars` does, per instrument --
1552 // and `feed` settles, so an order can leave the process from inside a poll. Computing the gate
1553 // afterwards would mean the startup gate and the staleness gate were installed one poll too
1554 // late, and on the very first iteration they would not be installed at all: a bar arriving
1555 // before the first order-list read could place an order against a position a predecessor may
1556 // have left, which is the exact hazard the startup gate exists for.
1557 hold(gate_for(stop, iteration.at_, report, /*may_read=*/true));
1558
1559 for (const auto kind : all_poll_kinds) {
1560 if (!is_due(kind, iteration.at_)) {
1561 continue;
1562 }
1563 // **Re-established immediately before every poll**, because a poll is a point at which an order
1564 // can leave the process and anything that ran before it may have replaced the calendar the phase
1565 // was derived from. Three things can have: a previous poll in this same loop, the calendar poll
1566 // in particular since it is polled first, and `gate_for`'s out-of-band `read_order_list` above.
1567 //
1568 // Before rather than after, which is the correction: "after every poll" closed the first two and
1569 // left the third, because the pre-loop install runs *before* `gate_for` and nothing re-derived
1570 // the phase between that call and the first poll. Placing it here makes the rule positional
1571 // instead of enumerated -- every send site is preceded by an install, and no list of which polls
1572 // can change what has to stay correct.
1573 //
1574 // The cost is a date comparison per due poll.
1575 install_phase(iteration);
1576 // **And the gate, for the same reason.** The polls before this one may have read the account: on
1577 // a cold start the hold for an account never read was installed before the polls, and without
1578 // re-evaluating it here it would still stand when the bars read after this iteration's holdings
1579 // are fed -- withholding the first decision against an account that has, by then, been read.
1580 // `may_read` is false: the order list is attempted once per iteration, above.
1581 hold(gate_for(stop, iteration.at_, report, /*may_read=*/false));
1582 const auto watch = watch_phase(iteration);
1583 // **Sampled before the poll, not after it.** A poll can send: `Runner::feed` settles, and a
1584 // working-order poll now feeds `on_order_update` as well as the bar and quote polls feeding
1585 // their handlers. An order a handler places during the working-order poll was sent *after*
1586 // the listing was fetched and cannot be in it -- so counting placements afterwards would
1587 // stamp that listing as though it covered the order, and `closing_source_verified` would
1588 // then approve a flatten against a picture missing a resting order. The count taken here is
1589 // the one that was true when the listing was read.
1590 const auto placements_before_poll = runner_->placements_at_broker();
1591 const auto outcome = source_->poll(kind, *runner_);
1592 if (kind == PollKind::WORKING_ORDERS) {
1593 working_asked_at_ = clock_.monotonic_now();
1594 } else if (kind == PollKind::HOLDINGS) {
1595 holdings_asked_at_ = clock_.monotonic_now();
1596 }
1597 note_revised_phase(std::string(poll_kind_name(kind)) + " poll", watch, report);
1598 // `UNCHANGED` counts, because a listing that is identical to the last one is still a listing
1599 // -- but only on a source that publishes them at all, which `note_working_orders_read`
1600 // enforces: a broker without the inquiry answers `UNCHANGED` without asking anybody.
1601 if (kind == PollKind::WORKING_ORDERS &&
1602 (outcome == PollOutcome::UPDATED || outcome == PollOutcome::UNCHANGED)) {
1603 note_working_orders_read(iteration.at_, placements_before_poll);
1604 }
1605 ++report.attempted_[static_cast<int>(kind)];
1606 iteration.polls_.emplace_back(kind, outcome);
1607 schedule_next(kind, iteration.at_);
1608 note_outcome(kind, outcome, report);
1609 if (outcome == PollOutcome::UPDATED) {
1610 iteration.fed_ = true;
1611 }
1612 if (report.abandoned_) {
1613 // Recorded before returning, so the report does not claim sending was permitted while the
1614 // guard `note_outcome` just halted says otherwise.
1615 iteration.gate_ = runner_->guard().blocking_reason();
1616 return iteration;
1617 }
1618 }
1619
1620 // Observed after the polls, not before: a poll can refresh the listing without a bar arriving,
1621 // and that is precisely when an order of ours begins showing as resting. Observing first would
1622 // read every iteration's portfolio one iteration late. A repeat observation of unchanged state
1623 // concludes nothing, so a quiet iteration costs a scan and teaches the probe nothing.
1624 runner_->observe_sellable_convention();
1625
1626 // Re-evaluated after the polls so the report and the next iteration reflect what those polls
1627 // changed -- a holdings refresh should lift the staleness hold now, not one iteration later.
1628 // The order list is deliberately not attempted a second time: one attempt per iteration, and
1629 // the gate simply stands until the next one.
1630 // The last send site: the announcement below settles what `on_phase` asks for, and the final poll
1631 // may have replaced the calendar after its own pre-poll install.
1632 install_phase(iteration);
1633 hold(gate_for(stop, iteration.at_, report, /*may_read=*/false));
1634
1635 // Announced last, so the handler runs on everything this iteration established. A change is
1636 // announced once, not every iteration it persists for.
1637 if (announced_ != iteration.phase_) {
1638 // **The clock moves with the phase, and only forwards.** `Runner::feed(MarketPhase)` does not
1639 // advance `Context::now_`; only a bar or a quote does, from a broker's stamp. So a daily-bar run
1640 // would hand `on_phase` an instant hours old, and `RiskGuard` files a send at
1641 // `GuardInputs::now_` and counts sends within `rate_window_` of it -- the phase order and the
1642 // next bar's order would be filed at what the guard thinks are two unrelated moments, and two
1643 // orders would pass a one-order window. The replay's step (`backtest/engine.hpp`) documents the same hazard
1644 // for the replay.
1645 //
1646 // Forwards only: a broker stamp ahead of this machine's clock must not be walked backwards,
1647 // because a handler deciding at an instant *earlier* than the bar it has already been shown is
1648 // the same incoherence with the sign flipped. The skew is the broker's and is not this loop's
1649 // to quietly correct.
1650 //
1651 // A bar fed earlier in this same iteration set `now_` to its own period end, which on a daily bar
1652 // is yesterday's close. Its order is still counted in the rate window at this iteration's
1653 // reading, beside this one: the runner counts a send at the later of the decision instant and
1654 // the reading this loop gave it.
1655 if (runner_->context().now() < iteration.at_) {
1656 runner_->context().set_now(iteration.at_);
1657 }
1658 announced_ = iteration.phase_;
1659 iteration.phase_changed_ = true;
1660 // Watched for the same reason a poll is: `Context::set_calendar` is public, so an `on_phase`
1661 // handler can revise the calendar and submit in one call, and `settle` runs before this returns.
1662 const auto watch = watch_phase(iteration);
1663 runner_->feed(iteration.phase_);
1664 note_revised_phase("on_phase handler", watch, report);
1665 }
1666
1667 // **Last of all the iteration's decisions**, so every handler that could restate the instrument --
1668 // the bars, the quotes, the listing's updates, and `on_phase` just above -- has spoken first, and
1669 // its instruction has replaced the retained target; a retry before `on_phase` could send an entry
1670 // the phase change was about to countermand. And after every account poll and the gate they
1671 // refreshed, so what is retried is sized against this iteration's holdings, listing and budget,
1672 // under this iteration's holds -- inside the holdings poll it ran before the gate lifted and before
1673 // buying power was read.
1674 runner_->retry_retained_targets();
1675
1676 // Reported from the guard rather than from what the loop just computed, so the report names the
1677 // most fundamental hold in force whoever owns it -- an operator's halt, or a journal that
1678 // failed, outranks anything the loop decided.
1679 iteration.gate_ = runner_->guard().blocking_reason();
1680 return iteration;
1681 }

1684 struct PhaseWatch {
1685 std::chrono::system_clock::time_point at_{};
1686 strategy::MarketPhase phase_{strategy::MarketPhase::UNKNOWN};

1690 // Records the phase and the send count, for comparison after a call into user code.
1691 PhaseWatch watch_phase(const IterationReport& iteration) const {
1692 return PhaseWatch{iteration.at_, iteration.phase_, runner_->guard().total_sent()};
1693 }
1694
1695 // Counts and names a call that sent an order and left the phase changed behind it.
1696 //
1697 // **The one hazard here the loop cannot prevent and can only witness.** It re-derives the phase before
1698 // every call it makes that could send, but nothing of its runs *inside* such a call: a
1699 // source that replaces the calendar and then feeds, or an `on_phase` handler that does the same before
1700 // submitting, settles an order against the phase as it stood when the call began. So the loop looks
1701 // afterwards, which turns an invisible violation into a line in the report -- the same choice
1702 // `closing_source_unverified_` makes about a listing it could not trust.
1703 //
1704 // **Both halves are required, and each one removes a wrong answer.** The phase must have moved, or an
1705 // ordinary send would be reported. Something must have been sent, or an ordinary calendar poll would be
1706 // reported on every iteration -- which is why this no longer excludes the calendar poll by kind. That
1707 // exclusion was a false-positive workaround and it hid this case: a source obeying
1708 // the letter of the contract, writing the calendar in the calendar poll and *then* feeding.
1709 //
1710 // `total_sent()` rather than the guard's rate-window deque, because that one prunes and can fall, so
1711 // it cannot be differenced across a call.
1712 void note_revised_phase(const std::string& what, const PhaseWatch& before, LoopReport& report) {
1713 if (runner_->guard().total_sent() == before.sends_) {
1714 return;
1715 }
1716 if (phase_at(before.at_) == before.phase_) {
1717 return;
1718 }
1719 ++report.sends_under_a_revised_phase_;
1720 if (report.sends_under_a_revised_phase_ == 1) {
1721 // Once per run, not once per call: a strategy doing this on every bar would otherwise bury every
1722 // other note under thousands of copies of this one.
1723 // **Worded to exactly what was observed**, which is co-occurrence and not causation. Two cases
1724 // the stronger claim would have got wrong: a call that sent *first* and revised the calendar
1725 // afterwards, where no order used the stale answer; and a call that only cancelled, since
1726 // `Runner::send_cancel` never consults the phase at all. Both still break the writer contract and
1727 // are worth reporting, so the note states the facts and names the check rather than the verdict.
1728 report.notes_.push_back(
1729 "the " + what + " changed the market phase and sent a request in the same call; check whether "
1730 "that request used the earlier phase. Only the calendar poll may write the trading calendar, "
1731 "and it must not send");
1732 }
1733 }
1734
1735 // Computes the phase for this iteration and installs it, so nothing can send under a stale one.
1736 //
1737 // **Called immediately before every point at which this loop hands control to the source**: once
1738 // before `gate_for`, which performs an out-of-band order-list read; once before each due poll; and
1739 // once before the phase announcement. So the invariant is exactly this, and no stronger:
1740 // **whenever the loop calls into the source, `Context::phase_` is derived from the calendar as it
1741 // stood at that call.**
1742 //
1743 // **What that deliberately does not promise**, because it cannot: a source is free to replace the
1744 // calendar and then feed the runner inside a single `poll`, and no loop code runs in between, so
1745 // that order would settle against the phase as it stood when the poll began. `PollKind::CALENDAR`
1746 // documents the contract that rules it out, `SessionSource` satisfies it -- it writes the calendar in
1747 // the calendar poll and nowhere else -- and `note_revised_phase` reports a source that
1748 // breaks it, because a violation nobody is told about is one nobody fixes. The stronger at-send-site
1749 // invariant does not hold.
1750 //

1752 // took a second does not move the session boundary mid-iteration.

1756 // does: "this loop has never known the phase" is a claim about the loop, not about one call.
1757 if (iteration.phase_ != strategy::MarketPhase::UNKNOWN) {
1758 phase_ever_known_ = true;
1759 }
1760 runner_->context().set_phase(iteration.phase_);

1766 // per-bar answer. An uncovered date is `UNKNOWN` and never "closed": `TradingCalendar` draws that

1771 // answers from it, so the day the calendar is asked about and the day the session boundaries are
1772 // computed for cannot disagree.

1792 // Says so when a run never established the phase, and names which of the three reasons applies.
1793 //
1794 // Silent otherwise, like `note_sellable_convention`: a line on every healthy run is how a report
1795 // stops being read. But a run that placed nothing because the phase was never known must not look
1796 // like a quiet market -- and this is **not** a hold on the guard, deliberately. A hold would also
1797 // stop cancellations, which `Runner::send_cancel` refuses on any blocking reason while never
1798 // consulting the phase at all; taking one because a reference-data poll failed would make an
1799 // unknown phase more restrictive than a closed one, and a withdrawal only ever reduces exposure.
1800 // `RiskGuard` already refuses every placement that opens or increases a position with
1801 // `PHASE_NOT_PERMITTED`, and permits one that closes a held position -- which is exactly what a
1802 // hold would wrongly stop, for the same reason it would wrongly stop a cancellation.
1803 void note_unestablished_phase(LoopReport& report) const {
1804 if (report.iterations_ == 0 || report.final_phase_ != strategy::MarketPhase::UNKNOWN) {
1805 return;

1807 // **"Never" is a claim, so it is tracked rather than derived.** A loop can establish the phase and
1808 // then lose it -- the local date rolls past what the calendar covers, or the broker revises it --
1809 // and telling an operator it was never established would send them looking for a configuration
1810 // fault that is not there.

1812 // This used to read `report.phase_transitions_ > 0`, which was wrong: `announced_`
1813 // outlives a `run` while the report's counter restarts at zero, so a run with nothing left to
1814 // announce reports no transitions and would claim the phase had never been known -- of a loop that
1815 // spent its first run open. The verdict was right in the case that was tested and the reasoning was
1816 // wrong, which is the worse of the two.
1817 const bool ever_known = phase_ever_known_;

1838 std::to_string(date.month_) + "-" + std::to_string(date.day_);

1849 // Attempts the startup order-list read once, and returns whether it succeeded.
1850 //
1851 // The startup gate's read and a pending stop's both come here, so the two cannot drift apart.

1872 // does not, it reads order history for the journal and never populates `working_`, so

1878 // Returns which of the loop's *own* conditions blocks sending, reading the order list if due.
1879 //
1880 // Only conditions the loop computes, because a hold is released by whoever took it. `OPERATOR`
1881 // belongs to a person and `LOCAL_FAILURE` to the runner; a loop that took either would be a
1882 // second holder of someone else's hold -- the defect `HaltReason` exists to prevent,
1883 // reintroduced one layer up.
1884 //
1885 // **Always evaluated, even while someone else's hold stands.** Short-circuiting on a foreign hold
1886 // would leave the loop's own hold un-maintained underneath it, so the moment an operator resumed,
1887 // the guard would be wide open against a portfolio nobody had rechecked.
1888 //
1889 // Called twice per iteration -- before the polls, which is what actually protects anything, and
1890 // after them so the report is current. `may_read` is false on the second call so the order-list
1891 // read is attempted at most once per iteration.
1892 std::optional<strategy::HaltReason> gate_for(
1893 StopSource& stop,
1894 std::chrono::system_clock::time_point at,
1895 LoopReport& report,
1896 bool may_read
1897 ) {
1898 if (stop.requested()) {
1899 // **A withdraw or a flatten reads the broker's order list first, if it never has.** Each builds
1900 // its instructions from the picture, and a stop pending before the first iteration -- a process
1901 // started with `flatten` in its control file -- used to return here before the startup read, so
1902 // its closing pass was unverified and a flatten closed nothing, for any position. That also left
1903 // no way to flatten a position behind a resting stop: the in-process flatten withdraws the stop
1904 // and, rightly, sends nothing beside it, and the restart that should finish the job could not.
1905 // The read is the startup gate's own, under its conditions; a quiesce sends nothing and needs no
1906 // picture, so it does not spend the request.
1907 //
1908 // **`STOPPING` on every path, the failed read included.** `ORDER_LIST_UNREAD` lets reducing
1909 // orders through (`halt_blocks`), and nothing but the stop's own may go out once one is pending.
1910 // Held before the read as well, because the source is a template parameter and may feed.
1911 const auto intent = stop.intent();
1912 // Once a stop is waiting, on the wait's own floor: a read that failed is not asked again at
1913 // every tick. The iteration that takes the stop up reads as any other.
1914 const bool on_the_floor = !closing_ || !asked_within_floor(PollKind::WORKING_ORDERS);
1915 if (intent && *intent != StopIntent::QUIESCE && !order_list_read_ && may_read && on_the_floor &&
1916 !runner_->guard().halted(strategy::HaltReason::LOCAL_FAILURE)) {
1917 hold(strategy::HaltReason::STOPPING);
1918 read_order_list_once(at, report);
1919 }
1920 return strategy::HaltReason::STOPPING;
1921 }
1922 if (!order_list_read_) {
1923 // Not attempted when the caller says not to, and not attempted at all while the runner's
1924 // journal is broken: a runner that cannot record what it does must not be sent looking for
1925 // more work to fail at. Either way the gate stands, which is the safe answer.
1926 //
1927 // `STOPPING` rather than `ORDER_LIST_UNREAD` on each path here while an order nobody can account
1928 // for is past the limit (below): the stronger hold, which also refuses a withdrawal, must not be
1929 // traded for the weaker one because a read failed. And held before the read, since the source is
1930 // a template parameter and its read may feed.
1931 const bool owed = owed_past_limit(clock_.now());
1932 const auto unread = owed ? strategy::HaltReason::STOPPING : strategy::HaltReason::ORDER_LIST_UNREAD;
1933 if (!may_read || runner_->guard().halted(strategy::HaltReason::LOCAL_FAILURE)) {
1934 return unread;
1935 }
1936 if (owed) {
1937 hold(strategy::HaltReason::STOPPING);
1938 }
1939 if (!read_order_list_once(at, report)) {
1940 return unread;
1941 }
1942 }
1943 // **An order nobody can account for, owed too long, stops the run's placements**
1944 // (`set_owed_order_limit`), in two steps. Once the limit has passed on the clock, this gate holds
1945 // `STOPPING`; it is evaluated before every poll, so an order-list read that begins after a check
1946 // that found the limit passed, and whatever its handlers ask for, runs under the hold (what was
1947 // already under way at the limit runs on the check before it). Once a read in an iteration that
1948 // began after the limit has succeeded and left the order unaccounted for, the stop is requested:
1949 // that iteration's every check found the limit passed, so that read ran under the hold. A read
1950 // whose sweep names the order lifts the hold instead.
1951 if (auto owed = owed_too_long()) {
1952 report.notes_.push_back(owed_stop_note(*owed));
1953 report.owed_stop_ = std::move(owed);
1954 stop.request(StopIntent::QUIESCE);
1955 return strategy::HaltReason::STOPPING;
1956 }
1957 if (owed_past_limit(clock_.now())) {
1958 return strategy::HaltReason::STOPPING;
1959 }
1960 // Measured from the snapshot's own `as_of_` rather than from when the poll returned, because
1961 // that is when the broker says the positions were true. A poll that succeeded at noon carrying
1962 // a snapshot the broker took at 11:00 is an hour stale, and the poll's own timestamp would call
1963 // it fresh.
1964 const auto as_of = runner_->context().portfolio().as_of_;
1965 if (as_of == std::chrono::system_clock::time_point{}) {
1966 // A default-constructed `Portfolio` carries the epoch, and no broker ever reports a snapshot
1967 // taken in 1970, so the epoch means **never populated** -- which is not the same as old, and
1968 // is the more dangerous of the two: an empty portfolio is the *permissive* answer to every
1969 // question the send path asks it, so a run that has never read the account sizes an entry as
1970 // though it were flat.
1971 //
1972 // **No longer inside the age check.** It was, and hoisting it was not a one-line change:
1973 // the hold would reach cancellations, and an order that cannot be withdrawn keeps resting and
1974 // can fill. That objection is answered rather than avoided -- `strategy::halt_blocks` lets a
1975 // named withdrawal through a hold that only means this runner is uninformed, and
1976 // `Runner::plan_withhold_reason` stops the *derived* cancellation that objection overlooked.
1977 //
1978 // The opt-out reaches the age comparison below as well, not just this hold. The epoch is 1970
1979 // and therefore older than any bound an operator would state, so an absence opt-out that did
1980 // not skip the comparison would simply move the refusal to `PORTFOLIO_STALE`.
1981 if (!allow_absent_portfolio_) {
1982 return strategy::HaltReason::PORTFOLIO_ABSENT;
1983 }
1984 // Opted out, so the age comparison is skipped rather than fallen through to -- and the checks
1985 // after this one still run, because an absent portfolio says nothing about buying power.
1986 }
1987
1988 // **A run must state how old a snapshot may be, or state that it accepts any age.**
1989 //
1990 // This was optional, and optional meant that a run which never thought about holdings sent
1991 // orders sized against whatever the portfolio happened to contain -- including nothing at all.
1992 // Fail-open by default was the defect. The library states no default
1993 // duration of its own, because any constant it picked would be a risk policy it has no basis
1994 // for: how stale a position may be before it must not be sized against is the operator's
1995 // judgement about their own market and cadence, not this library's.
1996 // **Checked after absence, not before it.** Both can be true at once, and "you have never read
1997 // this account" is the more useful of the two to be told: the missing policy is a configuration
1998 // fact the note below states in full, while absence is a fact about the data every ceiling and
1999 // every target subtraction is about to be measured against.
2000 if (!maximum_portfolio_age_ && !allow_unbounded_portfolio_age_) {
2001 if (!said_no_age_policy_) {
2002 said_no_age_policy_ = true;
2003 report.notes_.emplace_back(
2004 "no portfolio age policy: call set_maximum_portfolio_age or allow_unbounded_portfolio_age"
2005 );
2006 }
2007 return strategy::HaltReason::PORTFOLIO_STALE;
2008 }
2009 if (maximum_portfolio_age_ && as_of != std::chrono::system_clock::time_point{} &&
2010 at - as_of > *maximum_portfolio_age_) {
2011 return strategy::HaltReason::PORTFOLIO_STALE;
2012 }
2013 // **A runner that keeps its orders until the broker accounts for them needs the listing.** A holdings
2014 // poll refreshes positions and keeps `working_`; only a working-order poll replaces it. A runner
2015 // that sends to a broker keeps each order it sent in `working_` until that listing arrives
2016 // (`Runner::persist_reservation`), so with no listing -- none published, none polled, or one
2017 // older than the policy -- an order that expired or was refused would stay "working" for ever,
2018 // a target would stop being chased, and a filled one would be cancelled again on every pass.
2019 // Held, with the missing piece named once, whatever the portfolio age policy: opting out of a
2020 // holdings age is not opting out of a listing.
2021 if constexpr (runner_keeps_sent_orders<Runner>::value) {
2022 if (const auto missing = listing_missing(at)) {
2023 if ((said_listing_ & missing->first) == 0) {
2024 said_listing_ |= missing->first;
2025 report.notes_.emplace_back(missing->second);
2026 }
2027 return strategy::HaltReason::PORTFOLIO_STALE;
2028 }
2029 }
2030 if (runner_->guard().limits().enforce_buying_power_) {
2031 // Checked only when the caller asked for the limit. A loop with no buying-power cadence and
2032 // no check configured is an ordinary loop, and must not be held for a reading nobody wants.
2033 if (const auto unread = buying_power_unread(at)) {
2034 return *unread;
2035 }
2036 }
2037 return std::nullopt;
2038 }

2041 //
2042 // **Every configured currency must have its own fresh reading**, not merely one of them. A source
2043 // that names its currencies is asked, because an account trading two and answering for one is a
2044 // partial answer: the per-order guard would still refuse the unread currency, but the loop would
2045 // go on reporting an open gate while a budget it was told to watch never arrived at all.
2046 //
2047 // A source that names none falls back to "any reading at all", which is the most the loop can
2048 // check without knowing what was asked for.
2049 std::optional<strategy::HaltReason> buying_power_unread(
2050 std::chrono::system_clock::time_point at
2051 ) const {
2052 const auto& readings = runner_->context().portfolio().buying_power_;
2053 // Aged against the reading's own stamp, for the same reason the portfolio is: a figure that
2054 // arrived an hour ago is an hour old however recently the poll succeeded.
2055 const auto usable = [&](const BuyingPower& reading) {
2056 return !maximum_buying_power_age_ ||

2058 };
2059 if constexpr (source_names_buying_power_currencies<Source>::value) {
2060 const auto& wanted = source_->buying_power_currencies();

2075 }
2076 }

2083 }
2084 }

2145 // Enters a stop, once per run: the hold, the strategy's notice, and the end of whatever it had pending.

2148 closing_ = ClosingWait{};
2149 closing_->began_ = clock_.monotonic_now();
2150 closing_->deadline_ = closing_->began_ + std::chrono::duration_cast<std::chrono::steady_clock::duration>(
2151 closing_wait_ > std::chrono::nanoseconds::zero() ? closing_wait_ : std::chrono::nanoseconds::zero());
2152 hold(strategy::HaltReason::STOPPING);
2153 // Whenever the stop was requested -- before the iteration or during one of its polls -- nothing
2154 // retained before it is to be replayed after it, by this run or a resumed one. A target waiting on
2155 // a resting stop is the same kind of work.

2162 // strategy's order would be planned against the same snapshot and settled in the same window,
2163 // going through none of the sizing the closing pass does -- a second full-size close beside the
2164 // first, or a placement during a `WITHDRAW`, which promises none.

2166 // the calendar like any other handler, but `hold(STOPPING)` above is already in force, so every
2167 // instruction it asks for is refused before anything is sent -- which `stop_handler_sent_` asserts as

2169 // *is* answered, one call later: the closing derive below picks it up.

2172 // `source_->read_order_list`, `source_->buying_power_currencies` (a const accessor that is handed no
2173 // runner and so can neither feed nor write a calendar), `runner_->observe_sellable_convention` (which
2174 // reads the portfolio and dispatches nothing), this one, and `settle_closing`. The two that can send

2209 // Carries out `intent`, which is where the three stops actually differ.
2210 void finish_stop(StopIntent intent, LoopReport& report) {
2211 report.closing_waited_ = std::chrono::duration_cast<std::chrono::nanoseconds>(
2212 clock_.monotonic_now() - closing_->began_);
2213 closing_->reads_wanted_ = false;
2214 if (intent == StopIntent::QUIESCE) {
2215 return;
2216 }
2217 // **Whatever a strategy queued without settling goes nowhere.** The closing instructions below are
2218 // settled with the loop's hold lifted, and anything else still on the queue would be settled with
2219 // them. Settled here, under the hold, it is refused like the rest of what a strategy asks during a stop.
2220 {
2221 const auto stale = runner_->settle_closing();
2222 report.stop_handler_refused_ += stale.refused_.size() + stale.refused_cancel_.size() +
2223 stale.rejected_.size() + stale.withheld_.size() + stale.withheld_plans_.size() + stale.unavailable_.size();
2224 report.stop_handler_sent_ += stale.would_place_.size() + stale.would_cancel_.size();
2225 }
2226 // Nothing decided under the hold is kept, so this clears nothing today. It stands so that a change
2227 // to that rule cannot let kept work out with the closing orders below.
2228 runner_->clear_retained_targets();
2229 runner_->clear_deferred_targets();

2721 // Two sources, both required. The broker's own order list must have been read once, which is what
2722 // the startup gate exists for, and what a stop requested before the first iteration attempts itself
2723 // (a failed attempt leaves it unread). And the
2724 // holdings snapshot -- which is where a `WITHDRAW`'s working orders and a `FLATTEN`'s positions
2725 // actually come from -- must exist, be within the configured age, and be **no older than the
2726 // working-order listing**: a fill between the two reads leaves the listing before it reaches the
2727 // holdings, so a flatten would see neither the order nor the shares it bought.
2728 bool closing_source_verified(std::chrono::system_clock::time_point at) const {
2729 if (!order_list_read_) {
2730 return false;
2731 }
2732 // The working-order picture is its own source with its own freshness, and `as_of_` is not its
2733 // timestamp: `SessionSource::poll_holdings` refreshes positions and `as_of_` while *keeping* the
2734 // working orders it already had, and only a working-order poll replaces those. So a stop can
2735 // have a snapshot stamped a second ago whose resting orders were last read an hour ago.
2736 if (!working_orders_at_) {
2737 return false;
2738 }
2739 // And the decisive one, which no timestamp catches: an order this runner sent after that read
2740 // cannot be in it. A listing is out of date the moment the next placement is accepted, and an
2741 // indeterminate send counts too, because the order may exist.
2742 if (runner_->placements_at_broker() != placements_at_working_read_) {
2743 return false;
2744 }
2745 if (maximum_portfolio_age_ && at - *working_orders_at_ > *maximum_portfolio_age_) {
2746 return false;
2747 }
2748 const auto as_of = runner_->context().portfolio().as_of_;
2749 if (as_of == std::chrono::system_clock::time_point{}) {
2750 return false;
2751 }
2752 // On one clock: both stamps are the local arrival time of their replies, which
2753 // `working_orders_at_` above is not.
2754 if (runner_->context().portfolio().listing_newer_than_holdings()) {
2755 return false;
2756 }
2757 return !maximum_portfolio_age_ || at - as_of <= *maximum_portfolio_age_;
2758 }
2759
2760 // Records that the working-order picture was just refreshed, with what it could not contain.
2761 //
2762 // Does nothing when the source does not publish working orders, so a stop on such a broker stays
2763 // unverified rather than being told a history read or a no-op poll refreshed a listing it never
2764 // built.
2765 void note_working_orders_read(
2766 std::chrono::system_clock::time_point at, std::size_t placements_before_poll
2767 ) {
2768 if constexpr (source_provides_working_orders<Source>::value) {
2769 working_orders_at_ = at;
2770 order_list_at_ = at;
2771 // **The caller's count, taken before the poll ran.** Reading it here instead would include
2772 // an order an `on_order_update` handler placed *during* the poll -- an order sent after the
2773 // listing was fetched, which the listing therefore cannot contain.
2774 placements_at_working_read_ = placements_before_poll;
2775 } else {
2776 (void)at;
2777 (void)placements_before_poll;
2778 }
2779 }

2781 // Returns which half of the closing source is missing, for the operator's note.
2782 std::string unverified_source_reason() const {
2783 if constexpr (!source_provides_working_orders<Source>::value) {
2784 return "this source does not publish the broker's outstanding orders";
2785 }
2786 if (!order_list_read_ || !working_orders_at_) {
2787 return "the broker's order list has not been read";
2788 }
2789 if (runner_->placements_at_broker() != placements_at_working_read_) {
2790 return "an order was sent after the working orders were last read";
2791 }
2792 if (runner_->context().portfolio().as_of_ == std::chrono::system_clock::time_point{}) {
2793 return "no holdings snapshot has arrived";
2794 }
2795 if (runner_->context().portfolio().listing_newer_than_holdings()) {
2796 return "the holdings were read before the working orders, so a fill between them is in neither";
2797 }
2798 if (maximum_portfolio_age_ &&
2799 clock_.now() - *working_orders_at_ > *maximum_portfolio_age_) {
2800 return "the working orders were read longer ago than the configured maximum";
2801 }
2802 return "the holdings snapshot is older than the configured maximum";
2803 }

2805 // Returns whether `kind` is due at `at`.
2806 bool is_due(PollKind kind, std::chrono::system_clock::time_point at) const {
2807 // A waiting stop reads the two closing kinds on its own floor, whatever their cadence (a disabled
2808 // one included), and nothing reads them faster than that meanwhile. The iteration that takes the
2809 // stop up is not waiting yet: it reads as any other.
2810 if (closing_ && closing_->reads_wanted_ &&
2811 (kind == PollKind::WORKING_ORDERS || kind == PollKind::HOLDINGS)) {
2812 return closing_read_due(kind);
2813 }
2814 const auto interval = schedule_.interval_for(kind);
2815 if (!interval) {
2816 return false;
2817 }
2818 const auto& due = next_due_[static_cast<int>(kind)];
2819 return !due || at >= *due;
2820 }

2822 // Records when `kind` is next due.
2823 void schedule_next(PollKind kind, std::chrono::system_clock::time_point at) {
2824 const auto interval = schedule_.interval_for(kind);
2825 if (!interval) {
2826 return;
2827 }
2828 next_due_[static_cast<int>(kind)] =
2829 at + std::chrono::duration_cast<std::chrono::system_clock::duration>(*interval);
2830 }

2832 // Returns how long to wait before the next iteration.
2833 //
2834 // The soonest due poll, capped at the tick so a stop through the control file is noticed within
2835 // one tick however sparse the polling is. Never negative: a loop that has fallen behind catches up
2836 // rather than pausing once per overdue poll.
2837 std::chrono::nanoseconds until_next(std::chrono::system_clock::time_point at) const {
2838 auto wait = schedule_.tick_;
2839 for (const auto kind : all_poll_kinds) {
2840 const auto& due = next_due_[static_cast<int>(kind)];
2841 if (!due) {
2842 continue;
2843 }
2844 const auto remaining =
2845 std::chrono::duration_cast<std::chrono::nanoseconds>(*due - at);
2846 if (remaining < wait) {
2847 wait = remaining;
2848 }
2849 }
2850 return wait > std::chrono::nanoseconds::zero() ? wait : std::chrono::nanoseconds::zero();
2851 }
