171 static Panel from_bars(
172 const std::vector<BarSeries>& series,
173 BarField field,
174 Alignment alignment
175 ) {
176 Panel panel;
177 panel.field_ = field;
178 panel.alignment_ = alignment;
179 if (series.empty()) {
180 return panel;
181 }
182
183 std::vector<SeriesFrame> frames;
184 frames.reserve(series.size());
185 for (const auto& one : series) {
186 frames.push_back(analytics::from_bars(one));
187 panel.instruments_.push_back(one.instrument_);
188 }
189
190 auto stamps = alignment == Alignment::INTERSECTION ?
191 shared_instants(frames) :
192 combined_instants(frames);
193 panel.index_ = TimeIndex(std::move(stamps));
194
195 panel.values_ = math::Matrix::Constant(
196 static_cast<math::Index>(panel.index_.size()),
197 static_cast<math::Index>(frames.size()),
198 missing()
199 );
200
201 for (std::size_t column = 0; column < frames.size(); ++column) {
202 fill_column(panel, column, frames[column], field, alignment);
203 }
204 return panel;
205 }

207private:
208 // Every instant any frame reported, ascending and deduplicated.
209 static std::vector<TimeIndex::Instant> combined_instants(const std::vector<SeriesFrame>& frames) {
210 std::size_t total = 0;
211 for (const auto& frame : frames) {
212 total += frame.time().size();
213 }
214 std::vector<TimeIndex::Instant> all;
215 all.reserve(total);
216 for (const auto& frame : frames) {
217 const auto& stamps = frame.time().stamps();
218 all.insert(all.end(), stamps.begin(), stamps.end());
219 }
220 std::sort(all.begin(), all.end());
221 all.erase(std::unique(all.begin(), all.end()), all.end());
222 return all;
223 }
224
225 // Only the instants every frame reported.
226 static std::vector<TimeIndex::Instant> shared_instants(const std::vector<SeriesFrame>& frames) {
227 std::vector<TimeIndex::Instant> shared = frames.front().time().stamps();
228 std::vector<TimeIndex::Instant> narrowed;
229 narrowed.reserve(shared.size());
230 for (std::size_t position = 1; position < frames.size(); ++position) {
231 const auto& other = frames[position].time().stamps();
232 narrowed.clear();
233 std::set_intersection(
234 shared.begin(), shared.end(), other.begin(), other.end(), std::back_inserter(narrowed)
235 );
236 shared.swap(narrowed);
237 }
238 return shared;
239 }

241 // Writes one instrument's column under the chosen alignment, walking both ascending indices
242 // together in one pass.
243 static void fill_column(
244 Panel& panel,
245 std::size_t column,
246 const SeriesFrame& frame,
247 BarField field,
248 Alignment alignment
249 ) {
250 const auto& target = panel.index_.stamps();
251 const auto& source = frame.time().stamps();
252 if (target.empty() || source.empty()) {
253 return;
254 }
255 const auto series = frame.column(field);
256 const bool carry = alignment == Alignment::FORWARD_FILL;
257
258 std::size_t at = 0;
259 for (std::size_t row = 0; row < target.size(); ++row) {
260 const auto& when = target[row];
261 while (at + 1 < source.size() && !(when < source[at + 1])) {
262 ++at;
263 }
264 // `source[at]` is now the latest instant at or before `when`, unless the source
265 // begins after it — in which case there is nothing observed yet, and carrying
266 // backwards would be inventing a price the instrument never had.
267 if (when < source[at]) {
268 continue;
269 }
270 if (!carry && source[at] != when) {
271 continue;
272 }
273 panel.values_(static_cast<math::Index>(row), static_cast<math::Index>(column)) =
274 series(static_cast<math::Index>(at));
275 }
276 }
