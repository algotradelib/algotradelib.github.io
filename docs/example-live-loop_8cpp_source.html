88 runner::PollOutcome poll(runner::PollKind kind, LoopRunner& live_runner) {
89 if (kind == runner::PollKind::PRICE_LIMITS) {
90 // Today's permitted band, which reconciliation clamps a limit price into. A real source calls
91 // `session.price_limits(instrument)` per instrument here; what matters is that it lands on the
92 // runner rather than being fed as an event, and that it arrives **on a cadence** -- the band
93 // is a per-trading-day value, so one read at startup describes the day the process started on.
94 PriceLimits band;
95 band.currency_ = Currency{"KRW"};
96 band.upper_ = Decimal64::parse("91000");
97 band.lower_ = Decimal64::parse("49000");
98 live_runner.set_price_limits(samsung(), band);
99 return runner::PollOutcome::UPDATED;
100 }
101 if (kind != runner::PollKind::BARS) {
102 return runner::PollOutcome::UNCHANGED;
103 }
104 if (remaining_ <= 0) {
105 // Nothing new is the normal case between closes, and is not a failure.
106 return runner::PollOutcome::UNCHANGED;
107 }
108 --remaining_;
109
110 Bar bar;
111 bar.timestamp_ = Clock::time_point{} + std::chrono::hours(24 * (200 + fed_));
112 bar.close_ = Decimal64::parse(std::to_string(70000 + fed_ * 500));
113 bar.high_ = bar.close_;
114 bar.low_ = bar.close_;
115 bar.volume_ = Decimal64::parse("1000000");
116 ++fed_;
117
118 live_runner.context().set_last_price(samsung(), bar.close_);
119 live_runner.feed(
120 strategy::make_bar_event(samsung(), BarInterval::ONE_DAY, BarStamp::CLOSE, bar)
121 );
122 return runner::PollOutcome::UPDATED;
123 }

141void describe(const runner::LoopReport& report) {
142 std::cout << "iterations : " << report.iterations_ << '\n'
143 << "bar polls : " << report.attempted(runner::PollKind::BARS)
144 << " (" << report.failed(runner::PollKind::BARS) << " failed)\n"
145 << "blocked : " << report.blocked_ << " iteration(s)\n"
146 << "order list read : " << (report.order_list_read_ ? "yes" : "no") << '\n'
147 << "stopped by : "
148 << (report.stopped_by_ ?
149 runner::stop_intent_name(*report.stopped_by_) :
150 std::string_view("nothing"))
151 << '\n'
152 << "final gate : "
153 << (report.final_gate_ ?
154 strategy::halt_reason_name(*report.final_gate_) :
155 std::string_view("none"))
156 << '\n'
157 // The phase the loop derived for itself, and how many changes it announced. Reported
158 // because it decides most of what could be sent: `unknown` beside a default
159 // `permitted_phases_` means every placement that would open a position was refused, and
160 // an empty placement count on its own does not say that.
161 << "market phase : " << strategy::market_phase_name(report.final_phase_)
162 << " (" << report.phase_transitions_ << " transition(s))\n"
163 // And how many orders went out under a phase the limits do not permit. An order closing
164 // a held position is permitted in any phase -- which is what lets a flatten after the
165 // close reach the broker instead of being refused by the operator's own limit -- so this
166 // is the line that says it happened. Non-zero is not a fault; a nightly flatten is one.
167 << "sent off-phase : " << report.sends_outside_permitted_phase_ << '\n'
168 // And what the price-band policy decided. A band is a per-trading-day value, so a run
169 // that outlives the day it read one has to decide what to do with it: this library does
170 // not apply a band whose age nobody has bounded, and says so here rather than repricing
171 // an order against yesterday's ceiling in silence. Both stay zero in this example
172 // because `CountBars` places nothing -- the first counts plans, the second counts sent
173 // orders a broker refused after a band was withheld.
174 << "bands withheld : " << report.price_bands_withheld_ << '\n'
175 << "refused unclamped: " << report.unclamped_orders_rejected_ << '\n';
176 for (const auto& note : report.notes_) {
177 std::cout << "note : " << note << '\n';
178 }
179}

197int main() {
198 std::cout << "a live loop with no broker attached\n\n";
199
200 LoopRunner live_runner;
201 live_runner.context().set_instrument(samsung_reference());
202
203 // **The phase is the loop's, not this program's.** The loop derives it from its session window and
204 // the trading calendar, and overwrites anything set here; a fixed `set_phase(MarketPhase::OPEN)`
205 // would trade as though the market never closed, and `on_phase` would never fire.
206 //
207 // A real program gets that calendar from `PollKind::CALENDAR`, which is on by default. This example
208 // has no broker, so its source answers no calendar; the date is therefore uncovered, the phase is
209 // `UNKNOWN`, and the report says so in as many words. Seeding one here is what a run with a broker
210 // gets from its first calendar poll.
211 algotradelib::TradingCalendar calendar;
212 calendar.country_ = algotradelib::Country::KR;
213 calendar.received_at_ = start_time();
214 calendar.days_.push_back(algotradelib::TradingDay{
215 strategy::SessionWindow::krx().local_date(start_time()), true});
216 live_runner.context().set_calendar(calendar);
217
218 ScriptedSource source(4);
219
220 runner::PollSchedule schedule;
221 schedule.bars_ = std::chrono::minutes(1);
222 schedule.tick_ = std::chrono::minutes(1);
223 // No broker here, so nothing could answer a calendar request; a real program leaves this at its
224 // six-hour default.
225 schedule.calendar_.reset();
226 // **The band, on a cadence rather than once.** A band registered once at startup would clamp every
227 // limit order for as long as the process ran. Ten minutes here is arbitrary; the real constraint is
228 // that it be shorter than the session, because the value changes with the previous close.
229 schedule.price_limits_ = std::chrono::minutes(10);
230
231 runner::PollingLoop<LoopRunner, ScriptedSource, runner::ManualClock> loop(
232 live_runner, source, schedule, runner::ManualClock{start_time()});
233 // **Two statements every placement-enabled run owes, and this source cannot satisfy either.**
234 //
235 // A run states how old a holdings snapshot may be before it stops sizing against it, and a run
236 // that has never read one holds rather than treating an empty account as a current one -- an
237 // empty portfolio is the *permissive* answer to every ceiling and to every target subtraction, so a
238 // run that never thought about holdings would trade as though it held nothing.
239 //
240 // This example has no broker, so it reads no holdings ever. Saying so explicitly is what an
241 // operator in the same position does; leaving them out would hold this loop for the whole run.
242 loop.allow_unbounded_portfolio_age();
243 loop.allow_absent_portfolio();
244 // **The third statement, and the same shape as the first.** A run states how old a band may be, or
245 // that it accepts any age; with neither, no band is applied. Twenty
246 // minutes against a ten-minute cadence, so one failed poll is tolerated and a run of failures is
247 // not -- which is what the policy is actually bounding, since a failed poll deliberately leaves
248 // the last good band in place.
249 loop.set_maximum_price_limits_age(std::chrono::minutes(20));
250
251 runner::StopSource stop;
252 const auto report = loop.run(stop, 6);
253
254 describe(report);
255
256 std::cout << "\nthe strategy saw " << live_runner.strategy().bars_ << " bar(s)\n";
257
258 // The gate refused the first iteration because the order list had not been read. That is the
259 // cheapest half of reconciliation and it needs no persistence: a process that has just started
260 // cannot know whether a predecessor died mid-POST, and an empty ledger does not mean nothing is
261 // owed -- it means nothing is remembered.
262 std::cout << "the order list was read " << source.reads() << " time(s) before the gate opened\n";
263
264 std::cout << "\nhow a real operator would stop this:\n"
265 << " runner::SignalStop::install(stop); // SIGINT and SIGTERM quiesce; SIGHUP ignored\n"
266 << " loop.set_control_file(runner::FileStopChannel(\"/var/run/atl/stop\"));\n"
267 << " echo withdraw > /var/run/atl/stop // or quiesce, or flatten\n\n"
268 << "and the trap worth knowing: a supervisor's grace period must exceed one loop\n"
269 << "iteration plus one request timeout, or its SIGKILL turns the cleanest stop into\n"
270 << "the worst one -- a process killed mid-request, with an order whose fate is unknown.\n";
271
272 return 0;
