42 SeriesBuffer(std::size_t capacity, std::size_t fields = bar_field_count)

96 void push(Instant when, math::SeriesView row) {
97 if (static_cast<std::size_t>(row.size()) != fields_) {
98 throw std::invalid_argument("a series buffer observation must carry every field");
99 }
100 if (count_ > 0 && !(newest_instant() < when)) {
101 throw std::invalid_argument("a series buffer observation must be later than the last");
102 }
103
104 if (count_ == capacity_) {
105 ++begin_;
106 --count_;
107 }
108 if (begin_ + count_ == static_cast<std::size_t>(values_.rows())) {
109 compact();
110 }
111
112 const auto row_index = static_cast<math::Index>(begin_ + count_);
113 for (std::size_t field = 0; field < fields_; ++field) {
114 values_(row_index, static_cast<math::Index>(field)) =
115 row(static_cast<math::Index>(field));
116 }
117 stamps_[begin_ + count_] = when;
118 ++count_;
119 }
120
129 void push(Instant when, const std::vector<Scalar>& row) {
130 if (row.size() != fields_) {
131 throw std::invalid_argument("a series buffer observation must carry every field");
132 }
133 const math::Vector values = math::Vector::Map(row.data(), static_cast<math::Index>(row.size()));
134 push(when, values);
135 }

147 void push(const Bar& bar) {
148 if (fields_ != bar_field_count) {
149 throw std::invalid_argument("pushing a bar needs a buffer over every bar field");
150 }
151 math::Vector row = math::unobserved(bar_field_count);
152 if (bar.open_) {
153 row(static_cast<math::Index>(bar_field_index(BarField::OPEN))) = to_scalar(*bar.open_);
154 }
155 row(static_cast<math::Index>(bar_field_index(BarField::HIGH))) = to_scalar(bar.high_);
156 row(static_cast<math::Index>(bar_field_index(BarField::LOW))) = to_scalar(bar.low_);
157 row(static_cast<math::Index>(bar_field_index(BarField::CLOSE))) = to_scalar(bar.close_);
158 row(static_cast<math::Index>(bar_field_index(BarField::VOLUME))) = to_scalar(bar.volume_);
159 if (bar.traded_value_) {
160 row(static_cast<math::Index>(bar_field_index(BarField::TRADED_VALUE))) =
161 to_scalar(*bar.traded_value_);
162 }
163 push(bar.timestamp_, row);
164 }

241 // Moves the held observations to the front. Source and destination never overlap: this runs
242 // only when the held run reaches the end of storage, so begin_ = 2 * capacity - count_ >= count_.
243 void compact() {
244 const auto rows = static_cast<math::Index>(count_);
245 const auto from = static_cast<math::Index>(begin_);
246 values_.topRows(rows) = values_.middleRows(from, rows);
247 for (std::size_t position = 0; position < count_; ++position) {
248 stamps_[position] = stamps_[begin_ + position];
249 }
250 begin_ = 0;
251 ++compactions_;
252 }
