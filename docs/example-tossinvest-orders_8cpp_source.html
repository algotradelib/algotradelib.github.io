19void print_error(std::string_view operation, const atl::BrokerError& error) {
20 std::cerr << operation << " failed";
21 if (!error.broker_code_.empty()) {
22 std::cerr << " [" << error.broker_code_ << ']';
23 }
24 if (!error.request_id_.empty()) {
25 std::cerr << " request=" << error.request_id_;
26 }
27 if (error.http_status_ != 0) {
28 std::cerr << " HTTP " << error.http_status_;

83std::int64_t parse_account_sequence(const char* value) {
84 const std::string text(value);
85 std::size_t parsed = 0;
86 const auto sequence = std::stoll(text, &parsed);
87 if (parsed != text.size() || sequence <= 0) {
88 throw std::invalid_argument("account sequence must be a positive integer");
89 }
90 return sequence;
91}

96int main(int argc, char* argv[]) {
97 try {
98 atl::tossinvest::TossConfig config;
99 config.from_env();
100
101 ssl::context tls(ssl::context::tls_client);
102 tls.set_default_verify_paths();
103 tls.set_verify_mode(ssl::verify_peer);
104
105 atl::tossinvest::TossBlockingClient client(tls, std::move(config));
106
107 std::int64_t sequence = 0;
108 if (argc > 1) {
109 sequence = parse_account_sequence(argv[1]);
110 } else {
111 auto accounts_result = client.accounts();
112 if (!accounts_result) {
113 print_error("account discovery", accounts_result.error());
114 return 1;
115 }
116 if (accounts_result.value().empty()) {
117 std::cout << "no accounts available\n";
118 return 0;
119 }
120 sequence = accounts_result.value().front().sequence_;
121 }
122 std::cout << "selected account: " << sequence << '\n';
123
124 atl::tossinvest::TossOrderQuery open_query;
125 open_query.account_sequence_ = sequence;
126 open_query.scope_ = atl::tossinvest::TossOrderScope::OPEN;
127 auto open_result = client.orders(open_query);
128 if (!open_result) {
129 print_error("open order listing", open_result.error());
130 return 1;
131 }
132 std::cout << "open orders: " << open_result.value().orders_.size() << '\n';
133 for (const auto& order : open_result.value().orders_) {
134 print_order(order);
135 }
136
137 // Fetch only the first closed page. Following every cursor automatically
138 // can consume the ORDER_HISTORY request budget; callers decide explicitly
139 // whether and when to request another page.
140 atl::tossinvest::TossOrderQuery closed_query;
141 closed_query.account_sequence_ = sequence;
142 closed_query.scope_ = atl::tossinvest::TossOrderScope::CLOSED;
143 closed_query.limit_ = 20;
144 auto closed_result = client.orders(closed_query);
145 if (!closed_result) {
146 print_error("closed order listing", closed_result.error());
147 return 1;
148 }
149 std::cout << "closed orders on first page: " << closed_result.value().orders_.size() << '\n';
150 std::cout << "more closed orders: "
151 << (closed_result.value().next_cursor_ ? "yes" : "no") << '\n';
152 return 0;
153 } catch (const std::exception& exception) {
154 std::cerr << "example failed: " << exception.what() << '\n';
155 return 1;
156 }
