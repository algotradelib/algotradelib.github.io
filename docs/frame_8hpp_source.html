212inline SeriesFrame from_bars(const BarSeries& series) {
213 std::vector<TimeIndex::Instant> stamps;
214 stamps.reserve(series.bars_.size());
215 for (const auto& bar : series.bars_) {
216 stamps.push_back(bar.timestamp_);
217 }
218
219 SeriesFrame frame(TimeIndex(std::move(stamps)), series.instrument_);
220 for (std::size_t row = 0; row < series.bars_.size(); ++row) {
221 const auto& bar = series.bars_[row];
222 // An absent open is a trading day on which nothing traded. It stays unobserved
223 // rather than becoming zero or the previous close.
224 if (bar.open_) {
225 frame.set(row, BarField::OPEN, to_scalar(*bar.open_));
226 }
227 frame.set(row, BarField::HIGH, to_scalar(bar.high_));
228 frame.set(row, BarField::LOW, to_scalar(bar.low_));
229 frame.set(row, BarField::CLOSE, to_scalar(bar.close_));
230 frame.set(row, BarField::VOLUME, to_scalar(bar.volume_));
231 if (bar.traded_value_) {
232 frame.set(row, BarField::TRADED_VALUE, to_scalar(*bar.traded_value_));
233 }
234 }
235 return frame;
236}
