247inline SeriesFrame from_bars(const BarSeries& series) {
248 std::vector<TimeIndex::Instant> stamps;
249 stamps.reserve(series.bars_.size());
250 for (const auto& bar : series.bars_) {
251 stamps.push_back(bar.timestamp_);
252 }
253
254 SeriesFrame frame(TimeIndex(std::move(stamps)), series.instrument_);
255 for (std::size_t row = 0; row < series.bars_.size(); ++row) {
256 const auto& bar = series.bars_[row];
257 // An absent open is a trading day on which nothing traded. It stays unobserved
258 // rather than becoming zero or the previous close.
259 if (bar.open_) {
260 frame.set(row, BarField::OPEN, to_scalar(*bar.open_));
261 }
262 frame.set(row, BarField::HIGH, to_scalar(bar.high_));
263 frame.set(row, BarField::LOW, to_scalar(bar.low_));
264 frame.set(row, BarField::CLOSE, to_scalar(bar.close_));
265 frame.set(row, BarField::VOLUME, to_scalar(bar.volume_));
266 if (bar.traded_value_) {
267 frame.set(row, BarField::TRADED_VALUE, to_scalar(*bar.traded_value_));
268 }
269 }
270 return frame;
271}
