228inline SeriesFrame from_bars(const BarSeries& series) {
229 std::vector<TimeIndex::Instant> stamps;
230 stamps.reserve(series.bars_.size());
231 for (const auto& bar : series.bars_) {
232 stamps.push_back(bar.timestamp_);
233 }
234
235 SeriesFrame frame(TimeIndex(std::move(stamps)), series.instrument_);
236 for (std::size_t row = 0; row < series.bars_.size(); ++row) {
237 const auto& bar = series.bars_[row];
238 // An absent open is a trading day on which nothing traded. It stays unobserved
239 // rather than becoming zero or the previous close.
240 if (bar.open_) {
241 frame.set(row, BarField::OPEN, to_scalar(*bar.open_));
242 }
243 frame.set(row, BarField::HIGH, to_scalar(bar.high_));
244 frame.set(row, BarField::LOW, to_scalar(bar.low_));
245 frame.set(row, BarField::CLOSE, to_scalar(bar.close_));
246 frame.set(row, BarField::VOLUME, to_scalar(bar.volume_));
247 if (bar.traded_value_) {
248 frame.set(row, BarField::TRADED_VALUE, to_scalar(*bar.traded_value_));
249 }
250 }
251 return frame;
252}
