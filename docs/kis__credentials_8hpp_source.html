52inline std::optional<std::filesystem::path> take_directory(int& argc, char** argv) {
53 std::optional<std::filesystem::path> directory;
54 int kept = 1;
55 for (int i = 1; i < argc; ++i) {
56 if (std::string_view(argv[i]) == "--credentials") {
57 if (i + 1 >= argc || std::string_view(argv[i + 1]).empty() ||
58 std::string_view(argv[i + 1]).rfind("--", 0) == 0) {
59 throw std::runtime_error("--credentials needs a directory");
60 }
61 directory = argv[++i];
62 } else {
63 argv[kept++] = argv[i];
64 }
65 }
66 argc = kept;
67 return directory ? directory : environment_directory();
68}
