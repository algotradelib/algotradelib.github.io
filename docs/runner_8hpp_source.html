31// Whether an executor can ask the broker what may be sold of one instrument.
32//
33// Detected rather than required, for the same reason the strategy's handlers are: a dry runner, a
34// replay's queueing executor and a test double have no broker to ask, and none of them should have
35// to grow a method that returns nothing. An executor bound to a broker whose holdings already carry
36// the figure declares it too and answers nothing, so the runner's own condition stays one test.

41struct fills_sellable<
42 Executor,
43 std::void_t<
44 decltype(std::declval<Executor&>().sellable_quantity(std::declval<const InstrumentId&>()))
45 >
46> : std::true_type {};

48// Whether `Executor` states what its sellable *inquiry* answer is evidence of.
49//
50// Detected rather than required, and the default is what matters: an executor that answers a
51// sellable quantity and says nothing about its basis is treated as having established nothing, so
52// the guard refuses rather than measuring a sell against a figure nobody characterized. A consumer's
53// own executor gets that default without being told it exists, which is the point -- the alternative
54// default would silently enforce against whatever its author happened to return.
55template <typename Executor, typename = void>
56struct states_sellable_basis : std::false_type {};
57
58template <typename Executor>
59struct states_sellable_basis<
60 Executor,
61 std::void_t<decltype(Executor::sellable_inquiry_basis())>
62> : std::true_type {};

101// Whether `Executor` can say **why** the last instruction failed.
102//
103// Detected rather than required, on the same terms as the two above: a dry runner refuses nothing
104// and a replay's queueing executor has no broker to be refused by, so neither should have to grow a
105// method that answers nothing. An executor that binds a session declares
106// `const std::optional<BrokerError>& last_error() const` and its refusals then reach
107// `Runner::last_broker_refusal`, where a report can carry the broker's own code and message instead
108// of a count with nothing to chase.
109//
110// **The *shape* is detected, not the name.** An earlier version asked only whether `last_error()`
111// was a valid expression, which would have selected this branch for any executor that happened to
112// have a method of that name -- one returning `std::optional<std::string>`, say -- and then failed
113// to compile on `broker_code_`. A name is not a contract; the return type is.
114template <typename Executor, typename = void>
115struct states_last_error : std::false_type {};

118struct states_last_error<
119 Executor,
120 std::enable_if_t<
121 std::is_same_v<
122 std::decay_t<decltype(std::declval<const Executor&>().last_error())>,
123 std::optional<BrokerError>
124 >
125 >
126> : std::true_type {};

1118 void confirm_from_history(const std::vector<OrderSnapshot>& evidence) {
1119 if (serialized()) {
1120 confirm_serialized_from_history(evidence);
1121 return;
1122 }
1123 confirm_reservations();
1124 const auto now = freshness_instant();
1125 for (auto& entry : unconfirmed_) {
1126 entry.consulted_ = awaits_evidence(entry, now);
1127 if (!entry.consulted_) {
1128 continue;
1129 }
1130 // An entry not yet named matches no row: a row whose number is missing would read as equal.
1131 if (entry.ref_.primary_.empty()) {
1132 continue;
1133 }
1134 const auto* row = history_row_for(
1135 evidence, entry.ref_.primary_, entry.instrument_, entry.side_,
1136 entry.sent_at_ - order_time_tolerance_, entry.sent_at_ + order_time_tolerance_
1137 );
1138 if (row != nullptr && (entry.session_ended_ || !strategy::order_is_open(row->status_))) {
1139 if (decimal_less(row->execution_.filled_quantity_, entry.expected_)) {
1140 entry.expected_ = row->execution_.filled_quantity_;
1141 }
1142 entry.settled_ = true;
1143 // Holdings get a full timeout from here to show it (`confirm_reservations`).
1144 entry.unexplained_since_ = now;
1145 }
1146 }
1147 // Past the timeout and the history had no answer: noted, and judged by `confirm_reservations` against
1148 // the next holdings read, as a settled order is. Marking here, on the clock, marked a fill one
1149 // holdings read early on a loop polling slower than the timeout (the listing, and so this history,
1150 // is read before holdings in each iteration).
1151 for (auto& entry : unconfirmed_) {
1152 if (entry.consulted_ && !entry.settled_ && timed_out(entry, now)) {
1153 entry.history_missed_ = true;
1154 }
1155 }
1156 // Retires what the history just settled, and withholds what the holdings in hand already condemn.
1157 confirm_reservations();
1158 }

1197 Report retry_retained_targets() {
1198 // Once per iteration, so the context shows what this iteration's holdings confirmed even when no
1199 // pass runs (`unconfirmed_orders`).
1200 confirm_reservations();
1201 if (guard_.halted(strategy::HaltReason::OPERATOR) ||
1202 guard_.halted(strategy::HaltReason::LOCAL_FAILURE)) {
1203 retained_targets_.erase(
1204 std::remove_if(
1205 retained_targets_.begin(), retained_targets_.end(),
1206 [](const RetainedTarget& kept) { return kept.for_phase_; }
1207 ),
1208 retained_targets_.end()
1209 );
1210 kept_intents_.clear();
1211 }
1212 // **A statement still queued is newer than anything retained.** Releasing retained work appends it to
1213 // the queue, behind what a caller driving the runner itself has queued, where it would be settled as
1214 // the latest word on its instrument. So what a queued target or order speaks about is forgotten first,
1215 // as `settle` would forget it (a `CancelIntent` names an order, not an instrument, and forgets nothing).
1216 for (const auto& queued : context_.queued()) {
1217 forget_retained(instruction_instrument(queued));
1218 }
1219 if ((retained_targets_.empty() && kept_intents_.empty()) ||
1220 context_.portfolio().as_of_ == std::chrono::system_clock::time_point{} ||
1221 context_.portfolio().listing_newer_than_holdings() || guard_.blocking_reason()) {
1222 return Report{};
1223 }
1224 const bool phase_permits = guard_.permits_phase(context_.phase());
1225 std::vector<RetainedTarget> kept;
1226 bool released = false;
1227 const auto as_of = context_.portfolio().as_of_;
1228 // Kept because the broker could not be reached: not offered again until it has answered since.
1229 const auto awaiting_broker = [&](const auto& entry) { return entry.after_ && !(*entry.after_ < as_of); };
1230 for (auto& entry : retained_targets_) {
1231 if ((entry.for_phase_ && !phase_permits) || awaiting_broker(entry) ||
1232 (entry.until_accounted_ && owes(entry.target_.instrument_))) {
1233 kept.push_back(std::move(entry));
1234 continue;
1235 }
1236 context_.submit(std::move(entry.target_));
1237 released = true;
1238 }
1239 retained_targets_ = std::move(kept);
1240 // Kept intents follow, in order, once the phase permits; each goes out only if the holding it was
1241 // decided against is unchanged (and, for a stop, no stop has come to rest meanwhile).
1242 if (phase_permits && !kept_intents_.empty()) {
1243 auto intents = std::move(kept_intents_);
1244 kept_intents_.clear();
1245 for (auto& entry : intents) {
1246 if (awaiting_broker(entry)) {
1247 kept_intents_.push_back(std::move(entry));
1248 continue;
1249 }
1250 const auto held = context_.portfolio().net_quantity_of(entry.intent_.instrument());

1259 continue;
1260 }

1262 released = true;
1263 }
1264 }

1266 return Report{};
1267 }
1268 return settle();
1269 }

1694private:
1695 // Drains the queue and turns it into a report.
1696 //
1697 // **Cancellations are collected before any order is placed**, across the whole queue, rather than
1698 // per instruction. That is the sequencing `Plan` documents as the caller's obligation, and here
1699 // is the caller: a replacement placed before its predecessor is withdrawn is briefly a double
1700 // position, while cancelling first leaves the instrument unrepresented for a moment instead.
1701 //
1702 // A guard refusal drops that instruction and nothing else. One instrument breaching a position
1703 // ceiling must not silence the rest of a cross-sectional rebalance.
1704 Report settle() {
1705 // Before anything is planned, because this pass has not yet added anything of its own: the
1706 // probe must not read this runner's intentions as the broker's report. The portfolio can still
1707 // carry a reservation an *earlier* pass persisted, which is why those rows are marked and why
1708 // the probe skips them rather than relying on when it is called.
1709 observe_sellable_convention();
1710 // After the probe, which must read the broker's report rather than this runner's rows, and before
1711 // anything is planned: whatever replaced `working_` since the last pass, each order this runner
1712 // sent is counted again at what nothing yet explains (`unconfirmed_orders`).
1713 confirm_reservations();
1714
1715 Report report;
1716 auto instructions = context_.drain();
1717 kept_this_pass_.clear();
1718 drop_superseded_deferred(instructions);
1719 std::vector<strategy::PlannedOrder> pending;
1720 // The instruction each pending order came from, aligned with `pending`, so a phase refusal can
1721 // keep the target that produced it (`retain_for_phase`). Points into `instructions`, which lives
1722 // for the whole pass.
1723 std::vector<const Instruction<Tag>*> origin;
1724 // Whether a later instruction of this pass speaks about `source`'s instrument (`source` is one of
1725 // `instructions`): the latest statement wins.
1726 const auto restated_later = [&instructions](const Instruction<Tag>& source) {
1727 const auto instrument = instruction_instrument(source);
1728 const auto* const end = instructions.data() + instructions.size();
1729 for (const auto* later = &source + 1; later < end; ++later) {
1730 if (instruction_instrument(*later) == instrument) {
1731 return true;
1732 }
1733 }
1734 return false;
1735 };
1736
1737 // **One withdrawal per order per pass.** A strategy that clears what is resting and then states
1738 // a target asks twice for the same order: the explicit `CancelIntent` goes out, and then
1739 // `plan_for` reconciles the target against `context_.portfolio()` -- which does not carry the
1740 // cancellation just sent -- and plans a withdrawal for the same row. Two requests, and a broker
1741 // answering the second with "already cancelled" then reads as a rejected withdrawal in the
1742 // report.
1743 //
1744 // **Only the duplicate send is suppressed; the picture planning reads is untouched.** Removing
1745 // the row from that view is the tempting fix and it is unsafe: `SendOutcome::ACCEPTED` on a
1746 // withdrawal means the broker took the *request*, and `PENDING_CANCEL` counts as open in
1747 // `strategy::order_is_open`. Sizing a replacement against an emptied book would place a fresh
1748 // order of the full target while the original may still fill -- doubling a position to save a
1749 // request. Reconciliation keeps treating a possibly-working order as working, which is the
1750 // right answer until a listing says otherwise.
1751 //
1752 // Keyed on *attempted*, not on accepted: no outcome here means the order is gone, and a
1753 // withdrawal that was refused locally will be refused the same way a second time.
1754 std::vector<OrderRef<Tag>> attempted;
1755 const auto attempt_cancel = [this, &attempted, &report](const OrderRef<Tag>& ref) {
1756 // An unaddressed reference is exempt. Two of them compare equal, so deduping would collapse
1757 // two distinct unsendable withdrawals into one report entry and hide the second.
1758 if (!ref.primary_.empty() || !ref.secondary_.empty()) {
1759 for (const auto& seen : attempted) {
1760 if (seen == ref) {
1761 return;
1762 }
1763 }
1764 attempted.push_back(ref);
1765 }
1766 send_cancel(ref, report);
1767 };
1768 // Serialized: the instruments a plan in this pass places on, so a second plan for one of them is
1769 // withheld before its withdrawals go out (`serial_hold`).
1770 std::vector<InstrumentId> claimed;
1771
1772 for (const auto& instruction : instructions) {
1773 // **A later target or order replaces a retained target**: the strategy has spoken again about this
1774 // instrument, and replaying what it said before would override it. A withheld target is retained
1775 // again below, as the latest. A `CancelIntent` names an order, not an instrument, and leaves it.
1776 forget_retained(instruction_instrument(instruction));
1777 if (const auto* cancel = std::get_if<strategy::CancelIntent<Tag>>(&instruction)) {
1778 attempt_cancel(cancel->ref_);
1779 continue;
1780 }
1781
1782 const auto planned = plan_for(instruction);
1783 // **Checked before a single cancellation goes out, and on the plan rather than on its parts.**
1784 //
1785 // `plan_for` derives both halves from `context_.portfolio()`: which rows oppose the target, and
1786 // what remains to be placed. So a plan is only as good as the snapshot under it, and the
1787 // cancellations are sent below -- before the guard has seen the placements at all. Deciding the
1788 // halves separately is how a reprice becomes a bare withdrawal.
1789 //
1790 // Only a plan that would cancel something is stopped. One that only places is left to the guard
1791 // exactly as before, so a direct `OrderIntent` keeps reporting as a refusal rather than quietly
1792 // changing bucket.
1793 const bool would_cancel = !planned.cancel_.empty();
1794 if (const auto reason = plan_withhold_reason(would_cancel)) {
1795 // **A missing account stops the whole plan; a hold on placements stops only a plan that
1796 // would also withdraw something.**
1797 //
1798 // The difference is which of the two the *guard* can be relied on to catch. A held placement
1799 // is refused by `RiskGuard` a few lines below, with the hold named on its own report entry,
1800 // and that is the better result -- so a plan with nothing to cancel is left to it. What the
1801 // guard cannot catch is data: a snapshot nobody has read is not a hold, and the loop's hold
1802 // for it is not yet installed at every point an instruction can be settled. A source's
1803 // `read_order_list` runs before `PollingLoop` installs any hold and is handed a mutable
1804 // runner, so a target settled inside that call reaches here with the guard clear. Deciding
1805 // absence and staleness here, on the snapshot itself, is what makes the rule hold at every
1806 // send site rather than at most of them.
1807 // A missing-account reason is only ever returned when it should stop this plan, so it needs
1808 // no second test here. A hold on placements stops only a plan that would also withdraw
1809 // something: a placement-only plan is left to `RiskGuard` below, which refuses it with the
1810 // hold named on its own report entry, and that is the better result.
1811 const bool data_is_missing = *reason != WithheldPlanReason::PLACEMENT_HELD;
1812 if (data_is_missing || would_cancel) {
1813 // The whole plan, before any of its parts were counted separately. **Four reasons reach
1814 // here** and this counts them together: an account never read, one older than the stated
1815 // maximum, one read before the listing it would be reconciled with, and a hold refusing
1816 // placements for a plan that also wanted to withdraw something -- the last two on a perfectly
1817 // fresh portfolio. Counted at all because the loop's own gate
1818 // does not always describe it: with `allow_absent_portfolio` the gate can be open while this
1819 // fires.
1820 ++plans_withheld_;
1821 std::string detail = withhold_detail(*reason);
1822 if (data_is_missing) {
1823 if (retain_withheld(instruction)) {
1824 detail = "retained for the next holdings read";
1825 } else if (*reason == WithheldPlanReason::PORTFOLIO_SUPERSEDED &&
1826 std::holds_alternative<strategy::OrderIntent>(instruction)) {
1827 detail = "dropped: an order intent states a size against the mis-ordered picture";
1828 }
1829 }
1830 report.withheld_plans_.push_back(WithheldPlan<Tag>{
1831 *reason,
1832 instruction_instrument(instruction),
1833 instruction_tag(instruction),
1834 planned.cancel_,
1835 planned.place_.size(),
1836 std::move(detail),
1837 });
1838 continue;
1839 }
1840 }
1841 // **Serialized: decided on the plan, before any of its withdrawals go.** A plan whose placement will
1842 // be withheld must not withdraw first: a reprice would become a bare withdrawal, and a protective
1843 // order could be withdrawn with its replacement left waiting. A plan that only withdraws is not
1844 // held.
1845 // A target behind a resting stop places later: its plan withdraws the stop and waits
1846 // (`STOP_ORDER_RESTING`), so it is decided here too, or the stop could go with nothing to follow it.
1847 const bool defers = !planned.cancel_.empty() &&
1848 std::any_of(planned.rejected_.begin(), planned.rejected_.end(), [](const strategy::Rejection& rejection) {
1849 return rejection.reason_ == strategy::RejectReason::STOP_ORDER_RESTING;
1850 });
1851 if (serialized() && (!planned.place_.empty() || defers)) {
1852 std::optional<std::string> hold;
1853 for (const auto& order : planned.place_) {
1854 hold = serial_hold(order.instrument_, claimed);
1855 if (hold) {
1856 break;
1857 }
1858 }
1859 if (!hold && defers) {
1860 hold = serial_hold(instruction_instrument(instruction), claimed);
1861 }
1862 if (hold) {
1863 ++plans_withheld_;
1864 if (retain_withheld(instruction, /*for_phase=*/false, context_.portfolio().as_of_)) {
1865 *hold += "; retained for a later holdings read";
1866 }
1867 report.withheld_plans_.push_back(WithheldPlan<Tag>{
1868 WithheldPlanReason::AWAITING_CONFIRMATION,
1869 instruction_instrument(instruction),
1870 instruction_tag(instruction),
1871 planned.cancel_,
1872 planned.place_.size(),
1873 std::move(*hold),
1874 });
1875 continue;
1876 }
1877 for (const auto& order : planned.place_) {
1878 claimed.push_back(order.instrument_);
1879 }
1880 if (defers) {
1881 claimed.push_back(instruction_instrument(instruction));
1882 }
1883 }
1884 // **Serialized: a target that could only withdraw an order this runner still owes is kept**, and its
1885 // withdrawals go out all the same (`withdraws_owed`). Nothing is reported as withheld: nothing was.
1886 if (serialized() && planned.place_.empty() && !defers && withdraws_owed(instruction, planned.cancel_)) {
1887 retain_withheld(instruction, /*for_phase=*/false, context_.portfolio().as_of_, /*until_accounted=*/true);
1888 }
1889 for (const auto& ref : planned.cancel_) {
1890 attempt_cancel(ref);
1891 }
1892 defer_if_waiting_on_stop(instruction, planned);
1893 for (const auto& rejection : planned.rejected_) {
1894 // Reconciliation's own refusal: it could not build a valid instruction at all, which is a
1895 // different fact from a broker saying no. **Not a diagnosis**: the same bucket holds a
1896 // quantity that rounded away on the lot, which is a deliberate drop. The reason is on the
1897 // entry.
1898 ++orders_unbuildable_;
1899 last_unbuildable_ = rejection;
1900 report.rejected_.push_back(rejection);
1901 }
1902 for (const auto& order : planned.place_) {
1903 pending.push_back(order);
1904 origin.push_back(&instruction);
1905 }
1906 }
1907
1908 strategy::GuardInputs inputs;
1909 // When the order actually goes out (`send_instant`): a target kept at a ledger mark, or one
1910 // released at a holdings read, is sent between bars, and a send recorded at the bar that decided it
1911 // would already look a window old to the next bar.
1912 inputs.now_ = send_instant();
1913 inputs.phase_ = context_.phase();
1914
1915 // **Every order in this pass is checked against a view that already contains the orders this
1916 // pass has sent.** Without that, a limit is re-evaluated from the same starting point for every
1917 // order, and two orders that each fit can breach it together: with a hundred sellable and no
1918 // resting order, two sells of sixty both pass and twenty shares are sold that were not there.
1919 //
1920 // A copy per pass, mutated as orders are accepted, closes that within one `settle`.
1921 // `persist_reservation` closes the same hazard *across* passes: always for a runner that sends to
1922 // a broker listing working orders, and otherwise for the limits that read what is outstanding.
1923 // Nothing has to recognise its own order in a later listing: a working-order poll assigns
1924 // `working_` wholesale (a holdings poll keeps it), so the broker's own answer replaces the
1925 // reservation when it arrives. Between a placement and that poll, the context portfolio is
1926 // therefore the broker's last report **plus** what this runner has sent since -- which is what a
1927 // target and a limit have to size against, and why the rows are marked so that anything inferring
1928 // the broker's behaviour from the listing can tell them apart.
1929 auto view = context_.portfolio();
1930 // Resolved once for the pass. The currencies and marks do not move as orders are accepted --
1931 // only quantities do, and the guard reads those from the view -- and resolving per order would
1932 // repeat a map lookup per instrument for every instrument. Built only when the ceiling is on, so
1933 // a caller who has not asked for it pays nothing.
1934 if (guard_.limits().max_gross_exposure_) {
1935 inputs.exposure_basis_ = exposure_basis_for(view, pending);
1936 }
1937 for (std::size_t at = 0; at < pending.size(); ++at) {
1938 auto& order = pending[at];
1939 const auto& source_instruction = *origin[at];
1940 // Checked before the guards, because an unresolved instrument is a safety hold rather than a
1941 // limit breach: reporting a notional ceiling for an order that was never going to be sent
1942 // would point an operator at the wrong problem.
1943 // Checked before the ledger and before the guards, and checked *per order* rather than once
1944 // at the top. A journal that failed on the first order of a batch will fail on the second, so
1945 // a batch that carried on would produce one unexplained refusal per instrument. This is also
1946 // what makes the condition hold for a direct `feed()`, which never reaches `PollingLoop`'s
1947 // gate at all.
1948 if (guard_.halted(strategy::HaltReason::LOCAL_FAILURE)) {
1949 ++instructions_unavailable_;
1950 report.unavailable_.push_back(
1951 WithheldOrder{
1952 WithheldReason::LOCAL_FAILURE,
1953 order,
1954 guard_.hold_detail(strategy::HaltReason::LOCAL_FAILURE),
1955 }
1956 );
1957 continue;
1958 }
1959 if (ledger_.requires_reconcile(order.instrument_)) {
1960 // The indeterminacy ledger, not a guard: this instrument has an outcome nobody knows, so
1961 // nothing further goes out for it until a listing settles the question. Counted apart from
1962 // every other origin because the operator's response is to reconcile rather than to relax a
1963 // limit or fix a price.
1964 ++orders_reconcile_withheld_;
1965 std::string detail(ledger_.detail(order.instrument_));
1966 // **Serialized: the target is kept**, as at the gate below, and planned again at a later holdings
1967 // read: withheld again while the instrument is marked, sent once a listing has settled it. Not when
1968 // a later instruction of this pass spoke about the instrument: every instruction is planned before
1969 // any order is sent, so keeping this one here would undo the later one.
1970 if (serialized() && !restated_later(source_instruction) &&
1971 retain_withheld(source_instruction, /*for_phase=*/false, context_.portfolio().as_of_)) {
1972 detail += "; retained for a later holdings read";
1973 }
1974 report.withheld_.push_back(WithheldOrder{WithheldReason::RECONCILE_REQUIRED, order, std::move(detail)});
1975 continue;
1976 }
1977
1978 // **Serialized, again at the send**: an order placed earlier in this pass on the same instrument is
1979 // an order not yet accounted for. The baseline is taken here, once, and the same value goes to the
1980 // journal and to the order's record (`track_unconfirmed`).
1981 std::optional<Decimal64> baseline;
1982 if (serialized()) {
1983 if (auto hold = serial_hold(order.instrument_, {})) {
1984 ++orders_awaiting_confirmation_;
1985 if (retain_withheld(source_instruction, /*for_phase=*/false, context_.portfolio().as_of_)) {
1986 *hold += "; retained for a later holdings read";
1987 }
1988 report.withheld_.push_back(WithheldOrder{WithheldReason::AWAITING_CONFIRMATION, order, std::move(*hold)});
1989 continue;
1990 }
1991 baseline = context_.portfolio().net_quantity_of(order.instrument_);
1992 }
1993
1994 fill_sellable_if_needed(view, order);
1995
1996 inputs.reference_price_ = context_.last_price(order.instrument_);
1997 // The currency the order is priced in, which only the reference data knows. A buying-power
1998 // limit is per currency, and an unresolved instrument leaves it absent rather than defaulting
1999 // to one: guessing here would spend one currency's budget on another's order.
2000 const auto reference = context_.instrument(order.instrument_);
2001 inputs.order_currency_ =
2002 reference ? std::optional<Currency>(reference->currency_) : std::nullopt;
2003 // `evaluate` rather than `check`, for one fact `check` drops: whether the phase limit admitted
2004 // this order only because it closes a position. Nothing downstream can recover that -- the
2005 // order, the portfolio and the report all look the same either way -- and it is the one thing
2006 // an operator needs to see, because it is this library sending into a session the operator's
2007 // own limits said not to trade in.
2008 auto decision = guard_.evaluate(order, view, inputs);
2009 if (decision.rejection_) {
2010 // **Counted here and not at the send site, because there is no send.** A buy left above a
2011 // withheld ceiling is the larger notional, so every limit that multiplies quantity by price
2012 // sees a bigger number than the clamped order would have shown it. (A sell left below a
2013 // withheld floor is the smaller notional; these limits are not what stops that one.)
2014 if (order.price_would_have_been_clamped_) {
2015 ++unclamped_orders_refused_;
2016 }
2017 ++orders_guard_refused_;
2018 if (decision.rejection_->reason_ == strategy::GuardReason::PHASE_NOT_PERMITTED) {
2019 retain_for_phase(source_instruction);
2020 }
2021 report.refused_.push_back(std::move(*decision.rejection_));
2022 continue;
2023 }
2024
2025 // **The route it is sent with, written into the order before anything records it.** A
2026 // strategy names no venue, and a binding may send one anyway (`SessionExecutor::default_route`).
2027 // Resolved here rather than only at the wire so the journal line, the row `reserve` keeps --
2028 // which `cancel_all` and every other instruction built from it restate -- and the report all
2029 // name the venue the broker was asked for, not the blank the strategy left.
2030 order.venue_ = requested_route(order.venue_);
2031 const auto outcome = place_order(order, baseline);
2032 // **Counted where the outcome is known, not where the exception was granted.** An admitted
2033 // order can still be refused by the sellable check, a buying-power limit or either ceiling and
2034 // never leave the process, so counting at the guard would report a proxy for the harm rather
2035 // than the harm. `REJECTED` counts: the request went out and the broker answered it, which is
2036 // exactly the nightly-rejection case this exception was chosen knowing about, and a witness
2037 // that stayed at zero through every one of them would be silent in the case it exists for.
2038 // `REFUSED_LOCALLY` and `UNAVAILABLE` do not: nothing was sent. This is the same division
2039 // `note_sent` already makes, for the same reason.
2040 if (decision.admitted_outside_phase_ && send_outcome_reached_broker(outcome)) {
2041 ++sends_outside_permitted_phase_;
2042 }
2043 // **Split on whether a broker was asked, which is the division `send_outcome_reached_broker`
2044 // already makes.** `REJECTED` means the exchange saw this price and refused it.
2045 // `REFUSED_LOCALLY` means the binding did -- and that is a price-dependent refusal too: a
2046 // limit price no adapter will send is a limit price the withheld band might have moved into
2047 // range. The concrete case is a sell at zero that `Price::checked` admits
2048 // because zero sits on a stated grid. Counting only the first left that invisible.
2049 // `INDETERMINATE` is excluded because it is not a refusal at all -- the request may well be
2050 // resting -- and `UNAVAILABLE` because it is local machinery being unavailable, which no price
2051 // affects.
2052 if (order.price_would_have_been_clamped_) {

2054 ++unclamped_orders_rejected_;

2056 ++unclamped_orders_refused_;
2057 }

2059 // After every send, not only after a refusal. A terminal journal write that fails leaves the
2060 // broker's own outcome intact -- the send happened, and saying otherwise would be a lie -- but
2061 // it still means the journal is broken, and the runner must stop.

2066 // guard refused or the ledger withheld does not spend budget.

2070 // money is spent. An `INDETERMINATE` send is counted the same way below, with no reference.
2071 reserve(
2072 view, order, inputs.reference_price_, inputs.order_currency_, last_ref(),
2073 guard_.limits().enforce_buying_power_
2074 );

2078 break;
2079 case SendOutcome::REJECTED:

2081 // still consumed a request, which is what the broker will have counted.
2082 guard_.note_sent(inputs.now_);
2083 ++orders_broker_refused_;
2084 // An exit the phase limit let out of session, refused by a broker not taking orders then:
2085 // the target waits for the session instead of for its next bar, which may be a day away.
2086 if (decision.admitted_outside_phase_) {
2087 retain_for_phase(source_instruction, /*broker_answered=*/true);
2088 } else if (last_error_is(broker_rate_limited)) {
2089 // **Turned away for the broker's rate limit, which is about the moment.** Kept as an
2090 // unreachable send is, and offered again after a newer holdings read: dropped, a daily
2091 // strategy's opening order or protective stop refused once at a busy open waited a day
2092 // (in a simulated run, one `EGW00201` at 09:00 moved a buy to the next session and left a position
2093 // without its stop all day). The broker processed nothing, so a retry cannot double it.
2094 retain_for_phase(source_instruction, /*broker_answered=*/true, context_.portfolio().as_of_);
2095 }

2097 report.rejected_.push_back(
2098 strategy::Rejection{
2099 strategy::RejectReason::BROKER_REFUSED, order.instrument_, order.tag_
2100 }
2101 );
2102 break;
2103 case SendOutcome::INDETERMINATE:
2104 guard_.note_sent(inputs.now_);
2105 // Counted like an acceptance for the purpose below: the order may exist, and a listing
2106 // read before this moment cannot show it either way.
2107 ++placements_at_broker_;
2108 ledger_.mark(order.instrument_, "an order request has an unknown outcome");
2109 ++orders_indeterminate_;
2110 // **Counted as if accepted, until the sweep says what became of it.** The ledger
2111 // withholds the instrument meanwhile; but once a sweep finds the order in the broker's
2112 // listing it clears the ledger, and a market order that filled is then in no listing while
2113 // holdings may not show it yet, so a target sent it again (twice the target, in one
2114 // process). Reserved and tracked here with the holding at send as the baseline, and handed
2115 // its reference by `resolve_indeterminate_placement`. **No reference** is passed on purpose:
2116 // `last_ref()` still names the previous accepted order, and a row carrying it could be
2117 // withdrawn in that order's place.
2118 // A serialized runner records it whatever its executor reports: it owns the instrument until
2119 // accounted for.

2125 );

2128 }
2129 }
2130 report.indeterminate_.push_back(std::move(order));
2131 break;
2132 case SendOutcome::REFUSED_LOCALLY:
2133 // **No `note_sent`**: the binding refused it, so no request went out and the broker
2134 // counted nothing. Spending rate-limit budget here would refuse a later order that would
2135 // have been allowed, for a request nobody made. Reported as a rejection because that is
2136 // what it is from the order's point of view -- it does not exist, and nothing needs
2137 // reconciling -- and the runner keeps trading, because the next instruction may be
2138 // perfectly sendable. The reason is carried on the entry itself rather than left to
2139 // `last_error()`: a later accepted order in the same pass clears that, so a report can
2140 // outlive the only record of which refusal this was.
2141 ++orders_binding_refused_;
2142 if (decision.admitted_outside_phase_) {
2143 retain_for_phase(source_instruction);
2144 } else if (last_error_is(broker_unreachable)) {
2145 // **The broker could not be reached, so the decision is kept rather than lost.** Dropped,
2146 // a daily strategy's target or protective stop released at the open into a dropped
2147 // connection waited for its next bar, a day away (in a simulated run, a 30-minute outage at 09:00 moved a
2148 // buy to the next session and left a position without its stop all day). Kept as a phase
2149 // refusal is -- discarded by a stop, an operator's halt or a local failure, replaced by a

2157 strategy::Rejection{
2158 strategy::RejectReason::BINDING_REFUSED, order.instrument_, order.tag_
2159 }
2160 );
2161 break;
2162 case SendOutcome::UNAVAILABLE:

2164 // budget was spent. The ledger is not marked either -- an instruction that was never
2165 // attempted leaves no doubt about the position, and withholding the instrument would be
2166 // withholding it for the wrong reason.
2167 ++instructions_unavailable_;
2168 report.unavailable_.push_back(
2169 WithheldOrder{
2170 WithheldReason::LOCAL_FAILURE,
2171 order,
2172 guard_.halted(strategy::HaltReason::LOCAL_FAILURE) ?
2173 guard_.hold_detail(strategy::HaltReason::LOCAL_FAILURE) :
2174 std::string("local machinery refused to send"),

2176 );
2177 break;
2178 }

2364 // The serialized half of `confirm_reservations`: one order per instrument, so no group arithmetic.
2365 //
2366 // Each order is accounted for, with a `CONFIRMED` line written first, when it is named, no listing
2367 // shows it open, and either its history capped what it filled at zero, or a holdings read shows
2368 // exactly its baseline plus what it filled. Holdings past that hold it for an operator. Anything else
2369 // rebuilds its reserved row from what nothing explains yet; a timeout is counted once and releases
2370 // nothing (only an order not yet named marks the ledger).
2371 void confirm_serialized(strategy::Portfolio<Tag>& portfolio, std::chrono::system_clock::time_point now) {
2372 const auto zero = Decimal64::parse("0");
2373 const auto epoch = std::chrono::system_clock::time_point{};
2374 const auto equal = [](const Decimal64& a, const Decimal64& b) { return !decimal_less(a, b) && !decimal_less(b, a); };
2375 for (auto at = unconfirmed_.begin(); at != unconfirmed_.end();) {
2376 auto& entry = *at;
2377 persist_hold(entry);
2378 // An actual read, and not one older than the read the order was decided against.
2379 const bool read = portfolio.as_of_ != epoch && !(portfolio.as_of_ < entry.baseline_at_);
2380 // What a listing still shows of it: that part cannot have reached holdings.
2381 Decimal64 listed = zero;
2382 entry.listed_now_ = false;
2383 std::optional<Decimal64> listed_filled;
2384 if (const auto* working = broker_row_for(portfolio, entry)) {
2385 entry.listed_ = true;
2386 entry.listed_now_ = true;
2387 listed = decimal_subtract(working->order_.quantity_, working->order_.execution_.filled_quantity_).value_or(zero);
2388 listed_filled = working->order_.execution_.filled_quantity_;
2389 }
2390 // A listing of its own reporting more filled than it was for, or than its history said it ended with,
2391 // contradicts what is known: held for an operator before the listing can move on and the contradiction
2392 // be forgotten. (Holding is the safe direction, so an undated row may prove it.)
2393 if (listed_filled && !entry.operator_only_ &&
2394 (decimal_less(entry.row_.order_.quantity_, *listed_filled) ||
2395 (entry.settled_ && decimal_less(entry.expected_, *listed_filled)))) {
2396 hold_for_operator(
2397 entry, "an order this runner sent" + named(entry) + " is listed with " + listed_filled->str() +
2398 " filled, more than " + (entry.settled_ ? "its history said it ended with (" + entry.expected_.str() + ")" :
2399 "it was for")
2400 );
2401 }
2402 // The holding, and its movement in the order's direction since the baseline, from an actual read.
2403 std::optional<Decimal64> held;
2404 std::optional<Decimal64> moved;
2405 if (read) {
2406 held = portfolio.net_quantity_of(entry.instrument_);
2407 if (held) {
2408 moved = entry.side_ == OrderSide::BUY ? decimal_subtract(*held, entry.baseline_) :
2409 decimal_subtract(entry.baseline_, *held);
2410 }
2411 }
2412 const bool is_named = !entry.ref_.primary_.empty();
2413 if (!entry.operator_only_ && is_named && entry.identified_ && !entry.listed_now_) {
2414 if (entry.expected_.coefficient() == 0) {
2415 if (discharge(entry, std::nullopt, "its own history row shows it ended with nothing filled")) {
2416 at = unconfirmed_.erase(at);
2417 continue;
2418 }
2419 } else if (moved && equal(*moved, entry.expected_)) {
2420 if (discharge(entry, held, "holdings show " + held->str() + ": its baseline " + entry.baseline_.str() +
2421 " and the " + entry.expected_.str() + " it filled")) {
2422 set_boundary(entry.instrument_, portfolio.as_of_, /*strict=*/false);
2423 at = unconfirmed_.erase(at);
2424 continue;
2425 }
2426 } else if (moved && decimal_less(entry.expected_, *moved)) {
2427 hold_for_operator(
2428 entry, "holdings moved " + moved->str() + " from the baseline " + entry.baseline_.str() + ", more than the " +
2429 entry.expected_.str() + " an order this runner sent" + named(entry) + " can explain"
2430 );
2431 }
2432 }
2433
2434 // What nothing explains yet: what it expects, less what holdings show of it, less what is listed.
2435 Decimal64 covered = zero;
2436 const auto fillable = decimal_subtract(entry.expected_, listed).value_or(zero);
2437 if (!entry.operator_only_ && moved && decimal_less(zero, *moved) && decimal_less(zero, fillable)) {
2438 covered = decimal_less(*moved, fillable) ? *moved : fillable;
2439 }
2440 auto residual = decimal_subtract(fillable, covered).value_or(entry.expected_);
2441 if (!decimal_less(zero, residual)) {
2442 mark_explained(entry);
2443 ++at;
2444 continue;
2445 }
2446 entry.residual_ = residual;
2447 // Not from the epoch: a restored order's clock starts with the first real instant.
2448 if (!entry.unexplained_since_ && now != epoch) {
2449 entry.unexplained_since_ = now;
2450 }
2451 if (!entry.noticed_ && !entry.operator_only_ && (entry.settled_ || entry.history_missed_ || !is_named) &&
2452 timed_out(entry, now) && portfolio.as_of_ - *entry.unexplained_since_ > confirmation_timeout_) {
2453 // A notice, not a release: the order still holds its instrument, and later evidence can still
2454 // account for it. A named one marks no ledger: the gate already withholds the instrument while it
2455 // is owed, and a mark would outlive the holdings that account for it. An unnamed one is accounted
2456 // for only by an operator.
2457 entry.noticed_ = true;
2458 ++orders_unconfirmed_timed_out_;
2459 if (!is_named) {
2460 ledger_.mark(
2461 entry.instrument_,
2462 "an order this runner sent" + named(entry) +
2463 " was not accounted for by holdings within the confirmation timeout; it still holds the instrument"
2464 );
2465 }
2466 }
2467 // Not shown while the ledger holds an order not yet named; see the overlapping branch.
2468 if (!is_named && ledger_.requires_reconcile(entry.instrument_)) {
2469 ++at;
2470 continue;
2471 }
2472 rebuild_residual_row(portfolio, entry, residual);
2473 ++at;
2474 }

3309 // still covers every other pass before a listing (a runner fed by hand, two names in one bar

3913 const auto& portfolio = context_.portfolio();

3915 // accept an account nobody read or one read too long ago; this one was read, recently, and before
3916 // the listing it would be reconciled with -- which is the order in which a fill between the two
3917 // reads makes a position vanish. Exempt only while a deferred target is being released: that
3918 // release fires on the stop's own history showing it ended unfilled, so the listing change behind
3919 // it is by construction not a fill.
3920 if (!releasing_deferred_ && portfolio.listing_newer_than_holdings()) {
3921 return WithheldPlanReason::PORTFOLIO_SUPERSEDED;
3922 }

3924 // 1970. The same sentinel the loop uses; both writers of `as_of_` in this library stamp it.

3926 const bool too_old = !never_read && maximum_portfolio_age_ &&
3927 freshness_instant() - portfolio.as_of_ > *maximum_portfolio_age_;
3928

3930 // **A plan that would withdraw something is stopped whatever was opted out of.** The opt-outs
3931 // say the operator accepts *sizing* against an account this library cannot vouch for. They do
3932 // not say it should withdraw orders it cannot account for, and the two are not the same
3933 // concession: an account that merely looks flat makes a resting protective order look opposed.
3934 if (would_cancel) {
3935 return never_read ? WithheldPlanReason::PORTFOLIO_ABSENT : WithheldPlanReason::PORTFOLIO_STALE;
3936 }
3937 // A placement is the half the opt-outs are about, and it is also the half a driving loop's
3938 // hold would otherwise catch -- so it is stopped here only for a run that asked for the
3939 // account to be known.
3940 if (require_known_portfolio_ && !(never_read && allow_absent_portfolio_)) {
3941 return never_read ? WithheldPlanReason::PORTFOLIO_ABSENT : WithheldPlanReason::PORTFOLIO_STALE;

3944 // check below: returning early here was a defect, because it let a plan out under an operator's

3948 // while the withdrawal has already gone, and every reason that refuses a placement does that.

4002 void send_cancel(const OrderRef<Tag>& ref, Report& report) {
4003 // Gated like a placement, and in the same order. A cancellation is the one instruction that
4004 // cannot make an unresolved *position* worse, which is why the ledger does not gate it -- but it
4005 // still goes through the journal, and a journal that cannot record it must not let it be sent
4006 // unrecorded.
4007 if (guard_.halted(strategy::HaltReason::LOCAL_FAILURE)) {
4008 ++instructions_unavailable_;
4009 report.unavailable_cancel_.push_back(ref);
4010 return;
4011 }
4012 // And a halt stops this runner *sending*, which a cancellation is -- but only a halt that is
4013 // about this instruction. The argument for refusing here is `StopIntent` reaching this runner
4014 // from a signal and from a file: exempting cancellations wholesale would let
4015 // `echo withdraw > /var/run/atl/stop` send an instruction that an operator's own halt forbade,
4016 // and a kill switch a control file can partly bypass is not one. An operator who wants out
4017 // resumes first, which is one deliberate act and is visible in the report either way.
4018 //
4019 // **That argument covers the holds somebody set, and not the holds that merely mean this runner
4020 // is uninformed.** Refusing to withdraw a resting order because the *holdings* poll failed is
4021 // worse than the exposure it guards: the order keeps resting and can fill, and the reference it
4022 // would be withdrawn by came from a working-order read that succeeded. So the question asked is
4023 // `InstructionClass::WITHDRAWAL`, and `strategy::halt_blocks` holds the division and its
4024 // reasoning. The row is still resolved below, and a reference the snapshot cannot place is
4025 // reported rather than sent blind.
4026 if (const auto holding =
4027 guard_.blocking_reason_for(strategy::InstructionClass::WITHDRAWAL)) {
4028 ++cancels_refused_;
4029 report.refused_cancel_.push_back(RefusedCancel<Tag>{ref, *holding});
4030 return;
4031 }
4032 // Resolved against the portfolio first, because building a cancellation needs the order's own
4033 // quantity on a broker that requires one. A reference not in the snapshot means the snapshot is
4034 // stale, and sending a cancellation for an order the runner cannot describe is exactly the kind
4035 // of blind instruction that should be visible instead.
4036 // The broker's own row first: a reserved row for the same order is this runner's arithmetic.
4037 const WorkingOrder<Tag>* working = nullptr;
4038 for (const auto& row : context_.portfolio().working_) {
4039 if (row.ref_ == ref && (working == nullptr || (working->reserved_ && !row.reserved_))) {
4040 working = &row;
4041 }
4042 }
4043 if (working == nullptr) {
4044 ++cancels_unresolved_;
4045 report.unresolved_cancel_.push_back(ref);
4046 return;
4047 }
4048 // **Serialized: a reserved row whose reference the broker did not give is not sent.** A restored
4049 // order, or one a sweep named, carries the order id alone, which some brokers cannot cancel by; and a
4050 // row with no id addresses nothing. Refused as the binding would, before anything is sent.
4051 if (working->reserved_ && serialized() && !cancellable_reservation(*working)) {
4052 ++cancels_rejected_;
4053 report.rejected_cancel_.push_back(RejectedCancel<Tag>{ref, /*by_broker=*/false});
4054 return;
4055 }
4056 const auto outcome = executor_.cancel(*working);
4057 note_executor_failure(send_outcome_is_local_failure(outcome));
4058 // Switched over rather than tested one outcome at a time, deliberately. Three different outcomes
4059 // once fell through to `would_cancel_` -- the bucket that says the
4060 // broker withdrew the order -- because each fix handled the case it was told about and left the
4061 // rest. A switch with no default makes the next added enumerator a compile error in the library's
4062 // test and example builds (`-Werror=switch`), which is the only version of this that stays fixed.
4063 switch (outcome) {
4064 case SendOutcome::ACCEPTED:
4065 // The broker took the request. Whether the order stops working is a later fact, which only
4066 // its own listing reports: cancellation is asynchronous, and a broker that answers here can
4067 // still reject the withdrawal afterwards.
4068 //
4069 // **Counted against the rate limit, like a placement.** A withdrawal is a request the
4070 // broker counted, and a window that does not see it is not the window the operator
4071 // configured -- a `WITHDRAW` over thirty resting orders is thirty requests the guard was
4072 // blind to, and the next placement is then checked against a window it believes is empty.
4073 // It is still never *refused* by that limit: `send_cancel` calls no `check`, because a rate
4074 // limit must never be the reason an order cannot be withdrawn.
4075 guard_.note_sent(send_instant());
4076 // A sent order still counted may have filled before the withdrawal; its history decides.
4077 note_withdrawal_accepted(ref);
4078 report.would_cancel_.push_back(ref);
4079 return;
4080 case SendOutcome::REJECTED:
4081 // The broker refused the request, which is an answer: nothing needs reconciling. It is not
4082 // a withdrawal that happened, and it is not proof the order is working either -- a fill is
4083 // one of the ordinary reasons a withdrawal is refused. It still spent a request.
4084 guard_.note_sent(send_instant());
4085 ++cancels_rejected_;
4086 report.rejected_cancel_.push_back(RejectedCancel<Tag>{ref, /*by_broker=*/true});
4087 return;
4088 case SendOutcome::INDETERMINATE:
4089 // Marked *and* reported apart from `would_cancel_`. The mark was always here; reporting it
