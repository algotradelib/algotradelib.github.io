217 static Result<KisAmendment> to_amendment(
218 const Ref& ref,
219 const AmendInstruction& instruction
220 ) {
221 if (!instruction.quantity_) {
222 BrokerError error;
223 error.category_ = ErrorCategory::INVALID_ARGUMENT;
224 error.reached_broker_ = false;
225 error.broker_code_ = "quantity_required";
226 error.message_ =
227 "KIS requires an explicit quantity on a modification or cancellation, including a full "
228 "cancellation; read the amendable quantity from working orders first";
229 return Result<KisAmendment>(std::move(error));
230 }
231
232 KisAmendment amendment;
233 amendment.order_ = from_ref(ref);
234 // What the shape leaves unset stays as it is, which the resting shape says.
235 // KIS composes one `ORD_DVSN` from all three, so it cannot send an
236 // amendment with a condition or session it does not know -- an unset field
237 // with no resting shape to resolve it from is refused rather than defaulted,
238 // because defaulting it here is how `00` got sent for an IOC order.
239 const auto condition = instruction.shape_.time_in_force_
240 ? instruction.shape_.time_in_force_
241 : (instruction.resting_ ? instruction.resting_->time_in_force_ : std::nullopt);
242 const auto session = instruction.shape_.session_
243 ? instruction.shape_.session_
244 : (instruction.resting_ ? instruction.resting_->session_ : std::nullopt);
245 if (!condition || !session) {
246 BrokerError error;
247 error.category_ = ErrorCategory::INVALID_ARGUMENT;
248 error.reached_broker_ = false;
249 error.broker_code_ = "incomplete_shape";
250 error.message_ =
251 "KIS re-sends the order code on an amendment and cannot compose one without the order's "
252 "validity condition and session; state them on the shape or build the instruction with "
253 "reprice() or replace_with(), which take them from the resting order";
254 return Result<KisAmendment>(std::move(error));
255 }
256 amendment.type_ = instruction.shape_.type_;
257 amendment.time_in_force_ = *condition;
258 amendment.session_ = *session;
259 amendment.venue_ = instruction.venue_;
260 amendment.quantity_ = *instruction.quantity_;
261 amendment.price_ = instruction.shape_.price_;
262 amendment.trigger_price_ = amended_trigger_price(instruction);
263 amendment.all_remaining_ = instruction.all_remaining_;
264 return Result<KisAmendment>(std::move(amendment));
265 }

302 static Result<KisHistoryRange> to_history_query(
303 const KisAccount&,
304 const OrderHistoryQuery& query
305 ) {
306 BrokerError error;
307 error.category_ = ErrorCategory::INVALID_ARGUMENT;
308 error.reached_broker_ = false;
309 if (!query.from_ || !query.to_) {
310 error.broker_code_ = "range_required";
311 error.message_ = "KIS order history requires an inclusive from and to calendar date";
312 return Result<KisHistoryRange>(std::move(error));
313 }
314 if (query.symbol_) {
315 error.broker_code_ = "symbol_filter_unsupported";
316 error.message_ =
317 "KIS order history is not filtered by instrument by this adapter; filter the returned "
318 "orders instead";
319 return Result<KisHistoryRange>(std::move(error));
320 }
321 if (query.cursor_) {
322 error.broker_code_ = "cursor_unsupported";
323 error.message_ = "KIS order history pages internally and exposes no cursor";
324 return Result<KisHistoryRange>(std::move(error));
325 }
326 if (query.limit_) {
327 error.broker_code_ = "limit_unsupported";
328 error.message_ = "KIS order history returns every page and cannot honor a row limit";
329 return Result<KisHistoryRange>(std::move(error));
330 }
331 return Result<KisHistoryRange>(KisHistoryRange{*query.from_, *query.to_});
332 }
