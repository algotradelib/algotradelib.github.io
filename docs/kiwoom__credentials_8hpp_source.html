44inline bool valid_label(std::string_view label) {
45 if (label.empty() || label.front() == '-') {
46 return false;
47 }
48 for (const char c : label) {
49 if (!((c >= 'a' && c <= 'z') || (c >= '0' && c <= '9') || c == '-')) {
50 return false;
51 }
52 }
53 return true;
54}

63inline std::optional<std::string> take_label(int& argc, char** argv) {
64 std::optional<std::string> label;
65 int kept = 1;
66 for (int i = 1; i < argc; ++i) {
67 if (std::string_view(argv[i]) == "--kiwoom-account") {
68 if (i + 1 >= argc || std::string_view(argv[i + 1]).rfind("--", 0) == 0) {
69 throw std::runtime_error("--kiwoom-account needs a label, such as kr");
70 }
71 label = argv[++i];
72 if (!valid_label(*label)) {
73 throw std::runtime_error("--kiwoom-account takes lowercase letters, digits and '-', not \"" + *label + "\"");
74 }
75 } else {
76 argv[kept++] = argv[i];
77 }
78 }
79 argc = kept;
80 return label;
81}
