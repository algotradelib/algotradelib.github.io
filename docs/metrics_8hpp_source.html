37inline Summary summarize(math::SeriesView values) {
38 const auto parts = math::observed_parts(values);
39 Summary result;
40 const Scalar count = parts.present_.sum();
41 result.count_ = static_cast<std::size_t>(count);
42 if (result.count_ == 0) {
43 return result;
44 }
45 result.mean_ = parts.values_.sum() / count;
46 result.squared_ = (parts.values_ - result.mean_ * parts.present_).squaredNorm();
47 return result;
48}

74inline MaxDrawdown max_drawdown(math::SeriesView levels) {
75 MaxDrawdown result;
76 result.magnitude_ = missing();
77 Scalar peak{0};
78 math::Index peak_row = -1;
79 std::size_t observations = 0;
80 for (math::Index row = 0; row < levels.size(); ++row) {
81 if (!observed(levels[row])) {
82 continue;
83 }
84 ++observations;
85 if (peak_row < 0 || peak < levels[row]) {
86 peak = levels[row];
87 peak_row = row;
88 }
89 if (peak <= Scalar{0}) {
90 continue;
91 }
92 const Scalar decline = (peak - levels[row]) / peak;
93 if (!observed(result.magnitude_) || result.magnitude_ < decline) {
94 result.magnitude_ = decline;
95 result.peak_row_ = peak_row;
96 result.trough_row_ = row;
97 }
98 }
99 // Levels were present but no ratio could be formed from them -- every running peak was at or
100 // below zero. That is "undefined", not "no data".
101 if (observations > 0 && !observed(result.magnitude_)) {
102 result.magnitude_ = math::undefined();
103 }
104 return result;
105}

118inline Scalar sharpe(math::SeriesView period_returns, Scalar risk_free = Scalar{0}) {
119 const auto summary = metric_detail::summarize(period_returns);
120 if (summary.count_ < 2) {
121 return missing();
122 }
123 const Scalar deviation =
124 std::sqrt(summary.squared_ / static_cast<Scalar>(summary.count_ - 1));
125 if (deviation == Scalar{0}) {
126 // Returns were present and never varied. `undefined()` rather than `missing()`, because a
127 // caller ranking strategies should be able to tell this from "not enough history".
128 return math::undefined();
129 }
130 return (summary.mean_ - risk_free) / deviation;
131}

144inline Scalar sortino(math::SeriesView period_returns, Scalar target = Scalar{0}) {
145 std::size_t count = 0;
146 std::size_t below = 0;
147 Scalar sum{0};
148 Scalar shortfall{0};
149 for (math::Index row = 0; row < period_returns.size(); ++row) {
150 const Scalar value = period_returns[row];
151 if (!observed(value)) {
152 continue;
153 }
154 ++count;
155 sum += value;
156 if (value < target) {
157 ++below;
158 const Scalar gap = target - value;
159 shortfall += gap * gap;
160 }
161 }
162 if (count < 2) {
163 return missing();
164 }
165 if (below == 0) {
166 // Nothing fell short of the target. There is no downside risk to divide by, and an infinity
167 // would survive every later comparison as the best strategy ever measured. `undefined()`
168 // says "computed, and does not exist" rather than "no data", so a caller ranking strategies
169 // can treat the two differently.
170 return math::undefined();
171 }
172 // Divided by the whole count, not by `below`. That is the definition, and the difference
173 // is a systematic flattery of strategies with rare losses.
174 const Scalar downside = std::sqrt(shortfall / static_cast<Scalar>(count));
175 if (downside == Scalar{0}) {
176 return math::undefined();
177 }
178 return (sum / static_cast<Scalar>(count) - target) / downside;
179}

217inline math::Vector one_way_turnover(
218 math::MatrixView weights,
219 TurnoverStart start = TurnoverStart::FROM_CASH
220) {
221 // Masked once for the whole panel, so each period is a plain vectorized row difference.
222 const math::Matrix held =
223 (weights.array() == weights.array()).select(weights, Scalar{0});
224 math::Vector result = math::unobserved(static_cast<std::size_t>(weights.rows()));
225 for (math::Index row = 0; row < held.rows(); ++row) {
226 if (row == 0) {
227 if (start == TurnoverStart::SKIP) {
228 continue;
229 }
230 result[row] = Scalar{0.5} * held.row(0).cwiseAbs().sum();
231 continue;
232 }
233 result[row] = Scalar{0.5} * (held.row(row) - held.row(row - 1)).cwiseAbs().sum();
234 }
235 return result;
236}
