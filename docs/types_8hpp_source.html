310 static Decimal64 parse(std::string_view input) {
311 if (input.empty()) {
312 throw std::invalid_argument("decimal value is empty");
313 }
314
315 bool negative = false;
316 std::size_t index = 0;
317 if (input.front() == '+' || input.front() == '-') {
318 negative = input.front() == '-';
319 index = 1;
320 }
321 if (index == input.size()) {
322 throw std::invalid_argument("decimal value has no digits");
323 }
324
325 std::uint64_t magnitude = 0;
326 std::uint8_t scale = 0;
327 bool decimal_point = false;
328 bool has_digit = false;
329 const std::uint64_t limit = negative ?
330 static_cast<std::uint64_t>(std::numeric_limits<std::int64_t>::max()) + 1ULL :
331 static_cast<std::uint64_t>(std::numeric_limits<std::int64_t>::max());
332
333 for (; index < input.size(); ++index) {
334 const char ch = input[index];
335 if (ch == '.') {
336 if (decimal_point) {
337 throw std::invalid_argument("decimal value contains multiple points");
338 }
339 decimal_point = true;
340 continue;
341 }
342 if (ch < '0' || ch > '9') {
343 throw std::invalid_argument("decimal value contains an invalid character");
344 }
345 has_digit = true;
346 const auto digit = static_cast<std::uint64_t>(ch - '0');
347 if (magnitude > (limit - digit) / 10ULL) {
348 throw std::out_of_range("decimal value exceeds Decimal64 range");
349 }
350 magnitude = magnitude * 10ULL + digit;
351 if (decimal_point) {
352 if (scale == std::numeric_limits<std::uint8_t>::max()) {
353 throw std::out_of_range("decimal scale exceeds Decimal64 range");
354 }
355 ++scale;
356 }
357 }
358 if (!has_digit) {
359 throw std::invalid_argument("decimal value has no digits");
360 }
361
362 while (scale > 0 && magnitude % 10ULL == 0) {
363 magnitude /= 10ULL;
364 --scale;
365 }
366
367 std::int64_t coefficient;
368 if (
369 negative &&
370 magnitude == static_cast<std::uint64_t>(std::numeric_limits<std::int64_t>::max()) + 1ULL
371 ) {
372 coefficient = std::numeric_limits<std::int64_t>::min();
373 } else {
374 coefficient = static_cast<std::int64_t>(magnitude);
375 if (negative) {
376 coefficient = -coefficient;
377 }
378 }
379 return Decimal64(coefficient, scale);
380 }
381
386 std::int64_t coefficient() const noexcept { return coefficient_; }
387
392 std::uint8_t scale() const noexcept { return scale_; }
393
398 std::string str() const {
399 const bool negative = coefficient_ < 0;
400 std::uint64_t magnitude = negative ?
401 static_cast<std::uint64_t>(-(coefficient_ + 1)) + 1ULL :
402 static_cast<std::uint64_t>(coefficient_);
403 std::string digits = std::to_string(magnitude);
404 if (scale_ > 0) {
405 if (digits.size() <= scale_) {
406 digits.insert(0, static_cast<std::size_t>(scale_) - digits.size() + 1, '0');
407 }
408 digits.insert(digits.size() - scale_, 1, '.');
409 }
410 if (negative) {
411 digits.insert(digits.begin(), '-');
412 }
413 return digits;
414 }

460inline bool align(
461 const Decimal64& lhs,
462 const Decimal64& rhs,
463 std::int64_t& left,
464 std::int64_t& right,
465 std::uint8_t& scale
466) {
467 scale = lhs.scale() > rhs.scale() ? lhs.scale() : rhs.scale();
468 left = lhs.coefficient();
469 right = rhs.coefficient();
470 for (std::uint8_t step = lhs.scale(); step < scale; ++step) {
471 if (left > std::numeric_limits<std::int64_t>::max() / 10 ||
472 left < std::numeric_limits<std::int64_t>::min() / 10) {
473 return false;
474 }
475 left *= 10;
476 }
477 for (std::uint8_t step = rhs.scale(); step < scale; ++step) {
478 if (right > std::numeric_limits<std::int64_t>::max() / 10 ||
479 right < std::numeric_limits<std::int64_t>::min() / 10) {
480 return false;
481 }
482 right *= 10;
483 }
484 return true;
485}

558inline std::optional<Decimal64> decimal_multiply(const Decimal64& lhs, const Decimal64& rhs) {
559 const auto left = lhs.coefficient();
560 const auto right = rhs.coefficient();
561 const int scale = static_cast<int>(lhs.scale()) + static_cast<int>(rhs.scale());
562 if (scale > std::numeric_limits<std::uint8_t>::max()) {
563 return std::nullopt;
564 }
565 if (left != 0 && right != 0) {
566 // Checked before multiplying rather than after, because signed overflow is undefined and a
567 // wrapped price is exactly the failure this type exists to prevent.
568 if (left > 0 ? (right > 0 ? left > std::numeric_limits<std::int64_t>::max() / right
569 : right < std::numeric_limits<std::int64_t>::min() / left)
570 : (right > 0 ? left < std::numeric_limits<std::int64_t>::min() / right
571 : left < std::numeric_limits<std::int64_t>::max() / right)) {
572 return std::nullopt;
573 }
574 }
575 return decimal_normalized(Decimal64(left * right, static_cast<std::uint8_t>(scale)));
576}

589inline std::optional<Decimal64> decimal_divide(
590 const Decimal64& lhs,
591 const Decimal64& rhs,
592 std::uint8_t scale
593) {
594 std::int64_t left = 0;
595 std::int64_t right = 0;
596 std::uint8_t common = 0;
597 if (!decimal_detail::align(lhs, rhs, left, right, common) || right == 0) {
598 return std::nullopt;
599 }
600 // Aligning cancels the scales, so `left / right` is the plain ratio. Each requested decimal
601 // place is one more factor of ten taken before the division.
602 for (std::uint8_t step = 0; step < scale; ++step) {
603 if (left > std::numeric_limits<std::int64_t>::max() / 10 ||
604 left < std::numeric_limits<std::int64_t>::min() / 10) {
605 return std::nullopt;
606 }
607 left *= 10;
608 }
609 if (left == std::numeric_limits<std::int64_t>::min() && right == -1) {
610 return std::nullopt;
611 }
612 return decimal_normalized(Decimal64(left / right, scale));
613}

626inline std::optional<Decimal64> decimal_floor_to_multiple(
627 const Decimal64& value,
628 const Decimal64& step
629) {
630 std::int64_t left = 0;
631 std::int64_t right = 0;
632 std::uint8_t scale = 0;
633 if (!decimal_detail::align(value, step, left, right, scale) || right <= 0) {
634 return std::nullopt;
635 }
636 auto rest = left % right;
637 if (rest < 0) {
638 rest += right;
639 }
640 if (left < std::numeric_limits<std::int64_t>::min() + rest) {
641 return std::nullopt;
642 }
643 return decimal_normalized(Decimal64(left - rest, scale));
644}

652inline bool decimal_less(const Decimal64& lhs, const Decimal64& rhs) {
653 std::int64_t left = 0;
654 std::int64_t right = 0;
655 std::uint8_t scale = 0;
656 if (decimal_detail::align(lhs, rhs, left, right, scale)) {
657 return left < right;
658 }
659 // Rescaling overflowed, so the values differ by more than any shared scale can
660 // express: the one with the smaller scale has the larger magnitude. Between two
661 // negatives the larger magnitude is the smaller value.
662 const bool left_negative = lhs.coefficient() < 0;
663 if (left_negative != (rhs.coefficient() < 0)) {
664 return left_negative;
665 }
666 return left_negative ? lhs.scale() < rhs.scale() : lhs.scale() > rhs.scale();
667}

1128inline Date date_from_epoch_days(std::int64_t days) {
1129 // Shifts the era so that the year starts in March, which is what removes the leap-day special
1130 // case from everything below: 1970-01-01 is 719468 days after 0000-03-01.
1131 const std::int64_t shifted = days + 719468;
1132 const std::int64_t era = (shifted >= 0 ? shifted : shifted - 146096) / 146097;
1133 const std::int64_t day_of_era = shifted - era * 146097;
1134 const std::int64_t year_of_era =
1135 (day_of_era - day_of_era / 1460 + day_of_era / 36524 - day_of_era / 146096) / 365;
1136 const std::int64_t year = year_of_era + era * 400;
1137 const std::int64_t day_of_year =
1138 day_of_era - (365 * year_of_era + year_of_era / 4 - year_of_era / 100);
1139 const std::int64_t shifted_month = (5 * day_of_year + 2) / 153;
1140 const std::int64_t day = day_of_year - (153 * shifted_month + 2) / 5 + 1;
1141 const std::int64_t month = shifted_month + (shifted_month < 10 ? 3 : -9);
1142 return Date{
1143 static_cast<int>(year + (month <= 2 ? 1 : 0)),
1144 static_cast<unsigned>(month),
1145 static_cast<unsigned>(day),
1146 };
1147}

1154inline bool is_valid_date(const Date& date) {
1155 if (date.year_ < 1 || date.year_ > 9999 || date.month_ < 1 || date.month_ > 12) {

1504 RateSelection rate_for(
1505 Country country,
1506 const VenueKey& asserted,
1507 const Date& date
1508 ) const {
1509 // A row for a market this library does not normalize can never be the answer to a question asked
1510 // in terms of `Country`, and it is kept so a caller can see that the broker sent it.
1511 const auto in_market = [&](const CommissionRate& rate) {
1512 return rate.normalized() && rate.country_ == country;
1513 };
1514 // **Each pass stops at the first row that decides it, and an unreadable row decides it too.**
1515 // Within a pass the first match has always won, so a damaged row *after* a definite match could
1516 // never have been the answer and is not a reason to refuse -- scanning the whole pass first would
1517 // throw away an answer this library knows. A damaged row reached *before* any match is the
1518 // opposite case: it may have been the match, so nothing later in the pass can be trusted to be
1519 // the winner.
1520 const auto pass = [&](bool venue_specific) -> std::optional<RateSelection> {
1521 for (const auto& rate : rates_) {
1522 if (!in_market(rate)) {
1523 continue;
1524 }
1525 if (!(venue_specific ? rate.venue_.matches(asserted) : rate.venue_.is_any())) {
1526 continue;
1527 }
1528 switch (rate.applies_on(date)) {
1529 case RateApplicability::NO:
1530 continue;
1531 case RateApplicability::YES: {
1532 RateSelection found;
1533 found.rate_ = rate;
1534 return found;
1535 }
1536 case RateApplicability::UNREADABLE: {
1537 RateSelection refused;
1538 refused.bounds_unreadable_ = true;
1539 return refused;
1540 }
1541 }
1542 }
1543 return std::nullopt;
1544 };
1545 // An undecided pass one falls through; a *refused* pass one does not. Pass two's precondition is
1546 // "no row for the asserted venue applies", and that is precisely what stopped being known.
1547 if (const auto venue_row = pass(true)) {
1548 return *venue_row;
1549 }
1550 if (const auto neutral_row = pass(false)) {
1551 return *neutral_row;
1552 }
1553 return RateSelection{};
1554 }
