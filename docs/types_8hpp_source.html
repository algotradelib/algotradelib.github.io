179 template <typename IdOf>
180 std::optional<std::size_t> find(const InstrumentId& id, const IdOf& id_of) const noexcept {
181 if (count_ == 0) {
182 return std::nullopt;
183 }
184 const std::size_t mask = slots_.size() - 1;
185 for (std::size_t at = InstrumentIdHash{}(id) & mask;; at = (at + 1) & mask) {
186 const std::size_t slot = slots_[at];
187 if (slot == 0) {
188 return std::nullopt;
189 }
190 if (id_of(slot - 1) == id) {
191 return slot - 1;
192 }
193 }
194 }

200 template <typename IdOf>
201 void insert(const InstrumentId& id, std::size_t position, const IdOf& id_of) {
202 if (2 * (count_ + 1) > slots_.size()) {
203 std::size_t size = slots_.empty() ? 32 : 2 * slots_.size();
204 while (2 * (count_ + 1) > size) {
205 size *= 2;
206 }
207 std::vector<std::size_t> grown(size, 0);
208 for (const auto slot : slots_) {
209 if (slot != 0) {
210 place(grown, InstrumentIdHash{}(id_of(slot - 1)), slot);
211 }
212 }
213 slots_.swap(grown);
214 }
215 place(slots_, InstrumentIdHash{}(id), position + 1);
216 ++count_;
217 }
218
219private:
220 static void place(std::vector<std::size_t>& slots, std::size_t hash, std::size_t slot) noexcept {
221 const std::size_t mask = slots.size() - 1;
222 std::size_t at = hash & mask;
223 while (slots[at] != 0) {
224 at = (at + 1) & mask;
225 }
226 slots[at] = slot;
227 }
228
229 // A position plus one; zero is an empty slot.
230 std::vector<std::size_t> slots_;
231 std::size_t count_{0};
232};

440 static Decimal64 parse(std::string_view input) {
441 if (input.empty()) {
442 throw std::invalid_argument("decimal value is empty");
443 }
444
445 bool negative = false;
446 std::size_t index = 0;
447 if (input.front() == '+' || input.front() == '-') {
448 negative = input.front() == '-';
449 index = 1;
450 }
451 if (index == input.size()) {
452 throw std::invalid_argument("decimal value has no digits");
453 }
454
455 std::uint64_t magnitude = 0;
456 std::uint8_t scale = 0;
457 bool decimal_point = false;
458 bool has_digit = false;
459 const std::uint64_t limit = negative ?
460 static_cast<std::uint64_t>(std::numeric_limits<std::int64_t>::max()) + 1ULL :
461 static_cast<std::uint64_t>(std::numeric_limits<std::int64_t>::max());
462
463 for (; index < input.size(); ++index) {
464 const char ch = input[index];
465 if (ch == '.') {
466 if (decimal_point) {
467 throw std::invalid_argument("decimal value contains multiple points");
468 }
469 decimal_point = true;
470 continue;
471 }
472 if (ch < '0' || ch > '9') {
473 throw std::invalid_argument("decimal value contains an invalid character");
474 }
475 has_digit = true;
476 const auto digit = static_cast<std::uint64_t>(ch - '0');
477 if (magnitude > (limit - digit) / 10ULL) {
478 throw std::out_of_range("decimal value exceeds Decimal64 range");
479 }
480 magnitude = magnitude * 10ULL + digit;
481 if (decimal_point) {
482 if (scale == std::numeric_limits<std::uint8_t>::max()) {
483 throw std::out_of_range("decimal scale exceeds Decimal64 range");
484 }
485 ++scale;
486 }
487 }
488 if (!has_digit) {
489 throw std::invalid_argument("decimal value has no digits");
490 }
491
492 while (scale > 0 && magnitude % 10ULL == 0) {
493 magnitude /= 10ULL;
494 --scale;
495 }
496
497 std::int64_t coefficient;
498 if (
499 negative &&
500 magnitude == static_cast<std::uint64_t>(std::numeric_limits<std::int64_t>::max()) + 1ULL
501 ) {
502 coefficient = std::numeric_limits<std::int64_t>::min();
503 } else {
504 coefficient = static_cast<std::int64_t>(magnitude);
505 if (negative) {
506 coefficient = -coefficient;
507 }
508 }
509 return Decimal64(coefficient, scale);
510 }
511
516 std::int64_t coefficient() const noexcept { return coefficient_; }
517
522 std::uint8_t scale() const noexcept { return scale_; }
523
528 std::string str() const {
529 const bool negative = coefficient_ < 0;
530 std::uint64_t magnitude = negative ?
531 static_cast<std::uint64_t>(-(coefficient_ + 1)) + 1ULL :
532 static_cast<std::uint64_t>(coefficient_);
533 std::string digits = std::to_string(magnitude);
534 if (scale_ > 0) {
535 if (digits.size() <= scale_) {
536 digits.insert(0, static_cast<std::size_t>(scale_) - digits.size() + 1, '0');
537 }
538 digits.insert(digits.size() - scale_, 1, '.');
539 }
540 if (negative) {
541 digits.insert(digits.begin(), '-');
542 }
543 return digits;
544 }
545
558 friend bool operator==(const Decimal64& lhs, const Decimal64& rhs) {
559 return lhs.coefficient_ == rhs.coefficient_ && lhs.scale_ == rhs.scale_;
560 }

590inline bool align(
591 const Decimal64& lhs,
592 const Decimal64& rhs,
593 std::int64_t& left,
594 std::int64_t& right,
595 std::uint8_t& scale
596) {
597 scale = lhs.scale() > rhs.scale() ? lhs.scale() : rhs.scale();
598 left = lhs.coefficient();
599 right = rhs.coefficient();
600 for (std::uint8_t step = lhs.scale(); step < scale; ++step) {
601 if (left > std::numeric_limits<std::int64_t>::max() / 10 ||
602 left < std::numeric_limits<std::int64_t>::min() / 10) {
603 return false;
604 }
605 left *= 10;
606 }
607 for (std::uint8_t step = rhs.scale(); step < scale; ++step) {
608 if (right > std::numeric_limits<std::int64_t>::max() / 10 ||
609 right < std::numeric_limits<std::int64_t>::min() / 10) {
610 return false;
611 }
612 right *= 10;
613 }
614 return true;
615}

688inline std::optional<Decimal64> decimal_multiply(const Decimal64& lhs, const Decimal64& rhs) {
689 const auto left = lhs.coefficient();
690 const auto right = rhs.coefficient();
691 const int scale = static_cast<int>(lhs.scale()) + static_cast<int>(rhs.scale());
692 if (scale > std::numeric_limits<std::uint8_t>::max()) {
693 return std::nullopt;
694 }
695 if (left != 0 && right != 0) {
696 // Checked before multiplying rather than after, because signed overflow is undefined and a
697 // wrapped price is exactly the failure this type exists to prevent.
698 if (left > 0 ? (right > 0 ? left > std::numeric_limits<std::int64_t>::max() / right
699 : right < std::numeric_limits<std::int64_t>::min() / left)
700 : (right > 0 ? left < std::numeric_limits<std::int64_t>::min() / right
701 : left < std::numeric_limits<std::int64_t>::max() / right)) {
702 return std::nullopt;
703 }
704 }
705 return decimal_normalized(Decimal64(left * right, static_cast<std::uint8_t>(scale)));
706}

719inline std::optional<Decimal64> decimal_divide(
720 const Decimal64& lhs,
721 const Decimal64& rhs,
722 std::uint8_t scale
723) {
724 std::int64_t left = 0;
725 std::int64_t right = 0;
726 std::uint8_t common = 0;
727 if (!decimal_detail::align(lhs, rhs, left, right, common) || right == 0) {
728 return std::nullopt;
729 }
730 // Aligning cancels the scales, so `left / right` is the plain ratio. Each requested decimal
731 // place is one more factor of ten taken before the division.
732 for (std::uint8_t step = 0; step < scale; ++step) {
733 if (left > std::numeric_limits<std::int64_t>::max() / 10 ||
734 left < std::numeric_limits<std::int64_t>::min() / 10) {
735 return std::nullopt;
736 }
737 left *= 10;
738 }
739 if (left == std::numeric_limits<std::int64_t>::min() && right == -1) {
740 return std::nullopt;
741 }
742 return decimal_normalized(Decimal64(left / right, scale));
743}

756inline std::optional<Decimal64> decimal_floor_to_multiple(
757 const Decimal64& value,
758 const Decimal64& step
759) {
760 std::int64_t left = 0;
761 std::int64_t right = 0;
762 std::uint8_t scale = 0;
763 if (!decimal_detail::align(value, step, left, right, scale) || right <= 0) {
764 return std::nullopt;
765 }
766 auto rest = left % right;
767 if (rest < 0) {
768 rest += right;
769 }
770 if (left < std::numeric_limits<std::int64_t>::min() + rest) {
771 return std::nullopt;
772 }
773 return decimal_normalized(Decimal64(left - rest, scale));
774}

782inline bool decimal_less(const Decimal64& lhs, const Decimal64& rhs) {
783 std::int64_t left = 0;
784 std::int64_t right = 0;
785 std::uint8_t scale = 0;
786 if (decimal_detail::align(lhs, rhs, left, right, scale)) {
787 return left < right;
788 }
789 // Rescaling overflowed, so the values differ by more than any shared scale can
790 // express: the one with the smaller scale has the larger magnitude. Between two
791 // negatives the larger magnitude is the smaller value.
792 const bool left_negative = lhs.coefficient() < 0;
793 if (left_negative != (rhs.coefficient() < 0)) {
794 return left_negative;
795 }
796 return left_negative ? lhs.scale() < rhs.scale() : lhs.scale() > rhs.scale();
797}

1303inline Date date_from_epoch_days(std::int64_t days) {
1304 // Shifts the era so that the year starts in March, which is what removes the leap-day special
1305 // case from everything below: 1970-01-01 is 719468 days after 0000-03-01.
1306 const std::int64_t shifted = days + 719468;
1307 const std::int64_t era = (shifted >= 0 ? shifted : shifted - 146096) / 146097;
1308 const std::int64_t day_of_era = shifted - era * 146097;
1309 const std::int64_t year_of_era =
1310 (day_of_era - day_of_era / 1460 + day_of_era / 36524 - day_of_era / 146096) / 365;
1311 const std::int64_t year = year_of_era + era * 400;
1312 const std::int64_t day_of_year =
1313 day_of_era - (365 * year_of_era + year_of_era / 4 - year_of_era / 100);
1314 const std::int64_t shifted_month = (5 * day_of_year + 2) / 153;
1315 const std::int64_t day = day_of_year - (153 * shifted_month + 2) / 5 + 1;
1316 const std::int64_t month = shifted_month + (shifted_month < 10 ? 3 : -9);
1317 return Date{
1318 static_cast<int>(year + (month <= 2 ? 1 : 0)),
1319 static_cast<unsigned>(month),
1320 static_cast<unsigned>(day),
1321 };
1322}

1329inline bool is_valid_date(const Date& date) {
1330 if (date.year_ < 1 || date.year_ > 9999 || date.month_ < 1 || date.month_ > 12) {
1331 return false;
1332 }
1333 static constexpr unsigned days_per_month[] = {
1334 31, 28, 31, 30, 31, 30, 31, 31, 30, 31, 30, 31,
1335 };
1336 unsigned days = days_per_month[date.month_ - 1];
1337 const bool leap = date.year_ % 4 == 0 && (date.year_ % 100 != 0 || date.year_ % 400 == 0);
1338 if (date.month_ == 2 && leap) {
1339 ++days;
1340 }
1341 return date.day_ >= 1 && date.day_ <= days;
1342}

1709 RateSelection rate_for(
1710 Country country,
1711 const VenueKey& asserted,
1712 const Date& date
1713 ) const {
1714 // A row for a market this library does not normalize can never be the answer to a question asked
1715 // in terms of `Country`, and it is kept so a caller can see that the broker sent it.
1716 const auto in_market = [&](const CommissionRate& rate) {
1717 return rate.normalized() && rate.country_ == country;
1718 };
1719 // **Each pass stops at the first row that decides it, and an unreadable row decides it too.**
1720 // Within a pass the first match has always won, so a damaged row *after* a definite match could
1721 // never have been the answer and is not a reason to refuse -- scanning the whole pass first would
1722 // throw away an answer this library knows. A damaged row reached *before* any match is the
1723 // opposite case: it may have been the match, so nothing later in the pass can be trusted to be
1724 // the winner.
1725 const auto pass = [&](bool venue_specific) -> std::optional<RateSelection> {
1726 for (const auto& rate : rates_) {
1727 if (!in_market(rate)) {
1728 continue;
1729 }
1730 if (!(venue_specific ? rate.venue_.matches(asserted) : rate.venue_.is_any())) {
1731 continue;
1732 }
1733 switch (rate.applies_on(date)) {
1734 case RateApplicability::NO:
1735 continue;
1736 case RateApplicability::YES: {
1737 RateSelection found;
1738 found.rate_ = rate;
1739 return found;
1740 }
1741 case RateApplicability::UNREADABLE: {
1742 RateSelection refused;
1743 refused.bounds_unreadable_ = true;
1744 return refused;
1745 }
1746 }
1747 }
1748 return std::nullopt;
1749 };
1750 // An undecided pass one falls through; a *refused* pass one does not. Pass two's precondition is
1751 // "no row for the asserted venue applies", and that is precisely what stopped being known.
1752 if (const auto venue_row = pass(true)) {
1753 return *venue_row;
1754 }
1755 if (const auto neutral_row = pass(false)) {
1756 return *neutral_row;
1757 }
1758 return RateSelection{};
1759 }
1760
1766 bool any_dated_rate_for(Country country) const {
1767 for (const auto& rate : rates_) {
1768 // A bound that could not be read is still a bound the source stated. Asking only whether the
1769 // optionals are filled would read a row dated *solely* by damage as undated, and hand the
1770 // caller the very "every candidate is the same whichever calendar named the day" assurance
1771 // that the damage makes impossible.
1772 const bool dated = rate.from_ || rate.until_ || rate.from_unreadable_ || rate.until_unreadable_;
1773 if (rate.normalized() && rate.country_ == country && dated) {
1774 return true;
1775 }
1776 }
1777 return false;
1778 }

1790 return rate_for(country, VenueKey::any(), date);
