323 static Decimal64 parse(std::string_view input) {
324 if (input.empty()) {
325 throw std::invalid_argument("decimal value is empty");
326 }
327
328 bool negative = false;
329 std::size_t index = 0;
330 if (input.front() == '+' || input.front() == '-') {
331 negative = input.front() == '-';
332 index = 1;
333 }
334 if (index == input.size()) {
335 throw std::invalid_argument("decimal value has no digits");
336 }
337
338 std::uint64_t magnitude = 0;
339 std::uint8_t scale = 0;
340 bool decimal_point = false;
341 bool has_digit = false;
342 const std::uint64_t limit = negative ?
343 static_cast<std::uint64_t>(std::numeric_limits<std::int64_t>::max()) + 1ULL :
344 static_cast<std::uint64_t>(std::numeric_limits<std::int64_t>::max());
345
346 for (; index < input.size(); ++index) {
347 const char ch = input[index];
348 if (ch == '.') {
349 if (decimal_point) {
350 throw std::invalid_argument("decimal value contains multiple points");
351 }
352 decimal_point = true;
353 continue;
354 }
355 if (ch < '0' || ch > '9') {
356 throw std::invalid_argument("decimal value contains an invalid character");
357 }
358 has_digit = true;
359 const auto digit = static_cast<std::uint64_t>(ch - '0');
360 if (magnitude > (limit - digit) / 10ULL) {
361 throw std::out_of_range("decimal value exceeds Decimal64 range");
362 }
363 magnitude = magnitude * 10ULL + digit;
364 if (decimal_point) {
365 if (scale == std::numeric_limits<std::uint8_t>::max()) {
366 throw std::out_of_range("decimal scale exceeds Decimal64 range");
367 }
368 ++scale;
369 }
370 }
371 if (!has_digit) {
372 throw std::invalid_argument("decimal value has no digits");
373 }
374
375 while (scale > 0 && magnitude % 10ULL == 0) {
376 magnitude /= 10ULL;
377 --scale;
378 }
379
380 std::int64_t coefficient;
381 if (
382 negative &&
383 magnitude == static_cast<std::uint64_t>(std::numeric_limits<std::int64_t>::max()) + 1ULL
384 ) {
385 coefficient = std::numeric_limits<std::int64_t>::min();
386 } else {
387 coefficient = static_cast<std::int64_t>(magnitude);
388 if (negative) {
389 coefficient = -coefficient;
390 }
391 }
392 return Decimal64(coefficient, scale);
393 }
394
399 std::int64_t coefficient() const noexcept { return coefficient_; }
400
405 std::uint8_t scale() const noexcept { return scale_; }
406
411 std::string str() const {
412 const bool negative = coefficient_ < 0;
413 std::uint64_t magnitude = negative ?
414 static_cast<std::uint64_t>(-(coefficient_ + 1)) + 1ULL :
415 static_cast<std::uint64_t>(coefficient_);
416 std::string digits = std::to_string(magnitude);
417 if (scale_ > 0) {
418 if (digits.size() <= scale_) {
419 digits.insert(0, static_cast<std::size_t>(scale_) - digits.size() + 1, '0');
420 }
421 digits.insert(digits.size() - scale_, 1, '.');
422 }
423 if (negative) {
424 digits.insert(digits.begin(), '-');
425 }
426 return digits;
427 }

473inline bool align(
474 const Decimal64& lhs,
475 const Decimal64& rhs,
476 std::int64_t& left,
477 std::int64_t& right,
478 std::uint8_t& scale
479) {
480 scale = lhs.scale() > rhs.scale() ? lhs.scale() : rhs.scale();
481 left = lhs.coefficient();
482 right = rhs.coefficient();
483 for (std::uint8_t step = lhs.scale(); step < scale; ++step) {
484 if (left > std::numeric_limits<std::int64_t>::max() / 10 ||
485 left < std::numeric_limits<std::int64_t>::min() / 10) {
486 return false;
487 }
488 left *= 10;
489 }
490 for (std::uint8_t step = rhs.scale(); step < scale; ++step) {
491 if (right > std::numeric_limits<std::int64_t>::max() / 10 ||
492 right < std::numeric_limits<std::int64_t>::min() / 10) {
493 return false;
494 }
495 right *= 10;
496 }
497 return true;
498}

571inline std::optional<Decimal64> decimal_multiply(const Decimal64& lhs, const Decimal64& rhs) {
572 const auto left = lhs.coefficient();
573 const auto right = rhs.coefficient();
574 const int scale = static_cast<int>(lhs.scale()) + static_cast<int>(rhs.scale());
575 if (scale > std::numeric_limits<std::uint8_t>::max()) {
576 return std::nullopt;
577 }
578 if (left != 0 && right != 0) {
579 // Checked before multiplying rather than after, because signed overflow is undefined and a
580 // wrapped price is exactly the failure this type exists to prevent.
581 if (left > 0 ? (right > 0 ? left > std::numeric_limits<std::int64_t>::max() / right
582 : right < std::numeric_limits<std::int64_t>::min() / left)
583 : (right > 0 ? left < std::numeric_limits<std::int64_t>::min() / right
584 : left < std::numeric_limits<std::int64_t>::max() / right)) {
585 return std::nullopt;
586 }
587 }
588 return decimal_normalized(Decimal64(left * right, static_cast<std::uint8_t>(scale)));
589}

602inline std::optional<Decimal64> decimal_divide(
603 const Decimal64& lhs,
604 const Decimal64& rhs,
605 std::uint8_t scale
606) {
607 std::int64_t left = 0;
608 std::int64_t right = 0;
609 std::uint8_t common = 0;
610 if (!decimal_detail::align(lhs, rhs, left, right, common) || right == 0) {
611 return std::nullopt;
612 }
613 // Aligning cancels the scales, so `left / right` is the plain ratio. Each requested decimal
614 // place is one more factor of ten taken before the division.
615 for (std::uint8_t step = 0; step < scale; ++step) {
616 if (left > std::numeric_limits<std::int64_t>::max() / 10 ||
617 left < std::numeric_limits<std::int64_t>::min() / 10) {
618 return std::nullopt;
619 }
620 left *= 10;
621 }
622 if (left == std::numeric_limits<std::int64_t>::min() && right == -1) {
623 return std::nullopt;
624 }
625 return decimal_normalized(Decimal64(left / right, scale));
626}

639inline std::optional<Decimal64> decimal_floor_to_multiple(
640 const Decimal64& value,
641 const Decimal64& step
642) {
643 std::int64_t left = 0;
644 std::int64_t right = 0;
645 std::uint8_t scale = 0;
646 if (!decimal_detail::align(value, step, left, right, scale) || right <= 0) {
647 return std::nullopt;
648 }
649 auto rest = left % right;
650 if (rest < 0) {
651 rest += right;
652 }
653 if (left < std::numeric_limits<std::int64_t>::min() + rest) {
654 return std::nullopt;
655 }
656 return decimal_normalized(Decimal64(left - rest, scale));
657}

665inline bool decimal_less(const Decimal64& lhs, const Decimal64& rhs) {
666 std::int64_t left = 0;
667 std::int64_t right = 0;
668 std::uint8_t scale = 0;
669 if (decimal_detail::align(lhs, rhs, left, right, scale)) {
670 return left < right;
671 }
672 // Rescaling overflowed, so the values differ by more than any shared scale can
673 // express: the one with the smaller scale has the larger magnitude. Between two
674 // negatives the larger magnitude is the smaller value.
675 const bool left_negative = lhs.coefficient() < 0;
676 if (left_negative != (rhs.coefficient() < 0)) {
677 return left_negative;
678 }
679 return left_negative ? lhs.scale() < rhs.scale() : lhs.scale() > rhs.scale();
680}

1186inline Date date_from_epoch_days(std::int64_t days) {
1187 // Shifts the era so that the year starts in March, which is what removes the leap-day special
1188 // case from everything below: 1970-01-01 is 719468 days after 0000-03-01.
1189 const std::int64_t shifted = days + 719468;
1190 const std::int64_t era = (shifted >= 0 ? shifted : shifted - 146096) / 146097;
1191 const std::int64_t day_of_era = shifted - era * 146097;
1192 const std::int64_t year_of_era =
1193 (day_of_era - day_of_era / 1460 + day_of_era / 36524 - day_of_era / 146096) / 365;
1194 const std::int64_t year = year_of_era + era * 400;
1195 const std::int64_t day_of_year =
1196 day_of_era - (365 * year_of_era + year_of_era / 4 - year_of_era / 100);
1197 const std::int64_t shifted_month = (5 * day_of_year + 2) / 153;
1198 const std::int64_t day = day_of_year - (153 * shifted_month + 2) / 5 + 1;
1199 const std::int64_t month = shifted_month + (shifted_month < 10 ? 3 : -9);
1200 return Date{
1201 static_cast<int>(year + (month <= 2 ? 1 : 0)),
1202 static_cast<unsigned>(month),
1203 static_cast<unsigned>(day),
1204 };
1205}

1212inline bool is_valid_date(const Date& date) {
1213 if (date.year_ < 1 || date.year_ > 9999 || date.month_ < 1 || date.month_ > 12) {
1214 return false;
1215 }
1216 static constexpr unsigned days_per_month[] = {
1217 31, 28, 31, 30, 31, 30, 31, 31, 30, 31, 30, 31,
1218 };
1219 unsigned days = days_per_month[date.month_ - 1];
1220 const bool leap = date.year_ % 4 == 0 && (date.year_ % 100 != 0 || date.year_ % 400 == 0);
1221 if (date.month_ == 2 && leap) {
1222 ++days;
1223 }
1224 return date.day_ >= 1 && date.day_ <= days;
1225}

1592 RateSelection rate_for(
1593 Country country,
1594 const VenueKey& asserted,
1595 const Date& date
1596 ) const {
1597 // A row for a market this library does not normalize can never be the answer to a question asked
1598 // in terms of `Country`, and it is kept so a caller can see that the broker sent it.
1599 const auto in_market = [&](const CommissionRate& rate) {
1600 return rate.normalized() && rate.country_ == country;
1601 };
1602 // **Each pass stops at the first row that decides it, and an unreadable row decides it too.**
1603 // Within a pass the first match has always won, so a damaged row *after* a definite match could
1604 // never have been the answer and is not a reason to refuse -- scanning the whole pass first would
1605 // throw away an answer this library knows. A damaged row reached *before* any match is the
1606 // opposite case: it may have been the match, so nothing later in the pass can be trusted to be
1607 // the winner.
1608 const auto pass = [&](bool venue_specific) -> std::optional<RateSelection> {
1609 for (const auto& rate : rates_) {
1610 if (!in_market(rate)) {
1611 continue;
1612 }
1613 if (!(venue_specific ? rate.venue_.matches(asserted) : rate.venue_.is_any())) {
1614 continue;
1615 }
1616 switch (rate.applies_on(date)) {
1617 case RateApplicability::NO:
1618 continue;
1619 case RateApplicability::YES: {
1620 RateSelection found;
1621 found.rate_ = rate;
1622 return found;
1623 }
1624 case RateApplicability::UNREADABLE: {
1625 RateSelection refused;
1626 refused.bounds_unreadable_ = true;
1627 return refused;
1628 }
1629 }
1630 }
1631 return std::nullopt;
1632 };
1633 // An undecided pass one falls through; a *refused* pass one does not. Pass two's precondition is
1634 // "no row for the asserted venue applies", and that is precisely what stopped being known.
1635 if (const auto venue_row = pass(true)) {
1636 return *venue_row;
1637 }
1638 if (const auto neutral_row = pass(false)) {
1639 return *neutral_row;
1640 }
1641 return RateSelection{};
1642 }
1643
1649 bool any_dated_rate_for(Country country) const {
1650 for (const auto& rate : rates_) {
1651 // A bound that could not be read is still a bound the source stated. Asking only whether the
1652 // optionals are filled would read a row dated *solely* by damage as undated, and hand the
1653 // caller the very "every candidate is the same whichever calendar named the day" assurance
1654 // that the damage makes impossible.
1655 const bool dated = rate.from_ || rate.until_ || rate.from_unreadable_ || rate.until_unreadable_;
1656 if (rate.normalized() && rate.country_ == country && dated) {
1657 return true;
1658 }
1659 }
1660 return false;
1661 }

1673 return rate_for(country, VenueKey::any(), date);

2478inline std::optional<std::chrono::system_clock::time_point> parse_iso8601_timestamp(
2479 std::string_view value
2480) {
2481 // The fixed layout first: `sscanf`'s `%d` would accept signs, spaces and short fields.
2482 const auto digits = [value](std::size_t from, std::size_t count) {
2483 for (std::size_t index = from; index < from + count; ++index) {
2484 if (value[index] < '0' || value[index] > '9') {
2485 return false;
2486 }
2487 }
2488 return true;
2489 };
2490 if (value.size() < 20 || !digits(0, 4) || value[4] != '-' || !digits(5, 2) || value[7] != '-' ||
2491 !digits(8, 2) || value[10] != 'T' || !digits(11, 2) || value[13] != ':' || !digits(14, 2) ||
2492 value[16] != ':' || !digits(17, 2)) {
2493 return std::nullopt;
2494 }
2495 int year, month, day, hour, minute, second;
2496 if (
2497 std::sscanf(
2498 std::string(value).c_str(), "%d-%d-%dT%d:%d:%d", &year, &month, &day, &hour, &minute, &second
2499 ) != 6
2500 ) {
2501 return std::nullopt;
2502 }
2503 if (!is_valid_date(Date{year, static_cast<unsigned>(month), static_cast<unsigned>(day)}) ||
2504 hour < 0 || hour > 23 || minute < 0 || minute > 59 || second < 0 || second > 60) {
2505 return std::nullopt;
2506 }
2507
2508 // An optional fraction of digits only, then the zone, then nothing.
2509 std::size_t timezone_position = 19;
2510 if (value[19] == '.') {
2511 timezone_position = 20;
2512 while (timezone_position < value.size() && value[timezone_position] >= '0' &&
2513 value[timezone_position] <= '9') {
2514 ++timezone_position;
2515 }
2516 if (timezone_position == 20) {
2517 return std::nullopt;
2518 }
2519 }
2520 if (timezone_position >= value.size()) {
2521 return std::nullopt;
2522 }
2523 const char zone = value[timezone_position];
2524 if (zone == 'Z') {
2525 if (timezone_position + 1 != value.size()) {
2526 return std::nullopt;
2527 }
2528 } else if (zone == '+' || zone == '-') {
2529 if (value.size() != timezone_position + 6 || !digits(timezone_position + 1, 2) ||
2530 value[timezone_position + 3] != ':' || !digits(timezone_position + 4, 2)) {
2531 return std::nullopt;
2532 }
2533 } else {
2534 return std::nullopt;
2535 }
2536
2537 std::tm time{};
2538 time.tm_year = year - 1900;
2539 time.tm_mon = month - 1;
2540 time.tm_mday = day;
2541 time.tm_hour = hour;
2542 time.tm_min = minute;
2543 time.tm_sec = second;
2544 time.tm_isdst = 0;
2545#ifdef _WIN32
2546 std::time_t utc = _mkgmtime(&time);
2547#else
2548 std::time_t utc = timegm(&time);
2549#endif
2550 if (utc == static_cast<std::time_t>(-1)) {
2551 return std::nullopt;
2552 }
2553
2554 if (value[timezone_position] != 'Z') {
2555 int offset_hour, offset_minute;
2556 if (
2557 std::sscanf(
2558 std::string(value.substr(timezone_position + 1)).c_str(),
2559 "%d:%d",
2560 &offset_hour,
2561 &offset_minute
2562 ) != 2
2563 ) {
2564 return std::nullopt;
2565 }
2566 if (offset_hour > 23 || offset_minute > 59) {
2567 return std::nullopt;
2568 }
2569 const int offset = offset_hour * 3600 + offset_minute * 60;
2570 utc += value[timezone_position] == '+' ? -offset : offset;
