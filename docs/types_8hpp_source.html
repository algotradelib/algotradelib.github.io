188 template <typename IdOf>
189 std::optional<std::size_t> find(const InstrumentId& id, const IdOf& id_of) const noexcept {
190 if (count_ == 0) {
191 return std::nullopt;
192 }
193 const std::size_t mask = slots_.size() - 1;
194 for (std::size_t at = InstrumentIdHash{}(id)&mask;; at = (at + 1) & mask) {
195 const std::size_t slot = slots_[at];
196 if (slot == 0) {
197 return std::nullopt;
198 }
199 if (id_of(slot - 1) == id) {
200 return slot - 1;
201 }

209 template <typename IdOf>
210 void insert(const InstrumentId& id, std::size_t position, const IdOf& id_of) {
211 if (2 * (count_ + 1) > slots_.size()) {
212 std::size_t size = slots_.empty() ? 32 : 2 * slots_.size();
213 while (2 * (count_ + 1) > size) {
214 size *= 2;
215 }
216 std::vector<std::size_t> grown(size, 0);
217 for (const auto slot : slots_) {
218 if (slot != 0) {
219 place(grown, InstrumentIdHash{}(id_of(slot - 1)), slot);
220 }
221 }
222 slots_.swap(grown);
223 }
224 place(slots_, InstrumentIdHash{}(id), position + 1);
225 ++count_;
226 }
227
228private:
229 static void place(std::vector<std::size_t>& slots, std::size_t hash, std::size_t slot) noexcept {
230 const std::size_t mask = slots.size() - 1;
231 std::size_t at = hash & mask;
232 while (slots[at] != 0) {
233 at = (at + 1) & mask;
234 }
235 slots[at] = slot;
236 }
237
238 // A position plus one; zero is an empty slot.
239 std::vector<std::size_t> slots_;
240 std::size_t count_{0};

466 static Decimal64 parse(std::string_view input) {
467 if (input.empty()) {
468 throw std::invalid_argument("decimal value is empty");
469 }
470
471 bool negative = false;
472 std::size_t index = 0;
473 if (input.front() == '+' || input.front() == '-') {
474 negative = input.front() == '-';
475 index = 1;
476 }
477 if (index == input.size()) {
478 throw std::invalid_argument("decimal value has no digits");
479 }
480
481 std::uint64_t magnitude = 0;
482 std::uint8_t scale = 0;
483 bool decimal_point = false;
484 bool has_digit = false;
485 const std::uint64_t limit = negative ?
486 static_cast<std::uint64_t>(std::numeric_limits<std::int64_t>::max()) + 1ULL :
487 static_cast<std::uint64_t>(std::numeric_limits<std::int64_t>::max());
488
489 for (; index < input.size(); ++index) {
490 const char ch = input[index];
491 if (ch == '.') {
492 if (decimal_point) {
493 throw std::invalid_argument("decimal value contains multiple points");
494 }
495 decimal_point = true;
496 continue;
497 }
498 if (ch < '0' || ch > '9') {
499 throw std::invalid_argument("decimal value contains an invalid character");
500 }
501 has_digit = true;
502 const auto digit = static_cast<std::uint64_t>(ch - '0');
503 if (magnitude > (limit - digit) / 10ULL) {
504 throw std::out_of_range("decimal value exceeds Decimal64 range");
505 }
506 magnitude = magnitude * 10ULL + digit;
507 if (decimal_point) {
508 if (scale == std::numeric_limits<std::uint8_t>::max()) {
509 throw std::out_of_range("decimal scale exceeds Decimal64 range");
510 }
511 ++scale;
512 }
513 }
514 if (!has_digit) {
515 throw std::invalid_argument("decimal value has no digits");
516 }
517
518 while (scale > 0 && magnitude % 10ULL == 0) {
519 magnitude /= 10ULL;
520 --scale;
521 }
522
523 std::int64_t coefficient;
524 if (negative && magnitude == static_cast<std::uint64_t>(std::numeric_limits<std::int64_t>::max()) + 1ULL) {
525 coefficient = std::numeric_limits<std::int64_t>::min();
526 } else {
527 coefficient = static_cast<std::int64_t>(magnitude);
528 if (negative) {
529 coefficient = -coefficient;
530 }
531 }
532 return Decimal64(coefficient, scale);
533 }
534
539 std::int64_t coefficient() const noexcept {
540 return coefficient_;
541 }
542
547 std::uint8_t scale() const noexcept {
548 return scale_;
549 }
550
555 std::string str() const {
556 const bool negative = coefficient_ < 0;
557 std::uint64_t magnitude =
558 negative ? static_cast<std::uint64_t>(-(coefficient_ + 1)) + 1ULL : static_cast<std::uint64_t>(coefficient_);
559 std::string digits = std::to_string(magnitude);
560 if (scale_ > 0) {
561 if (digits.size() <= scale_) {
562 digits.insert(0, static_cast<std::size_t>(scale_) - digits.size() + 1, '0');
563 }
564 digits.insert(digits.size() - scale_, 1, '.');
565 }
566 if (negative) {
567 digits.insert(digits.begin(), '-');
568 }
569 return digits;
570 }

616inline bool align(
617 const Decimal64& lhs, const Decimal64& rhs, std::int64_t& left, std::int64_t& right, std::uint8_t& scale
618) {
619 scale = lhs.scale() > rhs.scale() ? lhs.scale() : rhs.scale();
620 left = lhs.coefficient();
621 right = rhs.coefficient();
622 for (std::uint8_t step = lhs.scale(); step < scale; ++step) {
623 if (left > std::numeric_limits<std::int64_t>::max() / 10 || left < std::numeric_limits<std::int64_t>::min() / 10) {
624 return false;
625 }
626 left *= 10;
627 }
628 for (std::uint8_t step = rhs.scale(); step < scale; ++step) {
629 if (
630 right > std::numeric_limits<std::int64_t>::max() / 10 || right < std::numeric_limits<std::int64_t>::min() / 10
631 ) {
632 return false;
633 }
634 right *= 10;
635 }
636 return true;
637}

714inline std::optional<Decimal64> decimal_multiply(const Decimal64& lhs, const Decimal64& rhs) {
715 const auto left = lhs.coefficient();
716 const auto right = rhs.coefficient();
717 const int scale = static_cast<int>(lhs.scale()) + static_cast<int>(rhs.scale());
718 if (scale > std::numeric_limits<std::uint8_t>::max()) {
719 return std::nullopt;
720 }
721 if (left != 0 && right != 0) {
722 // Checked before multiplying rather than after, because signed overflow is undefined and a
723 // wrapped price is exactly the failure this type exists to prevent.
724 constexpr auto largest = std::numeric_limits<std::int64_t>::max();
725 constexpr auto smallest = std::numeric_limits<std::int64_t>::min();
726 if (
727 left > 0 ?
728 (right > 0 ? left > largest / right : right < smallest / left) :
729 (right > 0 ? left < smallest / right : left < largest / right)
730 ) {
731 return std::nullopt;
732 }
733 }
734 return decimal_normalized(Decimal64(left * right, static_cast<std::uint8_t>(scale)));
735}

748inline std::optional<Decimal64> decimal_divide(const Decimal64& lhs, const Decimal64& rhs, std::uint8_t scale) {
749 std::int64_t left = 0;
750 std::int64_t right = 0;
751 std::uint8_t common = 0;
752 if (!decimal_detail::align(lhs, rhs, left, right, common) || right == 0) {
753 return std::nullopt;
754 }
755 // Aligning cancels the scales, so `left / right` is the plain ratio. Each requested decimal
756 // place is one more factor of ten taken before the division.
757 for (std::uint8_t step = 0; step < scale; ++step) {
758 if (left > std::numeric_limits<std::int64_t>::max() / 10 || left < std::numeric_limits<std::int64_t>::min() / 10) {
759 return std::nullopt;
760 }
761 left *= 10;
762 }
763 if (left == std::numeric_limits<std::int64_t>::min() && right == -1) {
764 return std::nullopt;
765 }
766 return decimal_normalized(Decimal64(left / right, scale));
767}

780inline std::optional<Decimal64> decimal_floor_to_multiple(const Decimal64& value, const Decimal64& step) {
781 std::int64_t left = 0;
782 std::int64_t right = 0;
783 std::uint8_t scale = 0;
784 if (!decimal_detail::align(value, step, left, right, scale) || right <= 0) {
785 return std::nullopt;
786 }
787 auto rest = left % right;
788 if (rest < 0) {
789 rest += right;
790 }
791 if (left < std::numeric_limits<std::int64_t>::min() + rest) {
792 return std::nullopt;
793 }
794 return decimal_normalized(Decimal64(left - rest, scale));
795}

803inline bool decimal_less(const Decimal64& lhs, const Decimal64& rhs) {
804 std::int64_t left = 0;
805 std::int64_t right = 0;
806 std::uint8_t scale = 0;
807 if (decimal_detail::align(lhs, rhs, left, right, scale)) {
808 return left < right;
809 }
810 // Rescaling overflowed, so the values differ by more than any shared scale can
811 // express: the one with the smaller scale has the larger magnitude. Between two
812 // negatives the larger magnitude is the smaller value.
813 const bool left_negative = lhs.coefficient() < 0;
814 if (left_negative != (rhs.coefficient() < 0)) {
815 return left_negative;
816 }
817 return left_negative ? lhs.scale() < rhs.scale() : lhs.scale() > rhs.scale();
818}

1333inline Date date_from_epoch_days(std::int64_t days) {
1334 // Shifts the era so that the year starts in March, which is what removes the leap-day special
1335 // case from everything below: 1970-01-01 is 719468 days after 0000-03-01.
1336 const std::int64_t shifted = days + 719468;
1337 const std::int64_t era = (shifted >= 0 ? shifted : shifted - 146096) / 146097;
1338 const std::int64_t day_of_era = shifted - era * 146097;
1339 const std::int64_t year_of_era = (day_of_era - day_of_era / 1460 + day_of_era / 36524 - day_of_era / 146096) / 365;
1340 const std::int64_t year = year_of_era + era * 400;
1341 const std::int64_t day_of_year = day_of_era - (365 * year_of_era + year_of_era / 4 - year_of_era / 100);
1342 const std::int64_t shifted_month = (5 * day_of_year + 2) / 153;
1343 const std::int64_t day = day_of_year - (153 * shifted_month + 2) / 5 + 1;
1344 const std::int64_t month = shifted_month + (shifted_month < 10 ? 3 : -9);
1345 return Date{
1346 static_cast<int>(year + (month <= 2 ? 1 : 0)),
1347 static_cast<unsigned>(month),
1348 static_cast<unsigned>(day),
1349 };
1350}

1764 RateSelection rate_for(Country country, const VenueKey& asserted, const Date& date) const {
1765 // A row for a market this library does not normalize can never be the answer to a question asked
1766 // in terms of `Country`, and it is kept so a caller can see that the broker sent it.
1767 const auto in_market = [&](const CommissionRate& rate) {
1768 return rate.normalized() && rate.country_ == country;
1769 };
1770 // **Each pass stops at the first row that decides it, and an unreadable row decides it too.**
1771 // Within a pass the first match has always won, so a damaged row *after* a definite match could
1772 // never have been the answer and is not a reason to refuse -- scanning the whole pass first would
1773 // throw away an answer this library knows. A damaged row reached *before* any match is the
1774 // opposite case: it may have been the match, so nothing later in the pass can be trusted to be
1775 // the winner.
1776 const auto pass = [&](bool venue_specific) -> std::optional<RateSelection> {
1777 for (const auto& rate : rates_) {
1778 if (!in_market(rate)) {
1779 continue;
1780 }
1781 if (!(venue_specific ? rate.venue_.matches(asserted) : rate.venue_.is_any())) {
1782 continue;
1783 }
1784 switch (rate.applies_on(date)) {
1785 case RateApplicability::NO:
1786 continue;
1787 case RateApplicability::YES: {
1788 RateSelection found;
1789 found.rate_ = rate;
1790 return found;
1791 }
1792 case RateApplicability::UNREADABLE: {
1793 RateSelection refused;
1794 refused.bounds_unreadable_ = true;
1795 return refused;
1796 }
1797 }
1798 }
1799 return std::nullopt;
1800 };
1801 // An undecided pass one falls through; a *refused* pass one does not. Pass two's precondition is
1802 // "no row for the asserted venue applies", and that is precisely what stopped being known.
1803 if (const auto venue_row = pass(true)) {
1804 return *venue_row;
1805 }
1806 if (const auto neutral_row = pass(false)) {
1807 return *neutral_row;
1808 }
1809 return RateSelection{};
1810 }
1811
1817 bool any_dated_rate_for(Country country) const {
1818 for (const auto& rate : rates_) {
1819 // A bound that could not be read is still a bound the source stated. Asking only whether the
1820 // optionals are filled would read a row dated *solely* by damage as undated, and hand the
1821 // caller the very "every candidate is the same whichever calendar named the day" assurance
1822 // that the damage makes impossible.
1823 const bool dated = rate.from_ || rate.until_ || rate.from_unreadable_ || rate.until_unreadable_;
1824 if (rate.normalized() && rate.country_ == country && dated) {
1825 return true;
1826 }
1827 }
1828 return false;
1829 }

1841 return rate_for(country, VenueKey::any(), date);

2668 }

2679 }
2680 }

2682 return std::nullopt;
2683 }

2688 }
2689 } else if (zone == '+' || zone == '-') {
2690 if (
2691 value.size() != timezone_position + 6 || !digits(timezone_position + 1, 2) ||
2692 value[timezone_position + 3] != ':' || !digits(timezone_position + 4, 2)
2693 ) {
2694 return std::nullopt;
2695 }

2705 time.tm_min = minute;
2706 time.tm_sec = second;

2710#else

2715 }

2742 }
