182 template <typename IdOf>
183 std::optional<std::size_t> find(const InstrumentId& id, const IdOf& id_of) const noexcept {
184 if (count_ == 0) {
185 return std::nullopt;
186 }
187 const std::size_t mask = slots_.size() - 1;
188 for (std::size_t at = InstrumentIdHash{}(id) & mask;; at = (at + 1) & mask) {
189 const std::size_t slot = slots_[at];
190 if (slot == 0) {
191 return std::nullopt;
192 }
193 if (id_of(slot - 1) == id) {
194 return slot - 1;
195 }

203 template <typename IdOf>
204 void insert(const InstrumentId& id, std::size_t position, const IdOf& id_of) {
205 if (2 * (count_ + 1) > slots_.size()) {
206 std::size_t size = slots_.empty() ? 32 : 2 * slots_.size();
207 while (2 * (count_ + 1) > size) {
208 size *= 2;
209 }
210 std::vector<std::size_t> grown(size, 0);
211 for (const auto slot : slots_) {
212 if (slot != 0) {
213 place(grown, InstrumentIdHash{}(id_of(slot - 1)), slot);
214 }
215 }
216 slots_.swap(grown);
217 }
218 place(slots_, InstrumentIdHash{}(id), position + 1);
219 ++count_;
220 }
221
222private:
223 static void place(std::vector<std::size_t>& slots, std::size_t hash, std::size_t slot) noexcept {
224 const std::size_t mask = slots.size() - 1;
225 std::size_t at = hash & mask;
226 while (slots[at] != 0) {
227 at = (at + 1) & mask;
228 }
229 slots[at] = slot;
230 }
231
232 // A position plus one; zero is an empty slot.
233 std::vector<std::size_t> slots_;
234 std::size_t count_{0};

443 static Decimal64 parse(std::string_view input) {
444 if (input.empty()) {
445 throw std::invalid_argument("decimal value is empty");
446 }
447
448 bool negative = false;
449 std::size_t index = 0;
450 if (input.front() == '+' || input.front() == '-') {
451 negative = input.front() == '-';
452 index = 1;
453 }
454 if (index == input.size()) {
455 throw std::invalid_argument("decimal value has no digits");
456 }
457
458 std::uint64_t magnitude = 0;
459 std::uint8_t scale = 0;
460 bool decimal_point = false;
461 bool has_digit = false;
462 const std::uint64_t limit = negative ?
463 static_cast<std::uint64_t>(std::numeric_limits<std::int64_t>::max()) + 1ULL :
464 static_cast<std::uint64_t>(std::numeric_limits<std::int64_t>::max());
465
466 for (; index < input.size(); ++index) {
467 const char ch = input[index];
468 if (ch == '.') {
469 if (decimal_point) {
470 throw std::invalid_argument("decimal value contains multiple points");
471 }
472 decimal_point = true;
473 continue;
474 }
475 if (ch < '0' || ch > '9') {
476 throw std::invalid_argument("decimal value contains an invalid character");
477 }
478 has_digit = true;
479 const auto digit = static_cast<std::uint64_t>(ch - '0');
480 if (magnitude > (limit - digit) / 10ULL) {
481 throw std::out_of_range("decimal value exceeds Decimal64 range");
482 }
483 magnitude = magnitude * 10ULL + digit;
484 if (decimal_point) {
485 if (scale == std::numeric_limits<std::uint8_t>::max()) {
486 throw std::out_of_range("decimal scale exceeds Decimal64 range");
487 }
488 ++scale;
489 }
490 }
491 if (!has_digit) {
492 throw std::invalid_argument("decimal value has no digits");
493 }
494
495 while (scale > 0 && magnitude % 10ULL == 0) {
496 magnitude /= 10ULL;
497 --scale;
498 }
499
500 std::int64_t coefficient;
501 if (
502 negative &&
503 magnitude == static_cast<std::uint64_t>(std::numeric_limits<std::int64_t>::max()) + 1ULL
504 ) {
505 coefficient = std::numeric_limits<std::int64_t>::min();
506 } else {
507 coefficient = static_cast<std::int64_t>(magnitude);
508 if (negative) {
509 coefficient = -coefficient;
510 }
511 }
512 return Decimal64(coefficient, scale);
513 }
514
519 std::int64_t coefficient() const noexcept { return coefficient_; }
520
525 std::uint8_t scale() const noexcept { return scale_; }
526
531 std::string str() const {
532 const bool negative = coefficient_ < 0;
533 std::uint64_t magnitude = negative ?
534 static_cast<std::uint64_t>(-(coefficient_ + 1)) + 1ULL :
535 static_cast<std::uint64_t>(coefficient_);
536 std::string digits = std::to_string(magnitude);

538 if (digits.size() <= scale_) {
539 digits.insert(0, static_cast<std::size_t>(scale_) - digits.size() + 1, '0');
540 }
541 digits.insert(digits.size() - scale_, 1, '.');
542 }
543 if (negative) {
544 digits.insert(digits.begin(), '-');
545 }
546 return digits;
547 }

593inline bool align(
594 const Decimal64& lhs,
595 const Decimal64& rhs,
596 std::int64_t& left,
597 std::int64_t& right,
598 std::uint8_t& scale
599) {
600 scale = lhs.scale() > rhs.scale() ? lhs.scale() : rhs.scale();
601 left = lhs.coefficient();
602 right = rhs.coefficient();
603 for (std::uint8_t step = lhs.scale(); step < scale; ++step) {
604 if (left > std::numeric_limits<std::int64_t>::max() / 10 ||
605 left < std::numeric_limits<std::int64_t>::min() / 10) {
606 return false;
607 }
608 left *= 10;
609 }
610 for (std::uint8_t step = rhs.scale(); step < scale; ++step) {
611 if (right > std::numeric_limits<std::int64_t>::max() / 10 ||
612 right < std::numeric_limits<std::int64_t>::min() / 10) {
613 return false;
614 }
615 right *= 10;
616 }
617 return true;
618}

691inline std::optional<Decimal64> decimal_multiply(const Decimal64& lhs, const Decimal64& rhs) {
692 const auto left = lhs.coefficient();
693 const auto right = rhs.coefficient();
694 const int scale = static_cast<int>(lhs.scale()) + static_cast<int>(rhs.scale());
695 if (scale > std::numeric_limits<std::uint8_t>::max()) {
696 return std::nullopt;
697 }
698 if (left != 0 && right != 0) {
699 // Checked before multiplying rather than after, because signed overflow is undefined and a
700 // wrapped price is exactly the failure this type exists to prevent.
701 if (left > 0 ? (right > 0 ? left > std::numeric_limits<std::int64_t>::max() / right
702 : right < std::numeric_limits<std::int64_t>::min() / left)
703 : (right > 0 ? left < std::numeric_limits<std::int64_t>::min() / right
704 : left < std::numeric_limits<std::int64_t>::max() / right)) {
705 return std::nullopt;
706 }
707 }
708 return decimal_normalized(Decimal64(left * right, static_cast<std::uint8_t>(scale)));
709}

722inline std::optional<Decimal64> decimal_divide(
723 const Decimal64& lhs,
724 const Decimal64& rhs,
725 std::uint8_t scale
726) {
727 std::int64_t left = 0;
728 std::int64_t right = 0;
729 std::uint8_t common = 0;
730 if (!decimal_detail::align(lhs, rhs, left, right, common) || right == 0) {
731 return std::nullopt;
732 }
733 // Aligning cancels the scales, so `left / right` is the plain ratio. Each requested decimal
734 // place is one more factor of ten taken before the division.
735 for (std::uint8_t step = 0; step < scale; ++step) {
736 if (left > std::numeric_limits<std::int64_t>::max() / 10 ||
737 left < std::numeric_limits<std::int64_t>::min() / 10) {
738 return std::nullopt;
739 }
740 left *= 10;
741 }
742 if (left == std::numeric_limits<std::int64_t>::min() && right == -1) {
743 return std::nullopt;
744 }
745 return decimal_normalized(Decimal64(left / right, scale));
746}

759inline std::optional<Decimal64> decimal_floor_to_multiple(
760 const Decimal64& value,
761 const Decimal64& step
762) {
763 std::int64_t left = 0;
764 std::int64_t right = 0;
765 std::uint8_t scale = 0;
766 if (!decimal_detail::align(value, step, left, right, scale) || right <= 0) {
767 return std::nullopt;
768 }
769 auto rest = left % right;
770 if (rest < 0) {
771 rest += right;
772 }
773 if (left < std::numeric_limits<std::int64_t>::min() + rest) {
774 return std::nullopt;
775 }
776 return decimal_normalized(Decimal64(left - rest, scale));
777}

785inline bool decimal_less(const Decimal64& lhs, const Decimal64& rhs) {
786 std::int64_t left = 0;
787 std::int64_t right = 0;
788 std::uint8_t scale = 0;
789 if (decimal_detail::align(lhs, rhs, left, right, scale)) {
790 return left < right;
791 }
792 // Rescaling overflowed, so the values differ by more than any shared scale can
793 // express: the one with the smaller scale has the larger magnitude. Between two
794 // negatives the larger magnitude is the smaller value.
795 const bool left_negative = lhs.coefficient() < 0;
796 if (left_negative != (rhs.coefficient() < 0)) {
797 return left_negative;
798 }
799 return left_negative ? lhs.scale() < rhs.scale() : lhs.scale() > rhs.scale();
800}

1306inline Date date_from_epoch_days(std::int64_t days) {
1307 // Shifts the era so that the year starts in March, which is what removes the leap-day special
1308 // case from everything below: 1970-01-01 is 719468 days after 0000-03-01.
1309 const std::int64_t shifted = days + 719468;
1310 const std::int64_t era = (shifted >= 0 ? shifted : shifted - 146096) / 146097;
1311 const std::int64_t day_of_era = shifted - era * 146097;
1312 const std::int64_t year_of_era =
1313 (day_of_era - day_of_era / 1460 + day_of_era / 36524 - day_of_era / 146096) / 365;
1314 const std::int64_t year = year_of_era + era * 400;
1315 const std::int64_t day_of_year =
1316 day_of_era - (365 * year_of_era + year_of_era / 4 - year_of_era / 100);
1317 const std::int64_t shifted_month = (5 * day_of_year + 2) / 153;
1318 const std::int64_t day = day_of_year - (153 * shifted_month + 2) / 5 + 1;
1319 const std::int64_t month = shifted_month + (shifted_month < 10 ? 3 : -9);
1320 return Date{
1321 static_cast<int>(year + (month <= 2 ? 1 : 0)),
1322 static_cast<unsigned>(month),
1323 static_cast<unsigned>(day),
1324 };
1325}

1332inline bool is_valid_date(const Date& date) {
1333 if (date.year_ < 1 || date.year_ > 9999 || date.month_ < 1 || date.month_ > 12) {
1334 return false;
1335 }
1336 static constexpr unsigned days_per_month[] = {
1337 31, 28, 31, 30, 31, 30, 31, 31, 30, 31, 30, 31,
1338 };
1339 unsigned days = days_per_month[date.month_ - 1];
1340 const bool leap = date.year_ % 4 == 0 && (date.year_ % 100 != 0 || date.year_ % 400 == 0);
1341 if (date.month_ == 2 && leap) {
1342 ++days;
1343 }
1344 return date.day_ >= 1 && date.day_ <= days;
1345}

1712 RateSelection rate_for(
1713 Country country,
1714 const VenueKey& asserted,
1715 const Date& date
1716 ) const {
1717 // A row for a market this library does not normalize can never be the answer to a question asked
1718 // in terms of `Country`, and it is kept so a caller can see that the broker sent it.
1719 const auto in_market = [&](const CommissionRate& rate) {
1720 return rate.normalized() && rate.country_ == country;
1721 };
1722 // **Each pass stops at the first row that decides it, and an unreadable row decides it too.**
1723 // Within a pass the first match has always won, so a damaged row *after* a definite match could
1724 // never have been the answer and is not a reason to refuse -- scanning the whole pass first would
1725 // throw away an answer this library knows. A damaged row reached *before* any match is the
1726 // opposite case: it may have been the match, so nothing later in the pass can be trusted to be
1727 // the winner.
1728 const auto pass = [&](bool venue_specific) -> std::optional<RateSelection> {
1729 for (const auto& rate : rates_) {
1730 if (!in_market(rate)) {
1731 continue;
1732 }
1733 if (!(venue_specific ? rate.venue_.matches(asserted) : rate.venue_.is_any())) {
1734 continue;
1735 }
1736 switch (rate.applies_on(date)) {
1737 case RateApplicability::NO:
1738 continue;
1739 case RateApplicability::YES: {
1740 RateSelection found;
1741 found.rate_ = rate;
1742 return found;
1743 }
1744 case RateApplicability::UNREADABLE: {
1745 RateSelection refused;
1746 refused.bounds_unreadable_ = true;
1747 return refused;
1748 }
1749 }
1750 }
1751 return std::nullopt;
1752 };
1753 // An undecided pass one falls through; a *refused* pass one does not. Pass two's precondition is
1754 // "no row for the asserted venue applies", and that is precisely what stopped being known.
1755 if (const auto venue_row = pass(true)) {
1756 return *venue_row;
1757 }
1758 if (const auto neutral_row = pass(false)) {
1759 return *neutral_row;
1760 }
1761 return RateSelection{};
1762 }
1763
1769 bool any_dated_rate_for(Country country) const {
1770 for (const auto& rate : rates_) {
1771 // A bound that could not be read is still a bound the source stated. Asking only whether the
1772 // optionals are filled would read a row dated *solely* by damage as undated, and hand the
1773 // caller the very "every candidate is the same whichever calendar named the day" assurance
1774 // that the damage makes impossible.
1775 const bool dated = rate.from_ || rate.until_ || rate.from_unreadable_ || rate.until_unreadable_;
1776 if (rate.normalized() && rate.country_ == country && dated) {
1777 return true;
1778 }
1779 }
1780 return false;
1781 }
