320template <typename S, typename Ctx>
321void call_every_handler(
322 S& strategy, Ctx& context, const BarEvent& bar, const BarSlice& slice, const QuoteEvent& quote,
323 const Fill& fill, const OrderSnapshot& order, const MarketPhase& phase
324) {
325 dispatch::start(strategy, context);
326 dispatch::stop(strategy, context);
327 // The bar route the runner takes: slices for a strategy declaring `on_bars`, single bars otherwise.
328 if constexpr (declares_on_bars<S>) {
329 static_cast<void>(bar);
330 dispatch::bars(strategy, context, slice);
331 } else {
332 static_cast<void>(slice);
333 dispatch::bar(strategy, context, bar);
334 }
335 dispatch::quote(strategy, context, quote);
336 dispatch::fill(strategy, context, fill);
337 dispatch::order_update(strategy, context, order);
338 dispatch::phase(strategy, context, phase);
339}

345template <typename S, typename Ctx>

662template <typename S, typename Ctx>
663struct StrategyConformance {
664 // A strategy's handler names are looked up through a class derived from it, which a `final` class does
665 // not allow; read any other way, a handler the runner cannot call could be skipped in silence.
666 static_assert(
667 strategy_detail::names_can_be_read<S>,
668 "a final strategy cannot be checked for a handler the runner would skip, because its member names "
669 "cannot be looked up; remove `final` from the strategy class"
670 );
671 // A handler taking its context by value would submit to a copy, and what it submitted would be lost.
672 // Every context this library hands a strategy is move-only; a caller's own must be too -- from the
673 // mutable lvalue `dispatch` passes as well as from a `const` one, since a copy constructor may take either.
674 static_assert(
675 !std::is_copy_constructible_v<Ctx> && !std::is_constructible_v<Ctx, Ctx&>,
676 "a strategy context must not be copyable, so that a handler taking it by value does not compile "
677 "(what it submitted would go to the copy and be lost); delete the context's copy constructor"
678 );
679 static_assert(
680 strategy_detail::declares_any_handler<S>,
681 "This type declares no strategy handler. Expected at least one of on_start, on_bar, on_bars, "
682 "on_quote, on_fill, on_order_update, on_phase, or on_stop"
683 );
684 static_assert(
685 !(strategy_detail::declares_on_bar<S> && strategy_detail::declares_on_bars<S>),
686 "a strategy declares on_bar or on_bars, not both: on_bar receives each bar, on_bars every "
687 "instrument's bar of a period at once; keep the one this strategy decides with"
688 );
689
691 static constexpr bool value = strategy_detail::compiles_every_handler_call<S, Ctx>();
692};
