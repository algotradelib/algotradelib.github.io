604template <typename S, typename Ctx>
605struct StrategyConformance {
606 // A strategy's handler names are looked up through a class derived from it, which a `final` class does
607 // not allow; read any other way, a handler the runner cannot call could be skipped in silence.
608 static_assert(
609 strategy_detail::names_can_be_read<S>,
610 "a final strategy cannot be checked for a handler the runner would skip, because its member names "
611 "cannot be looked up; remove `final` from the strategy class"
612 );
613 // A handler taking its context by value would submit to a copy, and what it submitted would be lost.
614 // Every context this library hands a strategy is move-only; a caller's own must be too -- from the
615 // mutable lvalue `dispatch` passes as well as from a `const` one, since a copy constructor may take either.
616 static_assert(
617 !std::is_copy_constructible_v<Ctx> && !std::is_constructible_v<Ctx, Ctx&>,
618 "a strategy context must not be copyable, so that a handler taking it by value does not compile "
619 "(what it submitted would go to the copy and be lost); delete the context's copy constructor"
620 );
621 static_assert(
622 strategy_detail::declares_any_handler<S>,
623 "This type declares no strategy handler. Expected at least one of on_start, on_bar, on_quote, "
624 "on_fill, on_order_update, on_phase, or on_stop"
625 );
626
628 static constexpr bool value = strategy_detail::compiles_every_handler_call<S, Ctx>();
629};
