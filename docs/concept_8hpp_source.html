103// Whether `S::reads_bar_history` names an accessible static data member declared `const bool`. The declared type
104// is checked as well as the address's, since a class type can overload `operator&`.
105template <typename S, typename = void>
106struct names_static_bool_reads_bar_history : std::false_type {};
107template <typename S>
108struct names_static_bool_reads_bar_history<
109 S, std::void_t<decltype(S::reads_bar_history), decltype(&S::reads_bar_history)>
110>
111: std::bool_constant<
112 std::is_same_v<decltype(S::reads_bar_history), const bool> &&
113 std::is_same_v<decltype(&S::reads_bar_history), const bool*>
114> {};

363template <typename S, typename Ctx>
364void call_every_handler(
365 S& strategy, Ctx& context, const BarEvent& bar, const BarSlice& slice, const QuoteEvent& quote,
366 const Fill& fill, const OrderSnapshot& order, const MarketPhase& phase
367) {
368 dispatch::start(strategy, context);
369 dispatch::stop(strategy, context);
370 // The bar route the runner takes: slices for a strategy declaring `on_bars`, single bars otherwise.
371 if constexpr (declares_on_bars<S>) {
372 static_cast<void>(bar);
373 dispatch::bars(strategy, context, slice);
374 } else {
375 static_cast<void>(slice);
376 dispatch::bar(strategy, context, bar);
377 }
378 dispatch::quote(strategy, context, quote);
379 dispatch::fill(strategy, context, fill);
380 dispatch::order_update(strategy, context, order);
381 dispatch::phase(strategy, context, phase);
382}

388template <typename S, typename Ctx>
389constexpr bool compiles_every_handler_call() {
390 if constexpr (names_can_be_read<S>) {
391 static_cast<void>(&call_every_handler<S, Ctx>);
392 }
393 return true;
394}

705template <typename S, typename Ctx>
706struct StrategyConformance {
707 // A strategy's handler names are looked up through a class derived from it, which a `final` class does
708 // not allow; read any other way, a handler the runner cannot call could be skipped in silence.
709 static_assert(
710 strategy_detail::names_can_be_read<S>,
711 "a final strategy cannot be checked for a handler the runner would skip, because its member names "
712 "cannot be looked up; remove `final` from the strategy class"
713 );
714 // A handler taking its context by value would submit to a copy, and what it submitted would be lost.
715 // Every context this library hands a strategy is move-only; a caller's own must be too -- from the
716 // mutable lvalue `dispatch` passes as well as from a `const` one, since a copy constructor may take either.
717 static_assert(
718 !std::is_copy_constructible_v<Ctx> && !std::is_constructible_v<Ctx, Ctx&>,
719 "a strategy context must not be copyable, so that a handler taking it by value does not compile "
720 "(what it submitted would go to the copy and be lost); delete the context's copy constructor"
721 );
722 static_assert(
723 strategy_detail::declares_any_handler<S>,
724 "This type declares no strategy handler. Expected at least one of on_start, on_bar, on_bars, "
725 "on_quote, on_fill, on_order_update, on_phase, or on_stop"
726 );
727 static_assert(
728 !(strategy_detail::declares_on_bar<S> && strategy_detail::declares_on_bars<S>),
729 "a strategy declares on_bar or on_bars, not both: on_bar receives each bar, on_bars every "
730 "instrument's bar of a period at once; keep the one this strategy decides with"
731 );
732
733 // Instantiated so a malformed `reads_bar_history` is refused here too.
734 static_assert(strategy_detail::reads_bar_history<S> || !strategy_detail::reads_bar_history<S>);
735
737 static constexpr bool value = strategy_detail::compiles_every_handler_call<S, Ctx>();
738};
