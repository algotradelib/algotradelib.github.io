103// Whether `S::reads_bar_history` names an accessible static data member declared `const bool`. The declared type
104// is checked as well as the address's, since a class type can overload `operator&`.
105template <typename S, typename = void>
106struct names_static_bool_reads_bar_history : std::false_type {};
107template <typename S>
108struct names_static_bool_reads_bar_history<
109 S, std::void_t<decltype(S::reads_bar_history), decltype(&S::reads_bar_history)>
110>
111: std::bool_constant<
112 std::is_same_v<decltype(S::reads_bar_history), const bool> &&
113 std::is_same_v<decltype(&S::reads_bar_history), const bool*>
114 > {};

432template <typename S, typename Ctx>
433void call_every_handler(
434 S& strategy, Ctx& context, const BarEvent& bar, const BarSlice& slice, const QuoteEvent& quote, const Fill& fill,
435 const OrderSnapshot& order, const MarketPhase& phase, const DataEvent<data_record_t<S>>& data
436) {
437 dispatch::start(strategy, context);
438 dispatch::stop(strategy, context);
439 // The bar route the runner takes: slices for a strategy declaring `on_bars`, single bars otherwise.
440 if constexpr (declares_on_bars<S>) {
441 static_cast<void>(bar);
442 dispatch::bars(strategy, context, slice);
443 } else {
444 static_cast<void>(slice);
445 dispatch::bar(strategy, context, bar);
446 }
447 dispatch::quote(strategy, context, quote);
448 dispatch::fill(strategy, context, fill);
449 dispatch::order_update(strategy, context, order);
450 dispatch::phase(strategy, context, phase);
451 dispatch::data(strategy, context, data);
452}

813template <typename S, typename Ctx>
814struct StrategyConformance {
815 // A strategy's handler names are looked up through a class derived from it, which a `final` class does
816 // not allow; read any other way, a handler the runner cannot call could be skipped in silence.
817 static_assert(
818 strategy_detail::names_can_be_read<S>,
819 "a final strategy cannot be checked for a handler the runner would skip, because its member names "
820 "cannot be looked up; remove `final` from the strategy class"
821 );
822 // A handler taking its context by value would submit to a copy, and what it submitted would be lost.
823 // Every context this library hands a strategy is move-only; a caller's own must be too -- from the
824 // mutable lvalue `dispatch` passes as well as from a `const` one, since a copy constructor may take either.
825 static_assert(
826 !std::is_copy_constructible_v<Ctx> && !std::is_constructible_v<Ctx, Ctx&>,
827 "a strategy context must not be copyable, so that a handler taking it by value does not compile "
828 "(what it submitted would go to the copy and be lost); delete the context's copy constructor"
829 );
830 static_assert(
831 strategy_detail::declares_any_handler<S>,
832 "This type declares no strategy handler. Expected at least one of on_start, on_bar, on_bars, "
833 "on_quote, on_fill, on_order_update, on_phase, on_data, or on_stop"
834 );
835 static_assert(
836 !(strategy_detail::declares_on_bar<S> && strategy_detail::declares_on_bars<S>),
837 "a strategy declares on_bar or on_bars, not both: on_bar receives each bar, on_bars every "
838 "instrument's bar of a period at once; keep the one this strategy decides with"
839 );
840
841 // Instantiated so a malformed `reads_bar_history` is refused here too.
842 static_assert(strategy_detail::reads_bar_history<S> || !strategy_detail::reads_bar_history<S>);
843
845 static constexpr bool value = strategy_detail::compiles_every_handler_call<S, Ctx>();
846};
