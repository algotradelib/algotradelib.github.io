39inline Reduction reduce(const math::Observed& parts, math::Index first, math::Index length) {
40 Reduction result;
41 result.count_ = static_cast<std::size_t>(parts.present_.segment(first, length).sum());
42 result.sum_ = parts.values_.segment(first, length).sum();
43 return result;

69inline Dispersion dispersion(
70 const math::Observed& parts,
71 math::Index first,
72 math::Index length,
73 std::size_t needed
74) {
75 Dispersion result;
76 const auto reduced = reduce(parts, first, length);
77 result.count_ = reduced.count_;
78 if (reduced.count_ < needed || reduced.count_ < 2) {
79 return result;
80 }
81 result.mean_ = reduced.sum_ / static_cast<Scalar>(reduced.count_);
82 const Scalar squared = (parts.values_.segment(first, length) -
83 result.mean_ * parts.present_.segment(first, length)).squaredNorm();
84 result.deviation_ = std::sqrt(squared / static_cast<Scalar>(reduced.count_ - 1));
85 result.reported_ = true;
86 return result;
87}

128inline math::Vector rolling_sum(
129 math::SeriesView values,
130 std::size_t window,
131 std::size_t min_observed
132) {
133 rolling_detail::check(window, min_observed);
134 const auto parts = math::observed_parts(values);
135 math::Vector result = math::unobserved(static_cast<std::size_t>(values.size()));
136 for (math::Index row = 0; row < values.size(); ++row) {
137 const math::Index first = rolling_detail::window_start(row, window);
138 const auto reduced = rolling_detail::reduce(parts, first, row - first + 1);
139 if (reduced.count_ >= min_observed) {
140 result[row] = reduced.sum_;
141 }
142 }
143 return result;

252inline math::Vector rolling_min(
253 math::SeriesView values,
254 std::size_t window,
255 std::size_t min_observed
256) {
257 rolling_detail::check(window, min_observed);
258 const auto parts = math::observed_parts(values);
259 // Absent rows become `+inf` so they lose every comparison without needing a mask inside the
260 // window. A window with no observation at all is rejected by the count, never read from here.
261 const math::Vector ceiling =
262 math::observed_or(values, std::numeric_limits<Scalar>::infinity());
263 math::Vector result = math::unobserved(static_cast<std::size_t>(values.size()));
264 for (math::Index row = 0; row < values.size(); ++row) {
265 const math::Index first = rolling_detail::window_start(row, window);
266 const math::Index length = row - first + 1;
267 if (parts.present_.segment(first, length).sum() >= static_cast<Scalar>(min_observed)) {
268 result[row] = ceiling.segment(first, length).minCoeff();
269 }
270 }
271 return result;
272}

364inline math::Vector wma(
365 math::SeriesView values,
366 std::size_t window,
367 std::size_t min_observed
368) {
369 rolling_detail::check(window, min_observed);
370 const auto parts = math::observed_parts(values);
371 math::Vector weights(static_cast<math::Index>(window));
372 for (math::Index at = 0; at < weights.size(); ++at) {
373 weights[at] = static_cast<Scalar>(at + 1);
374 }
375
376 math::Vector result = math::unobserved(static_cast<std::size_t>(values.size()));
377 for (math::Index row = 0; row < values.size(); ++row) {
378 const math::Index first = rolling_detail::window_start(row, window);
379 const math::Index length = row - first + 1;
380 if (parts.present_.segment(first, length).sum() < static_cast<Scalar>(min_observed)) {
381 continue;
382 }
383 const auto used = weights.head(length);
384 const Scalar divisor = parts.present_.segment(first, length).dot(used);
385 if (divisor > Scalar{0}) {
386 result[row] = parts.values_.segment(first, length).dot(used) / divisor;
387 }
388 }
389 return result;
390}

484inline math::Vector ema(math::SeriesView values, Scalar alpha, std::size_t min_observed = 1) {
485 rolling_detail::check_alpha(alpha);
486 if (min_observed == 0) {
487 throw std::invalid_argument("ema must require at least one observation");
488 }
489 math::Vector result = math::unobserved(static_cast<std::size_t>(values.size()));
490 Scalar level{0};
491 std::size_t count = 0;
492 for (math::Index row = 0; row < values.size(); ++row) {
493 const Scalar value = values[row];
494 if (!observed(value)) {
495 continue;
496 }
497 // Seeded by count rather than by an unobserved level, so a level an infinity turned into `NaN`
498 // stays `NaN` instead of being mistaken for "no level yet".
499 level = count == 0 ? value : level + alpha * (value - level);
500 ++count;
501 if (count >= min_observed) {
502 result[row] = level;
503 }
504 }
505 return result;
506}
