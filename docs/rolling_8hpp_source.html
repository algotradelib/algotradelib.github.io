69inline Dispersion dispersion(
70 const math::Observed& parts,
71 math::Index first,
72 math::Index length,
73 std::size_t needed
74) {
75 Dispersion result;
76 const auto reduced = reduce(parts, first, length);
77 result.count_ = reduced.count_;
78 if (reduced.count_ < needed || reduced.count_ < 2) {
79 return result;
80 }
81 result.mean_ = reduced.sum_ / static_cast<Scalar>(reduced.count_);
82 const Scalar squared = (parts.values_.segment(first, length) -
83 result.mean_ * parts.present_.segment(first, length)).squaredNorm();
84 result.deviation_ = std::sqrt(squared / static_cast<Scalar>(reduced.count_ - 1));
85 result.reported_ = true;
86 return result;
87}

130inline math::Vector rolling_mean(
131 math::SeriesView values,
132 std::size_t window,
133 std::size_t min_observed
134) {
135 rolling_detail::check(window, min_observed);
136 const auto parts = math::observed_parts(values);
137 math::Vector result = math::unobserved(static_cast<std::size_t>(values.size()));
138 for (math::Index row = 0; row < values.size(); ++row) {
139 const math::Index first = rolling_detail::window_start(row, window);
140 const auto reduced = rolling_detail::reduce(parts, first, row - first + 1);
141 if (reduced.count_ >= min_observed) {
142 result[row] = reduced.sum_ / static_cast<Scalar>(reduced.count_);
143 }
144 }
145 return result;
146}

213inline math::Vector rolling_min(
214 math::SeriesView values,
215 std::size_t window,
216 std::size_t min_observed
217) {
218 rolling_detail::check(window, min_observed);
219 const auto parts = math::observed_parts(values);
220 // Absent rows become `+inf` so they lose every comparison without needing a mask inside the
221 // window. A window with no observation at all is rejected by the count, never read from here.
222 const math::Vector ceiling =
223 math::observed_or(values, std::numeric_limits<Scalar>::infinity());
224 math::Vector result = math::unobserved(static_cast<std::size_t>(values.size()));
225 for (math::Index row = 0; row < values.size(); ++row) {
226 const math::Index first = rolling_detail::window_start(row, window);
227 const math::Index length = row - first + 1;
228 if (parts.present_.segment(first, length).sum() >= static_cast<Scalar>(min_observed)) {
229 result[row] = ceiling.segment(first, length).minCoeff();
230 }
231 }
232 return result;
233}

363inline math::Vector ema(math::SeriesView values, Scalar alpha, std::size_t min_observed = 1) {
364 rolling_detail::check_alpha(alpha);
365 if (min_observed == 0) {
366 throw std::invalid_argument("ema must require at least one observation");
367 }
368 math::Vector result = math::unobserved(static_cast<std::size_t>(values.size()));
369 Scalar level{0};
370 std::size_t count = 0;
371 for (math::Index row = 0; row < values.size(); ++row) {
372 const Scalar value = values[row];
373 if (!observed(value)) {
374 continue;
375 }
376 // Seeded by count rather than by an unobserved level, so a level an infinity turned into `NaN`
377 // stays `NaN` instead of being mistaken for "no level yet".
378 level = count == 0 ? value : level + alpha * (value - level);
379 ++count;
380 if (count >= min_observed) {
381 result[row] = level;
382 }
383 }
384 return result;
385}
