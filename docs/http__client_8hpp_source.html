233 HttpResponse request(
234 http::verb method,
235 const std::string& target,
236 const std::string& body = {},
237 const std::map<std::string, std::string>& headers = {}
238 ) {
239 for (int attempt = 0;; ++attempt) {
240 beast::error_code error;
241 // Never replayed, so never written into a connection the server may have closed while it was
242 // idle; see `HttpClientOptions::unreplayed_idle_limit_`.
243 if (method != http::verb::get && idle_since_ &&
244 std::chrono::steady_clock::now() - *idle_since_ > options_.unreplayed_idle_limit_) {
245 // Dropped without waiting for the peer's close notification: a synchronous TLS shutdown is not
246 // bounded by the stream's deadline, and a live server may never answer one.
247 close_ssl_stream(*stream_);
248 stream_.reset();
249 buffer_.consume(buffer_.size());
250 idle_since_.reset();
251 }
252 ensure_connected(error);
253 if (!error) {
254 http::request<http::string_body> request_value{method, target, 11};
255 request_value.set(http::field::host, host_header());
256 request_value.set(http::field::user_agent, "algotradelib");
257 request_value.keep_alive(true);
258 for (const auto& header : headers) {
259 request_value.set(header.first, header.second);
260 }
261 request_value.body() = body;
262 if (!body.empty()) {
263 request_value.prepare_payload();
264 }
265
266 beast::get_lowest_layer(*stream_).expires_after(options_.timeout_);
267 http::write(*stream_, request_value, error);
268 }
269
270 http::response_parser<http::string_body> parser;
271 if (!error) {
272 parser.body_limit(options_.body_limit_);
273 http::read(*stream_, buffer_, parser, error);
274 }
275
276 if (!error) {
277 auto response = parser.release();
278 if (!response.keep_alive() || response.need_eof()) {
279 disconnect();
280 } else {
281 idle_since_ = std::chrono::steady_clock::now();
282 }
283 return response;
284 }
285
286 disconnect();
287 if (method != http::verb::get || attempt > 0) {
288 throw beast::system_error(error);
289 }
290 }
291 }

294 void close() { disconnect(); }

465 void close() {
466 net::post(strand_, [self = shared_from_this()] {
467 self->stopped_ = true;
468 self->resolver_.cancel();
469 self->resolve_timer_.cancel();
470 self->admission_timer_.cancel();
471 self->disconnect();
472 BrokerError error;
473 error.category_ = ErrorCategory::CANCELLED;
474 error.broker_code_ = "close";
475 error.message_ = "HTTP client closed";
476 // Each entry answers for itself, which is why `sent_` lives on the request rather than being
477 // inferred from the connection: a caller has to be able to tell an order that provably does
478 // not exist -- no reconciliation owed, no request spent -- from one that may be on the wire.
479 // Reading the client's state here would answer for none of them, since `disconnect()` has
480 // just cleared it.
481 while (!self->queue_.empty()) {
482 auto pending = std::move(self->queue_.front());
483 self->queue_.pop_front();
484 BrokerError reported = error;
485 reported.reached_broker_ = pending.sent_;
486 self->complete(std::move(pending.handler_), Result<HttpResponse>(std::move(reported)));
487 }
488 });
489 }

502 struct Pending {
503 HttpRequest request_;
504 CompletionHandler handler_;
505 bool retried_{false};
506 // Whether this request already went to a fresh connection because the last one sat idle. Once is
507 // enough, and it bounds the rule: a fresh connection starts its own idle clock, which with a limit
508 // of zero or below would otherwise send it straight back for another.
509 bool refreshed_{false};
510 // Whether any attempt at this request has begun writing it to the socket.
511 //
512 // The honest record of what a caller may be told about the broker having seen it, and
513 // deliberately not `retried_`, which counts a replay rather than a send: a GET that fails to
514 // *connect* is retried without a byte having left, and reading the retry as evidence would
515 // report an unsent request as one the broker may hold. Sticky once set, because a second
516 // attempt cannot unsend the first.
517 bool sent_{false};
518 // When this request began waiting for rate-limit admission; empty while it is not waiting. Kept
519 // on the request, not the client: a request inserted ahead of a waiting one is admitted on its
520 // own terms, and must neither be charged the other's wait nor restart it.
521 std::optional<std::chrono::steady_clock::time_point> waiting_since_;
522
523 Pending(HttpRequest request_value, CompletionHandler completion)
524 : request_(std::move(request_value)), handler_(std::move(completion)) {}
525 };

527 net::strand<net::any_io_executor> strand_;
528 tcp::resolver resolver_;

530 // The admission timer, wrapped so that arming it is counted by construction.
531 //
532 // **A counter beside the timer is not enough, and that is not hypothetical.** The first version of
533 // this observable incremented a member at the one call site that armed the timer, on the paced
534 // branch. `pump()` returns before that site when there is no budget, so a regression arming this
535 // timer on the *unpaced* branch would have delayed every request on an unmetered API while the
536 // counter still read zero -- and the test asserting zero would have passed.
537 //
538 // Making the count a property of the timer removes the bypass rather than documenting it: there
539 // is no way to arm *this* timer without `expires_after`, so there is no way to wait on it without
540 // saying so. What that does not cover is a delay built on some other timer; see
541 // `admission_waits()`.
542 //
543 // **A narrow wrapper for two call sites, not a general Asio timer substitute.** It discards the
544 // completion-token return values its callers do not use, and it counts an expiry reset even when
545 // no wait follows -- which is exact for the one place that arms this timer and immediately waits,
546 // and would need revisiting for any other use. Changing an expiry cancels a pending wait, whose
547 // handler then receives `operation_aborted`; this client's handler already returns on any error.
548 class CountedTimer {
549 public:
550 explicit CountedTimer(net::strand<net::any_io_executor>& strand) : timer_(strand) {}
551
552 // Arms the timer for `delay`, and counts the arm.
553 template <typename Rep, typename Period>
554 void expires_after(const std::chrono::duration<Rep, Period>& delay) {
555 timer_.expires_after(delay);
556 // Counted after the call, so this records an arm that happened rather than one attempted.
557 ++arms_;
558 }
559
560 // Waits, forwarding to the underlying timer.
561 template <typename Handler>
562 void async_wait(Handler&& handler) {
563 timer_.async_wait(std::forward<Handler>(handler));
564 }
565
566 // Cancels any outstanding wait. Not counted: a cancellation is not an admission wait.
567 std::size_t cancel() { return timer_.cancel(); }
568
569 // Returns how many times this timer has been armed.
570 std::uint64_t arms() const noexcept { return arms_; }
571
572 private:
573 net::steady_timer timer_;
574 std::uint64_t arms_{0};
575 };

594 void enqueue(Pending pending) {
595 if (stopped_) {
596 BrokerError error;
597 error.category_ = ErrorCategory::CANCELLED;
598 // Never queued, let alone sent: the instruction reached nobody.
599 error.reached_broker_ = false;
600 error.broker_code_ = "enqueue";
601 error.message_ = "HTTP client is closed";
602 complete(std::move(pending.handler_), Result<HttpResponse>(std::move(error)));
603 return;
604 }
605 if (queue_.size() >= options_.queue_limit_) {
606 if (limiter_) {
607 limiter_->record_shed(pending.request_.group_);
608 }
609 BrokerError error;
610 error.category_ = ErrorCategory::TRANSPORT;
611 // Refused at admission, so nothing was sent. `TRANSPORT` would otherwise read as a request
612 // that may be on the wire, and a runner would withhold the instrument to reconcile an order
613 // that was never built.
614 error.reached_broker_ = false;
615 error.broker_code_ = "queue_full";
616 error.message_ = "HTTP request queue limit exceeded";
617 complete(std::move(pending.handler_), Result<HttpResponse>(std::move(error)));
618 return;
619 }
620 queue_.insert(insertion_point(pending.request_.group_), std::move(pending));
621 pump();
622 }
623
624 // Returns where a request of `group` belongs in the pending queue.
625 //
626 // Higher priority is served first, and equal priority keeps arrival order, so
627 // a queue carrying one traffic class behaves exactly as a FIFO queue does.
628 //
629 // The front request is skipped while it is on the wire. Reordering around an
630 // in-flight request would detach it from the response being read for it, which
631 // is the one thing priority must never be allowed to do.
632 std::deque<Pending>::iterator insertion_point(RateLimitGroup group) {
633 auto position = queue_.begin();
634 if (front_in_flight() && position != queue_.end()) {
635 ++position;
636 }
637 const int priority = rate_limit_priority(group);
638 while (position != queue_.end() &&
639 rate_limit_priority(position->request_.group_) >= priority) {
640 ++position;
641 }
642 return position;
643 }
644
645 // Returns whether the front request has already been committed to the wire.
646 bool front_in_flight() const noexcept {
647 switch (state_) {
648 case State::WRITING:
649 case State::READING:
650 return true;
651 case State::DISCONNECTED:
652 case State::RESOLVING:
653 case State::CONNECTING:
654 case State::HANDSHAKING:
655 case State::READY:
656 return false;
657 }
658 return true;
659 }

661 void pump() {
662 if (stopped_ || queue_.empty()) {
663 return;
664 }
665 if (state_ == State::DISCONNECTED) {
666 start_resolve();
667 return;
668 }
669 if (state_ != State::READY) {
670 return;
671 }
672 // A request that is never replayed is not written into a connection the server may have closed
673 // while it sat idle; see `HttpClientOptions::unreplayed_idle_limit_`. Before admission, so the
674 // reconnect spends no rate-limit slot of its own.
675 if (queue_.front().request_.method_ != http::verb::get && !queue_.front().refreshed_ && idle_since_ &&
676 std::chrono::steady_clock::now() - *idle_since_ > options_.unreplayed_idle_limit_) {
677 queue_.front().refreshed_ = true;
678 disconnect();
679 // The handshake is not admission delay; the wait, if any, is measured again from here.
680 queue_.front().waiting_since_.reset();
681 start_resolve();
682 return;
683 }
684 if (!limiter_) {
685 start_write();
686 return;
687 }
688
689 // Admission is evaluated for whichever request is at the front *now*, so a
690 // trading request that arrives while a polling request is waiting is
691 // reconsidered immediately rather than inheriting the wait it did not earn.
692 const auto group = queue_.front().request_.group_;
693 const auto delay = limiter_->delay_until(group);
694 if (delay > RateLimitController::Duration::zero()) {
695 // Waiting longer than the operation's own deadline would queue a request
696 // only to time it out on the wire. Failing now says the same thing sooner,
697 // and says it before the request is sent rather than after.
698 if (delay > std::chrono::duration_cast<RateLimitController::Duration>(options_.timeout_)) {
699 limiter_->record_shed(group);
700 auto pending = std::move(queue_.front());
701 queue_.pop_front();
702 BrokerError error;
703 error.category_ = ErrorCategory::CANCELLED;
704 // Ordinarily nothing was sent -- admission is evaluated only while the connection is
705 // `READY`, never while the front request is in flight. A GET being replayed is the
706 // exception: it is admitted again, and its first attempt did reach the wire.
707 error.reached_broker_ = pending.sent_;
708 error.broker_code_ = "admission_deadline";
709 error.message_ = "request would wait for rate-limit admission beyond its deadline";
710 complete(std::move(pending.handler_), Result<HttpResponse>(std::move(error)));
711 pump();
712 return;
713 }
714 if (!queue_.front().waiting_since_) {
715 queue_.front().waiting_since_ = std::chrono::steady_clock::now();
716 }
717 admission_timer_.expires_after(delay);
718 admission_timer_.async_wait([self = shared_from_this()](beast::error_code ec) {
719 if (ec) {
720 return;
721 }
722 self->pump();
723 });
724 return;
725 }
726
727 if (auto& waiting_since = queue_.front().waiting_since_) {
728 limiter_->record_delay(
729 group,
730 std::chrono::duration_cast<RateLimitController::Duration>(
731 std::chrono::steady_clock::now() - *waiting_since
732 )
733 );
734 waiting_since.reset();
735 }
736 admission_timer_.cancel();
737 limiter_->acquire(group);
738 start_write();
739 }

864 }

866 const bool keep_alive = response.keep_alive() && !response.need_eof();
867 self->parser_.reset();
868 self->active_request_.reset();
869 auto pending = std::move(self->queue_.front());
870 self->queue_.pop_front();
871 if (keep_alive) {
872 self->state_ = State::READY;
873 self->idle_since_ = std::chrono::steady_clock::now();
874 } else {
875 self->disconnect();
876 }
877 self->complete(std::move(pending.handler_), Result<HttpResponse>(std::move(response)));
878 self->pump();
879 }
880 );
881 }

883 void fail(const char* stage, beast::error_code ec) {
884 // **Asked of the request, not of the connection** (`sent_` is kept on each pending request), for
885 // the reason `close()` gives: `disconnect()` resets the connection's state, and asking it afterwards
886 // would answer "not in flight" for every failure -- including one that happened halfway through
887 // writing an order. This is the same question from the other
888 // side: a resolve, connect, or handshake failure happens *before* `start_write()`, so an order
889 // has certainly not been sent, and reporting it as indeterminate spends a rate-limit slot and
890 // withholds the instrument to reconcile an order that does not exist.
891 disconnect();
892 if (queue_.empty()) {
893 return;
894 }
895 auto& pending = queue_.front();
896 if (
897 pending.request_.method_ == http::verb::get && !pending.retried_ &&
898 ec != net::error::operation_aborted
899 ) {
900 pending.retried_ = true;
901 pump();
902 return;
903 }
904 auto completed = std::move(pending);
905 queue_.pop_front();
906 auto error = transport_error(stage, ec);
907 error.reached_broker_ = completed.sent_;
908 complete(std::move(completed.handler_), Result<HttpResponse>(std::move(error)));
909 pump();
910 }

912 void disconnect() noexcept {
913 resolve_timer_.cancel();
914 if (stream_) {
915 close_ssl_stream(*stream_);
916 stream_.reset();
917 }
918 buffer_.consume(buffer_.size());
919 parser_.reset();
920 active_request_.reset();
921 state_ = State::DISCONNECTED;
922 idle_since_.reset();
923 }
