326inline Report summarize(
327 const ReplayResult& result,
328 const NextBarFillModel& fills,
329 std::optional<math::Scalar> periods_per_year = std::nullopt,
330 math::Scalar risk_free = math::Scalar{0}
331) {
332 Report report;
333 report.bars_ = result.bars_;
334 report.executions_ = result.executions_.size();
335 report.rejected_ = result.rejected_.size();
336 report.refused_ = result.refused_.size();
337
338 report.caveats_.limit_fill_ = fills.limit_price();
339 report.halted_ = result.halted_;
340 report.aborted_ = result.aborted_;
341 if (fills.commission().is_scheduled()) {
342 report.caveats_.asserted_venue_ = fills.commission().asserted_venue();
343 }
344 report.caveats_.commission_tariff_ = std::string(fills.commission().tariff());
345 report.caveats_.slippage_basis_points_ = fills.slippage().basis_points_;
346 report.caveats_.slippage_ticks_ = fills.slippage().ticks_;
347 report.caveats_.gaps_ = result.gaps_;
348 report.caveats_.unfilled_ = result.unfilled_;
349 report.caveats_.stop_releases_ = result.stop_releases_;
350 report.caveats_.stop_fills_intrabar_ = result.stop_fills_intrabar_;
351 report.caveats_.limit_fills_bounded_ = result.limit_fills_bounded_;
352 report.caveats_.quotes_ = result.quotes_;
353 report.caveats_.unchecked_fill_causality_ = result.unchecked_fill_causality_;
354 report.caveats_.unvalued_bars_ = result.unvalued_bars_;
355 report.caveats_.first_unvalued_ = result.first_unvalued_;
356 report.caveats_.overdrawn_bars_ = result.overdrawn_bars_;
357 report.caveats_.first_overdrawn_ = result.first_overdrawn_;
358 report.caveats_.lowest_cash_ = result.lowest_cash_;
359 report.caveats_.lowest_cash_at_ = result.lowest_cash_at_;
360 report.caveats_.expired_day_orders_ = result.expired_day_orders_;
361
362 // Costs summed exactly, because they are money -- and **per currency**, because a replay whose
363 // instruments span markets would otherwise have won added to dollars to make one scalar.
364 // Checked, and a refusal is recorded rather than swallowed: leaving the previous total in place
365 // and carrying on produced a plausible figure that was short by an unknown amount, which is the
366 // one outcome every exact-decimal path here exists to prevent. Two individually representable
367 // commissions whose sum does not fit is enough to reach it, and a probe did.
368 bool costs_fit = true;
369 const auto accumulate = [&costs_fit](std::vector<Money>& totals, const Money& amount) {
370 for (auto& held : totals) {
371 if (held.currency_ == amount.currency_) {
372 const auto sum = decimal_add(held.amount_, amount.amount_);
373 if (!sum) {
374 costs_fit = false;
375 return;
376 }
377 held.amount_ = *sum;
378 return;
379 }
380 }
381 totals.push_back(amount);
382 };
383
384 std::vector<InstrumentId> seen;
385 for (const auto& execution : result.executions_) {
386 accumulate(report.commission_by_currency_, Money{execution.commission_, execution.currency_});
387 if (execution.institution_fee_) {
388 accumulate(
389 report.institution_by_currency_, Money{*execution.institution_fee_, execution.currency_}
390 );
391 } else {
392 // The rate stated one figure, so this execution's commission is inclusive and the split total
393 // below cannot be completed. Counted rather than approximated.
394 ++report.caveats_.inclusive_commission_executions_;
395 }
396 // The rate that charged it, collapsed so an unchanged rate is one row with a count rather than
397 // one row per fill. Accumulated here, from the executions the ledger accepted, rather than from
398 // the schedule: a row the replay never used is not a rate this run applied.
399 bool counted = false;
400 for (auto& applied : report.caveats_.applied_rates_) {
401 if (applied == execution.applied_) {
402 ++applied.executions_;
403 counted = true;
404 break;
405 }
406 }
407 if (!counted) {
408 auto entry = execution.applied_;
409 entry.executions_ = 1;
410 report.caveats_.applied_rates_.push_back(std::move(entry));
411 }
412 // The tax, totalled per currency, over the executions in the market it was stated for: another
413 // market's fills pay none by construction, and counting them would add a zero row in their
414 // currency and cost a single-currency total its scalar.
415 if (fills.sell_tax() && execution.fill_.instrument_.country_ == fills.sell_tax()->country_) {
416 accumulate(report.tax_by_currency_, Money{execution.tax_, execution.currency_});
417 }
418 // Slippage is a per-share price difference, so its cost is that times the quantity -- and it
419 // is totalled per currency like every other cost, because a replay can hold fills in more
420 // than one and adding them would produce a figure denominated in nothing.
421 if (const auto cost = decimal_multiply(execution.slippage_, execution.fill_.quantity_)) {
422 accumulate(report.slippage_by_currency_, Money{*cost, execution.currency_});
423 } else {
424 costs_fit = false;
425 }
426 bool known = false;
427 for (const auto& id : seen) {
428 if (id == execution.fill_.instrument_) {
429 known = true;
430 break;
431 }
432 }
433 if (!known) {
434 seen.push_back(execution.fill_.instrument_);
435 }
436 }
437 report.caveats_.cost_totals_incomplete_ = !costs_fit;
438 // A scalar is stated only where one can be true: one currency, something executed, and every
439 // addition representable. On a mixed-currency replay the per-currency vectors are the whole
440 // answer, and on an arithmetic refusal nothing is stated at all rather than a figure short by an
441 // amount nobody can name.
442 if (costs_fit && report.commission_by_currency_.size() == 1) {
443 report.commission_paid_ = report.commission_by_currency_.front().amount_;
444 }
445 // Present only when every charged rate split the two components, so the total is complete rather
446 // than a sum over the subset that happened to state one.
447 if (costs_fit && report.caveats_.inclusive_commission_executions_ == 0 &&
448 report.institution_by_currency_.size() == 1) {
449 report.institution_fees_paid_ = report.institution_by_currency_.front().amount_;
450 }
451 if (costs_fit && report.slippage_by_currency_.size() == 1) {
452 report.slippage_paid_ = report.slippage_by_currency_.front().amount_;
453 }
454 report.caveats_.sell_tax_ = fills.sell_tax();
455 if (fills.sell_tax() && costs_fit && report.tax_by_currency_.size() == 1) {
456 report.tax_paid_ = report.tax_by_currency_.front().amount_;
457 }
458 report.caveats_.multiple_instruments_ = seen.size() > 1;
459 for (const auto& execution : result.executions_) {
460 if (!(execution.currency_ == result.currency_)) {
461 ++report.caveats_.other_currency_executions_;
462 }
463 }
464
465 if (result.equity_.empty()) {
466 // Nothing to measure. Every figure stays unobserved rather than becoming zero, which would read as
467 // a flat result rather than as an absent one.
468 report.total_return_ = math::missing();
469 report.sharpe_ = math::missing();
470 report.annualized_sharpe_ = math::missing();
471 report.sortino_ = math::missing();
472 report.max_drawdown_ = math::missing();
473 report.total_turnover_ = math::missing();
474 return report;
475 }
476
477 report.opening_equity_ = result.equity_.front().equity_;
478 report.closing_equity_ = result.equity_.back().equity_;
479
480 // The returns and the equity curve, as the columns the metrics take.
481 math::Vector returns(static_cast<math::Index>(result.equity_.size()));
482 math::Vector levels(static_cast<math::Index>(result.equity_.size()));
483 for (std::size_t at = 0; at < result.equity_.size(); ++at) {
484 returns[static_cast<math::Index>(at)] = result.equity_[at].period_return_;
485 levels[static_cast<math::Index>(at)] = analytics::to_scalar(result.equity_[at].equity_);
486 }
487
488 // From a negative opening the ratio's sign reverses (a replay may start with negative cash), so a
489 // non-positive opening has no return and no turnover, as a zero one has none.
490 const auto opening = analytics::to_scalar(report.opening_equity_);
491 report.total_return_ = !(opening > math::Scalar{0}) ?
492 math::undefined() :
493 analytics::to_scalar(report.closing_equity_) / opening - math::Scalar{1};
494
495 report.sharpe_ = analytics::sharpe(returns, risk_free);
496 report.sortino_ = analytics::sortino(returns, risk_free);
497 // Equity at or below zero before the end leaves the periods after it without a return
498 // (`EquityPoint::period_return_`), so no ratio describes the run.
499 for (std::size_t at = 0; at + 1 < result.equity_.size(); ++at) {
500 if (result.equity_[at].equity_.coefficient() <= 0) {
501 report.sharpe_ = math::undefined();
502 report.sortino_ = math::undefined();
503 break;
504 }
505 }
506 report.max_drawdown_ = analytics::max_drawdown(levels).magnitude_;
507
508 report.annualized_sharpe_ = periods_per_year ?
509 analytics::annualize_ratio(report.sharpe_, *periods_per_year) :
510 math::missing();
511
512 // Every trade outside the curve: it holds only the opening cash, and its figures describe no trading.
513 const auto outside = report.caveats_.other_currency_executions_;
514 if (outside > 0 && outside == result.executions_.size()) {
515 report.total_return_ = math::missing();
516 report.sharpe_ = math::missing();
517 report.annualized_sharpe_ = math::missing();
518 report.sortino_ = math::missing();
519 report.max_drawdown_ = math::missing();
520 }
521
522 // Turnover from the executions rather than from a weight matrix, because a replay knows what it
523 // actually traded. One-way, matching `analytics::one_way_turnover`: the fraction of equity that
524 // changed hands.
525 // Same money on both sides, or no figure at all.
526 report.total_turnover_ = !(opening > math::Scalar{0}) ? math::undefined() : math::Scalar{0};
527 if (outside > 0) {
528 report.total_turnover_ = math::missing();
529 }
530 if (math::observed(report.total_turnover_)) {
531 for (const auto& execution : result.executions_) {
532 const auto traded = decimal_multiply(execution.fill_.quantity_, execution.fill_.price_);
533 if (traded) {
534 report.total_turnover_ += analytics::to_scalar(*traded) / opening;
535 }
536 }
537 }
538
539 return report;
540}
