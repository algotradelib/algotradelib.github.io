108 runner::SendOutcome place(const strategy::PlannedOrder& order) {
109 if (order.client_order_id_) {
110 if constexpr (!Caps::has_client_order_id) {
111 last_error_ = BrokerError{};
112 last_error_->category_ = ErrorCategory::INVALID_ARGUMENT;
113 last_error_->reached_broker_ = false;
114 last_error_->message_ =
115 "this broker accepts no idempotency key; the intent asked for one";
116 // Refused here, not by the broker: nothing was sent, so nothing may be counted against the
117 // rate limit. Reporting this as a broker rejection spent budget for a request nobody made.
118 return runner::SendOutcome::REFUSED_LOCALLY;
119 }
120 }
121
122 OrderRequest request;
123 request.instrument_ = order.instrument_;
124 request.side_ = order.side_;
125 request.type_ = order.type_;
126 request.time_in_force_ = order.time_in_force_;
127 // All three, every time. A fact not copied here never reaches the wire, so
128 // the order placed is not the order the caller planned -- and it fails
129 // silently, because a default is a valid value rather than an empty one.
130 request.session_ = order.session_;
131 // The runner has already written the route in; applied again for an executor driven by hand.
132 request.venue_ = requested_route(order.venue_);
133 request.quantity_ = order.quantity_;
134 request.price_ = order.price_;
135 // A broker whose `has_trigger_price` is false is refused it by the session before its adapter runs,
136 // which `outcome_of` reports as a local refusal: nothing was sent.
137 request.trigger_price_ = order.trigger_price_;
138 if constexpr (Caps::has_client_order_id) {
139 request.client_order_id_ = order.client_order_id_;
140 }
141
142 auto result = session_->place(std::move(request));
143 if (result) {
144 last_ref_ = result.value().ref_;
145 last_error_.reset();
146 return runner::SendOutcome::ACCEPTED;
147 }
148 last_error_ = result.error();
149 return runner::outcome_of(*last_error_);
150 }

160 runner::SendOutcome cancel(const WorkingOrder<BrokerTag>& working) {
161 if (working.ref_.primary_.empty()) {
162 // **A reference naming nothing addresses nothing at a broker.** A row can reach here with an
163 // empty one: a runner reserves an order it has just sent so that later limits count it, and
164 // an executor that could not report what the broker issued leaves the reference blank.
165 // Sending that is a request which can only fail while looking like an attempt, and a
166 // cancellation failure is the one this library must not manufacture noise around. Refused as
167 // an answer rather than as an outage, because nothing is in doubt: the order was never
168 // addressed, so there is nothing to reconcile.
169 //
170 // The check is here and not in the runner because a reference is only meaningful to a
171 // broker. A replay's engine matches a cancellation on instrument and side, and refusing one
172 // there would leave a queued order to fill that the caller asked to withdraw.
173 last_error_ = BrokerError{};
174 last_error_->category_ = ErrorCategory::INVALID_ARGUMENT;
175 last_error_->reached_broker_ = false;
176 last_error_->message_ =
177 "no broker reference for this order; it cannot be withdrawn until a listing supplies one";
178 return runner::SendOutcome::REFUSED_LOCALLY;
179 }
180 auto result = session_->cancel(working.ref_, cancel_all(working));
181 if (result) {
182 last_error_.reset();
183 return runner::SendOutcome::ACCEPTED;
184 }
185 last_error_ = result.error();
186 return runner::outcome_of(*last_error_);
187 }
