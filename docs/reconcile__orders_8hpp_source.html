129inline bool same_value(const Decimal64& lhs, const Decimal64& rhs) {
130 return !decimal_less(lhs, rhs) && !decimal_less(rhs, lhs);
131}
132
134inline bool same_value(
135 const std::optional<Decimal64>& lhs, const std::optional<Decimal64>& rhs
136) {
137 if (!lhs.has_value() || !rhs.has_value()) {
138 return lhs.has_value() == rhs.has_value();
139 }
140 return same_value(*lhs, *rhs);
141}

209template <typename Enum>
210inline Evidence attribute_evidence(Enum row_value, Enum entry_value) noexcept {
211 if (row_value == Enum::UNKNOWN || entry_value == Enum::UNKNOWN) {
212 return Evidence::ABSENT;
213 }
214 return row_value == entry_value ? Evidence::CONFIRMS : Evidence::CONTRADICTS;
215}

232inline Evidence trigger_evidence(
233 const std::optional<Decimal64>& row_trigger,
234 const std::optional<Decimal64>& entry_trigger
235) {
236 if (!row_trigger) {
237 return entry_trigger ? Evidence::ABSENT : Evidence::CONFIRMS;
238 }
239 if (!entry_trigger) {
240 return Evidence::ABSENT;
241 }
242 return reconcile_detail::same_value(*row_trigger, *entry_trigger) ?
243 Evidence::CONFIRMS :
244 Evidence::CONTRADICTS;
245}

263inline Evidence time_evidence(
264 const OpenInstruction& instruction, const OrderSnapshot& row, const MatchWindow& window
265) {
266 if (row.ordered_date_assumed_) {
267 return Evidence::ABSENT;
268 }
269 if (!row.ordered_at_) {
270 return Evidence::CONFIRMS;
271 }
272 const auto begin = instruction.opened_at_ - window.clock_skew_;
273 const auto end = instruction_sent_by(instruction, window) + window.clock_skew_;
274 return (*row.ordered_at_ >= begin && *row.ordered_at_ <= end) ? Evidence::CONFIRMS
275 : Evidence::CONTRADICTS;
276}

288inline RouteEvidence route_evidence(
289 const OpenInstruction& instruction, const OrderSnapshot& row
290) {
291 const auto& asked = instruction.entry_.venue_;
292 if (!asked) {
293 // The line predates the field, so what it asked for is unknowable.
294 return RouteEvidence::ABSENT;
295 }
296 if (*asked == Venue::UNSPECIFIED) {
297 // It asked for nothing, so nothing about the row's venue can contradict it.
298 return RouteEvidence::CONFIRMS;
299 }
300 if (row.venue_ == Venue::UNSPECIFIED) {
301 // It asked for a venue and the broker did not say where the order went.
302 return RouteEvidence::ABSENT;
303 }
304 return row.venue_ == *asked ? RouteEvidence::CONFIRMS : RouteEvidence::CONTRADICTS;
305}

319inline bool row_corresponds(
320 const OpenInstruction& instruction, const OrderSnapshot& row, const MatchWindow& window
321) {
322 const auto& entry = instruction.entry_;
323 if (!(row.instrument_ == entry.instrument_)) {
324 return false;
325 }
326 if (row.side_ != entry.side_) {
327 return false;
328 }
329 // A known basis on both sides must agree. An `UNKNOWN` on either side is not
330 // a disagreement and is not a correspondence either: it is a row this
331 // library could not classify, and `row_might_correspond` below is what stops
332 // it being read as absence.
333 if (!shape_detail::both_known(row.type_, entry.type_)) {
334 return false;
335 }
336 if (row.type_ != entry.type_) {
337 return false;
338 }
339 if (!reconcile_detail::same_value(row.quantity_, entry.quantity_)) {
340 return false;
341 }
342 if (!reconcile_detail::same_value(row.price_, entry.price_)) {
343 return false;
344 }
345 // A match needs every attribute established, not merely not contradicted.
346 for (const auto evidence : {
347 attribute_evidence(row.time_in_force_, entry.time_in_force_),
348 attribute_evidence(row.session_, entry.session_),
349 trigger_evidence(row.trigger_price_, entry.trigger_price_),
350 route_evidence(instruction, row),
351 time_evidence(instruction, row, window),
352 }) {
353 if (evidence != Evidence::CONFIRMS) {
354 return false;
355 }
356 }
357 return true;
358}

374inline bool row_might_correspond(
375 const OpenInstruction& instruction, const OrderSnapshot& row, const MatchWindow& window
376) {
377 const auto& entry = instruction.entry_;
378 // Anything that positively disagrees rules the row out however much else is
379 // silent -- an unknown-code row at a venue the instruction did not use is not
380 // this order, and letting it force an ambiguity would stall recovery for
381 // nothing.
382 for (const auto evidence : {
383 attribute_evidence(row.type_, entry.type_),
384 attribute_evidence(row.time_in_force_, entry.time_in_force_),
385 attribute_evidence(row.session_, entry.session_),
386 trigger_evidence(row.trigger_price_, entry.trigger_price_),
387 route_evidence(instruction, row),
388 time_evidence(instruction, row, window),
389 }) {
390 if (evidence == Evidence::CONTRADICTS) {
391 return false;
392 }
393 }
394
395 const bool anything_silent =
396 attribute_evidence(row.type_, entry.type_) == Evidence::ABSENT ||
397 attribute_evidence(row.time_in_force_, entry.time_in_force_) == Evidence::ABSENT ||
398 attribute_evidence(row.session_, entry.session_) == Evidence::ABSENT ||
399 trigger_evidence(row.trigger_price_, entry.trigger_price_) == Evidence::ABSENT ||
400 route_evidence(instruction, row) == Evidence::ABSENT ||
401 time_evidence(instruction, row, window) == Evidence::ABSENT;
402 if (!anything_silent) {
403 // Everything comparable was compared, so `row_corresponds` has already
404 // given the definitive answer and this predicate has nothing to add.
405 return false;
406 }
407
408 if (!(row.instrument_ == entry.instrument_) || row.side_ != entry.side_) {
409 return false;
410 }
411 if (!reconcile_detail::same_value(row.quantity_, entry.quantity_)) {
412 return false;
413 }
414 // Compare what survived. A row whose *basis* is unreadable has no price
415 // either, because the parser discards it -- but one whose only silence is a
416 // route or a timestamp still has both, and a row differing there is plainly
417 // another order rather than an ambiguity worth withholding an instrument for.
418 if (attribute_evidence(row.type_, entry.type_) != Evidence::ABSENT) {
419 if (!reconcile_detail::same_value(row.price_, entry.price_)) {
420 return false;
421 }
422 }
423 return true;
424}

436inline std::string unresolvable_reason(const JournalEntry& entry) {
437 if (entry.action_ == JournalAction::CANCEL && entry.order_id_.empty()) {
438 return "a cancellation was journalled without the order id it names, so no row can be "
439 "identified as its target; an operator must say what became of it";
440 }
441 // **Placements only.** A cancellation is matched on the broker id alone --
442 // `row.order_id_ == entry.order_id_` and nothing else -- so its copy of the
443 // order's shape is never compared and an inconsistency in it cannot mislead
444 // anything. Applying the check to every action withheld an instrument whose
445 // identity the journal *had* recorded, for a cancellation that resolves
446 // without trouble: the shape is copied from the working order, so a
447 // reservation carrying an unpriced `LIMIT` or a reported `MARKET` price puts
448 // it there without the cancellation caring.
449 if (entry.action_ == JournalAction::PLACE && entry.type_ != OrderType::UNKNOWN) {
450 const bool priced = carries_caller_price(entry.type_);
451 if (priced && !entry.price_) {
452 return "a price-bearing order was journalled without its price, so no row can be matched "
453 "on it; an operator must say what became of it";
454 }
455 if (!priced && entry.price_) {
456 return "an order whose price the market sets was journalled carrying one, so no row can "
457 "be matched on it; an operator must say what became of it";
458 }
459 // The trigger's twin of the price rule, for the same reason: a line the writer emits before
460 // the binding could refuse it, which no row can be matched against.
461 const bool triggered = carries_trigger_price(entry.type_);
462 if (triggered && !entry.trigger_price_) {
463 return "a stop-limit order was journalled without its trigger price, so no row can be "
464 "matched on it; an operator must say what became of it";
465 }
466 if (!triggered && entry.trigger_price_) {
467 return "an order that takes no trigger price was journalled carrying one, so no row can be "
468 "matched on it; an operator must say what became of it";
469 }
470 }
471 return {};
472}

485inline std::vector<OpenInstruction> recorded_placements(const JournalReplay& replay) {
486 std::vector<OpenInstruction> found;
487 std::unordered_map<std::string, std::size_t> found_at;
488 std::vector<std::string> refused;
489 for (const auto& entry : replay.entries_) {
490 if (entry.action_ != JournalAction::PLACE || entry.requires_operator_) {
491 continue;
492 }
493 auto known = found_at.find(entry.id_);
494 if (known == found_at.end()) {
495 OpenInstruction placement;
496 placement.entry_ = entry;
497 placement.entry_.order_id_.clear();
498 placement.opened_at_ = entry.recorded_at_;
499 known = found_at.emplace(entry.id_, found.size()).first;
500 found.push_back(std::move(placement));
501 }
502 auto& placement = found[known->second];
503 if (entry.outcome_ == JournalOutcome::REJECTED) {
504 refused.push_back(entry.id_);
505 }
506 const bool answers_the_send = entry.outcome_ == JournalOutcome::ACCEPTED ||
507 entry.outcome_ == JournalOutcome::REJECTED || entry.outcome_ == JournalOutcome::INDETERMINATE;
508 if (answers_the_send && !placement.closed_at_) {
509 placement.closed_at_ = entry.recorded_at_;
510 }
511 const bool names_the_order = entry.outcome_ == JournalOutcome::ACCEPTED || entry.outcome_ == JournalOutcome::RESOLVED;
512 if (names_the_order && !entry.order_id_.empty() && placement.entry_.order_id_.empty()) {
513 placement.entry_.order_id_ = entry.order_id_;
514 }
515 }
516 found.erase(
517 std::remove_if(
518 found.begin(), found.end(),
519 [&](const OpenInstruction& placement) {
520 return placement.entry_.order_id_.empty() ||
521 std::find(refused.begin(), refused.end(), placement.entry_.id_) != refused.end();
522 }
523 ),
524 found.end()
525 );
526 return found;
527}

541inline const OpenInstruction* recorded_owner(
542 const OpenInstruction& instruction,
543 const OrderSnapshot& row,
544 const std::vector<OpenInstruction>& recorded,
545 const MatchWindow& window
546) {
547 if (row.order_id_.empty()) {
548 return nullptr;
549 }
550 for (const auto& other : recorded) {
551 if (other.entry_.id_ == instruction.entry_.id_ || other.entry_.order_id_ != row.order_id_) {
552 continue;
553 }
554 if (row_corresponds(other, row, window) || row_might_correspond(other, row, window)) {
555 return &other;
556 }
557 }
558 return nullptr;
559}

579inline OrderMatch match_against(
580 const OpenInstruction& instruction,
581 const std::vector<OrderSnapshot>& rows,
582 const std::vector<OpenInstruction>& recorded,
583 std::chrono::system_clock::time_point now,
584 const MatchWindow& window = {}
585) {
586 OrderMatch match;
587 match.instrument_ = instruction.entry_.instrument_;
588 match.entry_id_ = instruction.entry_.id_;
589
590 // **An instruction whose own line cannot support a conclusion.** These are
591 // written by this library -- `JournalingExecutor` appends the pending line
592 // before the binding runs, so a locally refused order is on disk before
593 // anything could object -- which is why the *reader* accepts them. What must
594 // not happen is resolving one: each lacks precisely the field the matcher
595 // would decide on, so a sweep would conclude from silence.
596 //
597 // An ordinary local refusal is unaffected: its own terminal line closes the
598 // pending one before any of this is reached. Only an instruction still open
599 // at recovery arrives here, and for those the honest answer is an operator.
600 if (const auto reason = unresolvable_reason(instruction.entry_); !reason.empty()) {
601 match.verdict_ = MatchVerdict::NEEDS_OPERATOR;
602 match.detail_ = reason;
603 return match;
604 }
605 if (instruction.entry_.requires_operator_) {
606 // Checked before anything else, so no window arithmetic runs on a timestamp that was never
607 // recorded. This is the entry the file-journal documentation calls "something nothing can ever
608 // resolve", and this check is what makes that true rather than aspirational.
609 match.verdict_ = MatchVerdict::NEEDS_OPERATOR;
610 match.detail_ = instruction.entry_.detail_.empty() ?
611 "recovered from a damaged journal line; only an operator can resolve it" :
612 instruction.entry_.detail_;
613 return match;
614 }
615
616 const OrderSnapshot* found = nullptr;
617 std::size_t unreadable = 0;
618 std::size_t claimed = 0;
619 const OrderSnapshot* claimed_row = nullptr;
620 const OpenInstruction* claimant = nullptr;
621 if (instruction.entry_.action_ == JournalAction::CANCEL) {
622 for (const auto& row : rows) {
623 if (!instruction.entry_.order_id_.empty() && row.order_id_ == instruction.entry_.order_id_) {
624 ++match.candidates_;
625 found = &row;
626 }
627 }
628 } else {
629 for (const auto& row : rows) {
630 const bool fits = row_corresponds(instruction, row, window);
631 // Counted separately, because these are not candidates for a *match* -- the
632 // library could not read their shape, so it cannot claim one is this order.
633 // They are candidates against concluding the order does not exist.
634 const bool might_fit = row_might_correspond(instruction, row, window);
635 if (!fits && !might_fit) {
636 continue;
637 }
638 // Another placement's order by the journal's own record: neither this order nor evidence
639 // that this order is missing.
640 if (const auto* owner = recorded_owner(instruction, row, recorded, window)) {
641 ++claimed;
642 claimed_row = &row;
643 claimant = owner;
644 continue;
645 }
646 if (fits) {
647 ++match.candidates_;
648 found = &row;
649 }
650 if (might_fit) {
651 ++unreadable;
652 }
653 }
654 }
655
656 if (claimed > 0) {
657 // Not `MATCHED`: the row is another placement's. Not `NOT_PLACED` either, even when nothing else
658 // fits, and not `MATCHED` on another row that fits: the record may be a sweep's mistaken name,
659 // and then the claimed row is this order and the other row is the claimant's.
660 match.verdict_ = MatchVerdict::AMBIGUOUS;
661 const auto others = match.candidates_ + unreadable;
662 match.candidates_ = others + claimed;
663 match.detail_ = "order " + claimed_row->order_id_ + " in the broker's listing fits this instruction, and the "
664 "journal records it for another placement (" + claimant->entry_.id_ + ")" +
665 (claimed > 1 ? ", as it does " + std::to_string(claimed - 1) + " more such row(s)" : std::string()) +
666 (others > 0 ? "; " + std::to_string(others) + " other row(s) fit as well" : std::string()) +
667 "; nothing was resolved";
668 return match;
669 }
670 if (match.candidates_ > 1) {
671 match.verdict_ = MatchVerdict::AMBIGUOUS;
672 match.detail_ = std::to_string(match.candidates_) +
673 " orders in the broker's listing could be this instruction; nothing was resolved";
674 return match;
675 }
676 if (unreadable > 0) {
677 // Report, never guess -- and **whether or not a readable row also matched**.
678 // A row whose order code this release does not publish is not evidence that
679 // the order is missing, so `NOT_PLACED` would clear the hold and send it
680 // again beside the one already resting. But it is equally not evidence that
681 // some *other* row is the order: with one readable row matching every
682 // attribute and one unreadable row matching every attribute that survives
683 // an unreadable shape, the unreadable one may be the instruction and the
684 // readable one another order on the same account. Resolving to the readable
685 // one would attribute the wrong order and clear the hold on it, which is
686 // the same harm by a different route.
687 match.verdict_ = MatchVerdict::AMBIGUOUS;
688 match.candidates_ += unreadable;
689 match.order_id_.clear();
690 // Says which it is. An operator reading "unrecognized order code" about a
691 // row whose code was perfectly readable, and whose *route* was the unknown,
692 // is being sent to look at the wrong field.
693 match.detail_ = std::to_string(unreadable) +
694 " order(s) in the broker's listing match this instruction on instrument, side, quantity "
695 "and time, but this library cannot confirm which order they are: an order code it does "
696 "not recognize, a route nobody recorded, or both" +
697 (found != nullptr ? ", and another row matches it fully" : "") +
698 "; nothing was resolved. Each row carries the broker's own code as shape_code_ and its "
699 "venue as venue_";
700 return match;
701 }
702 if (match.candidates_ == 1) {
703 match.verdict_ = MatchVerdict::MATCHED;
704 match.order_id_ = found->order_id_;
705 match.detail_ = instruction.entry_.action_ == JournalAction::CANCEL ?
706 "the order named by the cancellation is still in the broker's listing" :
707 "the broker's listing carries this order";
708 return match;
709 }
710 if (!match_window_closed(instruction, now, window)) {
711 match.verdict_ = MatchVerdict::UNDECIDED;
712 match.detail_ =
713 "no matching order yet, and not enough time has passed for the listing to show one";
714 return match;
715 }
716 match.verdict_ = MatchVerdict::NOT_PLACED;
717 match.detail_ = instruction.entry_.action_ == JournalAction::CANCEL ?
718 "the order named by the cancellation is no longer in the broker's listing" :
719 "no matching order was found among the rows read, within the matching window";
720 return match;
721}

767inline void drop_rows_fitting_another_placement(
768 std::vector<OrderSnapshot>& rows,
769 const JournalReplay& replay,
770 const std::vector<OpenInstruction>& matching,
771 const MatchWindow& window = {}
772) {
773 // The other placements on the instruments being matched, each with the span of its send.
774 std::vector<OpenInstruction> others;
775 std::unordered_map<std::string, std::size_t> other_of;
776 std::vector<std::string> refused;
777 for (const auto& entry : replay.entries_) {
778 if (entry.action_ != JournalAction::PLACE) {
779 continue;
780 }
781 const bool on_an_instrument_matched = std::any_of(
782 matching.begin(), matching.end(),
783 [&](const OpenInstruction& instruction) { return instruction.entry_.instrument_ == entry.instrument_; }
784 );
785 const bool being_matched = std::any_of(matching.begin(), matching.end(), [&](const OpenInstruction& instruction) {
786 return instruction.entry_.id_ == entry.id_;
787 });
788 if (!on_an_instrument_matched || being_matched) {
789 continue;
790 }
791 if (entry.outcome_ == JournalOutcome::REJECTED) {
792 refused.push_back(entry.id_);
793 }
794 auto known = other_of.find(entry.id_);
795 if (known == other_of.end()) {
796 OpenInstruction other;
797 other.entry_ = entry;
798 other.opened_at_ = entry.recorded_at_;
799 known = other_of.emplace(entry.id_, others.size()).first;
800 others.push_back(std::move(other));
801 }
802 auto& other = others[known->second];
803 const bool answers_the_send = entry.outcome_ == JournalOutcome::ACCEPTED ||
804 entry.outcome_ == JournalOutcome::REJECTED || entry.outcome_ == JournalOutcome::INDETERMINATE;
805 if (answers_the_send && !other.closed_at_) {
806 other.closed_at_ = entry.recorded_at_;
807 }
808 }
809 rows.erase(
810 std::remove_if(
811 rows.begin(), rows.end(),
812 [&](const OrderSnapshot& row) {
813 return std::any_of(others.begin(), others.end(), [&](const OpenInstruction& other) {
814 return std::find(refused.begin(), refused.end(), other.entry_.id_) == refused.end() &&
815 (row_corresponds(other, row, window) || row_might_correspond(other, row, window));
816 });
817 }
818 ),
819 rows.end()
820 );
821}

890template <typename Journal>
891ReconcileReport reconcile_against(
892 const std::vector<OpenInstruction>& open,
893 const std::vector<OrderSnapshot>& rows,
894 const JournalReplay& replay,
895 std::chrono::system_clock::time_point now,
896 IndeterminacyLedger& ledger,
897 Journal& journal,
898 const MatchWindow& window = {},
899 std::string_view unplaced_note = {}
900) {
901 ReconcileReport report;
902 report.matches_.reserve(open.size());
903 const auto recorded = recorded_placements(replay);
904
905 std::vector<InstrumentId> undecided;
906 std::vector<InstrumentId> seen;
907 for (const auto& instruction : open) {
908 auto match = match_against(instruction, rows, recorded, now, window);
909 const auto instrument = match.instrument_;
910 if (match.verdict_ == MatchVerdict::AMBIGUOUS) {
911 report.ambiguous_ = true;
912 }
913 if (match.verdict_ == MatchVerdict::NOT_PLACED && instruction.entry_.action_ == JournalAction::PLACE) {
914 match.detail_ += unplaced_note;
915 }
916 if (match_verdict_resolves(match.verdict_)) {
917 // Written before the ledger is cleared, for the same reason a placement is written before it
918 // is sent: the durable record has to be the conservative one. A sweep that cleared memory and
919 // failed to record why would leave the instruction owed on disk, and the next restart would
920 // withhold the instrument again for a question that had already been answered.
921 if (journal.append(resolution_entry(instruction, match, now))) {
922 ++report.recorded_;
923 } else {
924 report.unrecorded_ = true;
925 undecided.push_back(instrument);
926 match.detail_ += " (not recorded: the journal could not be written, so the hold stands)";
927 }
928 } else {
929 undecided.push_back(instrument);
930 }
931 report.matches_.push_back(std::move(match));
932
933 bool known = false;
934 for (const auto& id : seen) {
935 if (id == instrument) {
936 known = true;
937 break;
938 }
939 }
940 if (!known) {
941 seen.push_back(instrument);
942 }
943 }
944
945 for (const auto& instrument : seen) {

1065inline bool dates_listed_order(const OrderSnapshot& assumed, const OrderSnapshot& row) {
1066 if (row.ordered_date_assumed_ || !row.ordered_at_ || !assumed.ordered_at_ || assumed.order_id_.empty() ||
1067 row.order_id_ != assumed.order_id_ || !(row.instrument_ == assumed.instrument_)) {
1068 return false;
1069 }
1070 // Both stamped from a date and the same `HHmmss` in one zone, so one order's two rows differ by whole days.
1071 if ((*assumed.ordered_at_ - *row.ordered_at_) % std::chrono::hours(24) != std::chrono::hours(0)) {
1072 return false;
1073 }
1074 // A number reused on another day can belong to a different order at the same time of day.
1075 const bool venues_differ = row.venue_ != Venue::UNSPECIFIED && assumed.venue_ != Venue::UNSPECIFIED &&
1076 row.venue_ != assumed.venue_;
1077 return row.side_ == assumed.side_ &&
1078 reconcile_detail::same_value(row.quantity_, assumed.quantity_) &&
1079 reconcile_detail::same_value(row.price_, assumed.price_) &&
1080 attribute_evidence(row.type_, assumed.type_) != Evidence::CONTRADICTS &&
1081 attribute_evidence(row.time_in_force_, assumed.time_in_force_) != Evidence::CONTRADICTS &&
1082 attribute_evidence(row.session_, assumed.session_) != Evidence::CONTRADICTS &&
1083 trigger_evidence(row.trigger_price_, assumed.trigger_price_) != Evidence::CONTRADICTS &&
1084 !venues_differ;
1085}

1100inline void merge_dated_history(
1101 std::vector<OrderSnapshot>& into,
1102 const std::vector<OrderSnapshot>& rows
1103) {
1104 std::vector<OrderSnapshot> listing;
1105 listing.reserve(into.size() + rows.size());
1106 for (auto& listed : into) {
1107 const bool dated_by_history = listed.ordered_date_assumed_ &&
1108 std::any_of(rows.begin(), rows.end(), [&](const OrderSnapshot& row) { return dates_listed_order(listed, row); });
1109 if (!dated_by_history) {
1110 listing.push_back(std::move(listed));
1111 }
1112 }
1113 into = std::move(listing);
1114 const auto listed = into.size();
1115 for (const auto& row : rows) {
1116 bool known = false;
1117 for (std::size_t at = 0; at < listed; ++at) {
1118 // A listing row still assumed is no copy of anything: the dated row is the evidence, and both stay.
1119 if (!into[at].ordered_date_assumed_ && exact_copy(into[at], row)) {
1120 known = true;
1121 break;
1122 }
1123 }
1124 if (!known) {
1125 into.push_back(row);
1126 }
1127 }
1128}

1135inline void merge_rows(std::vector<OrderSnapshot>& into, const std::vector<OrderSnapshot>& rows) {
1136 for (const auto& row : rows) {
1137 bool known = false;
1138 for (const auto& held : into) {
1139 // Deduplicated by the broker's own id, because an order that is still open appears in *both*
1140 // listings. Two copies of one row would read as two candidates, and the matcher would report
1141 // `AMBIGUOUS` for an instruction that is not ambiguous at all.
1142 if (!row.order_id_.empty() && held.order_id_ == row.order_id_) {
1143 known = true;
1144 break;
1145 }
1146 }
1147 if (!known) {
1148 into.push_back(row);
1149 }
1150 }

1245inline RecoveryReport recover_into(const JournalReplay& replay, IndeterminacyLedger& ledger) {
1246 RecoveryReport report;
1247 report.truncated_ = replay.truncated_tail_;
1248 report.unreadable_lines_ = replay.unreadable_lines_;
1249 report.unreadable_file_ = !replay.readable();
1250 report.detail_ = replay.detail_;
1251 report.open_ = unresolved_instructions(replay);
1252
1253 for (const auto& instruction : report.open_) {
1254 const auto& instrument = instruction.entry_.instrument_;
1255 if (instrument.symbol_.empty()) {
1256 report.unattributable_ = true;
1257 continue;
1258 }

1260 ledger.mark(

1262 instruction.detail_.empty() ?

1264 instruction.detail_
1265 );
1266 if (!marked) {
1267 report.marked_.push_back(instrument);
1268 }
1269 }
1270 if (report.truncated_ && report.open_.empty()) {
1271 report.unattributable_ = true;
1272 }

1303template <typename Runner, typename Journal>
1304RecoveryReport recover_runner(Runner& runner_instance, const Journal& journal) {
1305 const auto replay = journal.replay();
1306 auto report = recover_into(replay, runner_instance.ledger());
1307 // A serialized runner also gets back every order it may still owe (`Runner::restore_obligations`),
1308 // and places nothing until this has run.
1309 runner_instance.restore_obligations(replay);
1310 if (report.unattributable_) {
1311 runner_instance.halt_locally(
1312 report.detail_.empty() ?
1313 "the order journal was damaged in a way that names no instrument" :
1314 report.detail_
1315 );
1316 }
1317 if (report.unreadable_lines_ > 0) {
1318 runner_instance.halt_locally(
1319 "the order journal holds " + std::to_string(report.unreadable_lines_) +
1320 " line(s) this library did not write, so it cannot account for what was in flight"
1321 );
1322 }
1323 if (report.unreadable_file_) {
1324 // Nothing about the file's contents is known, and unlike a missing journal this one may hold
1325 // every order a predecessor left in flight. There is nothing to withhold specifically.
1326 runner_instance.halt_locally(
1327 report.detail_.empty() ?
1328 "the order journal exists and could not be read" :
1329 report.detail_
1330 );
1331 }
1332 return report;
1333}
