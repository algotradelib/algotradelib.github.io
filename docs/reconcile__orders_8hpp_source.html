129inline bool same_value(const Decimal64& lhs, const Decimal64& rhs) {
130 return !decimal_less(lhs, rhs) && !decimal_less(rhs, lhs);
131}
132
134inline bool same_value(const std::optional<Decimal64>& lhs, const std::optional<Decimal64>& rhs) {
135 if (!lhs.has_value() || !rhs.has_value()) {
136 return lhs.has_value() == rhs.has_value();
137 }
138 return same_value(*lhs, *rhs);
139}

197inline constexpr bool evidence_confirms(Evidence evidence) noexcept {
198 switch (evidence) {
199 case Evidence::CONFIRMS:
200 return true;
201 case Evidence::CONTRADICTS:
202 case Evidence::ABSENT:

205 return false;

218 case Evidence::ABSENT:

231 switch (evidence) {

252inline Evidence attribute_evidence(Enum row_value, Enum entry_value) noexcept {
253 if (row_value == Enum::UNKNOWN || entry_value == Enum::UNKNOWN) {
254 return Evidence::ABSENT;

274inline Evidence trigger_evidence(
275 const std::optional<Decimal64>& row_trigger, const std::optional<Decimal64>& entry_trigger
276) {
277 if (!row_trigger) {
278 return entry_trigger ? Evidence::ABSENT : Evidence::CONFIRMS;
279 }
280 if (!entry_trigger) {
281 return Evidence::ABSENT;
282 }
283 return reconcile_detail::same_value(*row_trigger, *entry_trigger) ? Evidence::CONFIRMS : Evidence::CONTRADICTS;

302inline Evidence time_evidence(const OpenInstruction& instruction, const OrderSnapshot& row, const MatchWindow& window) {
303 if (row.ordered_date_assumed_) {
304 return Evidence::ABSENT;
305 }
306 if (!row.ordered_at_) {
307 return Evidence::CONFIRMS;
308 }
309 const auto begin = instruction.opened_at_ - window.clock_skew_;
310 const auto end = instruction_sent_by(instruction, window) + window.clock_skew_;
311 return (*row.ordered_at_ >= begin && *row.ordered_at_ <= end) ? Evidence::CONFIRMS : Evidence::CONTRADICTS;
312}

324inline RouteEvidence route_evidence(const OpenInstruction& instruction, const OrderSnapshot& row) {
325 const auto& asked = instruction.entry_.venue_;
326 if (!asked) {
327 // The line predates the field, so what it asked for is unknowable.
328 return RouteEvidence::ABSENT;
329 }
330 if (*asked == Venue::UNSPECIFIED) {
331 // It asked for nothing, so nothing about the row's venue can contradict it.
332 return RouteEvidence::CONFIRMS;
333 }
334 if (row.venue_ == Venue::UNSPECIFIED) {
335 // It asked for a venue and the broker did not say where the order went.
336 return RouteEvidence::ABSENT;
337 }
338 return row.venue_ == *asked ? RouteEvidence::CONFIRMS : RouteEvidence::CONTRADICTS;
339}

353inline bool row_corresponds(const OpenInstruction& instruction, const OrderSnapshot& row, const MatchWindow& window) {
354 const auto& entry = instruction.entry_;
355 if (!(row.instrument_ == entry.instrument_)) {
356 return false;
357 }
358 if (row.side_ != entry.side_) {
359 return false;
360 }
361 // A known basis on both sides must agree. An `UNKNOWN` on either side is not
362 // a disagreement and is not a correspondence either: it is a row this
363 // library could not classify, and `row_might_correspond` below is what stops
364 // it being read as absence.
365 if (!shape_detail::both_known(row.type_, entry.type_)) {
366 return false;
367 }
368 if (row.type_ != entry.type_) {
369 return false;
370 }
371 if (!reconcile_detail::same_value(row.quantity_, entry.quantity_)) {
372 return false;
373 }
374 if (!reconcile_detail::same_value(row.price_, entry.price_)) {
375 return false;
376 }
377 // A match needs every attribute established, not merely not contradicted.
378 for (
379 const auto evidence : {
380 attribute_evidence(row.time_in_force_, entry.time_in_force_),
381 attribute_evidence(row.session_, entry.session_),
382 trigger_evidence(row.trigger_price_, entry.trigger_price_),
383 route_evidence(instruction, row),
384 time_evidence(instruction, row, window),
385 }
386 ) {
387 if (!evidence_confirms(evidence)) {
388 return false;
389 }
390 }
391 return true;
392}

408inline bool row_might_correspond(
409 const OpenInstruction& instruction, const OrderSnapshot& row, const MatchWindow& window
410) {
411 const auto& entry = instruction.entry_;
412 // Anything that positively disagrees rules the row out however much else is
413 // silent -- an unknown-code row at a venue the instruction did not use is not
414 // this order, and letting it force an ambiguity would stall recovery for
415 // nothing.
416 for (
417 const auto evidence : {
418 attribute_evidence(row.type_, entry.type_),
419 attribute_evidence(row.time_in_force_, entry.time_in_force_),
420 attribute_evidence(row.session_, entry.session_),
421 trigger_evidence(row.trigger_price_, entry.trigger_price_),
422 route_evidence(instruction, row),
423 time_evidence(instruction, row, window),
424 }
425 ) {
426 if (evidence_contradicts(evidence)) {
427 return false;
428 }
429 }
430
431 const bool anything_silent = evidence_leaves_open(attribute_evidence(row.type_, entry.type_)) ||
432 evidence_leaves_open(attribute_evidence(row.time_in_force_, entry.time_in_force_)) ||
433 evidence_leaves_open(attribute_evidence(row.session_, entry.session_)) ||
434 evidence_leaves_open(trigger_evidence(row.trigger_price_, entry.trigger_price_)) ||
435 evidence_leaves_open(route_evidence(instruction, row)) ||
436 evidence_leaves_open(time_evidence(instruction, row, window));
437 if (!anything_silent) {
438 // Everything comparable was compared, so `row_corresponds` has already
439 // given the definitive answer and this predicate has nothing to add.
440 return false;
441 }
442
443 if (!(row.instrument_ == entry.instrument_) || row.side_ != entry.side_) {
444 return false;
445 }
446 if (!reconcile_detail::same_value(row.quantity_, entry.quantity_)) {
447 return false;
448 }
449 // Compare what survived. A row whose *basis* is unreadable has no price
450 // either, because the parser discards it -- but one whose only silence is a
451 // route or a timestamp still has both, and a row differing there is plainly
452 // another order rather than an ambiguity worth withholding an instrument for.
453 if (!evidence_leaves_open(attribute_evidence(row.type_, entry.type_))) {
454 if (!reconcile_detail::same_value(row.price_, entry.price_)) {
455 return false;
456 }
457 }
458 return true;
459}

471inline std::string unresolvable_reason(const JournalEntry& entry) {
472 const bool by_order_id = journal_action_matches_by_order_id(entry.action_);
473 if (by_order_id && entry.order_id_.empty()) {
474 return "a cancellation was journalled without the order id it names, so no row can be identified as its target; an operator must say what became of it";
475 }
476 // **Placements only.** A cancellation is matched on the broker id alone --
477 // `row.order_id_ == entry.order_id_` and nothing else -- so its copy of the
478 // order's shape is never compared and an inconsistency in it cannot mislead
479 // anything. Applying the check to every action withheld an instrument whose
480 // identity the journal *had* recorded, for a cancellation that resolves
481 // without trouble: the shape is copied from the working order, so a
482 // reservation carrying an unpriced `LIMIT` or a reported `MARKET` price puts
483 // it there without the cancellation caring.
484 if (!by_order_id && entry.type_ != OrderType::UNKNOWN) {
485 const bool priced = carries_caller_price(entry.type_);
486 if (priced && !entry.price_) {
487 return "a price-bearing order was journalled without its price, so no row can be matched on it; an operator must say what became of it";
488 }
489 if (!priced && entry.price_) {
490 return "an order whose price the market sets was journalled carrying one, so no row can be matched on it; an operator must say what became of it";
491 }
492 // The trigger's twin of the price rule, for the same reason: a line the writer emits before
493 // the binding could refuse it, which no row can be matched against.
494 const bool triggered = carries_trigger_price(entry.type_);
495 if (triggered && !entry.trigger_price_) {
496 return "a stop-limit order was journalled without its trigger price, so no row can be matched on it; an operator must say what became of it";
497 }
498 if (!triggered && entry.trigger_price_) {
499 return "an order that takes no trigger price was journalled carrying one, so no row can be matched on it; an operator must say what became of it";
500 }
501 }
502 return {};
503}

516inline std::vector<OpenInstruction> recorded_placements(const JournalReplay& replay) {
517 std::vector<OpenInstruction> found;
518 std::unordered_map<std::string, std::size_t> found_at;
519 std::vector<std::string> refused;
520 for (const auto& entry : replay.entries_) {
521 if (entry.action_ != JournalAction::PLACE || entry.requires_operator_) {
522 continue;
523 }
524 auto known = found_at.find(entry.id_);
525 if (known == found_at.end()) {
526 OpenInstruction placement;
527 placement.entry_ = entry;
528 placement.entry_.order_id_.clear();
529 placement.opened_at_ = entry.recorded_at_;
530 known = found_at.emplace(entry.id_, found.size()).first;
531 found.push_back(std::move(placement));
532 }
533 auto& placement = found[known->second];
534 if (journal_outcome_refuses(entry.outcome_)) {
535 refused.push_back(entry.id_);
536 }
537 if (journal_outcome_answers_send(entry.outcome_) && !placement.closed_at_) {
538 placement.closed_at_ = entry.recorded_at_;
539 }
540 if (journal_outcome_names_order(entry.outcome_) && !entry.order_id_.empty() && placement.entry_.order_id_.empty()) {
541 placement.entry_.order_id_ = entry.order_id_;
542 }
543 }
544 found.erase(
545 std::remove_if(
546 found.begin(), found.end(),
547 [&](const OpenInstruction& placement) {
548 return placement.entry_.order_id_.empty() ||
549 std::find(refused.begin(), refused.end(), placement.entry_.id_) != refused.end();
550 }
551 ),
552 found.end()
553 );
554 return found;
555}

569inline const OpenInstruction* recorded_owner(
570 const OpenInstruction& instruction, const OrderSnapshot& row, const std::vector<OpenInstruction>& recorded,
571 const MatchWindow& window
572) {
573 if (row.order_id_.empty()) {
574 return nullptr;
575 }
576 for (const auto& other : recorded) {
577 if (other.entry_.id_ == instruction.entry_.id_ || other.entry_.order_id_ != row.order_id_) {
578 continue;
579 }
580 if (row_corresponds(other, row, window) || row_might_correspond(other, row, window)) {
581 return &other;
582 }
583 }
584 return nullptr;

605inline OrderMatch match_against(
606 const OpenInstruction& instruction, const std::vector<OrderSnapshot>& rows,
607 const std::vector<OpenInstruction>& recorded, std::chrono::system_clock::time_point now,
608 const MatchWindow& window = {}
609) {
610 OrderMatch match;
611 match.instrument_ = instruction.entry_.instrument_;
612 match.entry_id_ = instruction.entry_.id_;
613
614 // **An instruction whose own line cannot support a conclusion.** These are
615 // written by this library -- `JournalingExecutor` appends the pending line
616 // before the binding runs, so a locally refused order is on disk before
617 // anything could object -- which is why the *reader* accepts them. What must
618 // not happen is resolving one: each lacks precisely the field the matcher
619 // would decide on, so a sweep would conclude from silence.
620 //
621 // An ordinary local refusal is unaffected: its own terminal line closes the
622 // pending one before any of this is reached. Only an instruction still open
623 // at recovery arrives here, and for those the honest answer is an operator.
624 if (const auto reason = unresolvable_reason(instruction.entry_); !reason.empty()) {
625 match.verdict_ = MatchVerdict::NEEDS_OPERATOR;
626 match.detail_ = reason;
627 return match;
628 }
629 if (instruction.entry_.requires_operator_) {
630 // Checked before anything else, so no window arithmetic runs on a timestamp that was never
631 // recorded. This is the entry the file-journal documentation calls "something nothing can ever
632 // resolve", and this check is what makes that true rather than aspirational.
633 match.verdict_ = MatchVerdict::NEEDS_OPERATOR;
634 match.detail_ = instruction.entry_.detail_.empty() ?
635 "recovered from a damaged journal line; only an operator can resolve it" :
636 instruction.entry_.detail_;
637 return match;
638 }
639
640 const OrderSnapshot* found = nullptr;
641 std::size_t unreadable = 0;
642 std::size_t claimed = 0;
643 const OrderSnapshot* claimed_row = nullptr;
644 const OpenInstruction* claimant = nullptr;
645 const bool by_order_id = journal_action_matches_by_order_id(instruction.entry_.action_);
646 if (by_order_id) {
647 for (const auto& row : rows) {
648 if (!instruction.entry_.order_id_.empty() && row.order_id_ == instruction.entry_.order_id_) {
649 ++match.candidates_;
650 found = &row;
651 }
652 }
653 } else {
654 for (const auto& row : rows) {
655 const bool fits = row_corresponds(instruction, row, window);
656 // Counted separately, because these are not candidates for a *match* -- the
657 // library could not read their shape, so it cannot claim one is this order.
658 // They are candidates against concluding the order does not exist.
659 const bool might_fit = row_might_correspond(instruction, row, window);
660 if (!fits && !might_fit) {
661 continue;
662 }
663 // Another placement's order by the journal's own record: neither this order nor evidence
664 // that this order is missing.
665 if (const auto* owner = recorded_owner(instruction, row, recorded, window)) {
666 ++claimed;
667 claimed_row = &row;
668 claimant = owner;
669 continue;
670 }
671 if (fits) {
672 ++match.candidates_;
673 found = &row;
674 }
675 if (might_fit) {
676 ++unreadable;
677 }
678 }
679 }
680
681 if (claimed > 0) {
682 // Not `MATCHED`: the row is another placement's. Not `NOT_PLACED` either, even when nothing else
683 // fits, and not `MATCHED` on another row that fits: the record may be a sweep's mistaken name,
684 // and then the claimed row is this order and the other row is the claimant's.
685 match.verdict_ = MatchVerdict::AMBIGUOUS;
686 const auto others = match.candidates_ + unreadable;
687 match.candidates_ = others + claimed;
688 match.detail_ = "order " + claimed_row->order_id_ +
689 " in the broker's listing fits this instruction, and the "
690 "journal records it for another placement (" +
691 claimant->entry_.id_ + ")" +
692 (claimed > 1 ? ", as it does " + std::to_string(claimed - 1) + " more such row(s)" : std::string()) +
693 (others > 0 ? "; " + std::to_string(others) + " other row(s) fit as well" : std::string()) +
694 "; nothing was resolved";
695 return match;
696 }
697 if (match.candidates_ > 1) {
698 match.verdict_ = MatchVerdict::AMBIGUOUS;
699 match.detail_ = std::to_string(match.candidates_) +
700 " orders in the broker's listing could be this instruction; nothing was resolved";
701 return match;
702 }
703 if (unreadable > 0) {
704 // Report, never guess -- and **whether or not a readable row also matched**.
705 // A row whose order code this release does not publish is not evidence that
706 // the order is missing, so `NOT_PLACED` would clear the hold and send it
707 // again beside the one already resting. But it is equally not evidence that
708 // some *other* row is the order: with one readable row matching every
709 // attribute and one unreadable row matching every attribute that survives
710 // an unreadable shape, the unreadable one may be the instruction and the
711 // readable one another order on the same account. Resolving to the readable
712 // one would attribute the wrong order and clear the hold on it, which is
713 // the same harm by a different route.
714 match.verdict_ = MatchVerdict::AMBIGUOUS;
715 match.candidates_ += unreadable;
716 match.order_id_.clear();
717 // Says which it is. An operator reading "unrecognized order code" about a
718 // row whose code was perfectly readable, and whose *route* was the unknown,
719 // is being sent to look at the wrong field.
720 match.detail_ = std::to_string(unreadable) +
721 " order(s) in the broker's listing match this instruction on instrument, side, quantity "
722 "and time, but this library cannot confirm which order they are: an order code it does "
723 "not recognize, a route nobody recorded, or both" +
724 (found != nullptr ? ", and another row matches it fully" : "") +
725 "; nothing was resolved. Each row carries the broker's own code as shape_code_ and its "
726 "venue as venue_";
727 return match;
728 }
729 if (match.candidates_ == 1) {
730 match.verdict_ = MatchVerdict::MATCHED;
731 match.order_id_ = found->order_id_;
732 match.detail_ = by_order_id ?
733 "the order named by the cancellation is still in the broker's listing" :
734 "the broker's listing carries this order";
735 return match;
736 }
737 if (!match_window_closed(instruction, now, window)) {
738 match.verdict_ = MatchVerdict::UNDECIDED;
739 match.detail_ = "no matching order yet, and not enough time has passed for the listing to show one";
740 return match;
741 }
742 match.verdict_ = MatchVerdict::NOT_PLACED;
743 match.detail_ = by_order_id ?
744 "the order named by the cancellation is no longer in the broker's listing" :
745 "no matching order was found among the rows read, within the matching window";
746 return match;
747}

791inline void drop_rows_fitting_another_placement(
792 std::vector<OrderSnapshot>& rows, const JournalReplay& replay, const std::vector<OpenInstruction>& matching,
793 const MatchWindow& window = {}
794) {
795 // The other placements on the instruments being matched, each with the span of its send.
796 std::vector<OpenInstruction> others;
797 std::unordered_map<std::string, std::size_t> other_of;
798 std::vector<std::string> refused;
799 for (const auto& entry : replay.entries_) {
800 if (entry.action_ != JournalAction::PLACE) {
801 continue;
802 }
803 const bool on_an_instrument_matched =
804 std::any_of(matching.begin(), matching.end(), [&](const OpenInstruction& instruction) {
805 return instruction.entry_.instrument_ == entry.instrument_;
806 });
807 const bool being_matched = std::any_of(matching.begin(), matching.end(), [&](const OpenInstruction& instruction) {
808 return instruction.entry_.id_ == entry.id_;
809 });
810 if (!on_an_instrument_matched || being_matched) {
811 continue;
812 }
813 if (journal_outcome_refuses(entry.outcome_)) {
814 refused.push_back(entry.id_);
815 }
816 auto known = other_of.find(entry.id_);
817 if (known == other_of.end()) {
818 OpenInstruction other;
819 other.entry_ = entry;
820 other.opened_at_ = entry.recorded_at_;
821 known = other_of.emplace(entry.id_, others.size()).first;
822 others.push_back(std::move(other));
823 }
824 auto& other = others[known->second];
825 if (journal_outcome_answers_send(entry.outcome_) && !other.closed_at_) {
826 other.closed_at_ = entry.recorded_at_;
827 }
828 }
829 rows.erase(
830 std::remove_if(
831 rows.begin(), rows.end(),
832 [&](const OrderSnapshot& row) {
833 return std::any_of(others.begin(), others.end(), [&](const OpenInstruction& other) {
834 return std::find(refused.begin(), refused.end(), other.entry_.id_) == refused.end() &&
835 (row_corresponds(other, row, window) || row_might_correspond(other, row, window));
836 });
837 }
838 ),
839 rows.end()
840 );
841}

910template <typename Journal>
911ReconcileReport reconcile_against(
912 const std::vector<OpenInstruction>& open, const std::vector<OrderSnapshot>& rows, const JournalReplay& replay,
913 std::chrono::system_clock::time_point now, IndeterminacyLedger& ledger, Journal& journal,
914 const MatchWindow& window = {}, std::string_view unplaced_note = {}
915) {
916 ReconcileReport report;
917 report.matches_.reserve(open.size());
918 const auto recorded = recorded_placements(replay);
919
920 std::vector<InstrumentId> undecided;
921 std::vector<InstrumentId> seen;
922 for (const auto& instruction : open) {
923 auto match = match_against(instruction, rows, recorded, now, window);
924 const auto instrument = match.instrument_;
925 if (match.verdict_ == MatchVerdict::AMBIGUOUS) {
926 report.ambiguous_ = true;
927 }
928 if (match.verdict_ == MatchVerdict::NOT_PLACED && instruction.entry_.action_ == JournalAction::PLACE) {
929 match.detail_ += unplaced_note;
930 }
931 if (match_verdict_resolves(match.verdict_)) {
932 // Written before the ledger is cleared, for the same reason a placement is written before it
933 // is sent: the durable record has to be the conservative one. A sweep that cleared memory and
934 // failed to record why would leave the instruction owed on disk, and the next restart would
935 // withhold the instrument again for a question that had already been answered.
936 if (journal.append(resolution_entry(instruction, match, now))) {
937 ++report.recorded_;
938 } else {
939 report.unrecorded_ = true;
940 undecided.push_back(instrument);
941 match.detail_ += " (not recorded: the journal could not be written, so the hold stands)";
942 }
943 } else {
944 undecided.push_back(instrument);
945 }
946 report.matches_.push_back(std::move(match));
947
948 bool known = false;
949 for (const auto& id : seen) {

965 break;

972 ledger.resolve_after_reading_orders(instrument);
973 report.resolved_.push_back(instrument);
974 }
975 return report;

1073inline bool dates_listed_order(const OrderSnapshot& assumed, const OrderSnapshot& row) {
1074 if (
1075 row.ordered_date_assumed_ || !row.ordered_at_ || !assumed.ordered_at_ || assumed.order_id_.empty() ||
1076 row.order_id_ != assumed.order_id_ || !(row.instrument_ == assumed.instrument_)
1077 ) {
1078 return false;
1079 }
1080 // Both stamped from a date and the same `HHmmss` in one zone, so one order's two rows differ by whole days.
1081 if ((*assumed.ordered_at_ - *row.ordered_at_) % std::chrono::hours(24) != std::chrono::hours(0)) {
1082 return false;
1083 }
1084 // A number reused on another day can belong to a different order at the same time of day.
1085 const bool venues_differ =
1086 row.venue_ != Venue::UNSPECIFIED && assumed.venue_ != Venue::UNSPECIFIED && row.venue_ != assumed.venue_;
1087 return row.side_ == assumed.side_ && reconcile_detail::same_value(row.quantity_, assumed.quantity_) &&
1088 reconcile_detail::same_value(row.price_, assumed.price_) &&
1089 !evidence_contradicts(attribute_evidence(row.type_, assumed.type_)) &&
1090 !evidence_contradicts(attribute_evidence(row.time_in_force_, assumed.time_in_force_)) &&
1091 !evidence_contradicts(attribute_evidence(row.session_, assumed.session_)) &&
1092 !evidence_contradicts(trigger_evidence(row.trigger_price_, assumed.trigger_price_)) && !venues_differ;
1093}

1108inline void merge_dated_history(std::vector<OrderSnapshot>& into, const std::vector<OrderSnapshot>& rows) {
1109 std::vector<OrderSnapshot> listing;
1110 listing.reserve(into.size() + rows.size());
1111 for (auto& listed : into) {
1112 const bool dated_by_history =
1113 listed.ordered_date_assumed_ && std::any_of(rows.begin(), rows.end(), [&](const OrderSnapshot& row) {
1114 return dates_listed_order(listed, row);
1115 });
1116 if (!dated_by_history) {
1117 listing.push_back(std::move(listed));
1118 }
1119 }
1120 into = std::move(listing);
1121 const auto listed = into.size();
1122 for (const auto& row : rows) {
1123 bool known = false;
1124 for (std::size_t at = 0; at < listed; ++at) {
1125 // A listing row still assumed is no copy of anything: the dated row is the evidence, and both stay.
1126 if (!into[at].ordered_date_assumed_ && exact_copy(into[at], row)) {
1127 known = true;
1128 break;
1129 }
1130 }
1131 if (!known) {
1132 into.push_back(row);
1133 }
1134 }
1135}

1142inline void merge_rows(std::vector<OrderSnapshot>& into, const std::vector<OrderSnapshot>& rows) {
1143 for (const auto& row : rows) {
1144 bool known = false;
1145 for (const auto& held : into) {
1146 // Deduplicated by the broker's own id, because an order that is still open appears in *both*
1147 // listings. Two copies of one row would read as two candidates, and the matcher would report
1148 // `AMBIGUOUS` for an instruction that is not ambiguous at all.
1149 if (!row.order_id_.empty() && held.order_id_ == row.order_id_) {
1150 known = true;
1151 break;
1152 }
1153 }
1154 if (!known) {
1155 into.push_back(row);
1156 }
1157 }

1191inline std::pair<Date, Date> history_range_for(const std::vector<OpenInstruction>& open, const MatchWindow& window) {

1249inline RecoveryReport recover_into(const JournalReplay& replay, IndeterminacyLedger& ledger) {
1250 RecoveryReport report;
1251 report.truncated_ = replay.truncated_tail_;
1252 report.unreadable_lines_ = replay.unreadable_lines_;
1253 report.unreadable_file_ = !replay.readable();
1254 report.detail_ = replay.detail_;
1255 report.open_ = unresolved_instructions(replay);
1256
1257 for (const auto& instruction : report.open_) {
1258 const auto& instrument = instruction.entry_.instrument_;
1259 if (instrument.symbol_.empty()) {
1260 report.unattributable_ = true;
1261 continue;
1262 }
1263 const auto marked = ledger.requires_reconcile(instrument);
1264 ledger.mark(
1265 instrument,
1266 instruction.detail_.empty() ?
1267 "a previous process left this instruction unresolved in the order journal" :
1268 instruction.detail_
1269 );
1270 if (!marked) {
1271 report.marked_.push_back(instrument);
1272 }
1273 }
1274 if (report.truncated_ && report.open_.empty()) {
1275 report.unattributable_ = true;
1276 }
1277 return report;
1278}

1307template <typename Runner, typename Journal>
1308RecoveryReport recover_runner(Runner& runner_instance, const Journal& journal) {
1309 const auto& replay = journal.replay();
1310 auto report = recover_into(replay, runner_instance.ledger());
1311 // A serialized runner also gets back every order it may still owe (`Runner::restore_obligations`),
1312 // and places nothing until this has run.
1313 runner_instance.restore_obligations(replay);
1314 if (report.unattributable_) {
1315 runner_instance.halt_locally(
1316 report.detail_.empty() ? "the order journal was damaged in a way that names no instrument" : report.detail_
1317 );
1318 }
1319 if (report.unreadable_lines_ > 0) {
1320 runner_instance.halt_locally(
1321 "the order journal holds " + std::to_string(report.unreadable_lines_) +
1322 " line(s) this library did not write, so it cannot account for what was in flight"
1323 );
1324 }
1325 if (report.unreadable_file_) {
1326 // Nothing about the file's contents is known, and unlike a missing journal this one may hold
1327 // every order a predecessor left in flight. There is nothing to withhold specifically.
1328 runner_instance.halt_locally(
1329 report.detail_.empty() ? "the order journal exists and could not be read" : report.detail_
1330 );
1331 }
1332 return report;
1333}
