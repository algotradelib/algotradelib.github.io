129inline bool same_value(const Decimal64& lhs, const Decimal64& rhs) {
130 return !decimal_less(lhs, rhs) && !decimal_less(rhs, lhs);

134inline bool same_value(
135 const std::optional<Decimal64>& lhs, const std::optional<Decimal64>& rhs
136) {
137 if (!lhs.has_value() || !rhs.has_value()) {
138 return lhs.has_value() == rhs.has_value();

204inline constexpr bool evidence_confirms(Evidence evidence) noexcept {
205 switch (evidence) {
206 case Evidence::CONFIRMS:
207 return true;
208 case Evidence::CONTRADICTS:
209 case Evidence::ABSENT:

221 switch (evidence) {

224 case Evidence::CONFIRMS:
225 case Evidence::ABSENT:

228 return false;

238 switch (evidence) {

281inline Evidence trigger_evidence(
282 const std::optional<Decimal64>& row_trigger,
283 const std::optional<Decimal64>& entry_trigger
284) {
285 if (!row_trigger) {
286 return entry_trigger ? Evidence::ABSENT : Evidence::CONFIRMS;
287 }

312inline Evidence time_evidence(
313 const OpenInstruction& instruction, const OrderSnapshot& row, const MatchWindow& window
314) {
315 if (row.ordered_date_assumed_) {
316 return Evidence::ABSENT;
317 }
318 if (!row.ordered_at_) {
319 return Evidence::CONFIRMS;
320 }
321 const auto begin = instruction.opened_at_ - window.clock_skew_;
322 const auto end = instruction_sent_by(instruction, window) + window.clock_skew_;
323 return (*row.ordered_at_ >= begin && *row.ordered_at_ <= end) ? Evidence::CONFIRMS
324 : Evidence::CONTRADICTS;
325}

337inline RouteEvidence route_evidence(
338 const OpenInstruction& instruction, const OrderSnapshot& row
339) {
340 const auto& asked = instruction.entry_.venue_;
341 if (!asked) {
342 // The line predates the field, so what it asked for is unknowable.
343 return RouteEvidence::ABSENT;
344 }
345 if (*asked == Venue::UNSPECIFIED) {
346 // It asked for nothing, so nothing about the row's venue can contradict it.
347 return RouteEvidence::CONFIRMS;
348 }
349 if (row.venue_ == Venue::UNSPECIFIED) {
350 // It asked for a venue and the broker did not say where the order went.
351 return RouteEvidence::ABSENT;
352 }
353 return row.venue_ == *asked ? RouteEvidence::CONFIRMS : RouteEvidence::CONTRADICTS;
354}

368inline bool row_corresponds(
369 const OpenInstruction& instruction, const OrderSnapshot& row, const MatchWindow& window
370) {
371 const auto& entry = instruction.entry_;
372 if (!(row.instrument_ == entry.instrument_)) {
373 return false;
374 }
375 if (row.side_ != entry.side_) {
376 return false;
377 }
378 // A known basis on both sides must agree. An `UNKNOWN` on either side is not
379 // a disagreement and is not a correspondence either: it is a row this
380 // library could not classify, and `row_might_correspond` below is what stops
381 // it being read as absence.
382 if (!shape_detail::both_known(row.type_, entry.type_)) {
383 return false;
384 }
385 if (row.type_ != entry.type_) {
386 return false;
387 }
388 if (!reconcile_detail::same_value(row.quantity_, entry.quantity_)) {
389 return false;
390 }
391 if (!reconcile_detail::same_value(row.price_, entry.price_)) {
392 return false;
393 }
394 // A match needs every attribute established, not merely not contradicted.
395 for (const auto evidence : {
396 attribute_evidence(row.time_in_force_, entry.time_in_force_),
397 attribute_evidence(row.session_, entry.session_),
398 trigger_evidence(row.trigger_price_, entry.trigger_price_),
399 route_evidence(instruction, row),
400 time_evidence(instruction, row, window),
401 }) {
402 if (!evidence_confirms(evidence)) {
403 return false;
404 }
405 }
406 return true;
407}

423inline bool row_might_correspond(
424 const OpenInstruction& instruction, const OrderSnapshot& row, const MatchWindow& window
425) {
426 const auto& entry = instruction.entry_;
427 // Anything that positively disagrees rules the row out however much else is
428 // silent -- an unknown-code row at a venue the instruction did not use is not
429 // this order, and letting it force an ambiguity would stall recovery for
430 // nothing.
431 for (const auto evidence : {
432 attribute_evidence(row.type_, entry.type_),
433 attribute_evidence(row.time_in_force_, entry.time_in_force_),
434 attribute_evidence(row.session_, entry.session_),
435 trigger_evidence(row.trigger_price_, entry.trigger_price_),
436 route_evidence(instruction, row),
437 time_evidence(instruction, row, window),
438 }) {
439 if (evidence_contradicts(evidence)) {
440 return false;
441 }
442 }
443
444 const bool anything_silent =
445 evidence_leaves_open(attribute_evidence(row.type_, entry.type_)) ||
446 evidence_leaves_open(attribute_evidence(row.time_in_force_, entry.time_in_force_)) ||
447 evidence_leaves_open(attribute_evidence(row.session_, entry.session_)) ||
448 evidence_leaves_open(trigger_evidence(row.trigger_price_, entry.trigger_price_)) ||
449 evidence_leaves_open(route_evidence(instruction, row)) ||
450 evidence_leaves_open(time_evidence(instruction, row, window));
451 if (!anything_silent) {
452 // Everything comparable was compared, so `row_corresponds` has already
453 // given the definitive answer and this predicate has nothing to add.
454 return false;
455 }
456
457 if (!(row.instrument_ == entry.instrument_) || row.side_ != entry.side_) {
458 return false;
459 }
460 if (!reconcile_detail::same_value(row.quantity_, entry.quantity_)) {
461 return false;
462 }
463 // Compare what survived. A row whose *basis* is unreadable has no price
464 // either, because the parser discards it -- but one whose only silence is a
465 // route or a timestamp still has both, and a row differing there is plainly
466 // another order rather than an ambiguity worth withholding an instrument for.
467 if (!evidence_leaves_open(attribute_evidence(row.type_, entry.type_))) {
468 if (!reconcile_detail::same_value(row.price_, entry.price_)) {
469 return false;
470 }
471 }
472 return true;
473}

485inline std::string unresolvable_reason(const JournalEntry& entry) {
486 const bool by_order_id = journal_action_matches_by_order_id(entry.action_);
487 if (by_order_id && entry.order_id_.empty()) {
488 return "a cancellation was journalled without the order id it names, so no row can be "
489 "identified as its target; an operator must say what became of it";
490 }
491 // **Placements only.** A cancellation is matched on the broker id alone --
492 // `row.order_id_ == entry.order_id_` and nothing else -- so its copy of the
493 // order's shape is never compared and an inconsistency in it cannot mislead
494 // anything. Applying the check to every action withheld an instrument whose
495 // identity the journal *had* recorded, for a cancellation that resolves
496 // without trouble: the shape is copied from the working order, so a
497 // reservation carrying an unpriced `LIMIT` or a reported `MARKET` price puts
498 // it there without the cancellation caring.
499 if (!by_order_id && entry.type_ != OrderType::UNKNOWN) {
500 const bool priced = carries_caller_price(entry.type_);
501 if (priced && !entry.price_) {
502 return "a price-bearing order was journalled without its price, so no row can be matched "
503 "on it; an operator must say what became of it";
504 }
505 if (!priced && entry.price_) {
506 return "an order whose price the market sets was journalled carrying one, so no row can "
507 "be matched on it; an operator must say what became of it";
508 }
509 // The trigger's twin of the price rule, for the same reason: a line the writer emits before
510 // the binding could refuse it, which no row can be matched against.
511 const bool triggered = carries_trigger_price(entry.type_);
512 if (triggered && !entry.trigger_price_) {
513 return "a stop-limit order was journalled without its trigger price, so no row can be "
514 "matched on it; an operator must say what became of it";
515 }
516 if (!triggered && entry.trigger_price_) {
517 return "an order that takes no trigger price was journalled carrying one, so no row can be "
518 "matched on it; an operator must say what became of it";
519 }
520 }
521 return {};
522}

535inline std::vector<OpenInstruction> recorded_placements(const JournalReplay& replay) {
536 std::vector<OpenInstruction> found;
537 std::unordered_map<std::string, std::size_t> found_at;
538 std::vector<std::string> refused;
539 for (const auto& entry : replay.entries_) {
540 if (entry.action_ != JournalAction::PLACE || entry.requires_operator_) {
541 continue;
542 }
543 auto known = found_at.find(entry.id_);
544 if (known == found_at.end()) {
545 OpenInstruction placement;
546 placement.entry_ = entry;
547 placement.entry_.order_id_.clear();
548 placement.opened_at_ = entry.recorded_at_;
549 known = found_at.emplace(entry.id_, found.size()).first;
550 found.push_back(std::move(placement));
551 }
552 auto& placement = found[known->second];
553 if (journal_outcome_refuses(entry.outcome_)) {
554 refused.push_back(entry.id_);
555 }
556 if (journal_outcome_answers_send(entry.outcome_) && !placement.closed_at_) {
557 placement.closed_at_ = entry.recorded_at_;
558 }
559 if (journal_outcome_names_order(entry.outcome_) && !entry.order_id_.empty() && placement.entry_.order_id_.empty()) {
560 placement.entry_.order_id_ = entry.order_id_;
561 }
562 }
563 found.erase(
564 std::remove_if(
565 found.begin(), found.end(),
566 [&](const OpenInstruction& placement) {
567 return placement.entry_.order_id_.empty() ||
568 std::find(refused.begin(), refused.end(), placement.entry_.id_) != refused.end();
569 }
570 ),
571 found.end()
572 );
573 return found;
574}

588inline const OpenInstruction* recorded_owner(
589 const OpenInstruction& instruction,
590 const OrderSnapshot& row,
591 const std::vector<OpenInstruction>& recorded,
592 const MatchWindow& window
593) {
594 if (row.order_id_.empty()) {
595 return nullptr;
596 }
597 for (const auto& other : recorded) {
598 if (other.entry_.id_ == instruction.entry_.id_ || other.entry_.order_id_ != row.order_id_) {
599 continue;
600 }
601 if (row_corresponds(other, row, window) || row_might_correspond(other, row, window)) {
602 return &other;
603 }
604 }
605 return nullptr;
606}

626inline OrderMatch match_against(
627 const OpenInstruction& instruction,
628 const std::vector<OrderSnapshot>& rows,
629 const std::vector<OpenInstruction>& recorded,
630 std::chrono::system_clock::time_point now,
631 const MatchWindow& window = {}
632) {
633 OrderMatch match;
634 match.instrument_ = instruction.entry_.instrument_;
635 match.entry_id_ = instruction.entry_.id_;
636
637 // **An instruction whose own line cannot support a conclusion.** These are
638 // written by this library -- `JournalingExecutor` appends the pending line
639 // before the binding runs, so a locally refused order is on disk before
640 // anything could object -- which is why the *reader* accepts them. What must
641 // not happen is resolving one: each lacks precisely the field the matcher
642 // would decide on, so a sweep would conclude from silence.
643 //
644 // An ordinary local refusal is unaffected: its own terminal line closes the
645 // pending one before any of this is reached. Only an instruction still open
646 // at recovery arrives here, and for those the honest answer is an operator.
647 if (const auto reason = unresolvable_reason(instruction.entry_); !reason.empty()) {
648 match.verdict_ = MatchVerdict::NEEDS_OPERATOR;
649 match.detail_ = reason;
650 return match;
651 }
652 if (instruction.entry_.requires_operator_) {
653 // Checked before anything else, so no window arithmetic runs on a timestamp that was never
654 // recorded. This is the entry the file-journal documentation calls "something nothing can ever
655 // resolve", and this check is what makes that true rather than aspirational.
656 match.verdict_ = MatchVerdict::NEEDS_OPERATOR;
657 match.detail_ = instruction.entry_.detail_.empty() ?
658 "recovered from a damaged journal line; only an operator can resolve it" :
659 instruction.entry_.detail_;
660 return match;
661 }
662
663 const OrderSnapshot* found = nullptr;
664 std::size_t unreadable = 0;
665 std::size_t claimed = 0;
666 const OrderSnapshot* claimed_row = nullptr;
667 const OpenInstruction* claimant = nullptr;
668 const bool by_order_id = journal_action_matches_by_order_id(instruction.entry_.action_);
669 if (by_order_id) {
670 for (const auto& row : rows) {
671 if (!instruction.entry_.order_id_.empty() && row.order_id_ == instruction.entry_.order_id_) {
672 ++match.candidates_;
673 found = &row;
674 }
675 }
676 } else {
677 for (const auto& row : rows) {
678 const bool fits = row_corresponds(instruction, row, window);
679 // Counted separately, because these are not candidates for a *match* -- the
680 // library could not read their shape, so it cannot claim one is this order.
681 // They are candidates against concluding the order does not exist.
682 const bool might_fit = row_might_correspond(instruction, row, window);
683 if (!fits && !might_fit) {
684 continue;
685 }
686 // Another placement's order by the journal's own record: neither this order nor evidence
687 // that this order is missing.
688 if (const auto* owner = recorded_owner(instruction, row, recorded, window)) {
689 ++claimed;
690 claimed_row = &row;
691 claimant = owner;
692 continue;
693 }
694 if (fits) {
695 ++match.candidates_;
696 found = &row;
697 }
698 if (might_fit) {
699 ++unreadable;
700 }
701 }
702 }
703
704 if (claimed > 0) {
705 // Not `MATCHED`: the row is another placement's. Not `NOT_PLACED` either, even when nothing else
706 // fits, and not `MATCHED` on another row that fits: the record may be a sweep's mistaken name,
707 // and then the claimed row is this order and the other row is the claimant's.
708 match.verdict_ = MatchVerdict::AMBIGUOUS;
709 const auto others = match.candidates_ + unreadable;
710 match.candidates_ = others + claimed;
711 match.detail_ = "order " + claimed_row->order_id_ + " in the broker's listing fits this instruction, and the "
712 "journal records it for another placement (" + claimant->entry_.id_ + ")" +
713 (claimed > 1 ? ", as it does " + std::to_string(claimed - 1) + " more such row(s)" : std::string()) +
714 (others > 0 ? "; " + std::to_string(others) + " other row(s) fit as well" : std::string()) +
715 "; nothing was resolved";
716 return match;
717 }
718 if (match.candidates_ > 1) {
719 match.verdict_ = MatchVerdict::AMBIGUOUS;
720 match.detail_ = std::to_string(match.candidates_) +
721 " orders in the broker's listing could be this instruction; nothing was resolved";
722 return match;
723 }
724 if (unreadable > 0) {
725 // Report, never guess -- and **whether or not a readable row also matched**.
726 // A row whose order code this release does not publish is not evidence that
727 // the order is missing, so `NOT_PLACED` would clear the hold and send it
728 // again beside the one already resting. But it is equally not evidence that
729 // some *other* row is the order: with one readable row matching every
730 // attribute and one unreadable row matching every attribute that survives
731 // an unreadable shape, the unreadable one may be the instruction and the
732 // readable one another order on the same account. Resolving to the readable
733 // one would attribute the wrong order and clear the hold on it, which is
734 // the same harm by a different route.
735 match.verdict_ = MatchVerdict::AMBIGUOUS;
736 match.candidates_ += unreadable;
737 match.order_id_.clear();
738 // Says which it is. An operator reading "unrecognized order code" about a
739 // row whose code was perfectly readable, and whose *route* was the unknown,
740 // is being sent to look at the wrong field.
741 match.detail_ = std::to_string(unreadable) +
742 " order(s) in the broker's listing match this instruction on instrument, side, quantity "
743 "and time, but this library cannot confirm which order they are: an order code it does "
744 "not recognize, a route nobody recorded, or both" +
745 (found != nullptr ? ", and another row matches it fully" : "") +
746 "; nothing was resolved. Each row carries the broker's own code as shape_code_ and its "
747 "venue as venue_";
748 return match;
749 }
750 if (match.candidates_ == 1) {
751 match.verdict_ = MatchVerdict::MATCHED;
752 match.order_id_ = found->order_id_;
753 match.detail_ = by_order_id ?
754 "the order named by the cancellation is still in the broker's listing" :
755 "the broker's listing carries this order";
756 return match;
757 }
758 if (!match_window_closed(instruction, now, window)) {
759 match.verdict_ = MatchVerdict::UNDECIDED;
760 match.detail_ =
761 "no matching order yet, and not enough time has passed for the listing to show one";
762 return match;
763 }
764 match.verdict_ = MatchVerdict::NOT_PLACED;
765 match.detail_ = by_order_id ?
766 "the order named by the cancellation is no longer in the broker's listing" :
767 "no matching order was found among the rows read, within the matching window";
768 return match;
769}

815inline void drop_rows_fitting_another_placement(
816 std::vector<OrderSnapshot>& rows,
817 const JournalReplay& replay,
818 const std::vector<OpenInstruction>& matching,
819 const MatchWindow& window = {}
820) {
821 // The other placements on the instruments being matched, each with the span of its send.
822 std::vector<OpenInstruction> others;
823 std::unordered_map<std::string, std::size_t> other_of;
824 std::vector<std::string> refused;
825 for (const auto& entry : replay.entries_) {
826 if (entry.action_ != JournalAction::PLACE) {
827 continue;
828 }
829 const bool on_an_instrument_matched = std::any_of(
830 matching.begin(), matching.end(),
831 [&](const OpenInstruction& instruction) { return instruction.entry_.instrument_ == entry.instrument_; }
832 );
833 const bool being_matched = std::any_of(matching.begin(), matching.end(), [&](const OpenInstruction& instruction) {
834 return instruction.entry_.id_ == entry.id_;
835 });
836 if (!on_an_instrument_matched || being_matched) {
837 continue;
838 }
839 if (journal_outcome_refuses(entry.outcome_)) {
840 refused.push_back(entry.id_);
841 }
842 auto known = other_of.find(entry.id_);
843 if (known == other_of.end()) {
844 OpenInstruction other;
845 other.entry_ = entry;
846 other.opened_at_ = entry.recorded_at_;
847 known = other_of.emplace(entry.id_, others.size()).first;
848 others.push_back(std::move(other));
849 }
850 auto& other = others[known->second];
851 if (journal_outcome_answers_send(entry.outcome_) && !other.closed_at_) {
852 other.closed_at_ = entry.recorded_at_;
853 }
854 }
855 rows.erase(
856 std::remove_if(
857 rows.begin(), rows.end(),
858 [&](const OrderSnapshot& row) {
859 return std::any_of(others.begin(), others.end(), [&](const OpenInstruction& other) {
860 return std::find(refused.begin(), refused.end(), other.entry_.id_) == refused.end() &&
861 (row_corresponds(other, row, window) || row_might_correspond(other, row, window));
862 });
863 }
864 ),
865 rows.end()
866 );
867}

936template <typename Journal>
937ReconcileReport reconcile_against(
938 const std::vector<OpenInstruction>& open,
939 const std::vector<OrderSnapshot>& rows,
940 const JournalReplay& replay,
941 std::chrono::system_clock::time_point now,
942 IndeterminacyLedger& ledger,
943 Journal& journal,
944 const MatchWindow& window = {},
945 std::string_view unplaced_note = {}
946) {
947 ReconcileReport report;
948 report.matches_.reserve(open.size());
949 const auto recorded = recorded_placements(replay);
950
951 std::vector<InstrumentId> undecided;
952 std::vector<InstrumentId> seen;
953 for (const auto& instruction : open) {
954 auto match = match_against(instruction, rows, recorded, now, window);
955 const auto instrument = match.instrument_;
956 if (match.verdict_ == MatchVerdict::AMBIGUOUS) {
957 report.ambiguous_ = true;
958 }
959 if (match.verdict_ == MatchVerdict::NOT_PLACED && instruction.entry_.action_ == JournalAction::PLACE) {
960 match.detail_ += unplaced_note;
961 }
962 if (match_verdict_resolves(match.verdict_)) {
963 // Written before the ledger is cleared, for the same reason a placement is written before it
964 // is sent: the durable record has to be the conservative one. A sweep that cleared memory and
965 // failed to record why would leave the instruction owed on disk, and the next restart would
966 // withhold the instrument again for a question that had already been answered.
967 if (journal.append(resolution_entry(instruction, match, now))) {
968 ++report.recorded_;
969 } else {
970 report.unrecorded_ = true;
971 undecided.push_back(instrument);
972 match.detail_ += " (not recorded: the journal could not be written, so the hold stands)";
973 }
974 } else {
975 undecided.push_back(instrument);
976 }
977 report.matches_.push_back(std::move(match));
978
979 bool known = false;
980 for (const auto& id : seen) {
981 if (id == instrument) {
982 known = true;
983 break;
984 }
985 }
986 if (!known) {
987 seen.push_back(instrument);
988 }
989 }
990
991 for (const auto& instrument : seen) {
992 bool blocked = false;
993 for (const auto& id : undecided) {
994 if (id == instrument) {
995 blocked = true;
996 break;
997 }
998 }
999 if (blocked) {
1000 report.withheld_.push_back(instrument);
1001 continue;
1002 }
1003 ledger.resolve_after_reading_orders(instrument);
1004 report.resolved_.push_back(instrument);
1005 }
1006 return report;
1007}

1111inline bool dates_listed_order(const OrderSnapshot& assumed, const OrderSnapshot& row) {
1112 if (row.ordered_date_assumed_ || !row.ordered_at_ || !assumed.ordered_at_ || assumed.order_id_.empty() ||
1113 row.order_id_ != assumed.order_id_ || !(row.instrument_ == assumed.instrument_)) {
1114 return false;
1115 }
1116 // Both stamped from a date and the same `HHmmss` in one zone, so one order's two rows differ by whole days.
1117 if ((*assumed.ordered_at_ - *row.ordered_at_) % std::chrono::hours(24) != std::chrono::hours(0)) {
1118 return false;
1119 }
1120 // A number reused on another day can belong to a different order at the same time of day.
1121 const bool venues_differ = row.venue_ != Venue::UNSPECIFIED && assumed.venue_ != Venue::UNSPECIFIED &&
1122 row.venue_ != assumed.venue_;
1123 return row.side_ == assumed.side_ &&
1124 reconcile_detail::same_value(row.quantity_, assumed.quantity_) &&
1125 reconcile_detail::same_value(row.price_, assumed.price_) &&
1126 !evidence_contradicts(attribute_evidence(row.type_, assumed.type_)) &&
1127 !evidence_contradicts(attribute_evidence(row.time_in_force_, assumed.time_in_force_)) &&
1128 !evidence_contradicts(attribute_evidence(row.session_, assumed.session_)) &&
1129 !evidence_contradicts(trigger_evidence(row.trigger_price_, assumed.trigger_price_)) &&
1130 !venues_differ;
1131}

1146inline void merge_dated_history(
1147 std::vector<OrderSnapshot>& into,
1148 const std::vector<OrderSnapshot>& rows
1149) {
1150 std::vector<OrderSnapshot> listing;
1151 listing.reserve(into.size() + rows.size());
1152 for (auto& listed : into) {
1153 const bool dated_by_history = listed.ordered_date_assumed_ &&
1154 std::any_of(rows.begin(), rows.end(), [&](const OrderSnapshot& row) { return dates_listed_order(listed, row); });
1155 if (!dated_by_history) {
1156 listing.push_back(std::move(listed));
1157 }
1158 }
1159 into = std::move(listing);
1160 const auto listed = into.size();
1161 for (const auto& row : rows) {
1162 bool known = false;
1163 for (std::size_t at = 0; at < listed; ++at) {
1164 // A listing row still assumed is no copy of anything: the dated row is the evidence, and both stay.
1165 if (!into[at].ordered_date_assumed_ && exact_copy(into[at], row)) {
1166 known = true;
1167 break;
1168 }
1169 }
1170 if (!known) {
1171 into.push_back(row);
1172 }
1173 }
1174}

1181inline void merge_rows(std::vector<OrderSnapshot>& into, const std::vector<OrderSnapshot>& rows) {
1182 for (const auto& row : rows) {
1183 bool known = false;
1184 for (const auto& held : into) {
1185 // Deduplicated by the broker's own id, because an order that is still open appears in *both*
1186 // listings. Two copies of one row would read as two candidates, and the matcher would report
1187 // `AMBIGUOUS` for an instruction that is not ambiguous at all.
1188 if (!row.order_id_.empty() && held.order_id_ == row.order_id_) {
1189 known = true;
1190 break;
1191 }
1192 }
1193 if (!known) {
1194 into.push_back(row);
1195 }
1196 }

1204inline Date date_of(std::chrono::system_clock::time_point at) {
1205 const auto seconds = std::chrono::time_point_cast<std::chrono::seconds>(at);
1206 const std::time_t raw = std::chrono::system_clock::to_time_t(seconds);
1207 std::tm utc{};
1208#ifdef _WIN32
1209 gmtime_s(&utc, &raw);
1210#else
1211 gmtime_r(&raw, &utc);
1212#endif
1213 Date date;
1214 date.year_ = utc.tm_year + 1900;
1215 date.month_ = static_cast<unsigned>(utc.tm_mon + 1);
1216 date.day_ = static_cast<unsigned>(utc.tm_mday);
1217 return date;

1230inline std::pair<Date, Date> history_range_for(
1231 const std::vector<OpenInstruction>& open, const MatchWindow& window
1232) {

1246 date_of(latest + std::chrono::hours(24)),

1291inline RecoveryReport recover_into(const JournalReplay& replay, IndeterminacyLedger& ledger) {
1292 RecoveryReport report;
1293 report.truncated_ = replay.truncated_tail_;
1294 report.unreadable_lines_ = replay.unreadable_lines_;
1295 report.unreadable_file_ = !replay.readable();
1296 report.detail_ = replay.detail_;
1297 report.open_ = unresolved_instructions(replay);
1298
1299 for (const auto& instruction : report.open_) {
1300 const auto& instrument = instruction.entry_.instrument_;
1301 if (instrument.symbol_.empty()) {
1302 report.unattributable_ = true;
1303 continue;
1304 }
1305 const auto marked = ledger.requires_reconcile(instrument);
1306 ledger.mark(
1307 instrument,
1308 instruction.detail_.empty() ?
1309 "a previous process left this instruction unresolved in the order journal" :
1310 instruction.detail_
1311 );
1312 if (!marked) {
1313 report.marked_.push_back(instrument);
1314 }
1315 }
1316 if (report.truncated_ && report.open_.empty()) {
1317 report.unattributable_ = true;
1318 }
1319 return report;
1320}

1349template <typename Runner, typename Journal>
1350RecoveryReport recover_runner(Runner& runner_instance, const Journal& journal) {
1351 const auto replay = journal.replay();
1352 auto report = recover_into(replay, runner_instance.ledger());
1353 // A serialized runner also gets back every order it may still owe (`Runner::restore_obligations`),
1354 // and places nothing until this has run.
1355 runner_instance.restore_obligations(replay);
1356 if (report.unattributable_) {
1357 runner_instance.halt_locally(
1358 report.detail_.empty() ?
1359 "the order journal was damaged in a way that names no instrument" :
1360 report.detail_
1361 );
1362 }
1363 if (report.unreadable_lines_ > 0) {
1364 runner_instance.halt_locally(
1365 "the order journal holds " + std::to_string(report.unreadable_lines_) +
1366 " line(s) this library did not write, so it cannot account for what was in flight"
1367 );
1368 }
1369 if (report.unreadable_file_) {
1370 // Nothing about the file's contents is known, and unlike a missing journal this one may hold
1371 // every order a predecessor left in flight. There is nothing to withhold specifically.
1372 runner_instance.halt_locally(
1373 report.detail_.empty() ?
1374 "the order journal exists and could not be read" :
1375 report.detail_
1376 );
1377 }
1378 return report;
1379}
