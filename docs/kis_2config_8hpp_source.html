58static inline std::chrono::system_clock::time_point parse_expiry(const char* s) {
59 std::tm tm{};
60 if (
61 std::sscanf(s, "%d-%d-%d %d:%d:%d", &tm.tm_year, &tm.tm_mon, &tm.tm_mday, &tm.tm_hour, &tm.tm_min, &tm.tm_sec) != 6
62 ) {
63 return {};
64 }
65 tm.tm_year -= 1900;
66 tm.tm_mon -= 1;
67 tm.tm_isdst = 0;
68#ifdef _WIN32
69 const std::time_t utc = _mkgmtime(&tm) - 9 * 3600;
70#else
71 const std::time_t utc = timegm(&tm) - 9 * 3600;
72#endif
73 return std::chrono::system_clock::from_time_t(utc);
74}

131 void from_json_impl(const std::string& filename) {
132 std::ifstream ifs(filename);
133 if (!ifs.is_open()) {
134 throw std::runtime_error("Cannot open KIS credential file: " + filename);
135 }
136 const std::string content(std::istreambuf_iterator<char>(ifs), std::istreambuf_iterator<char>{});
137 boost::system::error_code ec;
138 const auto jv = json::parse(content, ec);
139 if (ec) {
140 throw std::runtime_error("Failed to parse KIS credentials file: " + filename + " — " + ec.message());
141 }
142 if (!jv.is_object()) {
143 throw std::runtime_error("Failed to parse KIS credentials file: " + filename + " — not a JSON object");
144 }
145 const auto& doc = jv.as_object();
146 const json::value* key = doc.if_contains("app_key");
147 const json::value* secret = doc.if_contains("app_secret");
148 if (!key || !secret || !key->is_string() || !secret->is_string()) {
149 throw std::runtime_error("Credential file should contain app_key and app_secret as strings");
150 }
151 // Everything is read and checked before anything is assigned, so a refused file leaves this
152 // config -- and the endpoint it already selected -- exactly as it was.
153 const bool demo = demo_from_json(doc, filename);
154 key_ = key->get_string().c_str();
155 secret_ = secret->get_string().c_str();
156 is_demo_ = demo;
157 }

164 static bool demo_from_env() {
165 const char* demo = std::getenv("KIS_DEMO");
166 if (!demo || std::string_view(demo) == "1") {
167 return true;
168 }
169 if (std::string_view(demo) == "0") {
170 return false;
171 }
172 throw std::runtime_error(
173 std::string("KIS_DEMO is \"") + demo +
174 "\"; set it to 1 for the demo endpoints or 0 to trade "
175 "live, or leave it unset for demo."
176 );
177 }

186 static bool demo_from_json(const json::object& doc, const std::string& filename) {
187 const json::value* demo = doc.if_contains("demo");
188 if (!demo) {

191 if (!demo->is_bool()) {
192 throw std::runtime_error(
193 "KIS credential file " + filename +
194 " has a \"demo\" that is not true or false; write "
195 "true for the demo endpoints or false to trade live, or leave it out for demo."
196 );
197 }
