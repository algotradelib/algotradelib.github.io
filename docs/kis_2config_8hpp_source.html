58static inline std::chrono::system_clock::time_point parse_expiry(const char* s) {
59 std::tm tm{};
60 if (
61 std::sscanf(
62 s,
63 "%d-%d-%d %d:%d:%d",
64 &tm.tm_year,
65 &tm.tm_mon,
66 &tm.tm_mday,
67 &tm.tm_hour,
68 &tm.tm_min,
69 &tm.tm_sec
70 ) != 6
71 ) {
72 return {};
73 }
74 tm.tm_year -= 1900;
75 tm.tm_mon -= 1;
76 tm.tm_isdst = 0;
77#ifdef _WIN32
78 const std::time_t utc = _mkgmtime(&tm) - 9 * 3600;
79#else
80 const std::time_t utc = timegm(&tm) - 9 * 3600;
81#endif
82 return std::chrono::system_clock::from_time_t(utc);
83}

140 void from_json_impl(
141 const std::string& filename
142 ) {
143 std::ifstream ifs(filename);
144 if (!ifs.is_open()) {
145 throw std::runtime_error("Cannot open KIS credential file: " + filename);
146 }
147 const std::string content(
148 std::istreambuf_iterator<char>(ifs), std::istreambuf_iterator<char>{}
149 );
150 boost::system::error_code ec;
151 const auto jv = json::parse(content, ec);
152 if (ec) {
153 throw std::runtime_error(
154 "Failed to parse KIS credentials file: " + filename + " — " + ec.message()
155 );
156 }
157 if (!jv.is_object()) {
158 throw std::runtime_error("Failed to parse KIS credentials file: " + filename + " — not a JSON object");
159 }
160 const auto& doc = jv.as_object();
161 const json::value* key = doc.if_contains("app_key");
162 const json::value* secret = doc.if_contains("app_secret");
163 if (!key || !secret || !key->is_string() || !secret->is_string()) {
164 throw std::runtime_error("Credential file should contain app_key and app_secret as strings");
165 }
166 // Everything is read and checked before anything is assigned, so a refused file leaves this
167 // config -- and the endpoint it already selected -- exactly as it was.
168 const bool demo = demo_from_json(doc, filename);
169 key_ = key->get_string().c_str();
170 secret_ = secret->get_string().c_str();
171 is_demo_ = demo;
172 }
