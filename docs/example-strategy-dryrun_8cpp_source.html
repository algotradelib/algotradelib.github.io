1// A strategy run end to end with no broker, no credentials, and no network.
2//
3// Bars are built in code, so the output is identical every time and the binary contains no code
4// path that could place an order -- `<algotradelib/runner>` links no TLS, which a test asserts.
5// That is what makes this the right place to see what a strategy decides before letting it near
6// a market.
7//
8// Run it as:
9//
10// example-strategy-dryrun

74strategy::BarEvent bar(std::size_t index) {
75 Bar value;
76 value.timestamp_ = std::chrono::system_clock::time_point{} +
77 std::chrono::hours(24 * (20000 + static_cast<int>(index)));
78 value.open_ = Decimal64::parse(closes()[index]);
79 value.high_ = Decimal64::parse(closes()[index]);
80 value.low_ = Decimal64::parse(closes()[index]);
81 value.close_ = Decimal64::parse(closes()[index]);
82 value.volume_ = Decimal64::parse("1200000");
83 return strategy::make_bar_event(samsung(), BarInterval::ONE_DAY, BarStamp::CLOSE, value);
84}

87template <typename Report>
88void print(std::size_t day, const Report& report) {
89 for (const auto& order : report.would_place_) {
90 std::cout << " day " << std::setw(2) << day << " "
91 << (order.side_ == OrderSide::BUY ? "BUY " : "SELL") << ' ' << std::setw(4)
92 << order.quantity_.str() << " shares";
93 if (order.price_) {
94 std::cout << " at " << order.price_->str();
95 } else {
96 std::cout << " at market";
97 }
98 std::cout << " [" << order.tag_ << "]\n";
99 }
100 for (const auto& rejection : report.rejected_) {
101 std::cout << " day " << std::setw(2) << day << " rejected: "
102 << strategy::reject_reason_name(rejection.reason_) << '\n';
103 }
104}

113using RefusalTally = std::map<std::string, std::pair<std::size_t, std::size_t>>;

115template <typename Report>
116void tally(std::size_t day, const Report& report, RefusalTally& counts) {
117 for (const auto& refusal : report.refused_) {
118 const std::string reason(strategy::guard_reason_name(refusal.reason_));
119 auto found = counts.find(reason);
120 if (found == counts.end()) {
121 counts.emplace(reason, std::pair<std::size_t, std::size_t>{day, 1});
122 } else {
123 ++found->second.second;
124 }
125 }
126}

135template <typename Strategy>
136void run(const char* title, Strategy instance, const strategy::RiskLimits& limits = {}) {
137 std::cout << title << '\n';
138
139 runner::DryRunner<Strategy> dry(std::move(instance), limits);
140 dry.context().set_instrument(samsung_reference());
141 dry.context().set_phase(strategy::MarketPhase::OPEN);
142 dry.set_portfolio_value(Decimal64::parse("100000000"));
143
144 Decimal64 held = Decimal64::parse("0");
145 RefusalTally refusals;
146 for (std::size_t day = 0; day < closes().size(); ++day) {
147 dry.context().set_last_price(samsung(), Decimal64::parse(closes()[day]));
148
149 strategy::Portfolio<runner::DryTag> portfolio;
150 HoldingPosition position;
151 position.instrument_ = samsung();
152 position.quantity_ = held;
153 // What the account may sell, which a broker's holdings state and the guard checks a sell against by
154 // default: here the whole holding. Without it a dry run refuses every sell (`quantity_not_sellable`).
155 position.sellable_quantity_ = held;
156 portfolio.positions_.push_back(position);
157 portfolio.sellable_basis_ = algotradelib::SellableBasis{
158 algotradelib::SellableBasis::Coverage::WHOLE_HOLDING,
159 algotradelib::SellableBasis::RestingTreatment::DEDUCT,
160 false,
161 };
162 dry.context().set_portfolio(portfolio);
163
164 const auto report = dry.feed(bar(day));
165 print(day, report);
166 tally(day, report, refusals);
167
168 for (const auto& order : report.would_place_) {
169 const auto signed_quantity = order.side_ == OrderSide::BUY ?
170 algotradelib::decimal_add(held, order.quantity_) :
171 algotradelib::decimal_subtract(held, order.quantity_);
172 if (signed_quantity) {
173 held = *signed_quantity;
174 }
175 }
176 }
177 for (const auto& entry : refusals) {
178 std::cout << " day " << std::setw(2) << entry.second.first << " refused: " << entry.first
179 << " (and on " << entry.second.second - 1 << " later bars)\n";
180 }
181 std::cout << " ending position: " << held.str() << " shares\n\n";
182}

186int main() {

190 vanilla::BuyAndHold holding;
191 holding.params_.instrument_ = samsung();
192 holding.params_.quantity_ = Decimal64::parse("100");
193 run("BuyAndHold: one target, then nothing, and it remembers nothing to do it", holding);
194
195#if defined(ATL_HAS_EIGEN)
196 vanilla::MovingAverageCrossover::Params crossing;
197 crossing.instrument_ = samsung();
198 crossing.fast_ = 3;
199 crossing.slow_ = 8;
200 crossing.quantity_ = Decimal64::parse("100");
201 run(
202 "SmaCrossover(3, 8): long while the fast average leads, flat while it lags",
203 vanilla::SmaCrossover{crossing}
204 );
205
206 vanilla::DonchianBreakout breaking;
207 breaking.params_.instrument_ = samsung();
208 breaking.params_.lookback_ = 5;
209 breaking.params_.quantity_ = Decimal64::parse("100");
210 run("DonchianBreakout(5): the channel excludes today, so a break is measurable", breaking);
211
212 vanilla::RsiThreshold indexed;
213 indexed.params_.instrument_ = samsung();
214 indexed.params_.period_ = 5;
215 indexed.params_.oversold_ = 35.0;
216 indexed.params_.overbought_ = 65.0;
217 indexed.params_.quantity_ = Decimal64::parse("100");
218 run("RsiThreshold(5, 35/65): Wilder's smoothing and Wilder's seed", indexed);
219#else
220 std::cout
221 << "The indicator strategies need a linear algebra backend and this build has none.\n"
222 << "Configure with -DATL_USE_EIGEN=ON to see SmaCrossover, DonchianBreakout and RsiThreshold.\n"
223 << "BuyAndHold and TimedLiquidation read no series and run either way.\n\n";
224#endif
225
226 // A guard refusing an order is part of the pipeline, not an error, so it belongs in the output.
227 strategy::RiskLimits tight;
228 tight.max_order_notional_ = Decimal64::parse("1000000");
229 vanilla::BuyAndHold oversized;
230 oversized.params_.instrument_ = samsung();
231 oversized.params_.quantity_ = Decimal64::parse("100");
232 run(
233 "BuyAndHold with a 1,000,000 KRW per-order ceiling: refused, never resized",
234 oversized,
235 tight
236 );
237
238 return 0;
239}
