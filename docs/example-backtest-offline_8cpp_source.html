1// A backtest with no broker, no credentials, and no network.
2//
3// Bars are built in code, so the output is identical every time. That makes this the place to see what
4// a backtest does and, more usefully, what it declines to tell you: the caveats print alongside the
5// figures because a reader who has to remember which assumptions applied to their run eventually will
6// not remember.
7//
8// The same strategies are replayed at daily and at minute granularity, and the engine is told nothing
9// about which it is getting. Neither KIS nor Toss makes intraday history easy, but that is a
10// limitation of those adapters rather than of backtesting.
11//
12// Run it as:
13//
14// example-backtest-offline
15

127backtest::ReplayConfig config_for() {
128 backtest::ReplayConfig config;
129 config.cash_ = Decimal64::parse("100000000");
130 config.instruments_ = {samsung_reference()};
131 config.series_capacity_ = 64;
132 // Any phase, so a minute replay's pre-open and post-close bars are not refused while the point here
133 // is the strategy rather than the operator's limits.
134 config.limits_.permitted_phases_ = std::nullopt;
135 return config;

139template <typename Strategy>
140backtest::Backtester<Strategy> replay(Strategy instance, BarInterval interval) {
141 backtest::Backtester<Strategy> engine(std::move(instance), config_for(), realistic_costs());
142 const auto start = std::chrono::system_clock::time_point{} + std::chrono::hours(24 * 20000);
143 for (std::size_t at = 0; at < closes().size(); ++at) {
144 const auto step = interval == BarInterval::ONE_MINUTE ?
145 std::chrono::minutes(static_cast<int>(at)) :
146 std::chrono::duration_cast<std::chrono::minutes>(std::chrono::hours(24 * static_cast<int>(at)));
147 Bar bar;
148 bar.timestamp_ = start + step;
149 bar.open_ = Decimal64::parse(std::to_string(closes()[at]));
150 bar.high_ = *bar.open_;
151 bar.low_ = *bar.open_;
152 bar.close_ = *bar.open_;
153 bar.volume_ = Decimal64::parse("1200000");
154 engine.feed(strategy::make_bar_event(samsung(), interval, BarStamp::CLOSE, bar));
155 }
156 engine.finish();
157 return engine;
158}

161std::string percent(math::Scalar value) {
162 if (!math::observed(value)) {
163 return math::is_undefined(value) ? "undefined" : " --";
164 }
165 std::ostringstream text;
166 text << std::fixed << std::setprecision(2) << std::setw(7) << value * 100.0 << " %";
167 return text.str();
168}

171std::string plain(math::Scalar value) {
172 if (!math::observed(value)) {
173 return math::is_undefined(value) ? "undefined" : " --";
174 }
175 std::ostringstream text;
176 text << std::fixed << std::setprecision(3) << std::setw(9) << value;
177 return text.str();
178}

182void print(const std::string& title, const backtest::Report& report) {
183 std::cout
184 << title << "\n"
185 << " bars " << report.bars_ << "\n"
186 << " executions " << report.executions_ << "\n"
187 << " total return " << percent(report.total_return_) << "\n"
188 << " Sharpe " << plain(report.sharpe_) << " (per bar)\n"
189 << " annualized " << plain(report.annualized_sharpe_) << "\n"
190 << " Sortino " << plain(report.sortino_) << "\n"
191 << " max drawdown " << percent(report.max_drawdown_) << "\n"
192 << " turnover " << percent(report.total_turnover_) << "\n"
193 << " commission " << money_or_absent(report.commission_paid_) << " KRW\n"
194 << " slippage " << money_or_absent(report.slippage_paid_) << " KRW\n";
195 std::cout
196 << " assumed limit fills " << backtest::limit_fill_price_name(report.caveats_.limit_fill_) << ", slippage "
197 << report.caveats_.slippage_ticks_ << " tick\n";
198 // The rates that actually charged something, rather than one number: a run spanning a rate change
199 // or costing by venue charges more than one, and printing the first would name an accident.
200 for (const auto& applied : report.caveats_.applied_rates_) {
201 std::cout
202 << " commission rate " << applied.rate_.str() << " with a " << applied.minimum_.str() << " floor, "
203 << commission_origin_name(applied.origin_) << ", charged " << applied.executions_ << " execution(s)\n";
204 }
205 if (report.caveats_.cost_totals_incomplete_) {
206 // The third reason a total can be absent, and the one a reader must not miss: it is not that
207 // the run spanned currencies or traded nothing, it is that an addition did not fit and the
208 // figures below it are short by an amount nobody can name.
209 std::cout
210 << " cost totals INCOMPLETE -- an exact sum did not fit; slippage is understated"
211 << " and the fee totals are withheld\n";
212 }
213 if (report.caveats_.inclusive_commission_executions_ > 0) {
214 std::cout
215 << " commission is inclusive of any institution charge on "
216 << report.caveats_.inclusive_commission_executions_ << " execution(s)\n";
217 }
218 std::cout << " not modelled volume, queue position, market impact, borrow\n";
219 if (report.caveats_.gaps_ > 0) {
220 std::cout << " gaps " << report.caveats_.gaps_ << " (not interpolated across)\n";
221 }
222 if (report.caveats_.unfilled_ > 0) {
223 std::cout << " unfilled " << report.caveats_.unfilled_ << " (no next bar to fill against)\n";
224 }
225 if (report.caveats_.expired_day_orders_ > 0) {
226 std::cout
227 << " expired " << report.caveats_.expired_day_orders_
228 << " DAY order(s) unfilled at their session's close, as at a broker\n";
229 }
230 if (report.caveats_.multiple_instruments_) {
231 std::cout
232 << " CAVEAT more than one instrument: this result may carry survivorship\n"
233 << " bias, because reference data lists what exists now\n";
234 }
235 std::cout << "\n";
236}

241int main() {
242 std::cout << "005930, " << closes().size() << " bars built in code -- no credentials, no network\n\n";

244#if defined(ATL_HAS_EIGEN)
245 const auto daily_periods = backtest::periods_per_year(BarInterval::ONE_DAY, strategy::SessionWindow::krx(), 245.0);
246 const auto minute_periods =
247 backtest::periods_per_year(BarInterval::ONE_MINUTE, strategy::SessionWindow::krx(), 245.0);
248 std::cout
249 << "periods per year: " << *daily_periods << " daily, " << *minute_periods
250 << " minute -- which is why annualizing takes the count rather than assuming it\n\n";
251
252 vanilla::BuyAndHold holding;
253 holding.params_.instrument_ = samsung();
254 holding.params_.quantity_ = Decimal64::parse("1000");
255 {
256 const auto engine = replay(holding, BarInterval::ONE_DAY);
257 print("BuyAndHold, daily", backtest::summarize(engine.result(), engine.fills(), *daily_periods));
258 }
259
260 vanilla::MovingAverageCrossover::Params crossing;
261 crossing.instrument_ = samsung();
262 crossing.fast_ = 3;
263 crossing.slow_ = 8;
264 crossing.quantity_ = Decimal64::parse("1000");
265 {
266 const auto engine = replay(vanilla::SmaCrossover{crossing}, BarInterval::ONE_DAY);
267 print("SmaCrossover(3, 8), daily", backtest::summarize(engine.result(), engine.fills(), *daily_periods));
268 }
269
270 // The same strategy over the same path, at a different granularity. The engine is told nothing about
271 // which it is getting; only the annualizing constant differs, and it differs because the caller says
272 // so rather than because the engine guessed.
273 {
274 const auto engine = replay(vanilla::SmaCrossover{crossing}, BarInterval::ONE_MINUTE);
275 print(
276 "SmaCrossover(3, 8), minute bars -- identical decisions, different annualizing",
277 backtest::summarize(engine.result(), engine.fills(), *minute_periods)
278 );
279 std::cout
280 << " ^ Note the annualized Sharpe against the daily run above. The per-bar figure is\n"
281 << " identical -- the decisions were identical -- and only the constant differs, because\n"
282 << " sqrt(95550) is about twenty times sqrt(245). The larger number is arithmetically\n"
283 << " right and practically meaningless here: this path moves like daily data, so calling\n"
284 << " its bars minutes claims a frequency the returns do not have. Two lessons. Mislabel\n"
285 << " your bar frequency and the annualized figure is wrong by the square root of the\n"
286 << " ratio. And annualizing assumes returns are serially uncorrelated, which a trending\n"
287 << " strategy violates by construction.\n\n";
288 }
289
290 vanilla::DonchianBreakout breaking;
291 breaking.params_.instrument_ = samsung();
292 breaking.params_.lookback_ = 5;
293 breaking.params_.quantity_ = Decimal64::parse("1000");
294 {
295 const auto engine = replay(breaking, BarInterval::ONE_DAY);
296 print("DonchianBreakout(5), daily", backtest::summarize(engine.result(), engine.fills(), *daily_periods));
297 }
298
299 // And the same strategy at no cost at all, to show what costs were doing.
300 {
301 backtest::Backtester<vanilla::SmaCrossover> engine(
302 vanilla::SmaCrossover{crossing}, config_for(), backtest::NextBarFillModel{}
303 );
304 const auto start = std::chrono::system_clock::time_point{} + std::chrono::hours(24 * 20000);
305 for (std::size_t at = 0; at < closes().size(); ++at) {
306 Bar bar;
307 bar.timestamp_ = start + std::chrono::hours(24 * static_cast<int>(at));
308 bar.open_ = Decimal64::parse(std::to_string(closes()[at]));
309 bar.high_ = *bar.open_;
310 bar.low_ = *bar.open_;
311 bar.close_ = *bar.open_;
312 bar.volume_ = Decimal64::parse("1200000");
313 engine.feed(strategy::make_bar_event(samsung(), BarInterval::ONE_DAY, BarStamp::CLOSE, bar));
314 }
315 engine.finish();
316 print(
317 "SmaCrossover(3, 8), daily, at zero cost -- the difference is what costs cost",
318 backtest::summarize(engine.result(), engine.fills(), *daily_periods)
319 );
320 }
321
322 std::cout
323 << "Read the caveats, not only the figures. A backtest is the easiest place to produce a\n"
324 << "confident wrong number, and the ones above assume every fill was available in full at\n"
325 << "the price shown, which no bar can actually tell us.\n";
326#else
327 std::cout
328 << "The report needs a linear algebra backend and this build has none.\n"
329 << "Configure with -DATL_USE_EIGEN=ON to see the metrics. The engine itself runs either way.\n";
330#endif
331
332 return 0;
