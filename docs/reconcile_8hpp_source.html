313 const Decimal64& quantity,
314 const std::optional<Decimal64>& lot,
315 bool& adjusted

317 adjusted = false;
318 if (!lot) {
319 return quantity;

341 const Decimal64& price,
342 OrderSide side,
343 const std::optional<PriceLimits>& limits,
344 bool& adjusted
345) {
346 adjusted = false;
347 if (!limits) {
348 return price;

351 adjusted = true;
352 return limits->upper_;

354 if (side == OrderSide::SELL && limits->lower_ && decimal_less(price, *limits->lower_)) {

418inline bool below_tolerance(
419 const Decimal64& drift,
420 const Decimal64& already_working,
421 const ReconcileInputs& inputs
422) {
423 const auto& tolerance = inputs.tolerance_;
424
425 if (tolerance.minimum_quantity_ && decimal_less(drift, *tolerance.minimum_quantity_)) {
426 return true;
427 }
428
429 if (tolerance.minimum_notional_ && inputs.reference_price_) {
430 const auto value = decimal_multiply(drift, *inputs.reference_price_);
431 if (value && decimal_less(*value, *tolerance.minimum_notional_)) {
432 return true;
433 }
434 }
435
436 // Only meaningful against something resting: with nothing working there is no queue position to
437 // give up, so a drift is simply a new order and this floor has nothing to say about it.
438 if (tolerance.minimum_fraction_of_working_ && already_working.coefficient() > 0) {
439 // Six places, which resolves a fraction far finer than any threshold an operator would state.
440 const auto fraction = decimal_divide(drift, already_working, 6);
441 if (fraction && decimal_less(*fraction, *tolerance.minimum_fraction_of_working_)) {
442 return true;
443 }
444 }
445
446 return false;
447}

488template <typename Broker>
489Plan<Broker> reconcile(
490 const TargetPosition& target,
491 const Portfolio<Broker>& portfolio,
492 const ReconcileInputs& inputs
493) {
494 Plan<Broker> plan;
495 const auto reject = [&](RejectReason reason) {
496 plan.cancel_.clear();
497 plan.place_.clear();
498 plan.rejected_.push_back(Rejection{reason, target.instrument_, target.tag_});
499 return plan;
500 };
501
502 if (inputs.instrument_ == nullptr) {
503 return reject(RejectReason::NO_REFERENCE_DATA);
504 }
505
506 const auto zero = Decimal64::parse("0");
507 // Net across every position row, as the guards read it: a holding can be split across lots.
508 const auto net = portfolio.net_quantity_of(target.instrument_);
509 if (!net) {
510 // A snapshot whose own rows will not add up cannot say what reaching a target would take, and
511 // `value_or(zero)` here would read a broken listing as flat and order the whole target.
512 return reject(RejectReason::NOT_REPRESENTABLE);
513 }
514 const auto held = *net;
515 const auto needed = decimal_subtract(target.target_quantity_, held);
516 if (!needed) {
517 return reject(RejectReason::NOT_REPRESENTABLE);
518 }
519
520 const bool buying = decimal_less(zero, *needed);
521 const bool selling = decimal_less(*needed, zero);
522
523 // **Stops first; see `RejectReason::STOP_ORDER_RESTING`.** A row is a stop when its basis says so
524 // or it carries a trigger -- an adapter records a positive trigger even on a row whose basis reads
525 // otherwise, and that row may be a released stop.
526 const auto is_stop = [](const OrderSnapshot& order) { return is_stop_order(order); };
527 bool stop_resting = false;
528 for (const auto& working : portfolio.working_) {
529 if (working.order_.instrument_ == target.instrument_ &&
530 order_is_open(working.order_.status_) && is_stop(working.order_)) {
531 stop_resting = true;
532 }
533 }
534 if (stop_resting) {
535 const bool met = !buying && !selling;
536 for (const auto& working : portfolio.working_) {
537 if (!(working.order_.instrument_ == target.instrument_) ||
538 !order_is_open(working.order_.status_)) {
539 continue;
540 }
541 if (is_stop(working.order_)) {
542 // Withdrawn only when a trade is needed, and not again once its withdrawal is pending: a
543 // second request for an order already on its way out can only fail.
544 if (!met && !order_status_cancel_pending(working.order_.status_)) {
545 plan.cancel_.push_back(working.ref_);
546 }
547 } else if (met) {
548 // The holding meets the target, so every *other* working order is unwanted, as before.
549 plan.cancel_.push_back(working.ref_);
550 }
551 }
552 if (!met) {
553 plan.rejected_.push_back(
554 Rejection{RejectReason::STOP_ORDER_RESTING, target.instrument_, target.tag_}
555 );
556 }
557 return plan;
558 }
559
560 // Step 2: withdraw what works against the target. With nothing needed, that is everything.
561 //
562 // Only orders that can still trade are withdrawn. A snapshot may carry an order that has since
563 // filled or been cancelled, and asking a broker to cancel one of those is a request that can
564 // only fail -- noise on a path where a real cancellation failure has to be noticed.
565 for (const auto& working : portfolio.working_) {
566 if (!(working.order_.instrument_ == target.instrument_) ||
567 !order_is_open(working.order_.status_)) {
568 continue;
569 }
570 const bool opposes = (buying && working.order_.side_ == OrderSide::SELL) ||
571 (selling && working.order_.side_ == OrderSide::BUY) || (!buying && !selling);
572 if (opposes) {
573 plan.cancel_.push_back(working.ref_);
574 }
575 }
576 if (!buying && !selling) {
577 return plan;
578 }
579
580 const OrderSide side = buying ? OrderSide::BUY : OrderSide::SELL;
581 const auto magnitude = buying ? *needed : decimal_subtract(zero, *needed).value_or(zero);
582 const auto already = portfolio.working_quantity(target.instrument_, side);
583
584 // Step 3: only the part not already in the market.
585 auto remaining = decimal_subtract(magnitude, already);
586 if (!remaining) {
587 return reject(RejectReason::NOT_REPRESENTABLE);
588 }
589
590 // Step 3a: a drift smaller than the tolerance is not worth an instruction.
591 //
592 // Measured on the *drift* -- how far the in-flight position is from the target -- and not on what
593 // would be sent. Those differ exactly in the case the tolerance exists for: 100 working against a
594 // target of 99 would withdraw the side and re-place 99, so the quantity sent is 99 while the drift
595 // is one share.
596 //
597 // Suppressed only when nothing has already been queued for cancellation. An order working the
598 // *wrong way* still has to be withdrawn however small the drift, because leaving it resting works
599 // against the target rather than merely failing to reach it. And returning an empty plan rather
600 // than the accumulated one matters for the same reason: withdrawing a working order and then
601 // declining to replace it would leave the position further from the target than doing nothing.
602 if (inputs.tolerance_.stated() && plan.cancel_.empty()) {
603 const auto drift = decimal_less(*remaining, zero) ?
604 decimal_subtract(zero, *remaining) :
605 std::optional<Decimal64>(*remaining);
606 if (drift && reconcile_detail::below_tolerance(*drift, already, inputs)) {
607 return Plan<Broker>{};
608 }
609 }
610
611 // Step 4: more working than wanted, so withdraw the side and send the whole remainder -- net of what
612 // this runner reserved for its own orders not yet shown by holdings, when those may have filled
613 // (`ReconcileInputs::reserved_rows_are_unconfirmed_`): their withdrawal is refused if they did, and a
614 // whole remainder beside them traded twice. One genuinely resting is withdrawn all the same;
615 // its history settles what it filled, and a later pass places the rest.
616 if (decimal_less(*remaining, zero)) {
617 Decimal64 reserved = zero;
618 for (const auto& working : portfolio.working_) {
619 if (working.order_.instrument_ == target.instrument_ &&
620 working.order_.side_ == side && order_is_open(working.order_.status_)) {
621 plan.cancel_.push_back(working.ref_);
622 if (inputs.reserved_rows_are_unconfirmed_ && working.reserved_) {
623 // A sum that will not compute refuses: undercounted, it would size the replacement larger.
624 const auto left = decimal_subtract(working.order_.quantity_, working.order_.execution_.filled_quantity_);
625 if (!left) {
626 return reject(RejectReason::NOT_REPRESENTABLE);
627 }
628 if (decimal_less(zero, *left)) {
629 const auto sum = decimal_add(reserved, *left);
630 if (!sum) {
631 return reject(RejectReason::NOT_REPRESENTABLE);
632 }
633 reserved = *sum;
634 }
635 }
636 }
637 }
638 remaining = decimal_subtract(magnitude, reserved);
639 if (!remaining) {
640 return reject(RejectReason::NOT_REPRESENTABLE);
641 }
642 if (!decimal_less(zero, *remaining)) {
643 return plan;
644 }
645 }
646 if (remaining->coefficient() == 0) {
647 return plan;
648 }
649
650 PlannedOrder order;
651 order.instrument_ = target.instrument_;
652 order.side_ = side;
653 order.time_in_force_ = TimeInForce::DAY;
654 order.tag_ = target.tag_;
655
656 const auto lot = reconcile_detail::lot_of(*inputs.instrument_);
657 const auto sized = reconcile_detail::round_to_lot(*remaining, lot, order.quantity_rounded_);
658 if (!sized) {
659 return reject(RejectReason::NOT_REPRESENTABLE);
660 }
661 if (sized->coefficient() == 0) {
662 // Rounded away entirely. A rejection rather than an empty plan, because the strategy asked
663 // for something and is getting nothing; silence here would look like a met target.
664 return reject(RejectReason::QUANTITY_ROUNDS_TO_ZERO);
665 }
666 order.quantity_ = *sized;
667
668 if (target.limit_) {
669 const auto priced = reconcile_detail::prepare_price(
670 target.limit_->value(),
671 side,
672 inputs.instrument_->price_rule_,
673 inputs.limits_,
674 order.price_clamped_,
675 order.price_rounded_
676 );
677 if (!priced) {
678 return reject(RejectReason::UNKNOWN_PRICE_GRID);
679 }
680 if (reconcile_detail::rounded_out_of_band(target.limit_->value(), *priced, side, inputs.limits_)) {
681 return reject(RejectReason::NO_PERMITTED_PRICE);
682 }
683 order.type_ = OrderType::LIMIT;
684 order.price_ = priced;
685 } else {
686 order.type_ = OrderType::MARKET;
687 }
688
689 plan.place_.push_back(std::move(order));
690 return plan;
691}

709template <typename Broker>
710Plan<Broker> reconcile(
711 const OrderIntent& intent,
712 const Portfolio<Broker>& portfolio,
713 const ReconcileInputs& inputs
714) {
715 (void)portfolio;
716 Plan<Broker> plan;
717 const auto reject = [&](RejectReason reason) {
718 plan.place_.clear();
719 plan.rejected_.push_back(Rejection{reason, intent.instrument(), intent.tag()});
720 return plan;
721 };
722
723 if (inputs.instrument_ == nullptr) {
724 return reject(RejectReason::NO_REFERENCE_DATA);
725 }
726
727 // A price to size against: the limit if there is one, otherwise whatever the runner supplied.
728 const auto sizing_price = intent.limit() ?
729 std::optional<Decimal64>(intent.limit()->value()) :
730 inputs.reference_price_;
731
732 Decimal64 wanted;
733 switch (intent.size().kind()) {
734 case SizeKind::QUANTITY:
735 if (intent.size().as_quantity().coefficient() < 0) {
736 return reject(RejectReason::NEGATIVE_SIZE);
737 }
738 wanted = intent.size().as_quantity();
739 break;
740 case SizeKind::NOTIONAL: {
741 if (intent.size().as_notional().amount_.coefficient() < 0) {
742 return reject(RejectReason::NEGATIVE_SIZE);
743 }
744 // Not positive is no price to size by: divided by a negative one, a buy became a negative quantity.
745 if (!sizing_price || sizing_price->coefficient() <= 0) {
746 return reject(RejectReason::NO_PRICE_AVAILABLE);
747 }
748 const auto shares =
749 decimal_divide(intent.size().as_notional().amount_, *sizing_price, 0);
750 if (!shares) {
751 return reject(RejectReason::NOT_REPRESENTABLE);
752 }
753 wanted = *shares;
754 break;
755 }
756 case SizeKind::TARGET_WEIGHT: {
757 // The guards `weight_target` applies: a weight states a holding, which cannot be negative, and is
758 // a fraction of a value and sized by a price that must both be positive.
759 if (intent.size().as_target_weight().coefficient() < 0) {
760 return reject(RejectReason::UNSUPPORTED_WEIGHT);
761 }
762 if (!inputs.portfolio_value_ || inputs.portfolio_value_->coefficient() <= 0) {
763 return reject(RejectReason::NO_PORTFOLIO_VALUE);
764 }
765 if (!sizing_price || sizing_price->coefficient() <= 0) {
766 return reject(RejectReason::NO_PRICE_AVAILABLE);
767 }
768 const auto allocation = decimal_multiply(
769 inputs.portfolio_value_.value(), intent.size().as_target_weight()
770 );
771 if (!allocation) {
772 return reject(RejectReason::NOT_REPRESENTABLE);
773 }
774 const auto shares = decimal_divide(*allocation, *sizing_price, 0);
775 if (!shares) {
776 return reject(RejectReason::NOT_REPRESENTABLE);
777 }
778 wanted = *shares;
779 break;
780 }
781 }
782
783 PlannedOrder order;
784 order.instrument_ = intent.instrument();
785 order.side_ = intent.side();
786 order.time_in_force_ = intent.time_in_force();
787 order.tag_ = intent.tag();
788 order.client_order_id_ = intent.client_order_id();
789
790 const auto lot = reconcile_detail::lot_of(*inputs.instrument_);
791 const auto sized = reconcile_detail::round_to_lot(wanted, lot, order.quantity_rounded_);
792 if (!sized) {
793 return reject(RejectReason::NOT_REPRESENTABLE);
794 }
795 if (sized->coefficient() == 0) {
796 return reject(RejectReason::QUANTITY_ROUNDS_TO_ZERO);
797 }
798 order.quantity_ = *sized;
799
800 if (intent.limit()) {
801 const auto priced = reconcile_detail::prepare_price(
802 intent.limit()->value(),
803 intent.side(),
804 inputs.instrument_->price_rule_,
805 inputs.limits_,
806 order.price_clamped_,
807 order.price_rounded_
808 );
809 if (!priced) {
810 return reject(RejectReason::UNKNOWN_PRICE_GRID);
811 }
812 if (reconcile_detail::rounded_out_of_band(intent.limit()->value(), *priced, intent.side(), inputs.limits_)) {
813 return reject(RejectReason::NO_PERMITTED_PRICE);
814 }
815 order.type_ = OrderType::LIMIT;
816 order.price_ = priced;
817 } else {
818 order.type_ = OrderType::MARKET;
819 }
820
821 if (intent.trigger()) {
822 // Checked, never moved; see `RejectReason::TRIGGER_OFF_GRID`. An unknown grid answers neither
823 // way and is refused as it is for a limit.
824 const auto aligned = inputs.instrument_->price_rule_.on_tick(intent.trigger()->value());
825 if (!aligned) {
826 return reject(RejectReason::UNKNOWN_PRICE_GRID);
827 }
828 if (!*aligned) {
829 return reject(RejectReason::TRIGGER_OFF_GRID);
830 }
831 order.type_ = OrderType::STOP_LIMIT;
832 order.trigger_price_ = intent.trigger()->value();
833 }
834
835 plan.place_.push_back(std::move(order));
836 return plan;
837}

862inline WeightTarget weight_target(const OrderIntent& intent, const ReconcileInputs& inputs) {
863 WeightTarget out;
864 const auto refuse = [&](RejectReason reason) {
865 out.refused_ = reason;
866 return out;
867 };
868 const auto& weight = intent.size().as_target_weight();
869 if (intent.side() != OrderSide::BUY || decimal_less(weight, Decimal64::parse("0")) ||
870 intent.trigger() || intent.time_in_force() != TimeInForce::DAY || intent.client_order_id()) {
871 return refuse(RejectReason::UNSUPPORTED_WEIGHT);
872 }
873 if (inputs.instrument_ == nullptr) {
874 return refuse(RejectReason::NO_REFERENCE_DATA);
875 }
876 // A value of zero or less is not one a holding can be a fraction of: sized against it, a weight
877 // would become a liquidation (zero) or a short (negative) -- an account whose read failed and came
878 // back empty, or a replay that borrowed below zero equity, must not sell on that.
879 if (!inputs.portfolio_value_ || inputs.portfolio_value_->coefficient() <= 0 ||
880 (inputs.portfolio_value_currency_ &&
881 !(*inputs.portfolio_value_currency_ == inputs.instrument_->currency_))) {
882 return refuse(RejectReason::NO_PORTFOLIO_VALUE);
883 }
884 const auto price = intent.limit() ?
885 std::optional<Decimal64>(intent.limit()->value()) :
886 inputs.reference_price_;
887 if (!price || price->coefficient() <= 0) {
888 return refuse(RejectReason::NO_PRICE_AVAILABLE);
889 }
890 const auto allocation = decimal_multiply(*inputs.portfolio_value_, weight);
891 const auto shares = allocation ? decimal_divide(*allocation, *price, 0) : std::nullopt;
892 bool adjusted = false;
893 const auto wanted = shares ?
894 reconcile_detail::round_to_lot(*shares, reconcile_detail::lot_of(*inputs.instrument_), adjusted) :
895 std::nullopt;
896 if (!wanted) {
897 return refuse(RejectReason::NOT_REPRESENTABLE);
898 }
899 TargetPosition target;
900 target.instrument_ = intent.instrument();
901 target.target_quantity_ = *wanted;
902 target.limit_ = intent.limit();
903 target.tag_ = intent.tag();
904 out.target_ = std::move(target);
905 return out;
906}
