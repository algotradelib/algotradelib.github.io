28struct VolatilityTargetSizing {
30 struct Params {
32 InstrumentId instrument_;
34 std::size_t window_{20};
36 Decimal64 risk_budget_;
38 Decimal64 max_quantity_;
40 std::string tag_{"volatility-target"};
41 };
42
44 Params params_;
45
50 std::size_t warmup_bars() const { return params_.window_ + 2; }
51
58 template <class Ctx>
59 void on_bar(Ctx& ctx, const BarEvent& bar) {
60 if (!(bar.instrument_ == params_.instrument_)) {
61 return;
62 }
63 if (params_.window_ < 2) {
64 return;
65 }
66 const auto* series = ctx.bars(bar.instrument_, bar.interval_);
67 if (series == nullptr) {
68 return;
69 }
70 const auto closes = series->column(analytics::BarField::CLOSE);
71 // One extra row, because a return consumes one: a deviation over `window_` returns needs
72 // `window_ + 1` closes.
73 const auto window = trailing(closes, params_.window_ + 1);
74 if (!window) {
75 return;
76 }
77
78 const auto period_returns = analytics::returns(*window);
79 const auto deviations = analytics::rolling_stdev(period_returns, params_.window_);
80 const auto deviation = deviations[deviations.size() - 1];
81 const auto close = (*window)[window->size() - 1];
82 if (!math::observed(deviation) || !math::observed(close)) {
83 return;
84 }
85 // A flat window has no volatility to size against; do not fall back to the maximum.
86 if (deviation <= math::Scalar{0} || close <= math::Scalar{0}) {
87 return;
88 }
89
90 const auto value_deviation = analytics::to_decimal(close * deviation, 4);
91 if (!value_deviation || value_deviation->coefficient() == 0) {
92 return;
93 }
94 const auto sized = decimal_divide(params_.risk_budget_, *value_deviation, 0);
95 if (!sized) {
96 return;
97 }
98
99 TargetPosition target;
100 target.instrument_ = params_.instrument_;
101 target.target_quantity_ =
102 decimal_less(params_.max_quantity_, *sized) ? params_.max_quantity_ : *sized;
103 target.tag_ = params_.tag_;
104 ctx.submit(target);
105 }
106};

142
149 template <class Ctx>
150 void on_phase(Ctx& ctx, MarketPhase phase) {
151 if (phase != params_.rebalance_at_ || params_.universe_.empty()) {
152 return;
153 }
154 // One divided by the count, truncated at the stated scale, so the arithmetic is exact and the
155 // sum is at most one rather than possibly over it.
156 const auto share = decimal_divide(
157 Decimal64::parse("1"),
158 Decimal64(static_cast<std::int64_t>(params_.universe_.size()), 0),
159 params_.weight_scale_
160 );
161 if (!share || share->coefficient() == 0) {
162 return;
163 }
164
165 for (const auto& instrument : params_.universe_) {
166 ctx.submit(
167 OrderIntent::market(instrument, OrderSide::BUY, SizeSpec::target_weight(*share))
168 .with_tag(params_.tag_)
169 );
170 }
171 }
