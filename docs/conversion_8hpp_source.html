60inline std::optional<Decimal64> to_decimal(Scalar value, std::uint8_t scale) {
61 if (!std::isfinite(value)) {
62 return std::nullopt;
63 }
64 Scalar scaled = value;
65 for (std::uint8_t step = 0; step < scale; ++step) {
66 scaled *= Scalar(10);
67 }
68 // `std::round` is half away from zero by definition. Adding 0.5 first rounds the sum: 0.49999999999999994
69 // became 1, and an odd integer near 2^52 moved up by one.
70 const Scalar rounded = std::round(scaled);
71 // `INT64_MAX` is not a double: it converts to 2^63, which does not fit, so that bound is exclusive.
72 if (
73 rounded >= static_cast<Scalar>(std::numeric_limits<std::int64_t>::max()) ||
74 rounded < static_cast<Scalar>(std::numeric_limits<std::int64_t>::min())
75 ) {
76 return std::nullopt;
77 }
78 return Decimal64(static_cast<std::int64_t>(rounded), scale);
79}

99inline std::optional<Decimal64> round_scalar_to_tick(
100 Scalar value,
101 const PriceRule& rule,
102 OrderSide side,
103 std::uint8_t scale = 0
104) {
105 const auto nearest = to_decimal(value, scale);
106 if (!nearest) {
107 return std::nullopt;
108 }
109 // `to_decimal` rounds to nearest, which can cross the computed value against the order. Step one
110 // unit back when it did by more than floating-point noise (a few units in the last place).
111 Scalar scaled = value;
112 for (std::uint8_t step = 0; step < scale; ++step) {
113 scaled *= Scalar(10);
114 }
115 // Each of the `scale` multiplications here can add half a unit in the last place, and a value that
116 // came through `to_scalar` within its bounds carries one more half unit from its division. The
117 // tolerance is wider than that sum: it also absorbs values computed from such inputs.
118 const Scalar noise = std::max(
119 Scalar(1e-9),
120 std::abs(scaled) * Scalar(2 * scale + 4) * std::numeric_limits<Scalar>::epsilon()
121 );
122 const auto coefficient = nearest->coefficient();
123 const Scalar crossed = static_cast<Scalar>(coefficient) - scaled;
124 auto exact = *nearest;
125 if (side == OrderSide::BUY && crossed > noise) {
126 if (coefficient == std::numeric_limits<std::int64_t>::min()) {
127 return std::nullopt;
128 }
129 exact = Decimal64(coefficient - 1, scale);
130 } else if (side == OrderSide::SELL && -crossed > noise) {
131 if (coefficient == std::numeric_limits<std::int64_t>::max()) {
132 return std::nullopt;
133 }
134 exact = Decimal64(coefficient + 1, scale);
135 }
136 return rule.round_to_tick(exact, side);
137}
