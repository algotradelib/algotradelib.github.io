49 struct Params {
51 InstrumentId instrument_;
53 std::size_t fast_{20};
55 std::size_t slow_{50};
57 AverageKind kind_{AverageKind::SIMPLE};
59 Decimal64 quantity_;
61 std::string tag_{"ma-crossover"};
62 };
63
65 Params params_;
66
71 std::size_t warmup_bars() const { return params_.slow_ + 1; }
72
79 template <class Ctx>
80 void on_bar(Ctx& ctx, const BarEvent& bar) {
81 if (!(bar.instrument_ == params_.instrument_)) {
82 return;
83 }
84 if (params_.fast_ == 0 || params_.slow_ <= params_.fast_) {
85 // A fast window that is not faster would invert the signal; do nothing instead.
86 return;
87 }
88 const auto* series = ctx.bars(bar.instrument_, bar.interval_);
89 if (series == nullptr) {
90 return;
91 }
92 const auto closes = series->column(analytics::BarField::CLOSE);
93 if (static_cast<std::size_t>(closes.size()) < warmup_bars()) {
94 return;
95 }
96
97 const auto fast = line(closes, params_.fast_);
98 const auto slow = line(closes, params_.slow_);
99 const auto fast_pair = last_two(fast);
100 const auto slow_pair = last_two(slow);
101 if (!fast_pair || !slow_pair) {
102 return;
103 }
104
105 TargetPosition target;
106 target.instrument_ = params_.instrument_;
107 target.target_quantity_ = slow_pair->current_ < fast_pair->current_ ?
108 params_.quantity_ :
109 Decimal64::parse("0");
110 target.tag_ = params_.tag_;
111 ctx.submit(target);
112 }
113
114private:
115 // The chosen average of `closes`, the whole cached series; see `AverageKind`.
116 math::Vector line(math::SeriesView closes, std::size_t span) const {
117 if (params_.kind_ == AverageKind::EXPONENTIAL) {
118 return exponential_line(closes, span);
119 }
120 // Only the window: slicing keeps the per-bar cost from growing with the cache. The slice is the
121 // slow window's for both lines, as it always was, so these numbers do not move.
122 return analytics::sma(closes.tail(static_cast<math::Index>(params_.slow_) + 1), span);
123 }
124};
