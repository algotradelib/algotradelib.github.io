56inline std::optional<math::Scalar> periods_per_year(
57 BarInterval interval,
58 const strategy::SessionWindow& session,
59 math::Scalar trading_days
60) {
61 if (!(trading_days > math::Scalar{0})) {
62 return std::nullopt;
63 }
64 if (interval == BarInterval::ONE_DAY) {
65 return trading_days;
66 }
67 const auto duration = interval_duration(interval);
68 if (!duration) {
69 return std::nullopt;
70 }
71 const auto session_seconds =
72 std::chrono::duration_cast<std::chrono::seconds>(session.close_ - session.open_).count();
73 if (session_seconds <= 0 || duration->count() <= 0) {
74 return std::nullopt;
75 }
76 const auto per_day = static_cast<math::Scalar>(session_seconds) /
77 static_cast<math::Scalar>(duration->count());
78 return per_day * trading_days;
79}
