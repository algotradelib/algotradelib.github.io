47inline std::optional<math::Scalar> periods_per_year(
48 BarInterval interval,
49 const strategy::SessionWindow& session,
50 math::Scalar trading_days
51) {
52 if (!(trading_days > math::Scalar{0})) {
53 return std::nullopt;
54 }
55 if (interval == BarInterval::ONE_DAY) {
56 return trading_days;
57 }
58 const auto duration = interval_duration(interval);
59 if (!duration) {
60 return std::nullopt;
61 }
62 const auto session_seconds =
63 std::chrono::duration_cast<std::chrono::seconds>(session.close_ - session.open_).count();
64 if (session_seconds <= 0 || duration->count() <= 0) {
65 return std::nullopt;
66 }
67 // Rounded up: the session's last bar may be shorter than the others.
68 const auto per_day = (session_seconds + duration->count() - 1) / duration->count();
69 return static_cast<math::Scalar>(per_day) * trading_days;
70}
