102inline SendOutcome outcome_of(const BrokerError& error) {
103 if (!error.reached_broker_) {
104 // Checked first: a request that never went out settles the question whatever the category.
105 return SendOutcome::REFUSED_LOCALLY;
106 }
107 switch (error.category_) {
108 case ErrorCategory::TRANSPORT:
109 case ErrorCategory::CANCELLED:
110 case ErrorCategory::PROTOCOL:
111 return SendOutcome::INDETERMINATE;
112 case ErrorCategory::HTTP:
113 return error.http_status_ >= 500 ? SendOutcome::INDETERMINATE : SendOutcome::REJECTED;
114 case ErrorCategory::AUTHENTICATION:
115 case ErrorCategory::RATE_LIMIT:
116 return SendOutcome::REJECTED;
117 case ErrorCategory::INVALID_ARGUMENT:
118 // The broker was asked and refused (a client's own validation was caught above).
119 return SendOutcome::REJECTED;
120 }
121 // An unrecognized category is treated as unknown, not refused.
122 return SendOutcome::INDETERMINATE;
123}

165// Whether `Executor` can put an order in front of a real broker.
166//
167// Declared by the executor, because it is the executor that knows. It decides one thing: whether a
168// runner built on it must have read the account before it places. That requirement has to hold for
169// every live path, and there are more of those than the two binding factories --
170// `doxygen/vignettes/binding-a-broker.md` shows a `Runner<..., SessionExecutor<Session>>` assembled
171// by hand, and a rule installed only by the factories would hold for their callers and nobody else.
172//
173// **The default is `true`, and an executor opts *out*.** That is the opposite of how a detection
174// trait beside it usually reads, and it is deliberate: the two directions fail differently. An unmarked
175// executor assumed harmless is a consumer's own adapter -- one whose `place` forwards to `SessionExecutor` or
176// straight to a broker client -- silently exempted from the rule, and it fails by sending a real order sized
177// against an account nobody read. An unmarked executor assumed live is a test double that refuses until it is
178// given a stamped portfolio or says `allow_absent_portfolio()`, and it fails by being noticed at once. Only
179// one of those two is discoverable, so silence means live.
180//
181// `RecordingExecutor` and `backtest::QueueingExecutor` therefore say `false` explicitly: a dry run
182// and a replay have no holdings inquiry that could have failed, and a replay stamps its own
183// portfolio. `JournalingExecutor` forwards whatever it wraps, because recording an order does not
184// change who receives it.
185template <typename Executor, typename = void>
186struct sends_to_a_broker : std::true_type {};
187
188template <typename Executor>
189struct sends_to_a_broker<Executor, std::void_t<decltype(Executor::reaches_a_broker)>>
190 : std::bool_constant<Executor::reaches_a_broker> {};
