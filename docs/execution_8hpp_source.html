117inline SendOutcome outcome_of(const BrokerError& error) {
118 if (!error.reached_broker_) {
119 // Checked first: a request that never went out settles the question whatever the category.
120 return SendOutcome::REFUSED_LOCALLY;
121 }
122 switch (error.category_) {
123 case ErrorCategory::TRANSPORT:
124 case ErrorCategory::CANCELLED:
125 case ErrorCategory::PROTOCOL:
126 return SendOutcome::INDETERMINATE;
127 case ErrorCategory::HTTP:
128 return error.http_status_ >= 500 ? SendOutcome::INDETERMINATE : SendOutcome::REJECTED;
129 case ErrorCategory::AUTHENTICATION:
130 case ErrorCategory::RATE_LIMIT:
131 return SendOutcome::REJECTED;
132 case ErrorCategory::INVALID_ARGUMENT:
133 // The broker was asked and refused (a client's own validation was caught above).
134 return SendOutcome::REJECTED;
135 }
136 // An unrecognized category is treated as unknown, not refused.
137 return SendOutcome::INDETERMINATE;
138}

233// Whether `Executor` can put an order in front of a real broker.
234//
235// Declared by the executor, because it is the executor that knows. It decides one thing: whether a
236// runner built on it must have read the account before it places. That requirement has to hold for
237// every live path, and there are more of those than the two binding factories --
238// `doxygen/vignettes/binding-a-broker.md` shows a `Runner<..., SessionExecutor<Session>>` assembled
239// by hand, and a rule installed only by the factories would hold for their callers and nobody else.
240//
241// **The default is `true`, and an executor opts *out*.** That is the opposite of how a detection
242// trait beside it usually reads, and it is deliberate: the two directions fail differently. An unmarked
243// executor assumed harmless is a consumer's own adapter -- one whose `place` forwards to `SessionExecutor` or
244// straight to a broker client -- silently exempted from the rule, and it fails by sending a real order sized
245// against an account nobody read. An unmarked executor assumed live is a test double that refuses until it is
246// given a stamped portfolio or says `allow_absent_portfolio()`, and it fails by being noticed at once. Only
247// one of those two is discoverable, so silence means live.
248//
249// `RecordingExecutor` and `backtest::QueueingExecutor` therefore say `false` explicitly: a dry run
250// and a replay have no holdings inquiry that could have failed, and a replay stamps its own
251// portfolio. `JournalingExecutor` forwards whatever it wraps, because recording an order does not
252// change who receives it.
253template <typename Executor, typename = void>
254struct sends_to_a_broker : std::true_type {};
255
256template <typename Executor>
257struct sends_to_a_broker<Executor, std::void_t<decltype(Executor::reaches_a_broker)>>
258 : std::bool_constant<Executor::reaches_a_broker> {};

519public:
525 void mark(const InstrumentId& id, std::string detail) {
526 for (auto& entry : pending_) {
527 if (entry.instrument_ == id) {
528 // Keep the first reason: the earliest unresolved instruction is the one in doubt.
529 return;
530 }
531 }
532 pending_.push_back(Entry{id, std::move(detail)});

534
540 bool requires_reconcile(const InstrumentId& id) const {
541 for (const auto& entry : pending_) {
542 if (entry.instrument_ == id) {
543 return true;
544 }
545 }
546 return false;
547 }
548
554 std::string_view detail(const InstrumentId& id) const {
555 for (const auto& entry : pending_) {
556 if (entry.instrument_ == id) {
557 return entry.detail_;
558 }
559 }
560 return {};
561 }
562
567 void resolve_after_reading_orders(const InstrumentId& id) {
568 for (auto at = pending_.begin(); at != pending_.end(); ++at) {
569 if (at->instrument_ == id) {
570 pending_.erase(at);
571 return;
572 }
573 }
574 }
575
580 std::vector<InstrumentId> pending() const {
581 std::vector<InstrumentId> ids;
582 ids.reserve(pending_.size());
583 for (const auto& entry : pending_) {
584 ids.push_back(entry.instrument_);
585 }
586 return ids;
587 }
588
593 bool empty() const noexcept { return pending_.empty(); }
594
595private:
596 struct Entry {
597 InstrumentId instrument_;
598 std::string detail_;
599 };
600
601 std::vector<Entry> pending_;
602};
