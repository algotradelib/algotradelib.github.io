29// Converts a broker's account listing into `AccountInfo` rows (shared by both sessions).
30template <typename Client, typename Items>
31Result<std::vector<AccountInfo<Client>>> to_account_info(Result<Items> raw) {
32 using Traits = BrokerTraits<Client>;
33 using Accounts = std::vector<AccountInfo<Client>>;
34 if (!raw) {
35 return Result<Accounts>(std::move(raw.error()));
36 }
37 Accounts listed;
38 listed.reserve(raw.value().size());
39 for (const auto& item : raw.value()) {
40 AccountInfo<Client> info;
41 info.display_number_ = Traits::display_of(item);
42 info.type_ = Traits::type_of(item);
43 info.selector_ = Traits::selector_of(item);
44 listed.push_back(std::move(info));
45 }
46 return Result<Accounts>(std::move(listed));
47}

61template <typename Client>
62std::optional<BrokerError> trigger_refusal(const std::optional<Decimal64>& trigger) {
63 if constexpr (BrokerCaps<Client>::has_trigger_price) {
64 return std::nullopt;
65 } else {
66 if (!trigger) {
67 return std::nullopt;
68 }
69 BrokerError error;
70 error.category_ = ErrorCategory::INVALID_ARGUMENT;
71 error.reached_broker_ = false;
72 error.broker_code_ = "unsupported_trigger_price";
73 error.message_ =
74 "this broker does not declare that it transmits a trigger price, so an order carrying one "
75 "would be sent without it; place the order without a trigger or through a broker whose "
76 "BrokerCaps::has_trigger_price is true";
77 return error;
78 }
79}

806 // Initiates one client call; normalization and the handler run once, on the handler's executor.
807 template <typename Out, typename Token, typename Call, typename Normalize>
808 static auto translate(Token&& token, Call call, Normalize normalize) {
809 return net::async_initiate<Token, void(Result<Out>)>(
810 [call = std::move(call), normalize = std::move(normalize)](auto&& handler) mutable {
811 auto executor = net::get_associated_executor(handler);
812 call(
813 net::bind_executor(
814 executor,
815 [
816 handler = std::forward<decltype(handler)>(handler),
817 normalize = std::move(normalize)
818 ](auto raw) mutable {
819 std::move(handler)(normalize(std::move(raw)));
820 }
821 )
822 );
823 },
824 token
825 );
826 }
827
828 // Completes with a pre-flight error, posted (never inline) to the handler's executor, or the
829 // system executor if it has none.
830 template <typename Out, typename Token>
831 static auto complete_error(Token&& token, BrokerError error) {
832 return net::async_initiate<Token, void(Result<Out>)>(
833 [error = std::move(error)](auto&& handler) mutable {
834 auto executor = net::get_associated_executor(handler, net::system_executor());
835 net::post(
836 executor,
837 [
838 handler = std::forward<decltype(handler)>(handler),
839 error = std::move(error)
840 ]() mutable {
841 std::move(handler)(Result<Out>(std::move(error)));
842 }
843 );
844 },
845 token
846 );
847 }
