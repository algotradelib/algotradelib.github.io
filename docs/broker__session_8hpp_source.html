29// Converts a broker's account listing into `AccountInfo` rows (shared by both sessions).
30template <typename Client, typename Items>
31Result<std::vector<AccountInfo<Client>>> to_account_info(Result<Items> raw) {
32 using Traits = BrokerTraits<Client>;
33 using Accounts = std::vector<AccountInfo<Client>>;
34 if (!raw) {
35 return Result<Accounts>(std::move(raw.error()));
36 }
37 Accounts listed;
38 listed.reserve(raw.value().size());
39 for (const auto& item : raw.value()) {
40 AccountInfo<Client> info;
41 info.display_number_ = Traits::display_of(item);
42 info.type_ = Traits::type_of(item);
43 info.selector_ = Traits::selector_of(item);
44 listed.push_back(std::move(info));
45 }
46 return Result<Accounts>(std::move(listed));
47}

61template <typename Client>
62std::optional<BrokerError> trigger_refusal(const std::optional<Decimal64>& trigger) {
63 if constexpr (BrokerCaps<Client>::has_trigger_price) {
64 return std::nullopt;
65 } else {
66 if (!trigger) {
67 return std::nullopt;
68 }
69 BrokerError error;
70 error.category_ = ErrorCategory::INVALID_ARGUMENT;
71 error.reached_broker_ = false;
72 error.broker_code_ = "unsupported_trigger_price";
73 error.message_ =
74 "this broker does not declare that it transmits a trigger price, so an order carrying one "
75 "would be sent without it; place the order without a trigger or through a broker whose "
76 "BrokerCaps::has_trigger_price is true";
77 return error;
78 }
79}

811 // Initiates one client call; normalization and the handler run once, on the handler's executor.
812 template <typename Out, typename Token, typename Call, typename Normalize>
813 static auto translate(Token&& token, Call call, Normalize normalize) {
814 return net::async_initiate<Token, void(Result<Out>)>(
815 [call = std::move(call), normalize = std::move(normalize)](auto&& handler) mutable {
816 auto executor = net::get_associated_executor(handler);
817 call(
818 net::bind_executor(
819 executor,
820 [handler = std::forward<decltype(handler)>(handler), normalize = std::move(normalize)](auto raw) mutable {
821 std::move(handler)(normalize(std::move(raw)));
822 }
823 )
824 );
825 },
826 token
827 );
828 }
829
830 // Completes with a pre-flight error, posted (never inline) to the handler's executor, or the
831 // system executor if it has none.
832 template <typename Out, typename Token>
833 static auto complete_error(Token&& token, BrokerError error) {
834 return net::async_initiate<Token, void(Result<Out>)>(
835 [error = std::move(error)](auto&& handler) mutable {
836 auto executor = net::get_associated_executor(handler, net::system_executor());
837 net::post(executor, [handler = std::forward<decltype(handler)>(handler), error = std::move(error)]() mutable {
838 std::move(handler)(Result<Out>(std::move(error)));
839 });
840 },
841 token
842 );
843 }
