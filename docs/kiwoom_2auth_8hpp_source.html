58 const std::string jsonbody = req_.to_jsonbody();
59 KiwoomHttpClient rest(ioc_, ctx_, req_.rest_host(), req_.rest_port());
60 auto response = rest.post(
61 "/oauth2/token", jsonbody, {{"Content-Type", "application/json;charset=UTF-8"}}
62 );
63 if (response.result_int() < 200 || response.result_int() >= 300) {
64 throw std::runtime_error(
65 "Token authentication failed with HTTP status " + std::to_string(response.result_int())
66 );
67 }
68 const std::string resp = std::move(response.body());
69 boost::system::error_code ec;
70 const auto jv = json::parse(resp, ec);
71 if (ec || !jv.is_object()) {
72 throw std::runtime_error("Kiwoom authentication failed: " + resp);
73 }
74 // Kiwoom states a business failure in the body of an HTTP 200, so a successful status is not
75 // a successful token. The classification lives in `kiwoom_declared_error` and is shared with
76 // every other endpoint rather than restated here: `8020` (no appkey or secretkey), `8011` and
77 // `8012` (a missing or wrong grant_type) all arrive this way.
78 const auto& doc = jv.as_object();
79 if (const auto declared = kiwoom_declared_error(doc)) {
80 throw std::runtime_error(
81 "Kiwoom authentication failed [" + declared->broker_code_ + "]: " + declared->message_
82 );
83 }
84 rest_resp_.from_json(doc);
85 }
