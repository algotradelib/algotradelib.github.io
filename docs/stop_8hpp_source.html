158class StopSource {
159public:
167 void request(StopIntent intent) noexcept {
168 const int desired = static_cast<int>(intent);
169 int current = requested_.load(std::memory_order_relaxed);
170 // A failed exchange reloads `current`, so the held intent's strength is read again each time.
171 while (current < 0 || stop_intent_strength(static_cast<StopIntent>(current)) < stop_intent_strength(intent)) {
172 if (requested_.compare_exchange_weak(
173 current, desired, std::memory_order_release, std::memory_order_relaxed)) {
174 return;
175 }
176 }
177 }
178
183 bool requested() const noexcept {
184 return requested_.load(std::memory_order_acquire) >= 0;
185 }
186
191 std::optional<StopIntent> intent() const noexcept {
192 const int held = requested_.load(std::memory_order_acquire);
193 if (held < 0) {
194 return std::nullopt;
195 }
196 return static_cast<StopIntent>(held);
197 }
198
200 void clear() noexcept { requested_.store(-1, std::memory_order_release); }
201
202private:
203 // Negative means no request; otherwise a `StopIntent`.
204 std::atomic<int> requested_{-1};
205
206 // Asserted rather than assumed, because the signal-safety argument above rests entirely on it. If
207 // this ever fails on some platform, `request()` is no longer callable from a handler and the
208 // signal installer is unsound -- which is worth a compile error rather than a subtle deadlock in a
209 // handler that fires once a month.
210 static_assert(
211 std::atomic<int>::is_always_lock_free,
212 "StopSource requires a lock-free atomic<int> to stay signal-safe");
213};

254 std::optional<StopIntent> poll() {
255 unrecognized_.clear();
256 if (path_.empty()) {
257 return std::nullopt;
258 }
259 std::FILE* handle = std::fopen(path_.c_str(), "rb");
260 if (handle == nullptr) {
261 return std::nullopt;
262 }
263 char buffer[64] = {};
264 const std::size_t read = std::fread(buffer, 1, sizeof(buffer) - 1, handle);
265 std::fclose(handle);
266
267 std::string_view text(buffer, read);
268 const auto begin = text.find_first_not_of(" \t\r\n");
269 if (begin == std::string_view::npos) {
270 return std::nullopt;
271 }
272 text.remove_prefix(begin);
273 const auto end = text.find_first_of(" \t\r\n");
274 if (end != std::string_view::npos) {
275 text = text.substr(0, end);
276 }
277
278 if (auto intent = stop_intent_from(text)) {
279 return intent;
280 }
281 unrecognized_.assign(text);
282 return std::nullopt;
283 }
284
289 const std::string& unrecognized() const noexcept { return unrecognized_; }
290
291private:
292 std::string path_;
293 std::string unrecognized_;
294};
