97class StopSource {
98public:
106 void request(StopIntent intent) noexcept {
107 const int desired = static_cast<int>(intent);
108 int current = requested_.load(std::memory_order_relaxed);
109 while (current < desired) {
110 if (requested_.compare_exchange_weak(
111 current, desired, std::memory_order_release, std::memory_order_relaxed)) {
112 return;
113 }
114 }
115 }
116
121 bool requested() const noexcept {
122 return requested_.load(std::memory_order_acquire) >= 0;
123 }
124
129 std::optional<StopIntent> intent() const noexcept {
130 const int held = requested_.load(std::memory_order_acquire);
131 if (held < 0) {
132 return std::nullopt;
133 }
134 return static_cast<StopIntent>(held);
135 }
136
138 void clear() noexcept { requested_.store(-1, std::memory_order_release); }
139
140private:
141 // Negative means no request; otherwise a `StopIntent`.
142 std::atomic<int> requested_{-1};
143
144 // Asserted rather than assumed, because the signal-safety argument above rests entirely on it. If
145 // this ever fails on some platform, `request()` is no longer callable from a handler and the
146 // signal installer is unsound -- which is worth a compile error rather than a subtle deadlock in a
147 // handler that fires once a month.
148 static_assert(
149 std::atomic<int>::is_always_lock_free,
150 "StopSource requires a lock-free atomic<int> to stay signal-safe");
151};
