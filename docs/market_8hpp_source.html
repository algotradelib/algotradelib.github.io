177 void advance_to(std::chrono::system_clock::time_point to) {
178 if (to < now_) {
179 throw std::invalid_argument("the mock market's clock does not move backwards");
180 }
181 if (halted_) {
182 throw std::runtime_error(*halted_);
183 }
184 for (;;) {
185 Listing* earliest = nullptr;
186 for (auto& listing : listings_) {
187 if (listing.next_ >= listing.series_.bars_.size()) {
188 continue;
189 }
190 if (earliest == nullptr ||
191 period_end(listing, listing.next_) < period_end(*earliest, earliest->next_)) {
192 earliest = &listing;
193 }
194 }
195 if (earliest == nullptr || to < period_end(*earliest, earliest->next_)) {
196 break;
197 }
198 now_ = period_end(*earliest, earliest->next_);
199 apply_bar(*earliest);
200 if (halted_) {
201 throw std::runtime_error(*halted_);
202 }
203 ++earliest->next_;
204 }
205 now_ = to;
206 }

214 Result<BarSeries> bars(const BarQuery& query) const {
215 const auto* listing = find_listing(query.instrument_);
216 if (listing == nullptr) {
217 return refused<BarSeries>("unknown_instrument", "the mock market has no bars of this instrument");
218 }
219 if (query.interval_ != listing->series_.interval_) {
220 return refused<BarSeries>("unsupported_interval", "the mock market has no bars of this interval");
221 }
222 if (query.count_ < 1) {
223 return refused<BarSeries>("bars_count", "a bar request asks for at least one bar");
224 }
225 BarSeries answer;
226 answer.instrument_ = listing->series_.instrument_;
227 answer.interval_ = listing->series_.interval_;
228 answer.stamp_ = listing->series_.stamp_;
229 answer.currency_ = listing->series_.currency_;
230 answer.adjusted_ = listing->series_.adjusted_;
231 answer.received_at_ = now_;
232 const auto& all = listing->series_.bars_;
233 std::size_t end = 0;
234 while (end < all.size() && !(now_ < period_end(*listing, end)) &&
235 (!query.before_ || !(*query.before_ < all[end].timestamp_))) {
236 ++end;
237 }
238 const auto count = static_cast<std::size_t>(query.count_);
239 const std::size_t begin = end > count ? end - count : 0;
240 answer.bars_.assign(all.begin() + static_cast<std::ptrdiff_t>(begin), all.begin() + static_cast<std::ptrdiff_t>(end));
241 return Result<BarSeries>(std::move(answer));
242 }

308 Result<HoldingsSnapshot> holdings() const {
309 HoldingsSnapshot snapshot;
310 snapshot.received_at_ = now_;
311 snapshot.sellable_basis_ = SellableBasis{
312 SellableBasis::Coverage::WHOLE_HOLDING,
313 SellableBasis::RestingTreatment::DEDUCT,
314 false,

316 for (const auto& lot : ledger_.positions()) {
317 if (lot.quantity_.coefficient() == 0) {
318 continue;
319 }
320 auto row = lot;
321 row.sellable_quantity_ = row.quantity_;
322 const auto currency = lot.market_value_.currency_;
323 const auto cost = decimal_multiply(row.quantity_, row.average_purchase_price_);
324 const auto gain = cost ? decimal_subtract(row.market_value_.amount_, *cost) : std::nullopt;
325 if (!gain || !add_to_totals(snapshot.totals_, row.market_value_.amount_, *cost, *gain, currency)) {
326 BrokerError error;
327 error.category_ = ErrorCategory::PROTOCOL;
328 error.broker_code_ = "out_of_range";
329 error.message_ = "a holding's value or the holdings' total does not fit an exact decimal";
330 return Result<HoldingsSnapshot>(std::move(error));
331 }
332 row.purchase_value_ = Money{*cost, currency};
333 row.unrealized_profit_loss_ = Money{*gain, currency};
334 snapshot.positions_.push_back(std::move(row));
335 }

382 Result<PlacedOrder<Broker>> place(const OrderRequest& order) {
383 if (halted_) {
384 return refused<PlacedOrder<Broker>>("market_halted", "the mock market halted and takes no orders");
385 }
386 const auto* listing = find_listing(order.instrument_);
387 if (listing == nullptr) {
388 return refused<PlacedOrder<Broker>>("unknown_instrument", "the mock market has no bars of this instrument");
389 }
390 if (order.type_ != OrderType::MARKET && order.type_ != OrderType::LIMIT) {
391 return refused<PlacedOrder<Broker>>("unsupported_order_type", "the mock market takes market and limit orders only");
392 }
393 switch (order.time_in_force_) {
394 case TimeInForce::DAY:
395 case TimeInForce::IMMEDIATE_OR_CANCEL:
396 case TimeInForce::FILL_OR_KILL:
397 break;
398 case TimeInForce::UNKNOWN:
399 return refused<PlacedOrder<Broker>>("unsupported_time_in_force", "the mock market needs a time in force");
400 }
401 const auto quantity = decimal_normalized(order.quantity_);
402 if (quantity.coefficient() <= 0 || quantity.scale() != 0) {
403 return refused<PlacedOrder<Broker>>("invalid_quantity", "an order's quantity is a positive whole number");
404 }
405 const bool limit = order.type_ == OrderType::LIMIT;
406 if (limit != order.price_.has_value() ||
407 (order.price_ && order.price_->coefficient() <= 0)) {
408 return refused<PlacedOrder<Broker>>("invalid_price", "a limit order has a positive price and a market order none");
409 }
410 const auto currency = listing->series_.currency_;
411 std::optional<Decimal64> reserve_price = order.price_;
412 if (!reserve_price) {
413 if (const auto last = last_closed(*listing)) {
414 reserve_price = listing->series_.bars_[*last].close_;
415 }
416 }
417 if (order.side_ == OrderSide::BUY) {
418 if (!reserve_price) {
419 return refused<PlacedOrder<Broker>>("no_price", "no bar of this instrument has ended to price a market buy at");
420 }
421 const auto cash = ledger_.cash_in(currency);
422 const auto spendable = cash ? spendable_in(currency, *cash) : std::nullopt;
423 const auto cost = decimal_multiply(quantity, *reserve_price);
424 if (!spendable || !cost || decimal_less(*spendable, *cost)) {
425 return refused<PlacedOrder<Broker>>("insufficient_cash", "the buy costs more than the buying power");
426 }
427 } else {
428 const auto held = ledger_.quantity_of(order.instrument_).value_or(Decimal64(0, 0));
429 const auto free = decimal_subtract(held, open_sells_of(order.instrument_));
430 if (!free || decimal_less(*free, quantity)) {
431 return refused<PlacedOrder<Broker>>("insufficient_holdings", "the sell is more than the shares held and not already offered");
432 }
433 }
434 MarketOrder accepted;
435 accepted.snapshot_.order_id_ = std::to_string(++last_id_);
436 accepted.snapshot_.instrument_ = order.instrument_;
437 accepted.snapshot_.side_ = order.side_;
438 accepted.snapshot_.type_ = order.type_;
439 accepted.snapshot_.time_in_force_ = order.time_in_force_;
440 accepted.snapshot_.session_ = order.session_;
441 accepted.snapshot_.venue_ = order.venue_;
442 accepted.snapshot_.status_ = OrderStatus::PENDING;
443 accepted.snapshot_.price_ = order.price_;
444 accepted.snapshot_.quantity_ = quantity;
445 accepted.snapshot_.currency_ = currency;
446 accepted.snapshot_.ordered_at_ = now_;
447 accepted.snapshot_.execution_.filled_quantity_ = Decimal64(0, 0);
448 accepted.reserve_price_ = reserve_price;
449 orders_.push_back(std::move(accepted));
450 PlacedOrder<Broker> placed;
451 placed.ack_.order_id_ = orders_.back().snapshot_.order_id_;
452 placed.ack_.accepted_at_ = now_;
453 placed.ref_.primary_ = placed.ack_.order_id_;
454 return Result<PlacedOrder<Broker>>(std::move(placed));
455 }

555 TradingCalendar calendar_of_bars() const {
556 using Days = std::chrono::duration<std::int64_t, std::ratio<86400>>;
557 const auto& session = settings_.session_;
558 const auto local_day = [&](std::chrono::system_clock::time_point at) {
559 return std::chrono::floor<Days>(at + session.utc_offset_).time_since_epoch().count();
560 };
561 TradingCalendar calendar;
562 std::set<std::int64_t> open;
563 for (const auto& listing : listings_) {
564 if (listing.series_.bars_.empty()) {
565 continue;
566 }
567 if (open.empty()) {
568 calendar.country_ = listing.series_.instrument_.country_;
569 }
570 for (const auto& bar : listing.series_.bars_) {
571 open.insert(local_day(bar.timestamp_));
572 }
573 }
574 if (open.empty()) {
575 return calendar;
576 }
577 for (auto day = *open.begin(); day <= *open.rbegin(); ++day) {
578 calendar.days_.push_back(TradingDay{date_from_epoch_days(day), open.count(day) > 0});
579 }
580 return calendar;
581 }

632 std::optional<Decimal64> spendable_in(const Currency& currency, const Decimal64& cash) const {
633 std::optional<Decimal64> left = cash;
634 for (const auto& order : orders_) {
635 if (!left) {
636 break;
637 }
638 if (order.snapshot_.status_ != OrderStatus::PENDING || order.snapshot_.side_ != OrderSide::BUY ||
639 !(order.snapshot_.currency_ == currency) || !order.reserve_price_) {
640 continue;
641 }

643 left = reserved ? decimal_subtract(*left, *reserved) : std::nullopt;
644 }

649 static bool add_to_totals(
650 std::vector<HoldingsTotals>& totals,
651 const Decimal64& market_value,
652 const Decimal64& cost,
653 const Decimal64& gain,
654 const Currency& currency
655 ) {
656 auto found = std::find_if(totals.begin(), totals.end(), [&](const HoldingsTotals& t) { return t.currency_ == currency; });
657 const auto zero = Decimal64(0, 0);
658 const auto purchase = decimal_add(found == totals.end() ? zero : found->purchase_value_, cost);
659 const auto market = decimal_add(found == totals.end() ? zero : found->market_value_, market_value);
660 const auto unrealized = decimal_add(found == totals.end() ? zero : found->unrealized_profit_loss_, gain);
661 if (!purchase || !market || !unrealized) {
662 return false;
663 }
664 if (found == totals.end()) {
665 HoldingsTotals fresh;
666 fresh.currency_ = currency;
667 totals.push_back(fresh);
668 found = totals.end() - 1;
669 }
670 found->purchase_value_ = *purchase;
671 found->market_value_ = *market;
672 found->unrealized_profit_loss_ = *unrealized;
673 return true;
674 }

697 void apply_bar(Listing& listing) {
698 const auto at = listing.next_;
699 const auto& series = listing.series_;
700 const auto& session = settings_.session_;
701 strategy::BarEvent event;
702 event.instrument_ = series.instrument_;
703 event.interval_ = series.interval_;
704 event.stamp_ = series.stamp_;
705 event.bar_ = series.bars_[at];
706 event.period_end_ = period_end(listing, at);
707 const auto start = period_start(listing, at);
708 const auto date = session.local_date(event.bar_.timestamp_);
709 const auto fill_date = session.local_date(event.period_end_);
710 const auto open_date = series.stamp_ == BarStamp::OPEN ? std::optional<Date>(date) : std::nullopt;
711 const auto closes = session.opens_at(event.bar_.timestamp_) - session.open_ + session.close_;
712 const bool ends_session = !bar_interval_length(series.interval_) || !(event.period_end_ < closes);
713 expire_day_orders(listing, date, false, event.period_end_);
714 for (auto& order : orders_) {
715 auto& snapshot = order.snapshot_;
716 if (snapshot.status_ != OrderStatus::PENDING || !(snapshot.instrument_ == series.instrument_) ||
717 start < *snapshot.ordered_at_) {
718 continue;
719 }
720 if (!order.session_) {
721 order.session_ = date;
722 }
723 strategy::PlannedOrder planned;
724 planned.instrument_ = snapshot.instrument_;
725 planned.side_ = snapshot.side_;
726 planned.type_ = snapshot.type_;
727 planned.time_in_force_ = snapshot.time_in_force_;
728 planned.quantity_ = snapshot.quantity_;
729 planned.price_ = snapshot.price_;
730 const auto decision = settings_.fills_.decide(planned, event, listing.reference_.price_rule_, fill_date, open_date);
731 if (decision.costing_error_) {
732 halted_ = "the mock market halted: order " + snapshot.order_id_ + "'s fill could not be costed (" +
733 decision.costing_error_->message_ + ")";
734 return;
735 }
736 if (decision.execution_) {
737 auto execution = *decision.execution_;
738 execution.fill_.order_id_ = snapshot.order_id_;
739 execution.currency_ = snapshot.currency_;
740 if (!ledger_.apply(execution, snapshot.currency_)) {
741 halted_ = "the mock market halted: order " + snapshot.order_id_ + "'s fill does not fit an exact decimal";
742 return;
743 }
744 record_fill(snapshot, execution, event.period_end_);
745 continue;
746 }
747 if (!decision.still_working_) {
748 cancel_at(order, event.period_end_);
749 }
750 }
751 expire_day_orders(listing, date, ends_session, event.period_end_);
752 if (!ledger_.mark(series.instrument_, event.bar_.close_)) {
753 halted_ = "the mock market halted: a position's value at the close of " + series.instrument_.symbol_ +
754 " does not fit an exact decimal";
755 }
756 }

759 static void record_fill(
760 OrderSnapshot& snapshot,
761 const backtest::Execution& execution,
762 std::chrono::system_clock::time_point at
763 ) {
764 const auto& fill = execution.fill_;
765 snapshot.status_ = OrderStatus::FILLED;
766 auto& done = snapshot.execution_;
767 done.filled_quantity_ = fill.quantity_;
768 done.average_filled_price_ = fill.price_;
769 if (const auto amount = decimal_multiply(fill.quantity_, fill.price_)) {
770 done.filled_amount_ = Money{*amount, snapshot.currency_};
771 }
772 auto commission = execution.commission_;
773 if (execution.institution_fee_) {
774 commission = decimal_add(commission, *execution.institution_fee_).value_or(commission);
775 }
776 done.commission_ = Money{commission, snapshot.currency_};
777 done.tax_ = Money{execution.tax_, snapshot.currency_};
778 done.filled_at_ = at;
779 }
