1// Shows how a request budget is spent, and what the trading reserve protects.
2//
3// Entirely offline: it drives the pacer with its own clock and issues no broker
4// request, so it needs no credentials and touches no network. The point is to
5// make a budget's behaviour inspectable before a strategy depends on it.
6//
7// example-rate-limit-stats [burst] [per-second] [trading-reserve]

58void print_allowances(const RateLimitBudget& budget) {
59 std::cout << "budget: burst=" << budget.burst_ << " per_second=" << budget.per_second_
60 << " trading_reserve=" << budget.trading_reserve_ << '\n';
61 std::cout << " interval between admissions: " << as_ms(Bucket(budget).interval()) << "ms\n";
62 for (const auto group : {
63 RateLimitGroup::AUTH,
64 RateLimitGroup::TRADING,
65 RateLimitGroup::ACCOUNT,
66 RateLimitGroup::MARKET_DATA,
67 RateLimitGroup::REFERENCE,
68 }) {
69 std::cout << " " << std::left << std::setw(12) << rate_limit_group_name(group)
70 << " burst allowance " << usable_burst(budget, group) << '\n';
71 }
72}

75void print_reserve(const RateLimitBudget& budget) {
76 Bucket bucket(budget);
77 const auto now = WalkClock::now();
78 const int polling = usable_burst(budget, RateLimitGroup::MARKET_DATA);
79 const int trading = usable_burst(budget, RateLimitGroup::TRADING);
80
81 int spent = 0;
82 while (bucket.try_acquire(now, polling)) {
83 ++spent;
84 }
85
86 std::cout << "\na saturated polling loop spends " << spent << " request(s) immediately\n";
87 std::cout << " its next request waits " << as_ms(bucket.delay_until(now, polling)) << "ms\n";
88 std::cout << " a cancellation waits " << as_ms(bucket.delay_until(now, trading)) << "ms";
89 std::cout << " with " << bucket.available(now, trading) << " reserved slot(s) left\n";
90}

93void print_corrections(const RateLimitBudget& budget) {
94 std::cout << "\nreactions to a broker refusal\n";
95
96 algotradelib::network::RateLimitController shared(budget);
97 shared.hold(std::chrono::seconds(1));
98 std::cout << " shared cap reached: allowance unchanged at " << shared.budget().per_second_
99 << "/s, admission held\n";
100
101 algotradelib::network::RateLimitController reported(budget);
102 reported.shrink(2, 2, std::chrono::seconds(1));
103 std::cout << " broker reported its allowance: adopted " << reported.budget().per_second_
104 << "/s\n";
105
106 std::cout << " broker reported no allowance: allowance unchanged, admission held\n";
107 std::cout << " (KIS states none, so its budget stays as configured and its"
108 << " refusals are counted)\n";
109}

112void print_sustained(const RateLimitBudget& budget) {
113 Bucket bucket(budget);
114 const auto start = WalkClock::now();
115 const int polling = usable_burst(budget, RateLimitGroup::MARKET_DATA);
116
117 int admitted = 0;
118 for (long long step = 0; step < 1000; step += 5) {
119 if (bucket.try_acquire(start + ms(step), polling)) {
120 ++admitted;
121 }
122 }
123 std::cout << "\none second of polling as fast as allowed: " << admitted << " request(s)\n";
124}
