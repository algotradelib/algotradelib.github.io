57 template <class Ctx>
58 void on_bar(Ctx& ctx, const BarEvent& bar) {
59 if (!(bar.instrument_ == params_.instrument_)) {
60 return;
61 }
62 if (params_.window_ < 2 || !(params_.entry_ > 0) || !(params_.exit_ > 0) ||
63 !(params_.exit_ < params_.entry_)) {
64 // An entry that is not below the mean, an exit that is not, or an exit at or beyond the entry
65 // is contradictory; do nothing.
66 return;
67 }
68 const auto* series = ctx.bars(bar.instrument_, bar.interval_);
69 if (series == nullptr) {
70 return;
71 }
72 const auto closes = series->column(analytics::BarField::CLOSE);
73 const auto window = trailing(closes, params_.window_);
74 if (!window) {
75 return;
76 }
77 const auto scores = analytics::zscore(*window, params_.window_);
78 const auto score = scores[scores.size() - 1];
79 // Unobserved covers both the warm-up and a window with no dispersion, where a score does not
80 // exist. Either way there is nothing to act on.
81 if (!math::observed(score)) {
82 return;
83 }
84
85 // Net across every position row: a spent lot of 0 beside a live one is not flat.
86 const auto held = ctx.portfolio().net_quantity_of(params_.instrument_);
87 if (!held) {
88 // Rows that do not add up say nothing about what is held; reading them as flat would enter.
89 return;
90 }
91 const bool holding = held->coefficient() != 0;
92
93 TargetPosition target;
94 target.instrument_ = params_.instrument_;
95 target.tag_ = params_.tag_;
96 if (!holding && score <= -params_.entry_) {
97 target.target_quantity_ = params_.quantity_;
98 } else if (holding && -params_.exit_ <= score) {
99 target.target_quantity_ = Decimal64::parse("0");
100 } else {
101 return;
102 }
103 ctx.submit(target);
104 }
105};

163 if (series == nullptr) {

167 if (static_cast<std::size_t>(closes.size()) < warmup_bars()) {
168 return;
169 }
170 // The whole cache, not a slice: Wilder's smoothing depends on its seed.
171 const auto index = analytics::rsi(closes, params_.period_);
172 const auto level = index[index.size() - 1];
173 if (!math::observed(level)) {
174 return;
175 }
176
177 // Net across every position row: a spent lot of 0 beside a live one is not flat.
178 const auto held = ctx.portfolio().net_quantity_of(params_.instrument_);
179 if (!held) {

181 return;
182 }
183 const bool holding = held->coefficient() != 0;

185 TargetPosition target;

187 target.tag_ = params_.tag_;
188 if (!holding && level <= params_.oversold_) {
189 target.target_quantity_ = params_.quantity_;
190 } else if (holding && params_.overbought_ <= level) {

234 template <class Ctx>
235 void on_bar(Ctx& ctx, const BarEvent& bar) {
236 if (!(bar.instrument_ == params_.instrument_)) {
237 return;
238 }
239 if (params_.lookback_ == 0) {
240 return;
241 }
242 const auto* series = ctx.bars(bar.instrument_, bar.interval_);
243 if (series == nullptr) {
244 return;
245 }
246 const auto closes = series->column(analytics::BarField::CLOSE);
247 const auto window = trailing(closes, params_.lookback_);
248 if (!window) {
249 return;
250 }
251
252 // The channel excludes the latest row, so a break is measurable at all.
253 const auto prior = window->head(window->size() - 1);
254 const auto highest = analytics::rolling_max(prior, params_.lookback_);
255 const auto lowest = analytics::rolling_min(prior, params_.lookback_);
256 const auto ceiling = highest[highest.size() - 1];
257 const auto floor = lowest[lowest.size() - 1];
258 const auto close = (*window)[window->size() - 1];
259 if (!math::observed(ceiling) || !math::observed(floor) || !math::observed(close)) {
260 return;
261 }
262
263 TargetPosition target;
264 target.instrument_ = params_.instrument_;
265 target.tag_ = params_.tag_;
266 if (ceiling < close) {
267 target.target_quantity_ = params_.quantity_;
268 } else if (close < floor) {
269 target.target_quantity_ = Decimal64::parse("0");
270 } else {
271 return;
272 }
273 ctx.submit(target);
274 }
275};
