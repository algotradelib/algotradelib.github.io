22inline bool order_is_open(OrderStatus status) {
23 switch (status) {
24 case OrderStatus::UNKNOWN:
25 case OrderStatus::PENDING:
26 case OrderStatus::PENDING_CANCEL:
27 case OrderStatus::PENDING_REPLACE:
28 case OrderStatus::PARTIAL_FILLED:
29 return true;
30 case OrderStatus::FILLED:
31 case OrderStatus::CANCELED:
32 case OrderStatus::REJECTED:
33 case OrderStatus::CANCEL_REJECTED:
34 case OrderStatus::REPLACE_REJECTED:
35 case OrderStatus::REPLACED:
36 return false;
37 }
38 return true;
39}

43// Adds a position row's quantity to a running net quantity; `false` if the sum does not fit.
44inline bool add_net(Decimal64& total, const HoldingPosition& position) {
45 const auto sum = decimal_add(total, position.quantity_);
46 if (!sum) {
47 return false;
48 }
49 total = *sum;
50 return true;
51}
52
53// Adds what a working row still has to fill, if it is open and on `side`; a row reporting more filled than
54// ordered, or a sum that does not fit, adds nothing.
55template <typename Working>
56void add_working(Decimal64& total, const Working& order, OrderSide side) {
57 if (order.order_.side_ != side || !order_is_open(order.order_.status_)) {
58 return;
59 }
60 const auto remaining = decimal_subtract(order.order_.quantity_, order.order_.execution_.filled_quantity_);
61 if (!remaining || decimal_less(*remaining, Decimal64(0, 0))) {
62 return;
63 }
64 if (const auto sum = decimal_add(total, *remaining)) {
65 total = *sum;
66 }
67}

222 SellableReading sellable_reading_for(const InstrumentId& instrument) const {
223 SellableReading reading;
224 bool seen = false;
225 for (const auto& position : positions_) {
226 if (!(position.instrument_ == instrument)) {
227 continue;
228 }
229 const SellableBasis basis =
230 position.sellable_basis_ ? *position.sellable_basis_ : sellable_basis_;
231 if (!seen) {
232 reading.quantity_ = position.sellable_quantity_;
233 reading.basis_ = basis;
234 seen = true;
235 continue;
236 }
237 // A second row of the same instrument. Figures are never summed, and a disagreement makes even
238 // the first row's figure uninterpretable, since nothing says which lot a sale draws from.
239 // First, a basis that disagrees.
240 if (basis.coverage_ != reading.basis_.coverage_ ||
241 basis.resting_ != reading.basis_.resting_) {
242 return SellableReading{reading.quantity_, SellableBasis{}};
243 }
244 // Then figures that disagree under one basis: two rows claiming the limit for the same shares
245 // with different numbers contradict each other. (KIS pages can repeat a `pdno`.)
246 // Compared as numbers: `operator==` also compares the scale, and "70" beside "70.0" is one figure.
247 if (reading.quantity_ && position.sellable_quantity_ &&
248 (decimal_less(*position.sellable_quantity_, *reading.quantity_) ||
249 decimal_less(*reading.quantity_, *position.sellable_quantity_))) {
250 return SellableReading{reading.quantity_, SellableBasis{}};
251 }
252 // A later row's figure where the first had none is the only figure stated.
253 if (!reading.quantity_) {
254 reading.quantity_ = position.sellable_quantity_;
255 }
256 }
257 return reading;
258 }

350 template <typename Broker>
351 explicit PortfolioRows(const Portfolio<Broker>& portfolio) {
352 for (std::size_t at = 0; at < portfolio.positions_.size(); ++at) {
353 rows_[portfolio.positions_[at].instrument_].positions_.push_back(at);
354 }
355 for (std::size_t at = 0; at < portfolio.working_.size(); ++at) {
356 rows_[portfolio.working_[at].order_.instrument_].working_.push_back(at);
357 }
358 }
359
365 const Rows* of(const InstrumentId& instrument) const {
366 const auto found = rows_.find(instrument);
367 return found == rows_.end() ? nullptr : &found->second;
368 }

370private:
371 std::unordered_map<InstrumentId, Rows, id_detail::InstrumentIdHash> rows_;
372};
