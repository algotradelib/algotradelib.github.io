20inline bool order_is_open(OrderStatus status) {
21 switch (status) {
22 case OrderStatus::UNKNOWN:
23 case OrderStatus::PENDING:
24 case OrderStatus::PENDING_CANCEL:
25 case OrderStatus::PENDING_REPLACE:
26 case OrderStatus::PARTIAL_FILLED:
27 return true;
28 case OrderStatus::FILLED:
29 case OrderStatus::CANCELED:
30 case OrderStatus::REJECTED:
31 case OrderStatus::CANCEL_REJECTED:
32 case OrderStatus::REPLACE_REJECTED:
33 case OrderStatus::REPLACED:
34 return false;
35 }
36 return true;
37}

165 Decimal64 working_quantity(const InstrumentId& instrument, OrderSide side) const {
166 Decimal64 total = Decimal64::parse("0");
167 for (const auto& order : working_) {
168 if (!(order.order_.instrument_ == instrument) || order.order_.side_ != side) {
169 continue;
170 }
171 if (!order_is_open(order.order_.status_)) {
172 continue;
173 }
174 const auto remaining = decimal_subtract(
175 order.order_.quantity_, order.order_.execution_.filled_quantity_
176 );
177 if (!remaining || decimal_less(*remaining, Decimal64::parse("0"))) {
178 continue;
179 }
180 const auto sum = decimal_add(total, *remaining);
181 if (!sum) {
182 continue;
183 }
184 total = *sum;
185 }
186 return total;

209 SellableReading sellable_reading_for(const InstrumentId& instrument) const {
210 SellableReading reading;
211 bool seen = false;
212 for (const auto& position : positions_) {
213 if (!(position.instrument_ == instrument)) {
214 continue;
215 }
216 const SellableBasis basis =
217 position.sellable_basis_ ? *position.sellable_basis_ : sellable_basis_;
218 if (!seen) {
219 reading.quantity_ = position.sellable_quantity_;
220 reading.basis_ = basis;
221 seen = true;
222 continue;
223 }
224 // A second row of the same instrument. Figures are never summed, and a disagreement makes even
225 // the first row's figure uninterpretable, since nothing says which lot a sale draws from.
226 // First, a basis that disagrees.
227 if (basis.coverage_ != reading.basis_.coverage_ ||
228 basis.resting_ != reading.basis_.resting_) {
229 return SellableReading{reading.quantity_, SellableBasis{}};
230 }
231 // Then figures that disagree under one basis: two rows claiming the limit for the same shares
232 // with different numbers contradict each other. (KIS pages can repeat a `pdno`.)
233 // Compared as numbers: `operator==` also compares the scale, and "70" beside "70.0" is one figure.
234 if (reading.quantity_ && position.sellable_quantity_ &&
235 (decimal_less(*position.sellable_quantity_, *reading.quantity_) ||
236 decimal_less(*reading.quantity_, *position.sellable_quantity_))) {
237 return SellableReading{reading.quantity_, SellableBasis{}};
238 }
239 // A later row's figure where the first had none is the only figure stated.
240 if (!reading.quantity_) {
241 reading.quantity_ = position.sellable_quantity_;
242 }
243 }
244 return reading;
245 }
