199 bool apply(const Execution& execution, const Currency& currency) {
200 const auto& fill = execution.fill_;
201 const auto zero = Decimal64::parse("0");
202
203 // Signed quantity: a sale reduces the position.
204 const auto signed_quantity = fill.side_ == OrderSide::BUY ?
205 std::optional<Decimal64>(fill.quantity_) :
206 decimal_subtract(zero, fill.quantity_);
207 if (!signed_quantity) {
208 return false;
209 }
210
211 const auto gross = decimal_multiply(fill.quantity_, fill.price_);
212 if (!gross) {
213 return false;
214 }
215 // Cash out on a buy, in on a sale, and commission out either way.
216 const auto cash_change = fill.side_ == OrderSide::BUY ?
217 decimal_subtract(zero, *gross) :
218 std::optional<Decimal64>(*gross);
219 if (!cash_change) {
220 return false;
221 }
222 const auto after_commission = decimal_subtract(*cash_change, execution.commission_);
223 if (!after_commission) {
224 return false;
225 }
226 // Both the broker's commission and the institution fee leave cash.
227 const auto after_fees = execution.institution_fee_ ?
228 decimal_subtract(*after_commission, *execution.institution_fee_) :
229 after_commission;
230 if (!after_fees) {
231 return false;
232 }
233 const auto after_tax = decimal_subtract(*after_fees, execution.tax_);
234 if (!after_tax) {
235 return false;
236 }
237
238 // Read, never opened. Opening the lot here is what made an early return leave a trace.
239 const auto* existing = find(fill.instrument_);
240 const auto held = existing != nullptr ? existing->quantity_ : zero;
241 const auto average = existing != nullptr ? existing->average_purchase_price_ : zero;
242
243 // A non-flat position cannot change currency; refuse rather than relabel it.
244 if (existing != nullptr && held.coefficient() != 0 &&
245 !(existing->market_value_.currency_ == currency)) {
246 return false;
247 }
248
249 const auto closing = closed_quantity(held, *signed_quantity);
250 if (!closing) {
251 return false;
252 }
253
254 // Realized profit exists only on the part that closed, and only against the old average.
255 std::optional<Decimal64> realized_total;
256 if (closing->coefficient() != 0) {
257 const auto moved = decimal_subtract(fill.price_, average);
258 if (!moved) {
259 return false;
260 }
261 auto banked = decimal_multiply(*closing, *moved);
262 if (!banked) {
263 return false;
264 }
265 // Closing a short banks the opposite sign: it profits when the price fell.
266 if (decimal_less(held, zero)) {
267 banked = decimal_subtract(zero, *banked);
268 if (!banked) {
269 return false;
270 }
271 }
272 realized_total = decimal_add(realized_in(currency), *banked);
273 if (!realized_total) {
274 return false;
275 }
276 }
277
278 const auto resulting = decimal_add(held, *signed_quantity);
279 if (!resulting) {
280 return false;
281 }
282
283 const auto valued = decimal_multiply(*resulting, fill.price_);
284 if (!valued) {
285 return false;
286 }
287
288 // The average price follows only the part that opened or added. A position crossing flat takes the
289 // fill price outright, because nothing of the old basis survives.
290 auto new_average = average;
291 if (resulting->coefficient() == 0) {
292 new_average = zero;
293 } else if (crossed_or_opened(held, *resulting)) {
294 new_average = fill.price_;
295 } else if (adding(held, *signed_quantity)) {
296 const auto blended = weighted_average(held, average, *signed_quantity, fill.price_);
297 if (!blended) {
298 return false;
299 }
300 new_average = *blended;
301 }
302 // Reducing without crossing leaves the average alone: selling half a position does not change
303 // what the remaining half cost.
304
305 const auto cash_total = decimal_add(
306 cash_in(currency).value_or(zero), *after_tax
307 );
308 if (!cash_total) {
309 return false;
310 }
311
312 // Room in `held_` for this lot is made before anything changes, so the insert below cannot allocate,
313 // whatever the list's history (a copy holds no spare room).
314 if (resulting->coefficient() != 0 && held_.size() == held_.capacity()) {
315 held_.reserve(std::max<std::size_t>(8, 2 * held_.capacity()));
316 }
317
318 // --- No arithmetic below can fail, and nothing below returns false -----------------------
319 auto* lot = find_or_open(fill.instrument_);
320 if ((lot->quantity_.coefficient() != 0) != (resulting->coefficient() != 0)) {
321 const auto at = static_cast<std::size_t>(lot - positions_.data());
322 const auto where = std::lower_bound(held_.begin(), held_.end(), at);
323 if (resulting->coefficient() != 0) {
324 held_.insert(where, at);
325 } else {
326 held_.erase(where);
327 }
328 }
329 lot->average_purchase_price_ = new_average;
330 lot->quantity_ = *resulting;
331 lot->last_price_ = fill.price_;
332 // **The currency is recorded here because this is the only place it is known.** `mark` is not
333 // told one and neither is `equity_in`'s caller beyond the currency it is asking about, so a
334 // position that never wrote its own currency down cannot be told apart from one denominated in
335 // whatever the caller happened to ask for. `Currency` default-constructs to KRW
336 // (`Currency`'s default constructor), which makes the untold case silently *answer* rather than be absent --
337 // a USD position would read as Korean won to anything summing per currency.
338 lot->market_value_ = Money{*valued, currency};
339 set_amount(cash_, currency, *cash_total);
340 if (realized_total) {
341 set_amount(realized_, currency, *realized_total);
342 }
343 executions_.push_back(execution);
344 return true;
345 }

355 bool mark(const InstrumentId& instrument, const Decimal64& price) {
356 // An instrument has at most one lot: `find_or_open` opens one only when none matches.
357 const auto slot = slot_of(instrument);
358 if (!slot) {
359 return true;
360 }
361 auto& lot = positions_[*slot];
362 // **Both fields move or neither does, and the caller is told which.** Two earlier versions
363 // each lost one half of that. Advancing the price and leaving the old `market_value_`
364 // handed a strategy reading `positions()` or `snapshot()` a stale number presented as
365 // current, with no error anywhere; skipping the row silently declined to apply a price the
366 // caller asked for. The currency is the one `apply` recorded -- a lot exists only because a
367 // fill opened it -- and it is the half nothing can invalidate.
368 const auto valued = decimal_multiply(lot.quantity_, price);
369 if (!valued) {
370 return false;
371 }
372 lot.last_price_ = price;
373 lot.market_value_ = Money{*valued, lot.market_value_.currency_};
374 return true;
375 }

389 strategy::Portfolio<BrokerTag> snapshot(
390 std::chrono::system_clock::time_point as_of
391 ) const {
392 strategy::Portfolio<BrokerTag> portfolio;
393 portfolio.positions_ = positions_;
394 // Every held share is sellable: a model consistent with crediting sales on the fill (no
395 // settlement); `ReportCaveats` discloses it.
396 for (auto& position : portfolio.positions_) {
397 position.sellable_quantity_ = position.quantity_;
398 }
399 // The basis a producer of sellable figures must state: whole holding, deduct resting, unverified.
400 portfolio.sellable_basis_ = SellableBasis{
401 SellableBasis::Coverage::WHOLE_HOLDING,
402 SellableBasis::RestingTreatment::DEDUCT,
403 false,
404 };
405 portfolio.cash_ = cash_;
406 // Buying power modelled as this ledger's cash, stamped at the snapshot time; the report states
407 // the model.
408 portfolio.buying_power_.reserve(cash_.size());
409 for (const auto& amount : cash_) {
410 portfolio.buying_power_.push_back(BuyingPower{amount, as_of});
411 }
412 portfolio.as_of_ = as_of;
413 return portfolio;
414 }

424 std::optional<Decimal64> equity_in(const Currency& currency) const {
425 auto total = cash_in(currency).value_or(Decimal64(0, 0));
426 // The lots with a non-zero quantity, in the order of `positions_`.
427 for (const auto at : held_) {
428 const auto& lot = positions_[at];
429 // Only positions in `currency` (recorded on `market_value_` by `apply`); no conversion.
430 if (!(lot.market_value_.currency_ == currency)) {
431 continue;
432 }
433 // Computed from the last mark rather than read from `market_value_`.
434 const auto value = decimal_multiply(lot.quantity_, lot.last_price_);
435 if (!value) {
436 return std::nullopt;
437 }
438 const auto sum = decimal_add(total, *value);
439 if (!sum) {
440 return std::nullopt;
441 }
442 total = *sum;
443 }
444 return total;
445 }

475 HoldingPosition* find_or_open(const InstrumentId& instrument) {
476 if (const auto slot = slot_of(instrument)) {
477 return &positions_[*slot];
478 }
479 HoldingPosition opened;
480 opened.instrument_ = instrument;
481 opened.quantity_ = Decimal64::parse("0");
482 opened.average_purchase_price_ = Decimal64::parse("0");
483 opened.last_price_ = Decimal64::parse("0");
484 positions_.push_back(std::move(opened));
485 if (index_.empty() && positions_.size() <= id_detail::scan_limit) {
486 return &positions_.back();
487 }
488 // Past `id_detail::scan_limit` lots the index holds every lot: built whole at the first one past it,
489 // extended after that. If it cannot be, the lot comes out again, so the two never disagree. Lots are never
490 // removed or reordered, so a position stays valid.
491 try {
492 if (index_.empty()) {
493 id_detail::PositionTable built;
494 for (std::size_t at = 0; at < positions_.size(); ++at) {
495 built.insert(positions_[at].instrument_, at, lot_ids());
496 }
497 index_ = std::move(built);
498 } else {
499 index_.insert(instrument, positions_.size() - 1, lot_ids());
500 }
501 } catch (...) {
502 positions_.pop_back();
503 throw;
504 }
505 return &positions_.back();
506 }

508 // Returns how much of `held` this trade closes, which is zero when it only adds.
509 static std::optional<Decimal64> closed_quantity(
510 const Decimal64& held,
511 const Decimal64& traded
512 ) {
513 const auto zero = Decimal64::parse("0");
514 const bool long_held = decimal_less(zero, held);
515 const bool short_held = decimal_less(held, zero);
516 const bool buying = decimal_less(zero, traded);
517 if ((long_held && buying) || (short_held && !buying) || held.coefficient() == 0) {
518 return zero;
519 }
520 const auto magnitude_held = long_held ? held : decimal_subtract(zero, held);
521 const auto magnitude_traded = buying ? traded : decimal_subtract(zero, traded);
522 if (!magnitude_held || !magnitude_traded) {
523 return std::nullopt;
524 }
525 return decimal_less(*magnitude_held, *magnitude_traded) ? *magnitude_held : *magnitude_traded;
526 }
527
528 // Returns whether the trade opened from flat or crossed through it.
529 static bool crossed_or_opened(const Decimal64& held, const Decimal64& resulting) {
530 const auto zero = Decimal64::parse("0");
531 if (held.coefficient() == 0) {
532 return true;
533 }
534 return decimal_less(zero, held) != decimal_less(zero, resulting);
535 }
536
537 // Returns whether the trade added to a position rather than reducing it.
538 static bool adding(const Decimal64& held, const Decimal64& traded) {
539 const auto zero = Decimal64::parse("0");
540 return decimal_less(zero, held) == decimal_less(zero, traded);
541 }

543 // Returns the average of two lots by magnitude, which is what adding to a position produces.
544 static std::optional<Decimal64> weighted_average(
545 const Decimal64& held,
546 const Decimal64& held_price,
547 const Decimal64& traded,
548 const Decimal64& traded_price
549 ) {
550 const auto zero = Decimal64::parse("0");
551 const auto held_magnitude = decimal_less(held, zero) ? decimal_subtract(zero, held) : held;
552 const auto traded_magnitude =
553 decimal_less(traded, zero) ? decimal_subtract(zero, traded) : traded;
554 if (!held_magnitude || !traded_magnitude) {
555 return std::nullopt;
556 }
557 const auto held_cost = decimal_multiply(*held_magnitude, held_price);
558 const auto traded_cost = decimal_multiply(*traded_magnitude, traded_price);
559 if (!held_cost || !traded_cost) {
560 return std::nullopt;
561 }
562 const auto total_cost = decimal_add(*held_cost, *traded_cost);
563 const auto total_quantity = decimal_add(*held_magnitude, *traded_magnitude);
564 if (!total_cost || !total_quantity || total_quantity->coefficient() == 0) {
565 return std::nullopt;
566 }
567 // Four places, which holds a tick on every grid this library knows and more than a price needs.
568 return decimal_divide(*total_cost, *total_quantity, 4);
569 }

571 // Writes `total` as the whole amount held in `currency`, appending the entry when there is none.
572 //
573 // Takes the **total** rather than a change, because the addition is the fallible half and `apply`
574 // has already done it before committing anything. What is left cannot fail, which is what lets
575 // `apply` promise the ledger is either wholly updated or untouched.
576 static void set_amount(
577 std::vector<Money>& amounts,
578 const Currency& currency,
579 const Decimal64& total
580 ) {
581 for (auto& amount : amounts) {
582 if (amount.currency_ == currency) {
583 amount.amount_ = total;
584 return;
585 }
586 }
587 amounts.push_back(Money{total, currency});
588 }
