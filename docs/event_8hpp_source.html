96 MarketPhase phase(
97 std::chrono::system_clock::time_point now,
98 std::optional<bool> trading_day
99 ) const {
100 if (!trading_day) {
101 return MarketPhase::UNKNOWN;
102 }
103 if (!*trading_day) {
104 return MarketPhase::CLOSED;
105 }
106 const auto local = now + utc_offset_;
107 const auto since_midnight =
108 std::chrono::duration_cast<std::chrono::minutes>(local - local_midnight(now));
109 if (since_midnight < open_) {
110 return MarketPhase::PRE_OPEN;
111 }
112 if (close_ <= since_midnight) {
113 return MarketPhase::POST_CLOSE;
114 }
115 return MarketPhase::OPEN;
116 }

140 std::chrono::system_clock::time_point opens_at(

146private:
147 // The market's local midnight before `now`, as a UTC-shifted instant. floor rounds toward
148 // negative infinity, so instants before the epoch work too. phase() and local_date() share it so
149 // they always agree on the day.
150 std::chrono::time_point<std::chrono::system_clock, std::chrono::seconds> local_midnight(
151 std::chrono::system_clock::time_point now

154 return std::chrono::time_point_cast<std::chrono::seconds>(

196inline std::chrono::system_clock::time_point bar_period_end(
197 const Bar& bar,
198 BarInterval interval,
199 BarStamp stamp
200) {
201 if (stamp == BarStamp::CLOSE) {
202 return bar.timestamp_;
203 }
204 switch (interval) {
205 case BarInterval::ONE_MINUTE:
206 return bar.timestamp_ + std::chrono::minutes(1);
207 case BarInterval::ONE_DAY:
208 return bar.timestamp_ + std::chrono::hours(24);
209 case BarInterval::ONE_WEEK:
210 return bar.timestamp_ + std::chrono::hours(24 * 7);
211 case BarInterval::ONE_MONTH:
212 // Nominal: a month has no fixed length. Documented on the function.
213 return bar.timestamp_ + std::chrono::hours(24 * 31);
214 case BarInterval::ONE_YEAR:
215 return bar.timestamp_ + std::chrono::hours(24 * 366);
216 }
217 return bar.timestamp_;
218}
