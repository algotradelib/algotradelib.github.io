132 MarketPhase phase(
133 std::chrono::system_clock::time_point now,
134 std::optional<bool> trading_day
135 ) const {
136 if (!trading_day) {
137 return MarketPhase::UNKNOWN;
138 }
139 if (!*trading_day) {
140 return MarketPhase::CLOSED;
141 }
142 const auto local = now + utc_offset_;
143 const auto since_midnight =
144 std::chrono::duration_cast<std::chrono::minutes>(local - local_midnight(now));
145 if (since_midnight < open_) {
146 return MarketPhase::PRE_OPEN;
147 }
148 if (close_ <= since_midnight) {
149 return MarketPhase::POST_CLOSE;
150 }
151 return MarketPhase::OPEN;
152 }

163 Date local_date(std::chrono::system_clock::time_point now) const {
164 using Days = std::chrono::duration<std::int64_t, std::ratio<86400>>;
165 return date_from_epoch_days(
166 std::chrono::duration_cast<Days>(local_midnight(now).time_since_epoch()).count()
167 );
168 }
169
176 std::chrono::system_clock::time_point opens_at(
177 std::chrono::system_clock::time_point now
178 ) const {
179 return local_midnight(now) - utc_offset_ + open_;
180 }
181
182private:
183 // The market's local midnight before `now`, as a UTC-shifted instant. floor rounds toward
184 // negative infinity, so instants before the epoch work too. phase() and local_date() share it so
185 // they always agree on the day.
186 std::chrono::time_point<std::chrono::system_clock, std::chrono::seconds> local_midnight(
187 std::chrono::system_clock::time_point now
188 ) const {
189 using Days = std::chrono::duration<std::int64_t, std::ratio<86400>>;
190 return std::chrono::time_point_cast<std::chrono::seconds>(
191 std::chrono::floor<Days>(now + utc_offset_)
192 );
193 }

233inline std::chrono::system_clock::time_point bar_period_end(
234 const Bar& bar,
235 BarInterval interval,
236 BarStamp stamp
237) {
238 if (stamp == BarStamp::CLOSE) {
239 return bar.timestamp_;
240 }
241 switch (interval) {
242 case BarInterval::ONE_MINUTE:
243 return bar.timestamp_ + std::chrono::minutes(1);
244 case BarInterval::FIVE_MINUTES:
245 return bar.timestamp_ + std::chrono::minutes(5);
246 case BarInterval::FIFTEEN_MINUTES:
247 return bar.timestamp_ + std::chrono::minutes(15);
248 case BarInterval::THIRTY_MINUTES:
249 return bar.timestamp_ + std::chrono::minutes(30);
250 case BarInterval::ONE_HOUR:
251 return bar.timestamp_ + std::chrono::hours(1);
252 case BarInterval::ONE_DAY:
253 return bar.timestamp_ + std::chrono::hours(24);
254 case BarInterval::ONE_WEEK:
255 return bar.timestamp_ + std::chrono::hours(24 * 7);
256 case BarInterval::ONE_MONTH:
257 // Nominal: a month has no fixed length. Documented on the function.
258 return bar.timestamp_ + std::chrono::hours(24 * 31);
259 case BarInterval::ONE_YEAR:
260 return bar.timestamp_ + std::chrono::hours(24 * 366);
261 }
262 return bar.timestamp_;
263}

294// Where each bar of a delivered slice is, by instrument, and the bars it was built over (their storage and
295// count). Only the period bucket fills it, on the slice it delivers; **a copy or a move of a slice starts with
296// none**, so the only slice that can use it is the one the library built and hands on only as a const reference
297// (to a handler, and from `live::SessionSource` to a runner), whose bars nobody changes. A copy, a move, or a slice

310 return *this;

318 table_.reset();

358 const BarEvent* find(const InstrumentId& id) const noexcept {

422 bool holds(const BarEvent& event) const noexcept {
423 for (const auto& slice : slices_) {
424 if (slice.interval_ == event.interval_) {
425 if (slice.bars_.size() > id_detail::scan_limit) {
426 if (const auto* entry = wide(event.interval_)) {

442 void add_checked(const BarEvent& event) {
443 if (slices_.empty()) {
444 period_end_ = event.period_end_;
445 }
446 for (auto& slice : slices_) {
447 if (slice.interval_ == event.interval_) {
448 slice.bars_.push_back(event);
449 return;
450 }
451 }
452 BarSlice slice;
453 slice.period_end_ = event.period_end_;
454 slice.interval_ = event.interval_;
455 slice.bars_.push_back(event);
456 slices_.push_back(std::move(slice));
457 }
458
460 void add(const BarEvent& event) {
461 if (slices_.empty()) {
462 period_end_ = event.period_end_;
463 }
464 for (auto& slice : slices_) {
465 if (slice.interval_ == event.interval_) {
466 slice.bars_.push_back(event);
467 // If the table cannot grow, the bar comes out again, so the two never disagree.
468 try {
469 if (slice.bars_.size() == id_detail::scan_limit + 1) {
470 widen(slice);
471 } else if (slice.bars_.size() > id_detail::scan_limit) {
472 if (auto* entry = wide(slice.interval_)) {
473 entry->table_.insert(event.instrument_, slice.bars_.size() - 1, ids_of(slice));
474 entry->count_ = slice.bars_.size();
475 }
476 }
477 } catch (...) {
478 slice.bars_.pop_back();
479 throw;
480 }
481 return;
482 }
483 }
484 BarSlice slice;
485 slice.period_end_ = event.period_end_;
486 slice.interval_ = event.interval_;
487 slice.bars_.push_back(event);
488 slices_.push_back(std::move(slice));
489 }
490
492 BarSlice take_first() {
493 // Moved out first, since a move does not carry the table (`SliceIndex`); the bars keep their storage.
494 BarSlice slice = std::move(slices_.front());
495 hand_index_to(slice);
496 slices_.erase(slices_.begin());
497 return slice;
498 }

501 std::vector<BarSlice> take() {
502 for (auto& slice : slices_) {
503 hand_index_to(slice);
504 }
505 std::vector<BarSlice> slices = std::move(slices_);
506 slices_.clear();
507 return slices;
508 }
509
511 void clear() noexcept { slices_.clear(); }
512
513private:
514 // An interval's table, which describes that interval's slice only while the slice holds more than
515 // `id_detail::scan_limit` bars: `widen` refills it when a slice passes that size and `add` extends it after,
516 // and nothing reads it while the slice is narrower. It outlives the slice, so the next wide slice of the
517 // interval reuses its storage; a bucket that never holds a wide slice has none.
518 struct Wide {
519 BarInterval interval_;
520 id_detail::PositionTable table_;
521 // How many of the slice's bars the table holds: all of them after `add` or `widen`, fewer after
522 // `add_checked`.
523 std::size_t count_{0};
524 };
525
526 const Wide* wide(BarInterval interval) const noexcept {
527 for (const auto& entry : wide_) {
528 if (entry.interval_ == interval) {
529 return &entry;
530 }
531 }
532 return nullptr;
533 }
534
535 Wide* wide(BarInterval interval) noexcept {
536 for (auto& entry : wide_) {
537 if (entry.interval_ == interval) {
538 return &entry;
539 }
540 }
541 return nullptr;
542 }
543
544 // Moves the table of a wide slice into it, with the bars it describes, so `BarSlice::find` can use it after
545 // the slice is delivered. Moving a slice keeps its bars' storage, so the table stays valid for it; the bucket's
546 // own table is rebuilt (`widen`) when a later period's slice grows wide again.
547 void hand_index_to(BarSlice& slice) noexcept {
548 if (slice.bars_.size() <= id_detail::scan_limit) {
549 return;
550 }
551 // Built now if `add_checked` left it unbuilt (or behind); a failed allocation only leaves the slice scanned.
552 const auto* built = wide(slice.interval_);
553 if (built == nullptr || built->count_ != slice.bars_.size() || built->table_.empty()) {
554 try {
555 widen(slice);
556 } catch (...) {
557 return;
558 }
559 }
560 if (auto* entry = wide(slice.interval_)) {
561 slice.index_.table_ = std::move(entry->table_);
562 entry->table_.reset();
563 entry->count_ = 0;
564 slice.index_.built_over_ = slice.bars_.data();
565 slice.index_.count_ = slice.bars_.size();
566 }
567 }
568
569 // Refills the interval's table from its slice, which has just passed `id_detail::scan_limit` bars (or is being
570 // delivered). On a failure the table is left empty, and the caller takes the bar back out or scans, so nothing
571 // reads it.
572 void widen(const BarSlice& slice) {
573 Wide* entry = wide(slice.interval_);
574 if (entry == nullptr) {
575 wide_.push_back(Wide{slice.interval_, {}});
576 entry = &wide_.back();
577 }
578 entry->table_.reset();
579 entry->count_ = 0;
580 try {
581 for (std::size_t at = 0; at < slice.bars_.size(); ++at) {
582 entry->table_.insert(slice.bars_[at].instrument_, at, ids_of(slice));
583 }
584 } catch (...) {
585 entry->table_.reset();
586 throw;
587 }
588 entry->count_ = slice.bars_.size();
589 }
590
591 std::chrono::system_clock::time_point period_end_{};
592 std::vector<BarSlice> slices_;
593 // At most one entry per interval that has had a wide slice.
594 std::vector<Wide> wide_;
595};
