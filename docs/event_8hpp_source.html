140 MarketPhase phase(std::chrono::system_clock::time_point now, std::optional<bool> trading_day) const {
141 if (!trading_day) {
142 return MarketPhase::UNKNOWN;
143 }
144 if (!*trading_day) {
145 return MarketPhase::CLOSED;
146 }
147 const auto local = now + utc_offset_;
148 const auto since_midnight = std::chrono::duration_cast<std::chrono::minutes>(local - local_midnight(now));
149 if (since_midnight < open_) {
150 return MarketPhase::PRE_OPEN;
151 }
152 if (close_ <= since_midnight) {
153 return MarketPhase::POST_CLOSE;
154 }
155 return MarketPhase::OPEN;
156 }

167 Date local_date(std::chrono::system_clock::time_point now) const {
168 using Days = std::chrono::duration<std::int64_t, std::ratio<86400>>;
169 return date_from_epoch_days(std::chrono::duration_cast<Days>(local_midnight(now).time_since_epoch()).count());
170 }
171
178 std::chrono::system_clock::time_point opens_at(std::chrono::system_clock::time_point now) const {
179 return local_midnight(now) - utc_offset_ + open_;
180 }
181
188 std::chrono::system_clock::time_point closes_at(std::chrono::system_clock::time_point now) const {
189 return local_midnight(now) - utc_offset_ + close_;
190 }
191
192private:
193 // The market's local midnight before `now`, as a UTC-shifted instant. floor rounds toward
194 // negative infinity, so instants before the epoch work too. phase() and local_date() share it so
195 // they always agree on the day.
196 std::chrono::time_point<std::chrono::system_clock, std::chrono::seconds> local_midnight(
197 std::chrono::system_clock::time_point now
198 ) const {
199 using Days = std::chrono::duration<std::int64_t, std::ratio<86400>>;
200 return std::chrono::time_point_cast<std::chrono::seconds>(std::chrono::floor<Days>(now + utc_offset_));
201 }

241inline std::chrono::system_clock::time_point bar_period_end(const Bar& bar, BarInterval interval, BarStamp stamp) {
242 if (stamp == BarStamp::CLOSE) {
243 return bar.timestamp_;
244 }
245 switch (interval) {
246 case BarInterval::ONE_MINUTE:
247 return bar.timestamp_ + std::chrono::minutes(1);
248 case BarInterval::FIVE_MINUTES:
249 return bar.timestamp_ + std::chrono::minutes(5);
250 case BarInterval::FIFTEEN_MINUTES:
251 return bar.timestamp_ + std::chrono::minutes(15);
252 case BarInterval::THIRTY_MINUTES:
253 return bar.timestamp_ + std::chrono::minutes(30);
254 case BarInterval::ONE_HOUR:
255 return bar.timestamp_ + std::chrono::hours(1);
256 case BarInterval::ONE_DAY:
257 return bar.timestamp_ + std::chrono::hours(24);
258 case BarInterval::ONE_WEEK:
259 return bar.timestamp_ + std::chrono::hours(24 * 7);
260 case BarInterval::ONE_MONTH:
261 // Nominal: a month has no fixed length. Documented on the function.
262 return bar.timestamp_ + std::chrono::hours(24 * 31);
263 case BarInterval::ONE_YEAR:
264 return bar.timestamp_ + std::chrono::hours(24 * 366);
265 }
266 return bar.timestamp_;
267}

348class SessionEdges {
349public:
356 SessionEdge next(const BarEvent& event, const SessionWindow& session) {
357 SessionEdge edge;
358 const auto end = event.period_end_;
359 const auto start = bar_period_start(event, session).value_or(end);
360 const auto date = session.local_date(start);
361 const auto open = session.opens_at(start);
362 const auto close = session.closes_at(start);
363 edge.point_ = start == end;
364 const bool reaches = start < close && (open < end || (edge.point_ && end == open));
365 const bool holds_close = start <= close && close <= end;
366 if (!reaches && !holds_close) {
367 // Before the open or after the close: pre-market or after-hours data, which still enters a strategy's series.
368 ++outside_count_;
369 }
370 if (reaches && !(opened_ && *opened_ == date)) {
371 edge.opens_ = true;
372 opened_ = date;
373 ++opened_count_;
374 }
375 if (holds_close && !(closed_ && *closed_ == date)) {
376 edge.closes_ = true;
377 closed_ = date;
378 ++closed_count_;
379 }
380 return edge;
381 }

388 return opened_count_;
389 }

396 return closed_count_;

578 void add_checked(const BarEvent& event) {
579 if (slices_.empty()) {
580 period_end_ = event.period_end_;
581 }
582 for (auto& slice : slices_) {
583 if (slice.interval_ == event.interval_) {
584 slice.bars_.push_back(event);
585 return;
586 }
587 }
588 BarSlice slice;
589 slice.period_end_ = event.period_end_;
590 slice.interval_ = event.interval_;
591 slice.bars_.push_back(event);
592 slices_.push_back(std::move(slice));
593 }
594
596 void add(const BarEvent& event) {
597 if (slices_.empty()) {
598 period_end_ = event.period_end_;
599 }
600 for (auto& slice : slices_) {
601 if (slice.interval_ == event.interval_) {
602 slice.bars_.push_back(event);
603 // If the table cannot grow, the bar comes out again, so the two never disagree.
604 try {
605 if (slice.bars_.size() == id_detail::scan_limit + 1) {
606 widen(slice);
607 } else if (slice.bars_.size() > id_detail::scan_limit) {
608 if (auto* entry = wide(slice.interval_)) {
609 entry->table_.insert(event.instrument_, slice.bars_.size() - 1, ids_of(slice));
610 entry->count_ = slice.bars_.size();
611 }
612 }
613 } catch (...) {
614 slice.bars_.pop_back();
615 throw;
616 }
617 return;
618 }
619 }
620 BarSlice slice;
621 slice.period_end_ = event.period_end_;
622 slice.interval_ = event.interval_;
623 slice.bars_.push_back(event);
624 slices_.push_back(std::move(slice));
625 }

628 BarSlice take_first() {
629 // Moved out first, since a move does not carry the table (`SliceIndex`); the bars keep their storage.
630 BarSlice slice = std::move(slices_.front());

651private:
652 // An interval's table, which describes that interval's slice only while the slice holds more than
653 // `id_detail::scan_limit` bars: `widen` refills it when a slice passes that size and `add` extends it after,
654 // and nothing reads it while the slice is narrower. It outlives the slice, so the next wide slice of the
655 // interval reuses its storage; a bucket that never holds a wide slice has none.
656 struct Wide {
657 BarInterval interval_;
658 id_detail::PositionTable table_;
659 // How many of the slice's bars the table holds: all of them after `add` or `widen`, fewer after
660 // `add_checked`.
661 std::size_t count_{0};
662 };
663
664 const Wide* wide(BarInterval interval) const noexcept {
665 for (const auto& entry : wide_) {
666 if (entry.interval_ == interval) {
667 return &entry;
668 }
669 }
670 return nullptr;
671 }
672
673 Wide* wide(BarInterval interval) noexcept {
674 for (auto& entry : wide_) {
675 if (entry.interval_ == interval) {
676 return &entry;
677 }
678 }
679 return nullptr;
680 }
681
682 // Moves the table of a wide slice into it, with the bars it describes, so `BarSlice::find` can use it after
683 // the slice is delivered. Moving a slice keeps its bars' storage, so the table stays valid for it; the bucket's
684 // own table is rebuilt (`widen`) when a later period's slice grows wide again.
685 void hand_index_to(BarSlice& slice) noexcept {
686 if (slice.bars_.size() <= id_detail::scan_limit) {
687 return;
688 }
689 // Built now if `add_checked` left it unbuilt (or behind); a failed allocation only leaves the slice scanned.
690 const auto* built = wide(slice.interval_);
691 if (built == nullptr || built->count_ != slice.bars_.size() || built->table_.empty()) {
692 try {
693 widen(slice);
694 } catch (...) {
695 return;
696 }
697 }
698 if (auto* entry = wide(slice.interval_)) {
699 slice.index_.table_ = std::move(entry->table_);
700 entry->table_.reset();
701 entry->count_ = 0;
702 slice.index_.built_over_ = slice.bars_.data();
703 slice.index_.count_ = slice.bars_.size();
704 }
705 }
706
707 // Refills the interval's table from its slice, which has just passed `id_detail::scan_limit` bars (or is being
708 // delivered). On a failure the table is left empty, and the caller takes the bar back out or scans, so nothing
709 // reads it.
710 void widen(const BarSlice& slice) {
711 Wide* entry = wide(slice.interval_);
712 if (entry == nullptr) {
713 wide_.push_back(Wide{slice.interval_, {}});
714 entry = &wide_.back();
715 }
716 entry->table_.reset();
717 entry->count_ = 0;
718 try {
719 for (std::size_t at = 0; at < slice.bars_.size(); ++at) {
720 entry->table_.insert(slice.bars_[at].instrument_, at, ids_of(slice));
721 }
722 } catch (...) {
723 entry->table_.reset();
724 throw;
725 }
726 entry->count_ = slice.bars_.size();
727 }
728
729 std::chrono::system_clock::time_point period_end_{};
730 std::vector<BarSlice> slices_;
731 // At most one entry per interval that has had a wide slice.
732 std::vector<Wide> wide_;
733};
