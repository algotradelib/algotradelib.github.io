97 MarketPhase phase(
98 std::chrono::system_clock::time_point now,
99 std::optional<bool> trading_day
100 ) const {
101 if (!trading_day) {
102 return MarketPhase::UNKNOWN;
103 }
104 if (!*trading_day) {
105 return MarketPhase::CLOSED;
106 }
107 const auto local = now + utc_offset_;
108 const auto since_midnight =
109 std::chrono::duration_cast<std::chrono::minutes>(local - local_midnight(now));
110 if (since_midnight < open_) {
111 return MarketPhase::PRE_OPEN;
112 }
113 if (close_ <= since_midnight) {
114 return MarketPhase::POST_CLOSE;
115 }
116 return MarketPhase::OPEN;
117 }

129 using Days = std::chrono::duration<std::int64_t, std::ratio<86400>>;
130 return date_from_epoch_days(
131 std::chrono::duration_cast<Days>(local_midnight(now).time_since_epoch()).count()
132 );
133 }

141 std::chrono::system_clock::time_point opens_at(
142 std::chrono::system_clock::time_point now
143 ) const {
144 return local_midnight(now) - utc_offset_ + open_;
145 }

147private:
148 // The market's local midnight before `now`, as a UTC-shifted instant. floor rounds toward
149 // negative infinity, so instants before the epoch work too. phase() and local_date() share it so
150 // they always agree on the day.
151 std::chrono::time_point<std::chrono::system_clock, std::chrono::seconds> local_midnight(
152 std::chrono::system_clock::time_point now
153 ) const {
154 using Days = std::chrono::duration<std::int64_t, std::ratio<86400>>;
155 return std::chrono::time_point_cast<std::chrono::seconds>(
156 std::chrono::floor<Days>(now + utc_offset_)

197inline std::chrono::system_clock::time_point bar_period_end(
198 const Bar& bar,
199 BarInterval interval,
200 BarStamp stamp
201) {
202 if (stamp == BarStamp::CLOSE) {
203 return bar.timestamp_;
204 }
205 switch (interval) {
206 case BarInterval::ONE_MINUTE:
207 return bar.timestamp_ + std::chrono::minutes(1);
208 case BarInterval::ONE_DAY:
209 return bar.timestamp_ + std::chrono::hours(24);
210 case BarInterval::ONE_WEEK:
211 return bar.timestamp_ + std::chrono::hours(24 * 7);
212 case BarInterval::ONE_MONTH:
213 // Nominal: a month has no fixed length. Documented on the function.
214 return bar.timestamp_ + std::chrono::hours(24 * 31);
215 case BarInterval::ONE_YEAR:
216 return bar.timestamp_ + std::chrono::hours(24 * 366);
217 }
218 return bar.timestamp_;
219}
