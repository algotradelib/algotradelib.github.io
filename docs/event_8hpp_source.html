132 MarketPhase phase(
133 std::chrono::system_clock::time_point now,
134 std::optional<bool> trading_day
135 ) const {
136 if (!trading_day) {
137 return MarketPhase::UNKNOWN;
138 }
139 if (!*trading_day) {
140 return MarketPhase::CLOSED;
141 }
142 const auto local = now + utc_offset_;
143 const auto since_midnight =
144 std::chrono::duration_cast<std::chrono::minutes>(local - local_midnight(now));
145 if (since_midnight < open_) {
146 return MarketPhase::PRE_OPEN;
147 }
148 if (close_ <= since_midnight) {
149 return MarketPhase::POST_CLOSE;
150 }
151 return MarketPhase::OPEN;
152 }

163 Date local_date(std::chrono::system_clock::time_point now) const {
164 using Days = std::chrono::duration<std::int64_t, std::ratio<86400>>;
165 return date_from_epoch_days(
166 std::chrono::duration_cast<Days>(local_midnight(now).time_since_epoch()).count()
167 );
168 }
169
176 std::chrono::system_clock::time_point opens_at(
177 std::chrono::system_clock::time_point now
178 ) const {
179 return local_midnight(now) - utc_offset_ + open_;
180 }
181
182private:
183 // The market's local midnight before `now`, as a UTC-shifted instant. floor rounds toward
184 // negative infinity, so instants before the epoch work too. phase() and local_date() share it so
185 // they always agree on the day.
186 std::chrono::time_point<std::chrono::system_clock, std::chrono::seconds> local_midnight(
187 std::chrono::system_clock::time_point now
188 ) const {
189 using Days = std::chrono::duration<std::int64_t, std::ratio<86400>>;
190 return std::chrono::time_point_cast<std::chrono::seconds>(
191 std::chrono::floor<Days>(now + utc_offset_)
192 );
193 }
194};

233inline std::chrono::system_clock::time_point bar_period_end(
234 const Bar& bar,
235 BarInterval interval,
236 BarStamp stamp
237) {
238 if (stamp == BarStamp::CLOSE) {
239 return bar.timestamp_;
240 }
241 switch (interval) {
242 case BarInterval::ONE_MINUTE:
243 return bar.timestamp_ + std::chrono::minutes(1);
244 case BarInterval::FIVE_MINUTES:
245 return bar.timestamp_ + std::chrono::minutes(5);
246 case BarInterval::FIFTEEN_MINUTES:
247 return bar.timestamp_ + std::chrono::minutes(15);
248 case BarInterval::THIRTY_MINUTES:
249 return bar.timestamp_ + std::chrono::minutes(30);
250 case BarInterval::ONE_HOUR:
251 return bar.timestamp_ + std::chrono::hours(1);
252 case BarInterval::ONE_DAY:
253 return bar.timestamp_ + std::chrono::hours(24);
254 case BarInterval::ONE_WEEK:
255 return bar.timestamp_ + std::chrono::hours(24 * 7);
256 case BarInterval::ONE_MONTH:
257 // Nominal: a month has no fixed length. Documented on the function.
258 return bar.timestamp_ + std::chrono::hours(24 * 31);
259 case BarInterval::ONE_YEAR:
260 return bar.timestamp_ + std::chrono::hours(24 * 366);
261 }
262 return bar.timestamp_;
263}

330 // The id at a position of a slice, for its table.
331 struct SliceIds {
332 const BarSlice* slice_;
333 const InstrumentId& operator()(std::size_t at) const noexcept { return slice_->bars_[at].instrument_; }
334 };
335 static SliceIds ids_of(const BarSlice& slice) noexcept { return SliceIds{&slice}; }
336
337public:
338 Bucket() = default;
339 Bucket(const Bucket&) = default;
340 Bucket(Bucket&&) = default;
341 Bucket& operator=(Bucket&&) = default;
342
344 Bucket& operator=(const Bucket& other) {
345 if (this != &other) {
346 Bucket copy(other);
347 *this = std::move(copy);
348 }
349 return *this;
350 }
351
353 bool empty() const noexcept { return slices_.empty(); }
354
356 std::chrono::system_clock::time_point period_end() const noexcept { return period_end_; }
357
359 bool holds(const BarEvent& event) const noexcept {
360 for (const auto& slice : slices_) {
361 if (slice.interval_ == event.interval_) {
362 if (slice.bars_.size() > id_detail::scan_limit) {
363 if (const auto* entry = wide(event.interval_)) {
364 return entry->table_.find(event.instrument_, ids_of(slice)).has_value();
365 }
366 }
367 return slice.find(event.instrument_) != nullptr;
368 }
369 }
370 return false;
371 }
372
374 void add(const BarEvent& event) {
375 if (slices_.empty()) {
376 period_end_ = event.period_end_;
377 }
378 for (auto& slice : slices_) {
379 if (slice.interval_ == event.interval_) {
380 slice.bars_.push_back(event);
381 // If the table cannot grow, the bar comes out again, so the two never disagree.
382 try {
383 if (slice.bars_.size() == id_detail::scan_limit + 1) {
384 widen(slice);
385 } else if (slice.bars_.size() > id_detail::scan_limit) {
386 if (auto* entry = wide(slice.interval_)) {
387 entry->table_.insert(event.instrument_, slice.bars_.size() - 1, ids_of(slice));
388 }
389 }
390 } catch (...) {
391 slice.bars_.pop_back();
392 throw;
393 }
394 return;
395 }
396 }
397 BarSlice slice;
398 slice.period_end_ = event.period_end_;
399 slice.interval_ = event.interval_;
400 slice.bars_.push_back(event);
401 slices_.push_back(std::move(slice));
402 }
403
405 BarSlice take_first() {
406 BarSlice slice = std::move(slices_.front());
407 slices_.erase(slices_.begin());
408 return slice;
409 }
410
412 std::vector<BarSlice> take() {
413 std::vector<BarSlice> slices = std::move(slices_);
414 slices_.clear();
415 return slices;
416 }
417
419 void clear() noexcept { slices_.clear(); }
420
421private:
422 // An interval's table, which describes that interval's slice only while the slice holds more than
423 // `id_detail::scan_limit` bars: `widen` refills it when a slice passes that size and `add` extends it after,
424 // and nothing reads it while the slice is narrower. It outlives the slice, so the next wide slice of the
425 // interval reuses its storage; a bucket that never holds a wide slice has none.
426 struct Wide {
427 BarInterval interval_;
428 id_detail::PositionTable table_;
429 };
430
431 const Wide* wide(BarInterval interval) const noexcept {
432 for (const auto& entry : wide_) {
433 if (entry.interval_ == interval) {
434 return &entry;
435 }
436 }
437 return nullptr;
438 }
439
440 Wide* wide(BarInterval interval) noexcept {
441 for (auto& entry : wide_) {
442 if (entry.interval_ == interval) {
443 return &entry;
444 }
445 }
446 return nullptr;
447 }
448
449 // Refills the interval's table from its slice, which has just passed `id_detail::scan_limit` bars. On a
450 // failure the table is left empty, and the caller takes the bar back out, so nothing reads it.
451 void widen(const BarSlice& slice) {
452 Wide* entry = wide(slice.interval_);
453 if (entry == nullptr) {
454 wide_.push_back(Wide{slice.interval_, {}});
455 entry = &wide_.back();
456 }
457 entry->table_.reset();
458 try {
459 for (std::size_t at = 0; at < slice.bars_.size(); ++at) {
460 entry->table_.insert(slice.bars_[at].instrument_, at, ids_of(slice));
461 }
462 } catch (...) {
463 entry->table_.reset();
464 throw;
465 }
466 }
467
468 std::chrono::system_clock::time_point period_end_{};
469 std::vector<BarSlice> slices_;
470 // At most one entry per interval that has had a wide slice.
471 std::vector<Wide> wide_;
472};
