65inline std::chrono::system_clock::time_point parse_compact_expiry(std::string_view text) {
66 if (text.size() != 14) {
67 return {};
68 }
69 for (const char digit : text) {
70 if (digit < '0' || digit > '9') {
71 return {};
72 }
73 }
74 const auto number = [text](std::size_t offset, std::size_t length) {
75 int value = 0;
76 for (std::size_t index = offset; index < offset + length; ++index) {
77 value = value * 10 + (text[index] - '0');
78 }
79 return value;
80 };
81 std::tm parts{};
82 parts.tm_year = number(0, 4) - 1900;
83 parts.tm_mon = number(4, 2) - 1;
84 parts.tm_mday = number(6, 2);
85 parts.tm_hour = number(8, 2);
86 parts.tm_min = number(10, 2);
87 parts.tm_sec = number(12, 2);
88 parts.tm_isdst = 0;
89#ifdef _WIN32
90 const std::time_t utc = _mkgmtime(&parts) - 9 * 3600;
91#else
92 const std::time_t utc = timegm(&parts) - 9 * 3600;
93#endif
94 return std::chrono::system_clock::from_time_t(utc);
95}

233 void from_json_impl(const std::string& filename) {
234 std::ifstream ifs(filename);
235 if (!ifs.is_open()) {
236 throw std::runtime_error("Cannot open Kiwoom credential file: " + filename);
237 }
238 const std::string content(
239 std::istreambuf_iterator<char>(ifs), std::istreambuf_iterator<char>{}
240 );
241 boost::system::error_code ec;
242 const auto jv = json::parse(content, ec);
243 if (ec || !jv.is_object()) {
244 throw std::runtime_error(
245 "Failed to parse Kiwoom credentials file: " + filename +
246 (ec ? " — " + ec.message() : std::string(" — not a JSON object"))
247 );
248 }
249 const auto& doc = jv.as_object();
250 const json::value* key = doc.if_contains("appkey");
251 const json::value* secret = doc.if_contains("secretkey");
252 if (!key || !secret || !key->is_string() || !secret->is_string()) {
253 throw std::runtime_error("Credential file should contain appkey and secretkey as strings");
254 }
255 // Everything is read and checked before anything is assigned, so a refused file leaves this
256 // config -- and the endpoint it already selected -- exactly as it was.
257 const bool demo = demo_from_json(doc, filename);
258 key_ = key->get_string().c_str();
259 secret_ = secret->get_string().c_str();
260 is_demo_ = demo;
261 }

268 static bool demo_from_env() {
269 const char* demo = std::getenv("KIWOOM_DEMO");
270 if (!demo || std::string_view(demo) == "1") {
271 return true;
272 }
273 if (std::string_view(demo) == "0") {
274 return false;
275 }
276 throw std::runtime_error(
277 std::string("KIWOOM_DEMO is \"") + demo + "\"; set it to 1 for the demo endpoints or 0 to trade "
278 "live, or leave it unset for demo."
279 );
280 }
281
289 static bool demo_from_json(const json::object& doc, const std::string& filename) {
290 const json::value* demo = doc.if_contains("demo");
291 if (!demo) {
292 return true;
293 }
294 if (!demo->is_bool()) {
295 throw std::runtime_error(
296 "Kiwoom credential file " + filename + " has a \"demo\" that is not true or false; write "
297 "true for the demo endpoints or false to trade live, or leave it out for demo."
298 );
299 }
300 return demo->get_bool();
301 }
