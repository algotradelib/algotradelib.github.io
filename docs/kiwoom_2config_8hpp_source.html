65inline std::chrono::system_clock::time_point parse_compact_expiry(std::string_view text) {
66 if (text.size() != 14) {
67 return {};
68 }
69 for (const char digit : text) {
70 if (digit < '0' || digit > '9') {
71 return {};
72 }
73 }
74 const auto number = [text](std::size_t offset, std::size_t length) {
75 int value = 0;
76 for (std::size_t index = offset; index < offset + length; ++index) {
77 value = value * 10 + (text[index] - '0');
78 }
79 return value;
80 };
81 std::tm parts{};
82 parts.tm_year = number(0, 4) - 1900;
83 parts.tm_mon = number(4, 2) - 1;
84 parts.tm_mday = number(6, 2);
85 parts.tm_hour = number(8, 2);
86 parts.tm_min = number(10, 2);
87 parts.tm_sec = number(12, 2);
88 parts.tm_isdst = 0;
89#ifdef _WIN32
90 const std::time_t utc = _mkgmtime(&parts) - 9 * 3600;
91#else
92 const std::time_t utc = timegm(&parts) - 9 * 3600;
93#endif
94 return std::chrono::system_clock::from_time_t(utc);
95}

250 void from_json_text_impl(std::string_view text, const std::string& origin) {
251 boost::system::error_code ec;
252 const auto jv = json::parse(text, ec);
253 if (ec || !jv.is_object()) {
254 throw std::runtime_error(
255 "Failed to parse Kiwoom credentials " + origin +
256 (ec ? " — " + ec.message() : std::string(" — not a JSON object"))
257 );
258 }
259 const auto& doc = jv.as_object();
260 const json::value* key = doc.if_contains("appkey");
261 const json::value* secret = doc.if_contains("secretkey");
262 if (!key || !secret || !key->is_string() || !secret->is_string()) {
263 throw std::runtime_error("Kiwoom credential " + origin + " should contain appkey and secretkey as strings");
264 }
265 // Everything is read and checked before anything is assigned, so a refused document leaves this
266 // config -- and the endpoint it already selected -- exactly as it was.
267 const bool demo = demo_from_json(doc, origin);
268 key_ = key->get_string().c_str();
269 secret_ = secret->get_string().c_str();
270 is_demo_ = demo;
271 }

278 static bool demo_from_env() {
279 const char* demo = std::getenv("KIWOOM_DEMO");
280 if (!demo || std::string_view(demo) == "1") {
281 return true;
282 }
283 if (std::string_view(demo) == "0") {
284 return false;
285 }
286 throw std::runtime_error(
287 std::string("KIWOOM_DEMO is \"") + demo +
288 "\"; set it to 1 for the demo endpoints or 0 to trade "
289 "live, or leave it unset for demo."
290 );
291 }
292
300 static bool demo_from_json(const json::object& doc, const std::string& origin) {
301 const json::value* demo = doc.if_contains("demo");
302 if (!demo) {
303 return true;
304 }
305 if (!demo->is_bool()) {
306 throw std::runtime_error(
307 "Kiwoom credential " + origin +
308 " has a \"demo\" that is not true or false; write "
309 "true for the demo endpoints or false to trade live, or leave it out for demo."
310 );
311 }
312 return demo->get_bool();
313 }
