365inline std::optional<std::string> salvage_string(std::string_view line, std::string_view key) {
366 std::string needle;
367 needle.reserve(key.size() + 4);
368 needle += '"';
369 needle.append(key);
370 needle += "\":\"";
371 const auto at = line.find(needle);
372 if (at == std::string_view::npos) {
373 return std::nullopt;
374 }
375 const auto begin = at + needle.size();
376 // **The closing quote is the first one not escaped**, not the first one. A value holding `\"`
377 // was cut short at the escape, and a symbol cut short is another symbol's key.
378 auto end = begin;
379 while (end < line.size() && line[end] != '"') {
380 end += line[end] == '\\' ? 2 : 1;
381 }
382 if (end >= line.size()) {
383 // The value itself was cut off mid-write. Reporting what is there would invent a symbol.
384 return std::nullopt;
385 }
386 // Decoded as the JSON string it is, so an escape reads as the character it stands for. A value
387 // whose escapes do not decode is not salvaged, for the same reason a cut-off one is not.
388 boost::system::error_code error;
389 const auto quoted = json::parse(line.substr(begin - 1, end - begin + 2), error);
390 if (error || !quoted.is_string()) {
391 return std::nullopt;
392 }
393 return std::string(quoted.as_string());
394}

411inline std::string checksum_text(std::uint32_t value) {
412 static constexpr char digits[] = "0123456789abcdef";
413 std::string text(8, '0');
414 for (int i = 7; i >= 0; --i) {
415 text[static_cast<std::size_t>(i)] = digits[value & 0xFu];
416 value >>= 4;
417 }
418 return text;
419}

422inline constexpr std::string_view kChecksumMember = ",\"crc\":\"";
423
425inline constexpr std::size_t kChecksumSuffixSize = kChecksumMember.size() + 8 + 2;
426
428inline bool ends_in_checksum(std::string_view line) {
429 if (line.size() <= kChecksumSuffixSize) {
430 return false;
431 }
432 const auto suffix = line.substr(line.size() - kChecksumSuffixSize);
433 if (suffix.substr(0, kChecksumMember.size()) != kChecksumMember ||
434 suffix.substr(kChecksumSuffixSize - 2) != "\"}") {
435 return false;
436 }
437 for (const char digit : suffix.substr(kChecksumMember.size(), 8)) {
438 if (!((digit >= '0' && digit <= '9') || (digit >= 'a' && digit <= 'f'))) {
439 return false;
440 }
441 }
442 return true;
443}
444
451inline bool checksum_matches(std::string_view line) {
452 std::string body(line.substr(0, line.size() - kChecksumSuffixSize));
453 body += '}';
454 const auto stated = line.substr(line.size() - kChecksumSuffixSize + kChecksumMember.size(), 8);
455 return checksum_text(line_checksum(body)) == stated;
456}

600inline std::string journal_entry_to_line(const JournalEntry& entry) {
601 json::object row;
602 // Ordered so that the fields a damaged line most needs to yield come first. A truncated write
603 // keeps its prefix, so putting the instrument ahead of the detail text is what makes salvage
604 // work in practice rather than only in principle.
605 row["id"] = entry.id_;
606 row["action"] = std::string(journal_action_name(entry.action_));
607 row["outcome"] = std::string(journal_outcome_name(entry.outcome_));
608 row["country"] = std::string(journal_detail::country_name(entry.instrument_.country_));
609 row["venue"] = std::string(journal_detail::venue_name(entry.instrument_.venue_));
610 row["symbol"] = entry.instrument_.symbol_;
611 row["side"] = std::string(journal_detail::side_name(entry.side_));
612 row["type"] = std::string(journal_detail::order_type_name(entry.type_));
613 row["time_in_force"] = std::string(journal_detail::time_in_force_name(entry.time_in_force_));
614 row["session"] = std::string(journal_detail::order_session_name(entry.session_));
615 // Distinct from `venue`, which is the instrument's identity venue. This is
616 // where the order actually goes -- the binding resolves a request that named
617 // no venue into the one the broker's default reaches -- and a line that omits
618 // the key entirely is
619 // one written before the field existed -- which is not the same as one that
620 // asked for no particular venue, and recovery treats the two differently.
621 if (entry.venue_) {
622 row["route"] = std::string(journal_detail::venue_name(*entry.venue_));
623 }
624 row["quantity"] = entry.quantity_.str();
625 if (entry.price_) {
626 row["price"] = entry.price_->str();
627 }
628 // Only when present, so a line that is not a stop carries no `trigger` key and every old journal
629 // reads unchanged. (Every new line ends in a `crc` member, added below; the fields before it are
630 // what an earlier writer produced.)
631 if (entry.trigger_price_) {
632 row["trigger"] = entry.trigger_price_->str();
633 }
634 // Only on a serialized runner's lines, so an overlapping runner's journal reads as before.
635 if (entry.baseline_) {
636 row["baseline"] = entry.baseline_->str();
637 }
638 // The authoritative instant, exact on every platform. `at_text` beside it is for the operator
639 // reading the file with `tail`, is only second-resolution, and is never read back: one field is
640 // the record and the other is the caption.
641 row["at"] = journal_detail::micros_of(entry.recorded_at_);
642 row["at_text"] = format_iso8601_utc(entry.recorded_at_);
643 if (!entry.order_id_.empty()) {
644 row["order_id"] = entry.order_id_;
645 }
646 if (!entry.detail_.empty()) {
647 row["detail"] = entry.detail_;
648 }
649 if (entry.requires_operator_) {
650 row["requires_operator"] = true;
651 }
652 if (!entry.scope_.empty()) {
653 row["scope"] = entry.scope_;
654 }
655 // **Last, and over the line as it reads without it** -- the whole object, closing brace included,
656 // which the member then replaces. See `journal_detail::line_checksum` for what it detects
657 // and `journal_entry_from_line` for what a line that fails it becomes.
658 auto line = json::serialize(row);
659 const auto checksum = journal_detail::checksum_text(journal_detail::line_checksum(line));
660 line.pop_back();
661 line.append(journal_detail::kChecksumMember);
662 line += checksum;
663 line += "\"}";
664 return line;
665}

695inline std::optional<JournalEntry> journal_entry_from_line(std::string_view line) {
696 boost::system::error_code error;
697 const auto value = json::parse(line, error);
698 if (error || !value.is_object()) {
699 return std::nullopt;
700 }
701 const auto& row = value.as_object();
702
703 const auto action = journal_action_from(journal_detail::string_field(row, "action"));
704 const auto outcome = journal_outcome_from(journal_detail::string_field(row, "outcome"));
705 // An operator-only line is exempt from the bound below, and deliberately: a
706 // salvaged fragment records the epoch precisely *because* it has no
707 // timestamp, and `requires_operator_` is what stops that epoch window
708 // resolving it. Repairing such a line writes it back, so rejecting the epoch
709 // here would make every repair unreadable on the next open. The bound is for
710 // a line that claims to be ordinary and carries an instant it cannot have.
711 const auto operator_only = row.if_contains("requires_operator");
712 const bool marked = operator_only != nullptr && operator_only->is_bool() &&
713 operator_only->as_bool();
714
715 const auto at = row.if_contains("at");
716 if (!action || !outcome || at == nullptr || !at->is_int64() ||
717 (!marked && !journal_detail::plausible_instant(at->as_int64()))) {
718 // A well-formed JSON object that is not a journal line. Treated as damage rather than as an
719 // empty entry, because an entry with a default instrument would be a claim about an order.
720 return std::nullopt;
721 }
722
723 // **A present value this reader cannot parse is damage, not a default.**
724 // Each of these has a documented default for an *absent* key -- a line that
725 // predates the field -- and applying that default to a token the line does
726 // carry turns damage into evidence: a damaged `"SELL"` reads as a buy, a
727 // damaged `"IOC"` as a day order, and a damaged `"COMBINED"` moves the entry
728 // to an instrument the runner is not trading, so the ledger marks the wrong
729 // key and the right one is never withheld at all.
730 //
731 // An unreadable value therefore produces an entry with **no instrument**,
732 // which `unattributable_journal_entry` documents and recovery already turns
733 // into a global halt. Marking a guessed instrument would not protect the real
734 // one, because they are different keys.
735 // Each field is asked two questions, because they have different answers.
736 // *Is the key there?* -- if not, the line predates the field and its
737 // documented default is a fact about such a line. *Is what it holds
738 // readable?* -- if not, the line says something this release cannot use, and
739 // guessing turns damage into evidence. `string_field` answers neither: it
740 // flattens absent, non-string and empty into the same `""`.
741 const auto read = [&row](std::string_view key) {
742 return journal_detail::readable_string(row, key);
743 };
744 // Present but unusable, for any reason: not a string, an empty string, or a
745 // token the reader could not parse. The first version of this asked only
746 // whether the *string* was readable, which let `"IOC"` damaged to `"IO?"`
747 // through -- that is a perfectly good string, and a perfectly bad condition.
748 const auto unusable = [&row](std::string_view key, bool parsed) {
749 return journal_detail::has_field(row, key) && !parsed;
750 };
751
752 // `std::optional::and_then` is C++23 and this library is C++17.
753 const auto parsed = [&read](std::string_view key, auto reader) {
754 using Result = decltype(reader(std::string_view{}));
755 const auto text = read(key);
756 return text ? reader(*text) : Result{};
757 };
758
759 // A `crc` member anywhere but last, or not eight lowercase hex digits, is not this writer's; one
760 // that is must match the line without it -- the object with its own closing brace. See
761 // `journal_detail::line_checksum`.
762 const bool checksum_holds = !journal_detail::has_field(row, "crc") ||
763 (journal_detail::ends_in_checksum(line) && journal_detail::checksum_matches(line));
764 bool unknown_key = false;
765 for (const auto& member : row) {
766 if (!journal_detail::known_entry_key(member.key())) {
767 unknown_key = true;
768 }
769 }
770 // **A line whose bytes changed is not read at all**, rather than read as a damaged entry. A damaged
771 // entry carries its own outcome and an id suffixed `~damaged`, which is right for a line whose
772 // *value* no longer parses -- but a damaged *terminal* line with an intact pending partner then
773 // pairs with nothing, is dropped as an orphan, and the pending line stays open as an ordinary one:
774 // the change was detected and then discarded before recovery decided anything. An unreadable line
775 // is counted by `parse_journal_bytes` and halts recovery outright, whatever it was.
776 if (!checksum_holds || unknown_key) {
777 return std::nullopt;
778 }
779
780 const auto symbol = read("symbol");
781 const auto country = parsed("country", journal_detail::known_country_from);
782 const auto venue = parsed("venue", journal_detail::known_venue_from);
783 const auto side = parsed("side", journal_detail::known_side_from);
784 const auto type = parsed("type", journal_detail::known_order_type_from);
785 const auto condition = parsed("time_in_force", journal_detail::known_time_in_force_from);
786 const auto session = parsed("session", journal_detail::known_order_session_from);
787 const auto quantity = parsed("quantity", journal_detail::known_decimal_from);
788 const auto price = parsed("price", journal_detail::known_decimal_from);
789 const auto trigger = parsed("trigger", journal_detail::known_decimal_from);
790 const auto baseline = parsed("baseline", journal_detail::known_decimal_from);
791
792 const bool damaged =
793 // Identity and the fields every complete line carries: required outright.
794 // A line this reader cannot attribute to an instrument must not be
795 // attributed to a guess, and `quantity` at a silent zero is a claim about
796 // an order's size.
797 !symbol || !country || !venue || !side || !type || !quantity ||
798 // Legacy-defaulted fields: absent is fine, present and unreadable is not.
799 unusable("time_in_force", condition.has_value()) ||
800 unusable("session", session.has_value()) ||
801 // A price is absent for a market order and required for a basis that
802 // carries one, so its presence is judged against the basis rather than
803 // assumed either way.
804 unusable("price", price.has_value()) ||
805 // The same rule for a stop's trigger: absent on every other order and on every line written
806 // before the field existed, and a present value that does not parse is damage.
807 unusable("trigger", trigger.has_value()) ||
808 unusable("baseline", baseline.has_value()) ||
809 // Checked above, where a changed line becomes unreadable rather than damaged; restated so the
810 // condition reads whole.
811 !checksum_holds ||
812 unknown_key;
813 // **A price missing from a priced basis, and a cancellation with no target
814 // id, are deliberately *not* damage here.** Both are lines this library's own
815 // writer emits: `JournalingExecutor` appends the pending line *before* the
816 // executor runs, so an order the binding then refuses locally -- a limit with
817 // no price, a cancellation of a reservation whose broker id is not known --
818 // has already been written by the time anyone could object. Rejecting them
819 // turned an ordinary local refusal into a global halt on the next restart,
820 // which is worse than the hazard the rejection was meant to close.
821 //
822 // The hazard is real and it lives one step later: such a line must not be
823 // *resolved* on evidence it does not have. `runner::match_against` refuses to
824 // conclude anything from one, which leaves it withheld for an operator while
825 // its own terminal line still closes it when the refusal was ordinary.
826 if (damaged) {
827 auto unattributable = unattributable_journal_entry(line);
828 const auto id = journal_detail::string_field(row, "id");
829 unattributable.id_ = (id.empty() ? std::string("unknown") : id) + "~damaged";
830 unattributable.action_ = *action;
831 // Its own outcome, not `PENDING`. Both lines of a pair carry the same id, so
832 // a damaged *terminal* line still closes its damaged pending partner --
833 // forcing `PENDING` left both open and halted a runner over an instruction
834 // that had already resolved.
835 unattributable.outcome_ = *outcome;

914 const auto symbol = journal_detail::salvage_string(line, "symbol");
915 if (!symbol || symbol->empty()) {
916 return std::nullopt;
917 }
918 JournalEntry entry;
919 // The flag that makes the promise in this comment block true. Without it the entry's missing
920 // timestamp reads as the epoch, every window test passes immediately, no real row falls inside
921 // the window, and the first sweep resolves it `NOT_PLACED`.
922 entry.requires_operator_ = true;
923 const auto id = journal_detail::salvage_string(line, "id");
924 entry.id_ = (id ? *id : std::string("unknown")) + "~truncated";
925 entry.action_ =
926 journal_action_from(journal_detail::salvage_string(line, "action").value_or("place"))
927 .value_or(JournalAction::PLACE);

929 // **Absent is a default; present and unreadable is damage** -- the same rule
930 // the strict reader follows, and for the same reason. A truncated line is
931 // missing its *tail*, so an absent field is expected and its default is a fact
932 // about what the line could have said. A field that is there and cannot be
933 // read is corruption in the middle, and guessing it hands this entry a
934 // different instrument than the order had: `requires_operator_` would then
935 // withhold a key the runner is not trading under, while the real one stays
936 // open. Such a line becomes unattributable instead, carrying no instrument at
937 // all, which recovery turns into a global halt.
938 // A key the truncation removed takes its default; a key that is *there* and
939 // unreadable is damage, and `salvage_string` cannot tell those apart on its
940 // own -- it recognizes `"key":"value"` and nothing else, so `null` and `7`
941 // both read as absent.

943 const auto text = journal_detail::salvage_string(line, key);
944 if (text) {
945 return std::optional<std::string>(*text);
946 }
947 return salvage_damaged(line, key) ? std::optional<std::string>()

949 };

951 using Result = decltype(reader(std::string_view{}));
952 const auto text = salvaged(key, fallback);
953 return text ? reader(*text) : Result{};
954 };

958 const auto side = parsed("side", "BUY", journal_detail::known_side_from);
959 const auto type = parsed("type", "UNKNOWN", journal_detail::known_order_type_from);
960 const auto condition =
961 parsed("time_in_force", "DAY", journal_detail::known_time_in_force_from);
962 const auto session = parsed("session", "REGULAR", journal_detail::known_order_session_from);
963 if (!country || !venue || !side || !type || !condition || !session) {
964 auto unattributable = unattributable_journal_entry(line);
965 unattributable.id_ = entry.id_;
966 unattributable.detail_ =
967 "a truncated journal line carries a value this release cannot read, so the order it "
968 "describes cannot be identified; nothing may be traded until an operator resolves it: " +
969 std::string(line.substr(0, line.size() < 160 ? line.size() : 160));

971 }
972 entry.instrument_ = InstrumentId{*country, *venue, *symbol};
973 entry.side_ = *side;
974 entry.type_ = *type;
975 entry.time_in_force_ = *condition;
976 entry.session_ = *session;
977 entry.venue_ = journal_detail::known_venue_from(
978 journal_detail::salvage_string(line, "route").value_or("")
979 );
980 if (const auto quantity = journal_detail::salvage_string(line, "quantity")) {
981 // A truncated line's last field is routinely half-written, so an unparseable
982 // number here is expected rather than exceptional -- and it must not throw
983 // out of a salvage path that exists to survive exactly this.
984 if (const auto value = journal_detail::known_decimal_from(*quantity)) {
985 entry.quantity_ = *value;
986 }
987 }
988 if (const auto price = journal_detail::salvage_string(line, "price")) {
989 if (const auto value = journal_detail::known_decimal_from(*price)) {

993 if (const auto trigger = journal_detail::salvage_string(line, "trigger")) {
994 if (const auto value = journal_detail::known_decimal_from(*trigger)) {
995 entry.trigger_price_ = *value;
996 }
997 }

1009inline JournalBytes read_journal_bytes(const std::string& path) {
1010 JournalBytes bytes;
1011 errno = 0;
1012 std::unique_ptr<std::FILE, int (*)(std::FILE*)> handle(
1013 std::fopen(path.c_str(), "rb"), [](std::FILE* file) { return file ? std::fclose(file) : 0; }
1014 );
1015 if (handle == nullptr) {
1016 if (errno == ENOENT) {
1017 bytes.status_ = JournalReadStatus::MISSING;
1018 return bytes;
1019 }
1020 // Every other reason -- a permission the process does not have, a directory where a file was
1021 // expected, a device error -- means the file exists and its contents are unknown. Reporting
1022 // that as "no journal" is how a process claims a path holding its predecessor's in-flight
1023 // orders and then trades on an empty ledger.
1024 bytes.status_ = JournalReadStatus::UNREADABLE;
1025 bytes.detail_ = "the order journal at " + path + " exists and could not be opened for reading";
1026 return bytes;
1027 }
1028
1029 char buffer[4096];
1030 while (const auto read = std::fread(buffer, 1, sizeof(buffer), handle.get())) {
1031 bytes.text_.append(buffer, read);
1032 }
1033 if (std::ferror(handle.get()) != 0) {
1034 // A read that stopped partway looks exactly like an ordinary end of file to the loop above, so
1035 // the truncated tail it produces would be a fiction about where the journal ends.
1036 bytes.status_ = JournalReadStatus::UNREADABLE;
1037 bytes.detail_ = "the order journal at " + path + " could not be read to the end";
1038 bytes.text_.clear();
1039 return bytes;
1040 }
1041 bytes.status_ = JournalReadStatus::READ;
1042 return bytes;
1043}

1056inline JournalReplay parse_journal_bytes(const JournalBytes& bytes) {
1057 JournalReplay replay;
1058 replay.status_ = bytes.status_;
1059 if (bytes.status_ != JournalReadStatus::READ) {
1060 replay.detail_ = bytes.detail_;
1061 return replay;
1062 }
1063
1064 const auto& text = bytes.text_;
1065 std::size_t begin = 0;
1066 // Set by the first checksummed entry line; see the check below.
1067 bool seen_checksum = false;
1068 while (begin < text.size()) {
1069 const auto end = text.find('\n', begin);
1070 if (end == std::string::npos) {
1071 // Everything after the last newline is a write that never finished. This is the whole reason
1072 // the format is line-oriented, and dropping it silently would defeat the feature.
1073 replay.truncated_tail_ = true;
1074 const std::string_view tail(text.data() + begin, text.size() - begin);
1075 if (auto salvaged = salvage_journal_line(tail)) {
1076 replay.detail_ = "the final journal line was truncated; it named " +
1077 salvaged->instrument_.symbol_ + " and is treated as unresolved";
1078 replay.entries_.push_back(std::move(*salvaged));
1079 } else {
1080 replay.detail_ =
1081 "the final journal line was truncated and named no instrument; something was in flight";
1082 // The same entry a writer's repair appends for this tail, so recovery halts on it whether or
1083 // not the file was repaired, and whatever else is open beside it.
1084 replay.entries_.push_back(unattributable_journal_entry(tail));
1085 }
1086 break;
1087 }
1088 const std::string_view line(text.data() + begin, end - begin);
1089 begin = end + 1;
1090 // **An empty line, or a header anywhere but first, is damage.** This writer produces neither: every
1091 // append is one non-empty line and its newline, and the header is written once, before anything
1092 // else. Skipping them was how an entry overwritten by a bare newline, or by a copy of the header,
1093 // vanished from recovery with nothing counted -- a changed line removed rather than detected.
1094 const auto unreadable = [&replay](const char* why) {
1095 ++replay.unreadable_lines_;
1096 if (replay.detail_.empty()) {
1097 replay.detail_ = why;
1098 }
1099 };
1100 if (line.empty()) {
1101 unreadable("the journal contains an empty line, which this library never writes");
1102 continue;
1103 }
1104 if (auto scope = journal_header_scope(line)) {
1105 if (replay.has_header_ || !replay.entries_.empty() || replay.unreadable_lines_ > 0) {
1106 unreadable("the journal contains a header after its first line, which this library never "
1107 "writes");
1108 continue;
1109 }
1110 // Not an instruction and not damage. Recorded so an opener can check ownership before it
1111 // writes anything at all.
1112 replay.has_header_ = true;
1113 replay.scope_ = std::move(*scope);
1114 // A header that declares checksums protects every entry after it, the first one included.
1115 seen_checksum = seen_checksum || journal_header_declares_checksums(line);
1116 continue;
1117 }
1118 const bool checksummed = journal_detail::ends_in_checksum(line);
1119 if (seen_checksum && !checksummed) {
1120 // **This writer never puts a checksum-less line after a checksummed one, or after a header that
1121 // declares checksums**, so this is a protected line that lost its checksum -- damage that removed
1122 // the whole member, which the line's own reader cannot tell from a line written before checksums
1123 // existed. Only the first checksummed line of a file begun by an earlier release is left without
1124 // this protection; see `journal_header_line`.
1125 ++replay.unreadable_lines_;
1126 if (replay.detail_.empty()) {
1127 replay.detail_ =
1128 "a journal line without a checksum follows a checksummed one, so it was changed after it "
1129 "was written";
1130 }
1131 continue;
1132 }
1133 seen_checksum = seen_checksum || checksummed;
1134 if (auto entry = journal_entry_from_line(line)) {
1135 replay.entries_.push_back(std::move(*entry));
1136 } else {
1137 // A complete line that is not a journal entry. Not expected from this writer, so it means
1138 // the file was edited or written by something else, or a repair was interrupted; counted
1139 // rather than ignored, and recovery halts on it.
1140 ++replay.unreadable_lines_;
1141 if (replay.detail_.empty()) {
1142 replay.detail_ = "the journal contains a line this library did not write";
1143 }
1144 }
1145 }
1146 return replay;
1147}

1306 // Returns whether this journal may write to the file, deciding ownership from the header alone.

1308 // **The header, not the records, and that distinction is the fix.** Reading ownership from the
1309 // records leaves a hole exactly where it matters most: the scope is one field among many on a
1310 // record, so a process killed partway through writing the *first* instruction leaves a file whose
1311 // only line is torn and unscoped. Another account could then open the path, repair that line into
1312 // its own journal, and reconcile the first account's instruction against its own rows.
1313 //
1314 // A header is written before any instruction is contemplated, so it is complete or the file is
1315 // empty. Three cases, and the third is the strict one:
1316 //
1317 // - **Empty or absent.** Ours to claim; the header is written before the first append.
1318 // - **Header present.** The scopes must agree.
1319 // - **Non-empty with no header.** Refused. Something else wrote this file, or it predates the
1320 // format, and either way its ownership cannot be established -- which is not a question to
1321 // guess at while holding a file that may name another account's live orders.
