27inline math::Vector returns(math::SeriesView prices) {
28 const auto rows = prices.size();
29 math::Vector result = math::unobserved(static_cast<std::size_t>(rows));
30 if (rows < 2) {
31 return result;
32 }
33 // One vectorized division over the whole series, then the guards. An unobserved price on either
34 // side makes the quotient NaN by itself, which is already the answer; a zero prior price would
35 // make it an infinity and a negative one reverses its sign, so those rows are replaced explicitly.
36 const auto current = prices.tail(rows - 1).array();
37 const auto previous = prices.head(rows - 1).array();
38 result.tail(rows - 1) =
39 (previous <= Scalar{0}).select(missing(), current / previous - Scalar{1});
40 return result;
41}

53inline math::Vector log_returns(math::SeriesView prices) {
54 const auto rows = prices.size();
55 math::Vector result = math::unobserved(static_cast<std::size_t>(rows));
56 if (rows < 2) {
57 return result;
58 }
59 const auto current = prices.tail(rows - 1).array();
60 const auto previous = prices.head(rows - 1).array();
61 // A non-positive price on either side is refused before the logarithm rather than left to
62 // produce a bare NaN with nothing to say which row or which side caused it.
63 const auto usable = (previous > Scalar{0}) && (current > Scalar{0});
64 result.tail(rows - 1) = usable.select((current / previous).log(), missing());
65 return result;
66}
