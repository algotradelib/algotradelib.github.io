1// A runner killed with an order in flight, and the restart that finds out what happened.
2//
3// No broker, no credentials, no network, and no signal raised. The "crash" is a scope that ends
4// without the terminal line ever being written, which is exactly what a `SIGKILL` leaves on disk,
5// and the "restart" is a fresh ledger reading the file its predecessor left behind.
6//
7// The point it demonstrates is the one sentence the whole feature rests on: **an empty ledger does
8// not mean nothing is owed -- it means nothing is remembered.**
9//
10// Run it as:
11//
12// example-order-journal
13

72void show_journal(const std::string& path) {
73 std::FILE* handle = std::fopen(path.c_str(), "rb");
74 if (handle == nullptr) {
75 std::cout << " (no journal)\n";
76 return;
77 }
78 std::string text;
79 char buffer[1024];
80 while (const auto read = std::fread(buffer, 1, sizeof(buffer), handle)) {
81 text.append(buffer, read);
82 }
83 std::fclose(handle);
84
85 std::size_t begin = 0;
86 while (begin < text.size()) {
87 const auto end = text.find('\n', begin);
88 const auto stop = end == std::string::npos ? text.size() : end;
89 std::cout << " " << text.substr(begin, stop - begin);
90 if (end == std::string::npos) {
91 std::cout << " <- no newline: this write never finished";
92 }
93 std::cout << "\n";
94 if (end == std::string::npos) {
95 break;
96 }
97 begin = end + 1;
98 }
99}

117int main() {
118 const std::string path = "example-order-journal.jsonl";
119 std::remove(path.c_str());
120
121 std::cout << "1. A runner sends one order, and is killed before it learns the outcome.\n\n";
122 {
123 runner::FileJournal journal(path, "example:demo-account");
124 if (!journal.open()) {
125 std::cout << "could not open " << path << " for appending: " << journal.rejection() << "\n";
126 return 1;
127 }
128 runner::JournalingExecutor<KilledExecutor<runner::DryTag>, runner::FileJournal,
129 runner::ManualClock>
130 executor{KilledExecutor<runner::DryTag>{}, journal, runner::ManualClock{sent_at()}};
131 executor.set_id_prefix("run-a");
132 try {
133 executor.place(buy_ten());
134 } catch (const std::exception& error) {
135 std::cout << " " << error.what() << "\n\n";
136 }
137 }
138
139 std::cout << " What the file holds:\n";
140 show_journal(path);
141 std::cout
142 << "\n One line, and it says `pending`. The order may be resting on the exchange or may\n"
143 << " never have arrived. Nothing in this process knows which, and the process is gone.\n\n";
144
145 std::cout << "2. A new process starts. Its ledger is empty.\n\n";
146 runner::IndeterminacyLedger ledger;
147 std::cout << " Before recovery, may it trade " << samsung().symbol_
148 << "? " << (ledger.requires_reconcile(samsung()) ? "no" : "yes")
149 << " <- and that is the trap\n";
150
151 const auto recovery = runner::recover_into(runner::read_journal_file(path), ledger);
152 std::cout << " After recovery: "
153 << (ledger.requires_reconcile(samsung()) ? "no" : "yes") << "\n";
154 std::cout << " Instruments marked: " << recovery.marked_.size()
155 << ", instructions owed: " << recovery.open_.size() << "\n\n";
156
157 std::cout << "3. The broker's own order list is read, and it is the only thing that resolves.\n\n";
158
159 // Too soon. An accepted write does not appear in a listing instantly, so an absent row this early
160 // is not evidence that the order does not exist.
161 runner::FileJournal journal(path, "example:demo-account");
162 const auto early = runner::reconcile_against(
163 recovery.open_, {}, sent_at() + std::chrono::seconds(5), ledger, journal
164 );
165 std::cout << " Five seconds later, an empty listing: "
166 << runner::match_verdict_name(early.matches_.front().verdict_)
167 << " -- still withheld\n";
168
169 // Later, the row is there. The order existed after all.
170 const auto settled = runner::reconcile_against(
171 recovery.open_, {listed_order()}, sent_at() + std::chrono::minutes(10), ledger, journal
172 );
173 std::cout << " Ten minutes later, the order is listed: "
174 << runner::match_verdict_name(settled.matches_.front().verdict_) << " as "
175 << settled.matches_.front().order_id_ << "\n";
176 std::cout << " May it trade " << samsung().symbol_ << " now? "
177 << (ledger.requires_reconcile(samsung()) ? "no" : "yes") << "\n\n";
178
179 std::cout << "4. The resolution is written back, so a second restart finds nothing owed.\n\n";
180 std::cout << " What the file holds now:\n";
181 show_journal(path);
182
183 runner::IndeterminacyLedger later;
184 const auto again = runner::recover_into(runner::read_journal_file(path), later);
185 std::cout << "\n A third process recovers: instructions owed: " << again.open_.size()
186 << ", may it trade " << samsung().symbol_ << "? "
187 << (later.requires_reconcile(samsung()) ? "no" : "yes") << "\n\n";
188
189 std::cout
190 << "Had the listing stayed empty past the window, the verdict would have been `not_placed`:\n"
191 << "no order found, and the ledger would have cleared just the same. A serialized runner goes\n"
192 << "on owing such an order all the same, since a list can lag an order it accepted. Had two\n"
193 << "rows matched, it would have been `ambiguous`, nothing would clear, and an operator would\n"
194 << "decide. The one thing that never happens is a guess.\n";
195
196 std::remove(path.c_str());
197 return 0;
198}
