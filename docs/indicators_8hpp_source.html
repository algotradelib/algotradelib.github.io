38inline math::Vector wilder(math::SeriesView values, std::size_t period) {
39 const auto span = static_cast<Scalar>(period);
40 math::Vector result = math::unobserved(static_cast<std::size_t>(values.size()));
41 Scalar average{0};
42 Scalar seed{0};
43 std::size_t count = 0;
44 for (math::Index row = 0; row < values.size(); ++row) {
45 const Scalar value = values[row];
46 if (!observed(value)) {
47 continue;
48 }
49 ++count;
50 if (count < period) {
51 seed += value;
52 continue;
53 }
54 if (count == period) {
55 average = (seed + value) / span;
56 } else {
57 average = (average * (span - Scalar{1}) + value) / span;
58 }
59 result[row] = average;
60 }
61 return result;
62}

80inline Paired pair_series(math::SeriesView left, math::SeriesView right) {
81 const auto left_parts = math::observed_parts(left);
82 const auto right_parts = math::observed_parts(right);
83 Paired paired;
84 paired.present_ = math::both_present(left_parts.present_, right_parts.present_);
85 paired.left_ = left_parts.values_.cwiseProduct(paired.present_);
86 paired.right_ = right_parts.values_.cwiseProduct(paired.present_);
87 return paired;
88}

99inline Pairing reduce_pair(
100 const Paired& paired,
101 math::Index first,
102 math::Index length,
103 std::size_t needed
104) {
105 Pairing result;
106 const Scalar count = paired.present_.segment(first, length).sum();
107 result.count_ = static_cast<std::size_t>(count);
108 if (result.count_ < needed || result.count_ < 2) {
109 return result;
110 }
111 result.mean_left_ = paired.left_.segment(first, length).sum() / count;
112 result.mean_right_ = paired.right_.segment(first, length).sum() / count;
113 result.reported_ = true;
114 return result;
115}

154inline math::Vector zscore(
155 math::SeriesView values,
156 std::size_t window,
157 std::size_t min_observed
158) {
159 rolling_detail::check(window, min_observed);
160 const auto parts = math::observed_parts(values);
161 math::Vector result = math::unobserved(static_cast<std::size_t>(values.size()));
162 for (math::Index row = 0; row < values.size(); ++row) {
163 if (!observed(values[row])) {
164 continue;
165 }
166 const math::Index first = rolling_detail::window_start(row, window);
167 const auto spread = rolling_detail::dispersion(parts, first, row - first + 1, min_observed);
168 if (!spread.reported_) {
169 continue;
170 }
171 if (spread.deviation_ == Scalar{0}) {
172 // The data was there and the statistic does not exist for it, which is a different
173 // answer from "no data yet" and now says so.
174 result[row] = math::undefined();
175 continue;
176 }
177 result[row] = (values[row] - spread.mean_) / spread.deviation_;
178 }
179 return result;
180}

220inline BollingerBands bollinger(
221 math::SeriesView values,
222 std::size_t window,
223 Scalar multiplier,
224 std::size_t min_observed
225) {
226 rolling_detail::check(window, min_observed);
227 const auto parts = math::observed_parts(values);
228 const auto rows = static_cast<std::size_t>(values.size());
229 BollingerBands bands{math::unobserved(rows), math::unobserved(rows), math::unobserved(rows)};
230 for (math::Index row = 0; row < values.size(); ++row) {
231 const math::Index first = rolling_detail::window_start(row, window);
232 const auto spread = rolling_detail::dispersion(parts, first, row - first + 1, min_observed);
233 if (!spread.reported_) {
234 continue;
235 }
236 bands.middle_[row] = spread.mean_;
237 bands.upper_[row] = spread.mean_ + multiplier * spread.deviation_;
238 bands.lower_[row] = spread.mean_ - multiplier * spread.deviation_;
239 }
240 return bands;
241}

275inline math::Vector rsi(math::SeriesView prices, std::size_t period = 14) {
276 if (period == 0) {
277 throw std::invalid_argument("rsi period must be at least one row");
278 }
279 const auto rows = static_cast<std::size_t>(prices.size());
280 math::Vector gains = math::unobserved(rows);
281 math::Vector losses = math::unobserved(rows);
282 for (math::Index row = 1; row < prices.size(); ++row) {
283 if (!observed(prices[row - 1]) || !observed(prices[row])) {
284 continue;
285 }
286 const Scalar change = prices[row] - prices[row - 1];
287 gains[row] = std::max(change, Scalar{0});
288 losses[row] = std::max(-change, Scalar{0});
289 }
290
291 const math::Vector average_gain = indicator_detail::wilder(gains, period);
292 const math::Vector average_loss = indicator_detail::wilder(losses, period);
293
294 math::Vector result = math::unobserved(rows);
295 for (math::Index row = 0; row < prices.size(); ++row) {
296 if (!observed(average_gain[row]) || !observed(average_loss[row])) {
297 continue;
298 }
299 if (average_loss[row] == Scalar{0}) {
300 result[row] = average_gain[row] == Scalar{0} ? Scalar{50} : Scalar{100};
301 continue;
302 }
303 const Scalar strength = average_gain[row] / average_loss[row];
304 result[row] = Scalar{100} - Scalar{100} / (Scalar{1} + strength);
305 }
306 return result;
307}

321inline math::Vector true_range(
322 math::SeriesView highs,
323 math::SeriesView lows,
324 math::SeriesView closes
325) {
326 indicator_detail::require_same_rows(highs, lows);
327 indicator_detail::require_same_rows(highs, closes);
328
329 math::Vector result = math::unobserved(static_cast<std::size_t>(highs.size()));
330 for (math::Index row = 0; row < highs.size(); ++row) {
331 if (!observed(highs[row]) || !observed(lows[row])) {
332 continue;
333 }
334 Scalar range = highs[row] - lows[row];
335 if (row > 0 && observed(closes[row - 1])) {
336 const Scalar previous = closes[row - 1];
337 range = std::max(range, std::abs(highs[row] - previous));
338 range = std::max(range, std::abs(lows[row] - previous));
339 }
340 result[row] = range;
341 }
342 return result;
343}

395inline MacdLines macd(
396 math::SeriesView values,
397 std::size_t fast = 12,
398 std::size_t slow = 26,
399 std::size_t signal_span = 9
400) {
401 if (fast >= slow) {
402 throw std::invalid_argument("macd fast span must be shorter than the slow span");
403 }
404 const math::Vector quick = ema(values, alpha_from_span(fast));
405 const math::Vector slower = ema(values, alpha_from_span(slow));
406
407 const auto rows = static_cast<std::size_t>(values.size());
408 MacdLines lines{math::unobserved(rows), math::unobserved(rows), math::unobserved(rows)};
409 for (math::Index row = 0; row < values.size(); ++row) {
410 if (observed(quick[row]) && observed(slower[row])) {
411 lines.macd_[row] = quick[row] - slower[row];
412 }
413 }
414 lines.signal_ = ema(lines.macd_, alpha_from_span(signal_span));
415 for (math::Index row = 0; row < values.size(); ++row) {
416 if (observed(lines.macd_[row]) && observed(lines.signal_[row])) {
417 lines.histogram_[row] = lines.macd_[row] - lines.signal_[row];
418 }
419 }
420 return lines;
421}

437inline math::Vector rolling_cov(
438 math::SeriesView left,
439 math::SeriesView right,
440 std::size_t window,
441 std::size_t min_observed
442) {
443 indicator_detail::require_same_rows(left, right);
444 rolling_detail::check(window, min_observed);
445 const auto paired = indicator_detail::pair_series(left, right);
446 math::Vector result = math::unobserved(static_cast<std::size_t>(left.size()));
447 for (math::Index row = 0; row < left.size(); ++row) {
448 const math::Index first = rolling_detail::window_start(row, window);
449 const math::Index length = row - first + 1;
450 const auto reduced = indicator_detail::reduce_pair(paired, first, length, min_observed);
451 if (!reduced.reported_) {
452 continue;
453 }
454 const math::Vector left_centred = indicator_detail::centred(
455 paired.left_, paired.present_, first, length, reduced.mean_left_
456 );
457 const math::Vector right_centred = indicator_detail::centred(
458 paired.right_, paired.present_, first, length, reduced.mean_right_
459 );
460 result[row] = left_centred.dot(right_centred) / static_cast<Scalar>(reduced.count_ - 1);
461 }
462 return result;
463}

497inline math::Vector rolling_corr(
498 math::SeriesView left,
499 math::SeriesView right,
500 std::size_t window,
501 std::size_t min_observed
502) {
503 indicator_detail::require_same_rows(left, right);
504 rolling_detail::check(window, min_observed);
505 const auto paired = indicator_detail::pair_series(left, right);
506 math::Vector result = math::unobserved(static_cast<std::size_t>(left.size()));
507 for (math::Index row = 0; row < left.size(); ++row) {
508 const math::Index first = rolling_detail::window_start(row, window);
509 const math::Index length = row - first + 1;
510 const auto reduced = indicator_detail::reduce_pair(paired, first, length, min_observed);
511 if (!reduced.reported_) {
512 continue;
513 }
514 const math::Vector left_centred = indicator_detail::centred(
515 paired.left_, paired.present_, first, length, reduced.mean_left_
516 );
517 const math::Vector right_centred = indicator_detail::centred(
518 paired.right_, paired.present_, first, length, reduced.mean_right_
519 );
520 const Scalar squared_left = left_centred.squaredNorm();
521 const Scalar squared_right = right_centred.squaredNorm();
522 if (squared_left == Scalar{0} || squared_right == Scalar{0}) {
523 // A constant series. Correlation with it is undefined, not zero.
524 result[row] = math::undefined();
525 continue;
526 }
527 result[row] = left_centred.dot(right_centred) / std::sqrt(squared_left * squared_right);
528 }
529 return result;
530}

563inline math::Vector beta(
564 math::SeriesView asset,
565 math::SeriesView market,
566 std::size_t window,
567 std::size_t min_observed
568) {
569 indicator_detail::require_same_rows(asset, market);
570 rolling_detail::check(window, min_observed);
571 const auto paired = indicator_detail::pair_series(asset, market);
572 math::Vector result = math::unobserved(static_cast<std::size_t>(asset.size()));
573 for (math::Index row = 0; row < asset.size(); ++row) {
574 const math::Index first = rolling_detail::window_start(row, window);
575 const math::Index length = row - first + 1;
576 const auto reduced = indicator_detail::reduce_pair(paired, first, length, min_observed);
577 if (!reduced.reported_) {
578 continue;
579 }
580 const math::Vector market_centred = indicator_detail::centred(
581 paired.right_, paired.present_, first, length, reduced.mean_right_
582 );
583 const Scalar squared_market = market_centred.squaredNorm();
584 if (squared_market == Scalar{0}) {
585 // No market movement to attribute the asset's movement to.
586 result[row] = math::undefined();
587 continue;
588 }
589 const math::Vector asset_centred = indicator_detail::centred(
590 paired.left_, paired.present_, first, length, reduced.mean_left_
591 );
592 result[row] = asset_centred.dot(market_centred) / squared_market;
593 }
594 return result;
595}
