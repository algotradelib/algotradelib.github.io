37inline math::Vector wilder(math::SeriesView values, std::size_t period) {
38 const auto span = static_cast<Scalar>(period);
39 math::Vector result = math::unobserved(static_cast<std::size_t>(values.size()));
40 Scalar average{0};
41 Scalar seed{0};
42 std::size_t count = 0;
43 for (math::Index row = 0; row < values.size(); ++row) {
44 const Scalar value = values[row];
45 if (!observed(value)) {
46 continue;
47 }
48 ++count;
49 if (count < period) {
50 seed += value;
51 continue;
52 }
53 if (count == period) {
54 average = (seed + value) / span;
55 } else {
56 average = (average * (span - Scalar{1}) + value) / span;
57 }
58 result[row] = average;
59 }
60 return result;
61}

134inline math::Vector rsi(math::SeriesView prices, std::size_t period = 14) {
135 if (period == 0) {
136 throw std::invalid_argument("rsi period must be at least one row");
137 }
138 const auto rows = static_cast<std::size_t>(prices.size());
139 math::Vector gains = math::unobserved(rows);
140 math::Vector losses = math::unobserved(rows);
141 for (math::Index row = 1; row < prices.size(); ++row) {
142 if (!observed(prices[row - 1]) || !observed(prices[row])) {
143 continue;
144 }
145 const Scalar change = prices[row] - prices[row - 1];
146 gains[row] = std::max(change, Scalar{0});
147 losses[row] = std::max(-change, Scalar{0});
148 }
149
150 const math::Vector average_gain = indicator_detail::wilder(gains, period);
151 const math::Vector average_loss = indicator_detail::wilder(losses, period);
152
153 math::Vector result = math::unobserved(rows);
154 for (math::Index row = 0; row < prices.size(); ++row) {
155 if (!observed(average_gain[row]) || !observed(average_loss[row])) {
156 continue;
157 }
158 if (average_loss[row] == Scalar{0}) {
159 result[row] = average_gain[row] == Scalar{0} ? Scalar{50} : Scalar{100};
160 continue;
161 }
162 const Scalar strength = average_gain[row] / average_loss[row];
163 result[row] = Scalar{100} - Scalar{100} / (Scalar{1} + strength);
164 }
165 return result;
166}
