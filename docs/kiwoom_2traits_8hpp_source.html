263 static Result<KiwoomAmendment> to_amendment(
264 const Ref& ref,
265 const AmendInstruction& instruction,
266 bool cancel
267 ) {
268 BrokerError error;
269 error.category_ = ErrorCategory::INVALID_ARGUMENT;
270 error.reached_broker_ = false;
271
272 if (!instruction.all_remaining_ && !instruction.quantity_) {
273 error.broker_code_ = "quantity_required";
274 error.message_ =
275 "a Kiwoom modification or cancellation affects a stated quantity; set all_remaining_ to "
276 "act on everything still resting, or build the instruction with cancel_all()";
277 return Result<KiwoomAmendment>(std::move(error));
278 }
279 if (auto refusal = refuse_shape_change(instruction, cancel)) {
280 return Result<KiwoomAmendment>(std::move(*refusal));
281 }
282
283 KiwoomAmendment amendment;
284 amendment.order_ = from_ref(ref);
285 amendment.venue_ = instruction.venue_;
286 // Kiwoom's own "everything still resting", which it evaluates when it processes the
287 // instruction rather than when the caller read the listing.
288 amendment.quantity_ =
289 instruction.all_remaining_ ? Decimal64() : *instruction.quantity_;
290 if (!cancel) {
291 // The basis is known here and nowhere after: `kt10002` carries no `trde_tp`, so this is the
292 // last point that can tell a trigger the order wants from one it would silently lose.
293 const auto trigger = amended_trigger_price(instruction);
294 if (auto invalid = kiwoom::invalid_trigger(instruction.shape_.type_, trigger)) {
295 return Result<KiwoomAmendment>(std::move(*invalid));
296 }
297 amendment.price_ = instruction.shape_.price_;
298 amendment.condition_price_ = trigger;
299 }
300 // **Never on a cancellation.** `kt10003` has no trigger field and the client refuses a
301 // cancellation carrying one; `cancel_all` restates the resting trigger, so copying it here would
302 // make every stop order impossible to withdraw through the port.
303 return Result<KiwoomAmendment>(std::move(amendment));
304 }

318 static std::optional<BrokerError> refuse_shape_change(
319 const AmendInstruction& instruction,
320 bool cancel
321 ) {
322 BrokerError error;
323 error.category_ = ErrorCategory::INVALID_ARGUMENT;
324 error.reached_broker_ = false;
325 error.broker_code_ = "shape_change_unsupported";
326
327 const auto refuse = [&error, cancel](const char* what) {
328 error.message_ = cancel ?
329 std::string("Kiwoom's cancellation carries a quantity and nothing else, so it cannot "
330 "change an order's ") + what + " on the way out" :
331 std::string("Kiwoom's amendment endpoints carry a quantity and a price only, so they "
332 "cannot change an order's ") + what +
333 "; cancel the order and place a new one instead";
334 return error;
335 };
336
337 // An unset field means "leave it as it is", so it is never a change. A set field is a change
338 // only when it differs from what the order already is -- and a set field with no resting shape
339 // to compare against cannot be shown to be a restatement, so it is refused rather than assumed
340 // harmless.
341 const auto differs = [](auto stated, auto current) {
342 return stated.has_value() && (!current.has_value() || *stated != *current);
343 };
344
345 if (cancel) {
346 // **A cancellation transmits a quantity and nothing else**, so every other stated field is
347 // one Kiwoom cannot act on. The price and the trigger are refused because acting on them is
348 // what a caller would expect; these three are refused for the same reason and it is not a
349 // lesser one -- reporting acceptance for a stated change that never reached the broker is
350 // the defect, whatever the field.
351 //
352 // `cancel_all()` and `cancel_quantity()` copy the resting order's whole shape into both
353 // `shape_` and `resting_`, so every comparison below sees a restatement and the ordinary
354 // withdrawal path is unaffected.
355 const bool restates_price = instruction.resting_ &&
356 instruction.resting_->price_ == instruction.shape_.price_;
357 if (instruction.shape_.price_ && !restates_price) {
358 error.broker_code_ = "price_on_cancellation";
359 error.message_ =
360 "Kiwoom's cancellation carries a quantity and no price, so it cannot reprice an order on "
361 "the way out; cancel it and place a new one instead";
362 return error;
363 }
364 // The same rule as the price: a restatement of the resting trigger -- which is what
365 // `cancel_all` writes -- is the ordinary withdrawal, and only a stated *change* is refused.
366 const bool restates_trigger = instruction.resting_ &&
367 instruction.resting_->trigger_price_ == instruction.shape_.trigger_price_;
368 if (instruction.shape_.trigger_price_ && !restates_trigger) {
369 error.broker_code_ = "trigger_on_cancellation";
370 error.message_ =
371 "Kiwoom's cancellation carries a quantity and no trigger price, so it cannot move an "
372 "order's stop on the way out; cancel it and place a new one instead";
373 return error;
374 }
375 } else if (!instruction.resting_) {
376 // **A modification with no resting shape is refused, and that is the conservative
377 // direction.** `OrderShape::type_` is a plain value defaulting to `LIMIT`, so this adapter
378 // cannot tell a caller who *means* `LIMIT` from one who left it alone -- and if the resting
379 // order is a best-limit order, accepting the instruction would send a price and leave the
380 // basis where it was, reporting success for a change that did not happen. KIS refuses the
381 // same case for the same reason, as `incomplete_shape`.
382 error.broker_code_ = "incomplete_shape";
383 error.message_ =
384 "Kiwoom's modification cannot change an order's price basis, validity condition or session, "
385 "and without the order's current shape this adapter cannot tell a restatement from a "
386 "change; build the instruction with reprice() or replace_with(), which take it from the "
387 "resting order";
388 return error;
389 }
390
391 const auto resting_condition = instruction.resting_ ?
392 instruction.resting_->time_in_force_ :
393 std::optional<TimeInForce>();
394 const auto resting_session = instruction.resting_ ?
395 instruction.resting_->session_ :
396 std::optional<OrderSession>();
397 if (differs(instruction.shape_.time_in_force_, resting_condition)) {
398 return refuse("validity condition");
399 }
400 if (differs(instruction.shape_.session_, resting_session)) {
401 return refuse("session");
402 }
403 // `OrderShape::type_` is a plain value rather than an optional, so "left alone" and "stated
404 // `LIMIT`" are the same bytes and the comparison has to be made in two steps.
405 if (instruction.resting_) {
406 if (instruction.shape_.type_ != instruction.resting_->type_) {
407 return refuse("price basis");
408 }
409 } else if (instruction.shape_.type_ != OrderShape().type_) {
410 // **With no resting shape, a value that differs from the default is still a stated change.**
411 // Only the default is ambiguous; `MARKET` on a cancellation is a caller saying something, and
412 // `kt10003` transmits no basis, so accepting it would report acceptance for a change the
413 // broker never saw. Compared against `OrderShape`'s own default rather than a literal, so the
414 // two cannot drift apart.
415 return refuse("price basis");
416 }
417 return std::nullopt;
418 }

456 static Result<KiwoomHistoryQuery> to_history_query(
457 const KiwoomAccount&,
458 const OrderHistoryQuery& query
459 ) {
460 BrokerError error;
461 error.category_ = ErrorCategory::INVALID_ARGUMENT;
462 error.reached_broker_ = false;
463 if (!query.from_) {
464 error.broker_code_ = "date_required";
465 error.message_ = "Kiwoom's order-history inquiry lists one calendar date, which must be given";
466 return Result<KiwoomHistoryQuery>(std::move(error));
467 }
468 if (query.to_ && !(query.to_->year_ == query.from_->year_ &&
469 query.to_->month_ == query.from_->month_ &&
470 query.to_->day_ == query.from_->day_)) {
471 error.broker_code_ = "single_date_only";
472 error.message_ =
473 "Kiwoom's order-history inquiry lists one calendar date; ask for each date separately "
474 "rather than for a range";
475 return Result<KiwoomHistoryQuery>(std::move(error));
476 }
477 if (query.cursor_) {
478 error.broker_code_ = "cursor_unsupported";
479 error.message_ = "Kiwoom order history pages internally and exposes no cursor";
480 return Result<KiwoomHistoryQuery>(std::move(error));
481 }
482 if (query.limit_) {
483 error.broker_code_ = "limit_unsupported";
484 error.message_ = "Kiwoom order history documents no row limit and cannot honor one";
485 return Result<KiwoomHistoryQuery>(std::move(error));
486 }
487 KiwoomHistoryQuery converted;
488 converted.date_ = *query.from_;
489 converted.symbol_ = query.symbol_ ? *query.symbol_ : std::string();
490 return Result<KiwoomHistoryQuery>(std::move(converted));
491 }
