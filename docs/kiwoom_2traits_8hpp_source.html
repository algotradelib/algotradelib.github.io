260 static Result<KiwoomAmendment> to_amendment(
261 const Ref& ref,
262 const AmendInstruction& instruction,
263 bool cancel
264 ) {
265 BrokerError error;
266 error.category_ = ErrorCategory::INVALID_ARGUMENT;
267 error.reached_broker_ = false;
268
269 if (!instruction.all_remaining_ && !instruction.quantity_) {
270 error.broker_code_ = "quantity_required";
271 error.message_ =
272 "a Kiwoom modification or cancellation affects a stated quantity; set all_remaining_ to "
273 "act on everything still resting, or build the instruction with cancel_all()";
274 return Result<KiwoomAmendment>(std::move(error));
275 }
276 if (auto refusal = refuse_shape_change(instruction, cancel)) {
277 return Result<KiwoomAmendment>(std::move(*refusal));
278 }
279
280 KiwoomAmendment amendment;
281 amendment.order_ = from_ref(ref);
282 amendment.venue_ = instruction.venue_;
283 // Kiwoom's own "everything still resting", which it evaluates when it processes the
284 // instruction rather than when the caller read the listing.
285 amendment.quantity_ =
286 instruction.all_remaining_ ? Decimal64() : *instruction.quantity_;
287 if (!cancel) {
288 // The basis is known here and nowhere after: `kt10002` carries no `trde_tp`, so this is the
289 // last point that can tell a trigger the order wants from one it would silently lose.
290 const auto trigger = amended_trigger_price(instruction);
291 if (auto invalid = kiwoom::invalid_trigger(instruction.shape_.type_, trigger)) {
292 return Result<KiwoomAmendment>(std::move(*invalid));
293 }
294 amendment.price_ = instruction.shape_.price_;
295 amendment.condition_price_ = trigger;
296 }
297 // **Never on a cancellation.** `kt10003` has no trigger field and the client refuses a
298 // cancellation carrying one; `cancel_all` restates the resting trigger, so copying it here would
299 // make every stop order impossible to withdraw through the port.
300 return Result<KiwoomAmendment>(std::move(amendment));
301 }

315 static std::optional<BrokerError> refuse_shape_change(
316 const AmendInstruction& instruction,
317 bool cancel
318 ) {
319 BrokerError error;
320 error.category_ = ErrorCategory::INVALID_ARGUMENT;
321 error.reached_broker_ = false;
322 error.broker_code_ = "shape_change_unsupported";
323
324 const auto refuse = [&error, cancel](const char* what) {
325 error.message_ = cancel ?
326 std::string("Kiwoom's cancellation carries a quantity and nothing else, so it cannot "
327 "change an order's ") + what + " on the way out" :
328 std::string("Kiwoom's amendment endpoints carry a quantity and a price only, so they "
329 "cannot change an order's ") + what +
330 "; cancel the order and place a new one instead";
331 return error;
332 };
333
334 // An unset field means "leave it as it is", so it is never a change. A set field is a change
335 // only when it differs from what the order already is -- and a set field with no resting shape
336 // to compare against cannot be shown to be a restatement, so it is refused rather than assumed
337 // harmless.
338 const auto differs = [](auto stated, auto current) {
339 return stated.has_value() && (!current.has_value() || *stated != *current);
340 };
341
342 if (cancel) {
343 // **A cancellation transmits a quantity and nothing else**, so every other stated field is
344 // one Kiwoom cannot act on. The price and the trigger are refused because acting on them is
345 // what a caller would expect; these three are refused for the same reason and it is not a
346 // lesser one -- reporting acceptance for a stated change that never reached the broker is
347 // the defect, whatever the field.
348 //
349 // `cancel_all()` and `cancel_quantity()` copy the resting order's whole shape into both
350 // `shape_` and `resting_`, so every comparison below sees a restatement and the ordinary
351 // withdrawal path is unaffected.
352 const bool restates_price = instruction.resting_ &&
353 instruction.resting_->price_ == instruction.shape_.price_;
354 if (instruction.shape_.price_ && !restates_price) {
355 error.broker_code_ = "price_on_cancellation";
356 error.message_ =
357 "Kiwoom's cancellation carries a quantity and no price, so it cannot reprice an order on "
358 "the way out; cancel it and place a new one instead";
359 return error;
360 }
361 // The same rule as the price: a restatement of the resting trigger -- which is what
362 // `cancel_all` writes -- is the ordinary withdrawal, and only a stated *change* is refused.
363 const bool restates_trigger = instruction.resting_ &&
364 instruction.resting_->trigger_price_ == instruction.shape_.trigger_price_;
365 if (instruction.shape_.trigger_price_ && !restates_trigger) {
366 error.broker_code_ = "trigger_on_cancellation";
367 error.message_ =
368 "Kiwoom's cancellation carries a quantity and no trigger price, so it cannot move an "
369 "order's stop on the way out; cancel it and place a new one instead";
370 return error;
371 }
372 } else if (!instruction.resting_) {
373 // **A modification with no resting shape is refused, and that is the conservative
374 // direction.** `OrderShape::type_` is a plain value defaulting to `LIMIT`, so this adapter
375 // cannot tell a caller who *means* `LIMIT` from one who left it alone -- and if the resting
376 // order is a best-limit order, accepting the instruction would send a price and leave the
377 // basis where it was, reporting success for a change that did not happen. KIS refuses the
378 // same case for the same reason, as `incomplete_shape`.
379 error.broker_code_ = "incomplete_shape";
380 error.message_ =
381 "Kiwoom's modification cannot change an order's price basis, validity condition or session, "
382 "and without the order's current shape this adapter cannot tell a restatement from a "
383 "change; build the instruction with reprice() or replace_with(), which take it from the "
384 "resting order";
385 return error;
386 }
387
388 const auto resting_condition = instruction.resting_ ?
389 instruction.resting_->time_in_force_ :
390 std::optional<TimeInForce>();
391 const auto resting_session = instruction.resting_ ?
392 instruction.resting_->session_ :
393 std::optional<OrderSession>();
394 if (differs(instruction.shape_.time_in_force_, resting_condition)) {
395 return refuse("validity condition");
396 }
397 if (differs(instruction.shape_.session_, resting_session)) {
398 return refuse("session");
399 }
400 // `OrderShape::type_` is a plain value rather than an optional, so "left alone" and "stated
401 // `LIMIT`" are the same bytes and the comparison has to be made in two steps.
402 if (instruction.resting_) {
403 if (instruction.shape_.type_ != instruction.resting_->type_) {
404 return refuse("price basis");
405 }
406 } else if (instruction.shape_.type_ != OrderShape().type_) {
407 // **With no resting shape, a value that differs from the default is still a stated change.**
408 // Only the default is ambiguous; `MARKET` on a cancellation is a caller saying something, and
409 // `kt10003` transmits no basis, so accepting it would report acceptance for a change the
410 // broker never saw. Compared against `OrderShape`'s own default rather than a literal, so the
411 // two cannot drift apart.
412 return refuse("price basis");
413 }
414 return std::nullopt;
415 }

453 static Result<KiwoomHistoryQuery> to_history_query(
454 const KiwoomAccount&,
455 const OrderHistoryQuery& query
456 ) {
457 BrokerError error;
458 error.category_ = ErrorCategory::INVALID_ARGUMENT;
459 error.reached_broker_ = false;
460 if (!query.from_) {
461 error.broker_code_ = "date_required";
462 error.message_ = "Kiwoom's order-history inquiry lists one calendar date, which must be given";
463 return Result<KiwoomHistoryQuery>(std::move(error));
464 }
465 if (query.to_ && !(query.to_->year_ == query.from_->year_ &&
466 query.to_->month_ == query.from_->month_ &&
467 query.to_->day_ == query.from_->day_)) {
468 error.broker_code_ = "single_date_only";
469 error.message_ =
470 "Kiwoom's order-history inquiry lists one calendar date; ask for each date separately "
471 "rather than for a range";
472 return Result<KiwoomHistoryQuery>(std::move(error));
473 }
474 if (query.cursor_) {
475 error.broker_code_ = "cursor_unsupported";
476 error.message_ = "Kiwoom order history pages internally and exposes no cursor";
477 return Result<KiwoomHistoryQuery>(std::move(error));
478 }
479 if (query.limit_) {
480 error.broker_code_ = "limit_unsupported";
481 error.message_ = "Kiwoom order history documents no row limit and cannot honor one";
482 return Result<KiwoomHistoryQuery>(std::move(error));
483 }
484 KiwoomHistoryQuery converted;
485 converted.date_ = *query.from_;
486 converted.symbol_ = query.symbol_ ? *query.symbol_ : std::string();
487 return Result<KiwoomHistoryQuery>(std::move(converted));
488 }
