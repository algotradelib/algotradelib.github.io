334 // Hashes an instrument id for the two per-instrument tables below. Both are looked up by id and
335 // never iterated, so their order is nobody's to observe.
336 struct InstrumentHash {
337 std::size_t operator()(const InstrumentId& id) const noexcept {
338 std::size_t seed = std::hash<std::string>{}(id.symbol_);
339 seed ^= static_cast<std::size_t>(id.country_) + 0x9e3779b97f4a7c15ULL + (seed << 6) + (seed >> 2);
340 seed ^= static_cast<std::size_t>(id.venue_) + 0x9e3779b97f4a7c15ULL + (seed << 6) + (seed >> 2);
341 return seed;
342 }
343 };
