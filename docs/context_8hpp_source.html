62// A count of a context's portfolio replacements that a move also advances. Moving a context's state into
63// another replaces the destination's portfolio, so the destination's count must not merely take the source's
64// value: it rises above both, and the source's (now emptied) count rises too. A replay comparing the count with
65// the one it last saw then rebuilds whatever a move brought in.
66struct Revision {
67 std::uint64_t value_{0};
68
69 Revision() = default;
70 Revision(const Revision&) = delete;
71 Revision& operator=(const Revision&) = delete;
72 Revision(Revision&& other) noexcept : value_(other.value_ + 1) { ++other.value_; }
73 Revision& operator=(Revision&& other) noexcept {
74 value_ = (value_ > other.value_ ? value_ : other.value_) + 1;
75 ++other.value_;
76 return *this;
77 }
78};

155// A context's bar history: each series' trailing bars, or, when the context keeps none, each series' newest bar
156// time only (to refuse a bar that is not later). Whether it keeps them belongs to the context object, not to
157// its contents: a move assignment keeps the destination's, so a context moved into a runner's place cannot turn
158// on reads the runner never records for, and what it brings is converted to the destination's form in place,
159// without allocating, so moving a context still cannot throw.
160struct BarHistory {
161 // One series of an instrument at an interval.
162 struct Entry {
163 // Its bars; never made in a context keeping no history.
164 std::optional<analytics::SeriesBuffer> series_;
165 // Its newest bar time where it has no bars: kept instead of them, or brought by a context keeping none,
166 // when it bounds the first bar pushed.
167 std::optional<std::chrono::system_clock::time_point> newest_;
168 };
169 using ByInterval = std::map<BarInterval, Entry>;
170
171 // Node-based both ways, so an entry never moves once made.
172 std::unordered_map<InstrumentId, ByInterval, id_detail::InstrumentIdHash> entries_;
173 bool kept_{true};
174
175 BarHistory() = default;
176 BarHistory(const BarHistory&) = delete;
177 BarHistory& operator=(const BarHistory&) = delete;
178 BarHistory(BarHistory&&) = default;
179 BarHistory& operator=(BarHistory&& other) noexcept {
180 entries_ = std::move(other.entries_);
181 if (!kept_) {
182 keep_times_only();
183 }
184 return *this;
185 }
186
187 // For a runner whose strategy reads no history.
188 void stop_keeping() noexcept {
189 kept_ = false;
190 keep_times_only();
191 }
192
193private:
194 // Replaces each series by its newest bar time, keeping the later where both exist.
195 void keep_times_only() noexcept {
196 for (auto& [id, by_interval] : entries_) {
197 for (auto& [interval, entry] : by_interval) {
198 if (entry.series_ && !entry.series_->empty()) {
199 const auto newest = entry.series_->newest_instant();
200 if (!entry.newest_ || *entry.newest_ < newest) {
201 entry.newest_ = newest;
202 }
203 }
204 entry.series_.reset();
205 }
206 }
207 }
208};
209#endif
