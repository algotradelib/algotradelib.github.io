63// A count of a context's portfolio replacements that a move also advances. Moving a context's state into
64// another replaces the destination's portfolio, so the destination's count must not merely take the source's
65// value: it rises above both, and the source's (now emptied) count rises too. A replay comparing the count with
66// the one it last saw then rebuilds whatever a move brought in.
67struct Revision {
68 std::uint64_t value_{0};
69
70 Revision() = default;
71 Revision(const Revision&) = delete;
72 Revision& operator=(const Revision&) = delete;
73 Revision(Revision&& other) noexcept : value_(other.value_ + 1) {
74 ++other.value_;
75 }
76 Revision& operator=(Revision&& other) noexcept {
77 value_ = (value_ > other.value_ ? value_ : other.value_) + 1;
78 ++other.value_;
79 return *this;
80 }
81};

164// A context's bar history: each series' trailing bars, or, when the context keeps none, each series' newest bar
165// time only (to refuse a bar that is not later). Whether it keeps them belongs to the context object, not to
166// its contents: a move assignment keeps the destination's, so a context moved into a runner's place cannot turn
167// on reads the runner never records for, and what it brings is converted to the destination's form in place,
168// without allocating, so moving a context still cannot throw.
169struct BarHistory {
170 // One series of an instrument at an interval.
171 struct Entry {
172 // Its bars; never made in a context keeping no history.
173 std::optional<analytics::SeriesBuffer> series_;
174 // Its newest bar time where it has no bars: kept instead of them, or brought by a context keeping none,
175 // when it bounds the first bar pushed.
176 std::optional<std::chrono::system_clock::time_point> newest_;
177 };
178 using ByInterval = std::map<BarInterval, Entry>;
179
180 // Node-based both ways, so an entry never moves once made.
181 std::unordered_map<InstrumentId, ByInterval, id_detail::InstrumentIdHash> entries_;
182 bool kept_{true};
183
184 BarHistory() = default;
185 BarHistory(const BarHistory&) = delete;
186 BarHistory& operator=(const BarHistory&) = delete;
187 BarHistory(BarHistory&&) = default;
188 BarHistory& operator=(BarHistory&& other) noexcept {
189 entries_ = std::move(other.entries_);
190 if (!kept_) {
191 keep_times_only();
192 }
193 return *this;
194 }
195
196 // For a runner whose strategy reads no history.
197 void stop_keeping() noexcept {
198 kept_ = false;
199 keep_times_only();
200 }
201
202private:
203 // Replaces each series by its newest bar time, keeping the later where both exist.
204 void keep_times_only() noexcept {
205 for (auto& [id, by_interval] : entries_) {
206 for (auto& [interval, entry] : by_interval) {
207 if (entry.series_ && !entry.series_->empty()) {
208 const auto newest = entry.series_->newest_instant();
209 if (!entry.newest_ || *entry.newest_ < newest) {
210 entry.newest_ = newest;
211 }
212 }
213 entry.series_.reset();
214 }
215 }
216 }
217};
218#endif
