82inline math::Vector adjusted_ema(math::SeriesView values, math::Scalar alpha, std::size_t min_observed = 1) {
83 if (!(math::Scalar{0} < alpha && alpha <= math::Scalar{1})) {
84 throw std::invalid_argument("an ema smoothing factor must be in (0, 1]");
85 }
86 if (min_observed == 0) {
87 throw std::invalid_argument("ema must require at least one observation");
88 }
89 math::Vector result = math::unobserved(static_cast<std::size_t>(values.size()));
90 const math::Scalar keep = math::Scalar{1} - alpha;
91 math::Scalar mean{0};
92 math::Scalar total{0};
93 std::size_t count = 0;
94 for (math::Index row = 0; row < values.size(); ++row) {
95 const math::Scalar value = values[row];
96 if (!math::observed(value)) {
97 continue;
98 }
99 total = math::Scalar{1} + keep * total;
100 // `alpha = 1` keeps only the newest value, exactly: `mean + (value - mean)` need not round back to it.
101 mean = count == 0 || keep == math::Scalar{0} ? value : mean + (value - mean) / total;
102 ++count;
103 if (count >= min_observed) {
104 result[row] = mean;
105 }
106 }
107 return result;
108}

122inline math::Vector exponential_line(math::SeriesView closes, std::size_t span) {
123 const auto alpha = analytics::alpha_from_span(span);
124 std::size_t observations = 0;
125 for (math::Index row = 0; row < closes.size(); ++row) {
126 observations += math::observed(closes[row]) ? 1 : 0;
127 }
128 if (observations < 3 * span) {
129 return adjusted_ema(closes, alpha, span);
130 }
131 return analytics::ema(closes, alpha, span);
