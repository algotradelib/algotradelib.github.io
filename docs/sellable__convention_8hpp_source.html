103template <typename Broker>
104class SellableConventionProbe {
105public:
110 void observe(const Portfolio<Broker>& portfolio) {
111 // **A figure of unknown coverage teaches nothing about resting orders, and would teach something
112 // false.** This probe reads the *distance* between what is held and what is sellable and watches
113 // how that distance moves. Both numbers have to describe the same shares for the distance to mean
114 // anything: if `rmnd_qty` is account-wide while `trde_able_qty` covers one venue -- exactly what
115 // `SellableBasis::Coverage::UNESTABLISHED` means at Kiwoom -- the distance is a venue split, and a
116 // position moving between venues would read as a resting convention. So an unenforceable basis is
117 // skipped whole rather than judged, and the verdict stays `UNDETERMINED`, which is true.
118 for (const auto& position : portfolio.positions_) {
119 const auto reading = portfolio.sellable_reading_for(position.instrument_);
120 if (!reading.basis_.is_enforceable()) {
121 // **Skipped, and the baseline is dropped with it.** Merely ignoring the reading would leave the
122 // previous one pending, so the *next* enforceable reading would be compared against a
123 // baseline from before the gap -- and the distance would appear to move by whatever happened
124 // in between. A declared 100/100/0, an undeclared 100/70/0, then a declared 100/70/30 would
125 // record a confident NET_OF_RESTING across a window the probe never observed.
126 //
127 // Resolved per instrument rather than per portfolio, because a per-instrument inquiry's basis
128 // lands on its own row: checking only the snapshot's would discard every Toss reading.
129 forget(position.instrument_);
130 continue;
131 }
132 if (!position.sellable_quantity_) {
133 // The broker did not publish it, so there is nothing to learn from this position.
134 continue;
135 }
136 const auto resting = reported_resting(portfolio, position.instrument_);
137 const Reading now{position.quantity_, *position.sellable_quantity_, resting};
138 if (auto* previous = find(last_, position.instrument_)) {
139 judge(position.instrument_, *previous, now);
140 *previous = now;
141 continue;
142 }
143 last_.push_back({position.instrument_, now});
144 }
145 }
146
152 void forget(const InstrumentId& instrument) {
153 for (auto it = last_.begin(); it != last_.end(); ++it) {
154 if (it->first == instrument) {
155 last_.erase(it);
156 return;
157 }
158 }
159 }

166 SellableEvidence evidence(const InstrumentId& instrument) const {
167 if (const auto* found = find(evidence_, instrument)) {
168 return *found;
169 }
170 return SellableEvidence{};
171 }

177 SellableEvidence overall() const {
178 SellableEvidence total;
179 for (const auto& entry : evidence_) {
180 total.net_ += entry.second.net_;
181 total.gross_ += entry.second.gross_;
182 }
183 return total;
184 }

200private:
201 // Returns the sell quantity **the broker reported** resting, ignoring the runner's reservations.
202 //
203 // A runner adds a row of its own for an order it has just sent, so that the next order in the
204 // pass cannot re-sell the same shares. Those rows are this library's arithmetic, and counting
205 // them here would be circular: the resting quantity would rise with no change in the broker's
206 // sellable figure, and the probe would record `GROSS_OF_RESTING` on evidence it manufactured
207 // itself -- every time, for any broker, whenever the sellable check happened to be enabled.
208 static Decimal64 reported_resting(
209 const Portfolio<Broker>& portfolio,
210 const InstrumentId& instrument
211 ) {
212 Decimal64 total = Decimal64::parse("0");
213 for (const auto& order : portfolio.working_) {
214 if (order.reserved_ || !(order.order_.instrument_ == instrument) ||
215 order.order_.side_ != OrderSide::SELL) {
216 continue;
217 }
218 if (!order_is_open(order.order_.status_)) {
219 continue;
220 }
221 const auto remaining = decimal_subtract(
222 order.order_.quantity_, order.order_.execution_.filled_quantity_
223 );
224 if (!remaining || decimal_less(*remaining, Decimal64::parse("0"))) {
225 continue;
226 }
227 if (const auto sum = decimal_add(total, *remaining)) {
228 total = *sum;
229 }
230 }
231 return total;
232 }

234 // One position as a poll reported it.
235 struct Reading {
236 Decimal64 held_;
237 Decimal64 sellable_;
238 Decimal64 resting_;
239 };
240
241 // Returns whether two decimals are numerically equal, across differing scales.
242 static bool same(const Decimal64& lhs, const Decimal64& rhs) {
243 // `operator==` compares the coefficient and the scale, so "70" and "70.0" are not equal by it.
244 // A broker is free to change the scale it reports between two polls, and a probe that read that
245 // as a change would conclude from noise.
246 return !decimal_less(lhs, rhs) && !decimal_less(rhs, lhs);
247 }
248
249 // Records what the move from `before` to `now` says, if it says anything.
250 void judge(const InstrumentId& instrument, const Reading& before, const Reading& now) {
251 if (!same(before.held_, now.held_)) {
252 return;
253 }
254 const auto grew = decimal_subtract(now.resting_, before.resting_);
255 if (!grew || !decimal_less(Decimal64::parse("0"), *grew)) {
256 return;
257 }
258 const auto gap_before = decimal_subtract(before.held_, before.sellable_);
259 const auto gap_now = decimal_subtract(now.held_, now.sellable_);
260 if (!gap_before || !gap_now) {
261 return;
262 }
263 const auto moved = decimal_subtract(*gap_now, *gap_before);
264 if (!moved) {
265 return;
266 }
267
268 auto* entry = find(evidence_, instrument);
269 if (entry == nullptr) {
270 evidence_.push_back({instrument, SellableEvidence{}});
271 entry = &evidence_.back().second;
272 }
273 if (same(*moved, *grew)) {
274 ++entry->net_;
275 } else if (same(*moved, Decimal64::parse("0"))) {
276 ++entry->gross_;
277 }
278 // Anything else moved by an amount neither rule predicts -- a settlement in the same window,
279 // most likely -- and is discarded rather than forced into one of the two answers.
280 }
281
282 // Returns the entry for `instrument`, or null.
283 template <typename Entries>
284 static auto find(Entries& entries, const InstrumentId& instrument)
285 -> decltype(&entries.front().second) {
286 for (auto& entry : entries) {
287 if (entry.first == instrument) {
288 return &entry.second;
289 }
290 }
291 return nullptr;
292 }

294 std::vector<std::pair<InstrumentId, Reading>> last_;
295 std::vector<std::pair<InstrumentId, SellableEvidence>> evidence_;
296};
