63template <typename Strategy, typename Session>
64BoundRunner<Strategy, Session> bind(
65 Session& session, Strategy strategy, strategy::RiskLimits limits = {},
66 std::size_t series_capacity = default_series_capacity, Venue default_route = Venue::UNSPECIFIED
67) {
68 BoundRunner<Strategy, Session> bound(
69 std::move(strategy), std::move(limits), series_capacity, SessionExecutor<Session>(session, default_route)
70 );
71 // **A runner that can reach a broker requires an account it has actually read** -- and that is not
72 // arranged here. It comes from `SessionExecutor::reaches_a_broker`, so a runner assembled by hand
73 // carries it too: `doxygen/vignettes/binding-a-broker.md` shows exactly that, and a rule this
74 // factory installed would hold for its own callers and for nobody else. A run that genuinely reads
75 // no holdings says `allow_absent_portfolio()` on the runner.
76 return bound;
77}
