308 FillDecision decide(
309 const strategy::PlannedOrder& order, const strategy::BarEvent& next, const PriceRule& rule, const Date& date,
310 const std::optional<Date>& open_date = std::nullopt, bool released = false,
311 const std::optional<Decimal64>& decision_price = std::nullopt,
312 const std::optional<std::chrono::system_clock::time_point>& decision_price_at = std::nullopt,
313 const std::optional<std::chrono::system_clock::time_point>& decided_at = std::nullopt
314 ) const {
315 FillDecision decision;
316 bool at_open = false;
317 std::optional<Decimal64> reference;
318 if (market_price_ && !limit_priced(order.type_)) {
319 reference = market_price_(MarketPriceInput{order, decision_price, decision_price_at, decided_at, next, rule});
320 if (!reference) {
321 // Not this bar: a DAY order rests for the rest of its session; any other is withdrawn.
322 decision.still_working_ = order.time_in_force_ == TimeInForce::DAY;
323 return decision;
324 }
325 if (reference->coefficient() <= 0) {
326 // A defect in the rule, not a price: filling at it would book a trade no market made.
327 BrokerError error;
328 error.category_ = ErrorCategory::INVALID_ARGUMENT;
329 error.reached_broker_ = false;
330 error.broker_code_ = "market_price_not_positive";
331 error.message_ = "the market-price rule '" + market_price_name_ + "' named a price that is not positive";
332 decision.costing_error_ = std::move(error);
333 return decision;
334 }
335 decision.priced_by_rule_ = true;
336 // Charged on the fill bar's own day, as an open's fill is.
337 at_open = true;
338 } else if (carries_trigger_price(order.type_) && !released) {
339 if (!order.trigger_price_ || !order.price_) {
340 // Not a stop this model can release (a caller driving the model directly).
341 return decision;
342 }
343 switch (release_of(order, next)) {
344 case Release::NONE:
345 // Dormant: the bar never reached the trigger.
346 decision.still_working_ = true;
347 return decision;
348 case Release::AT_OPEN:
349 // The whole bar is after the release, so it is priced as the limit it now is.
350 decision.released_ = true;
351 reference = reference_price(order, next, at_open);
352 // Dated from the open whatever the price policy credits: it traded at the open.
353 if (reference && limit_marketable_at(order, *next.bar_.open_)) {
354 at_open = true;
355 }
356 break;
357 case Release::INTRABAR:
358 decision.released_ = true;
359 reference = intrabar_release_price(order, next);
360 if (!reference) {
361 // The bar cannot show the price reached the limit after the release. Rests for the rest of
362 // its session (on daily bars, it expires unfilled).
363 decision.still_working_ = true;
364 return decision;
365 }
366 decision.intrabar_release_fill_ = true;
367 break;
368 }
369 } else {
370 reference = reference_price(order, next, at_open);
371 }
372 if (!reference) {
373 // A limit the bar never reached stays queued for the rest of its session.
374 decision.still_working_ = rests_unfilled(order);
375 return decision;
376 }
377
378 // A limit fill is credited a price its bar traded and never one through the limit: the bound is a
379 // sell's max(low, limit) or a buy's min(high, limit), applied before and after slippage.
380 std::optional<Decimal64> bound;
381 const bool limit_basis = limit_priced(order.type_);
382 if (limit_basis && order.price_) {
383 const auto& limit = *order.price_;
384 if (order.side_ == OrderSide::SELL) {
385 bound = decimal_less(limit, next.bar_.low_) ? next.bar_.low_ : limit;
386 if (decimal_less(*reference, *bound)) {
387 reference = bound;
388 decision.bounded_ = true;
389 }
390 } else {
391 bound = decimal_less(next.bar_.high_, limit) ? next.bar_.high_ : limit;
392 if (decimal_less(*bound, *reference)) {
393 reference = bound;
394 decision.bounded_ = true;
395 }
396 }
397 }
398 auto slipped = slippage_.apply(*reference, order.side_, rule);
399 if (slipped && decision.intrabar_release_fill_) {
400 // No favourable slippage after an intrabar release: the bar's other extreme may have come first.
401 const bool favourable = order.side_ == OrderSide::SELL ?
402 decimal_less(*reference, slipped->first) :
403 decimal_less(slipped->first, *reference);
404 if (favourable) {
405 slipped->first = *reference;
406 slipped->second = Decimal64::parse("0");
407 }
408 }
409 if (slipped && bound) {
410 // Negative slippage cannot carry a fill past the bar's opposite extreme either.
411 const auto& ceiling = order.side_ == OrderSide::SELL ? next.bar_.high_ : next.bar_.low_;
412 const bool past =
413 order.side_ == OrderSide::SELL ? decimal_less(ceiling, slipped->first) : decimal_less(slipped->first, ceiling);
414 if (past) {
415 const auto moved = order.side_ == OrderSide::SELL ?
416 decimal_subtract(*reference, ceiling) :
417 decimal_subtract(ceiling, *reference);
418 if (!moved) {
419 slipped.reset();
420 } else {
421 slipped->first = ceiling;
422 slipped->second = *moved;
423 decision.bounded_ = true;
424 }
425 }
426 }
427 if (slipped && bound) {
428 const bool beyond =
429 order.side_ == OrderSide::SELL ? decimal_less(slipped->first, *bound) : decimal_less(*bound, slipped->first);
430 if (beyond) {
431 const auto moved =
432 order.side_ == OrderSide::SELL ? decimal_subtract(*reference, *bound) : decimal_subtract(*bound, *reference);
433 if (!moved) {
434 slipped.reset();
435 } else {
436 slipped->first = *bound;
437 slipped->second = *moved;
438 decision.bounded_ = true;
439 }
440 }
441 }
442 if (!slipped) {
443 decision.still_working_ = rests_unfilled(order);
444 return decision;
445 }
446 const auto& charged_on = at_open && open_date ? *open_date : date;
447 auto charged = commission_.charge(order.quantity_, slipped->first, charged_on, order.instrument_.country_);
448 if (!charged) {
449 // A cost that cannot be resolved halts the engine; it is neither a non-fill nor silent.
450 decision.costing_error_ = std::move(charged.error());
451 return decision;
452 }
453
454 // The tax on a sale, where a rate was stated: of the sale's value, truncated to whole units.
455 std::optional<Decimal64> tax;
456 if (sell_tax_ && order.side_ == OrderSide::SELL && order.instrument_.country_ == sell_tax_->country_) {
457 const auto value = decimal_multiply(order.quantity_, slipped->first);
458 const auto charged = value ? decimal_multiply(*value, sell_tax_->rate_) : std::nullopt;
459 tax = charged ? decimal_floor_to_multiple(*charged, Decimal64::parse("1")) : std::nullopt;
460 if (!tax) {
461 // Not a tax of zero: the cost is unknown, so the engine halts.
462 BrokerError error;
463 error.category_ = ErrorCategory::INVALID_ARGUMENT;
464 error.reached_broker_ = false;
465 error.broker_code_ = "tax_not_representable";
466 error.message_ = "the tax on that sale does not fit an exact decimal";
467 decision.costing_error_ = std::move(error);
468 return decision;
469 }
470 }
471
472 Execution execution;
473 execution.fill_.instrument_ = order.instrument_;
474 execution.fill_.side_ = order.side_;
475 execution.fill_.quantity_ = order.quantity_;
476 execution.fill_.price_ = slipped->first;
477 execution.fill_.tag_ = order.tag_;
478 execution.fill_.at_ = next.period_end_;
479 execution.commission_ = charged.value().broker_;
480 execution.institution_fee_ = charged.value().institution_;
481 execution.applied_ = std::move(charged.value().applied_);
482 execution.slippage_ = slipped->second;
483 if (tax) {
484 execution.tax_ = *tax;
485 }
486 decision.execution_ = std::move(execution);
487 return decision;
488 }

491 // Where in a bar a dormant stop released, if it did.
492 enum class Release {
493 // The bar never reached the trigger.
494 NONE,
495 // The open was already through the trigger, so the whole bar followed the release.
496 AT_OPEN,
497 // The bar's extreme reached the trigger and its open did not.
498 INTRABAR,
499 };

501 // Returns whether, and where, `next` released a dormant stop.
502 //
503 // **A sell stop releases when the price falls to its trigger and a buy stop when it rises to it.**
504 // That is an assumption about the broker -- neither KIS nor Kiwoom documents the direction -- and
505 // it is the universal convention. `ReportCaveats::stop_releases_` counts how often it was used.
506 //
507 // A bar with no open cannot establish a release *at* the open, so a release there is intrabar,
508 // which is the answer that credits less.
509 static Release release_of(const strategy::PlannedOrder& order, const strategy::BarEvent& next) {
510 const auto& trigger = *order.trigger_price_;
511 const bool sell = order.side_ == OrderSide::SELL;
512 const bool reached = sell ? !decimal_less(trigger, next.bar_.low_) : !decimal_less(next.bar_.high_, trigger);
513 if (!reached) {
514 return Release::NONE;
515 }
516 if (next.bar_.open_) {
517 const auto& open = *next.bar_.open_;
518 const bool through = sell ? !decimal_less(trigger, open) : !decimal_less(open, trigger);
519 if (through) {
520 return Release::AT_OPEN;
521 }
522 }
523 return Release::INTRABAR;
524 }

526 // Returns whether the order's limit would trade at `price` -- a sell at or above it, a buy at or below.
527 static bool limit_marketable_at(const strategy::PlannedOrder& order, const Decimal64& price) {
528 const auto& limit = *order.price_;
529 return order.side_ == OrderSide::SELL ? !decimal_less(price, limit) : !decimal_less(limit, price);
530 }

532 // Returns what a stop released inside `next` is credited with, or nothing when the bar cannot show
533 // it traded after the release.
534 //
535 // At the release the price *was* the trigger. Two cases, using only prices the bar proves came after
536 // it -- the trigger itself and the close, which is always last:
537 //
538 // - **Marketable at the trigger** (a sell limited at or below it, a buy at or above): it trades. It is
539 // credited the **worst price the bar traded that the limit allows** -- a sell the higher of its
540 // limit and the bar's low, a buy the lower of its limit and the bar's high -- so the price is inside
541 // the bar and never through the limit. Crediting the limit itself could name a price the bar never
542 // reached; crediting the trigger would assume the book was there.
543 // - **Not marketable at the trigger** (a sell limited above it): it trades only if the price came back
544 // to the limit after the release. The close proves that when it is at or through the limit; then it
545 // is credited the limit. Otherwise the bar's other extreme may have come first, and nothing is.
546 static std::optional<Decimal64> intrabar_release_price(
547 const strategy::PlannedOrder& order, const strategy::BarEvent& next
548 ) {
549 const auto& trigger = *order.trigger_price_;
550 const auto& limit = *order.price_;
551 const bool sell = order.side_ == OrderSide::SELL;
552 if (limit_marketable_at(order, trigger)) {
553 if (sell) {
554 return decimal_less(limit, next.bar_.low_) ? next.bar_.low_ : limit;
555 }
556 return decimal_less(next.bar_.high_, limit) ? next.bar_.high_ : limit;
557 }
558 if (limit_marketable_at(order, next.bar_.close_)) {
559 return limit;
560 }
561 return std::nullopt;
562 }
563
564 // Returns the price before slippage, or nothing when the order does not trade against this bar.
565 //
566 // A market order takes the open. A limit trades only if the bar's range reached it — a buy needs the
567 // low at or below the limit, a sell needs the high at or above it — because a bar that never touched
568 // the price is a bar in which the order sat unfilled, however close it came.
569 //
570 // Sets `at_open` when the price is the bar's stated open -- a market order, or a limit under
571 // `AT_OPEN_IF_BETTER` whose open was better -- which is what lets `decide` date the charge from the
572 // open's day. False for a bar with no open, whose close stands in for it.
573 std::optional<Decimal64> reference_price(
574 const strategy::PlannedOrder& order, const strategy::BarEvent& next, bool& at_open
575 ) const {
576 at_open = false;
577 // A bar with no open prices a market order at its close, which is not the open's moment.
578 const bool has_open = next.bar_.open_.has_value();
579 const auto open = has_open ? *next.bar_.open_ : next.bar_.close_;
580
581 // A released stop is priced as the limit it has become.
582 const bool limit_basis = limit_priced(order.type_);
583 if (!limit_basis || !order.price_) {
584 at_open = has_open;
585 return open;
586 }
587 const auto& limit = *order.price_;
588 bool open_if_better = false;
589 switch (limit_price_) {
590 case LimitFillPrice::AT_OPEN_IF_BETTER:
591 open_if_better = true;
592 break;
593 case LimitFillPrice::AT_LIMIT:
594 break;
595 }
596
597 if (order.side_ == OrderSide::BUY) {
598 if (decimal_less(limit, next.bar_.low_)) {
599 return std::nullopt;
600 }
601 if (open_if_better && decimal_less(open, limit)) {
602 at_open = has_open;
603 return open;
604 }
605 return limit;
606 }
607
608 if (decimal_less(next.bar_.high_, limit)) {
609 return std::nullopt;
610 }
611 if (open_if_better && decimal_less(limit, open)) {
612 at_open = has_open;
613 return open;
614 }
615 return limit;
616 }
