178 static bool rests_unfilled(const strategy::PlannedOrder& order) noexcept {
179 switch (order.time_in_force_) {
180 case TimeInForce::DAY:
181 break;
182 // Filled or not on the next bar, and not kept after it.
183 case TimeInForce::IMMEDIATE_OR_CANCEL:
184 case TimeInForce::FILL_OR_KILL:
185 case TimeInForce::UNKNOWN:
186 return false;
187 }
188 // A stop rests while dormant and, once released, as the limit it has become.
189 return limit_priced(order.type_);

220 FillDecision decide(
221 const strategy::PlannedOrder& order,
222 const strategy::BarEvent& next,
223 const PriceRule& rule,
224 const Date& date,
225 const std::optional<Date>& open_date = std::nullopt,
226 bool released = false
227 ) const {
228 FillDecision decision;
229 bool at_open = false;
230 std::optional<Decimal64> reference;
231 if (carries_trigger_price(order.type_) && !released) {
232 if (!order.trigger_price_ || !order.price_) {
233 // Not a stop this model can release (a caller driving the model directly).
234 return decision;
235 }
236 switch (release_of(order, next)) {
237 case Release::NONE:
238 // Dormant: the bar never reached the trigger.
239 decision.still_working_ = true;
240 return decision;
241 case Release::AT_OPEN:
242 // The whole bar is after the release, so it is priced as the limit it now is.
243 decision.released_ = true;
244 reference = reference_price(order, next, at_open);
245 // Dated from the open whatever the price policy credits: it traded at the open.
246 if (reference && limit_marketable_at(order, *next.bar_.open_)) {
247 at_open = true;
248 }
249 break;
250 case Release::INTRABAR:
251 decision.released_ = true;
252 reference = intrabar_release_price(order, next);
253 if (!reference) {
254 // The bar cannot show the price reached the limit after the release. Rests for the rest of
255 // its session (on daily bars, it expires unfilled).
256 decision.still_working_ = true;
257 return decision;
258 }
259 decision.intrabar_release_fill_ = true;
260 break;
261 }
262 } else {
263 reference = reference_price(order, next, at_open);
264 }
265 if (!reference) {
266 // A limit the bar never reached stays queued for the rest of its session.
267 decision.still_working_ = rests_unfilled(order);
268 return decision;
269 }
270
271 // A limit fill is credited a price its bar traded and never one through the limit: the bound is a
272 // sell's max(low, limit) or a buy's min(high, limit), applied before and after slippage.
273 std::optional<Decimal64> bound;
274 const bool limit_basis = limit_priced(order.type_);
275 if (limit_basis && order.price_) {
276 const auto& limit = *order.price_;
277 if (order.side_ == OrderSide::SELL) {
278 bound = decimal_less(limit, next.bar_.low_) ? next.bar_.low_ : limit;
279 if (decimal_less(*reference, *bound)) {
280 reference = bound;
281 decision.bounded_ = true;
282 }
283 } else {
284 bound = decimal_less(next.bar_.high_, limit) ? next.bar_.high_ : limit;
285 if (decimal_less(*bound, *reference)) {
286 reference = bound;
287 decision.bounded_ = true;
288 }
289 }
290 }
291 auto slipped = slippage_.apply(*reference, order.side_, rule);
292 if (slipped && decision.intrabar_release_fill_) {
293 // No favourable slippage after an intrabar release: the bar's other extreme may have come first.
294 const bool favourable = order.side_ == OrderSide::SELL ?
295 decimal_less(*reference, slipped->first) :
296 decimal_less(slipped->first, *reference);
297 if (favourable) {
298 slipped->first = *reference;
299 slipped->second = Decimal64::parse("0");
300 }
301 }
302 if (slipped && bound) {
303 // Negative slippage cannot carry a fill past the bar's opposite extreme either.
304 const auto& ceiling = order.side_ == OrderSide::SELL ? next.bar_.high_ : next.bar_.low_;
305 const bool past = order.side_ == OrderSide::SELL ?
306 decimal_less(ceiling, slipped->first) :
307 decimal_less(slipped->first, ceiling);
308 if (past) {
309 const auto moved = order.side_ == OrderSide::SELL ?
310 decimal_subtract(*reference, ceiling) :
311 decimal_subtract(ceiling, *reference);
312 if (!moved) {
313 slipped.reset();
314 } else {
315 slipped->first = ceiling;
316 slipped->second = *moved;
317 decision.bounded_ = true;
318 }
319 }
320 }
321 if (slipped && bound) {
322 const bool beyond = order.side_ == OrderSide::SELL ?
323 decimal_less(slipped->first, *bound) :
324 decimal_less(*bound, slipped->first);
325 if (beyond) {
326 const auto moved = order.side_ == OrderSide::SELL ?
327 decimal_subtract(*reference, *bound) :
328 decimal_subtract(*bound, *reference);
329 if (!moved) {
330 slipped.reset();
331 } else {
332 slipped->first = *bound;
333 slipped->second = *moved;
334 decision.bounded_ = true;
335 }
336 }
337 }
338 if (!slipped) {
339 decision.still_working_ = rests_unfilled(order);
340 return decision;
341 }
342 const auto& charged_on = at_open && open_date ? *open_date : date;
343 auto charged =
344 commission_.charge(order.quantity_, slipped->first, charged_on, order.instrument_.country_);
345 if (!charged) {
346 // A cost that cannot be resolved halts the engine; it is neither a non-fill nor silent.
347 decision.costing_error_ = std::move(charged.error());
348 return decision;
349 }
350
351 // The tax on a sale, where a rate was stated: of the sale's value, truncated to whole units.
352 std::optional<Decimal64> tax;
353 if (sell_tax_ && order.side_ == OrderSide::SELL && order.instrument_.country_ == sell_tax_->country_) {
354 const auto value = decimal_multiply(order.quantity_, slipped->first);
355 const auto charged = value ? decimal_multiply(*value, sell_tax_->rate_) : std::nullopt;
356 tax = charged ? decimal_floor_to_multiple(*charged, Decimal64::parse("1")) : std::nullopt;
357 if (!tax) {
358 // Not a tax of zero: the cost is unknown, so the engine halts.
359 BrokerError error;
360 error.category_ = ErrorCategory::INVALID_ARGUMENT;
361 error.reached_broker_ = false;
362 error.broker_code_ = "tax_not_representable";
363 error.message_ = "the tax on that sale does not fit an exact decimal";
364 decision.costing_error_ = std::move(error);
365 return decision;
366 }
367 }
368
369 Execution execution;
370 execution.fill_.instrument_ = order.instrument_;
371 execution.fill_.side_ = order.side_;
372 execution.fill_.quantity_ = order.quantity_;
373 execution.fill_.price_ = slipped->first;
374 execution.fill_.tag_ = order.tag_;
375 execution.fill_.at_ = next.period_end_;
376 execution.commission_ = charged.value().broker_;
377 execution.institution_fee_ = charged.value().institution_;
378 execution.applied_ = std::move(charged.value().applied_);
379 execution.slippage_ = slipped->second;
380 if (tax) {
381 execution.tax_ = *tax;
382 }
383 decision.execution_ = std::move(execution);
384 return decision;
385 }

388 // Where in a bar a dormant stop released, if it did.
389 enum class Release {
390 // The bar never reached the trigger.
391 NONE,
392 // The open was already through the trigger, so the whole bar followed the release.
393 AT_OPEN,
394 // The bar's extreme reached the trigger and its open did not.
395 INTRABAR,
396 };
397
398 // Returns whether, and where, `next` released a dormant stop.
399 //
400 // **A sell stop releases when the price falls to its trigger and a buy stop when it rises to it.**
401 // That is an assumption about the broker -- neither KIS nor Kiwoom documents the direction -- and
402 // it is the universal convention. `ReportCaveats::stop_releases_` counts how often it was used.
403 //
404 // A bar with no open cannot establish a release *at* the open, so a release there is intrabar,
405 // which is the answer that credits less.
406 static Release release_of(const strategy::PlannedOrder& order, const strategy::BarEvent& next) {
407 const auto& trigger = *order.trigger_price_;
408 const bool sell = order.side_ == OrderSide::SELL;
409 const bool reached = sell ?
410 !decimal_less(trigger, next.bar_.low_) :
411 !decimal_less(next.bar_.high_, trigger);
412 if (!reached) {
413 return Release::NONE;
414 }
415 if (next.bar_.open_) {
416 const auto& open = *next.bar_.open_;
417 const bool through = sell ? !decimal_less(trigger, open) : !decimal_less(open, trigger);
418 if (through) {
419 return Release::AT_OPEN;
420 }
421 }
422 return Release::INTRABAR;
423 }

425 // Returns whether the order's limit would trade at `price` -- a sell at or above it, a buy at or below.
426 static bool limit_marketable_at(const strategy::PlannedOrder& order, const Decimal64& price) {
427 const auto& limit = *order.price_;
428 return order.side_ == OrderSide::SELL ? !decimal_less(price, limit) : !decimal_less(limit, price);
429 }

431 // Returns what a stop released inside `next` is credited with, or nothing when the bar cannot show
432 // it traded after the release.
433 //
434 // At the release the price *was* the trigger. Two cases, using only prices the bar proves came after
435 // it -- the trigger itself and the close, which is always last:
436 //
437 // - **Marketable at the trigger** (a sell limited at or below it, a buy at or above): it trades. It is
438 // credited the **worst price the bar traded that the limit allows** -- a sell the higher of its
439 // limit and the bar's low, a buy the lower of its limit and the bar's high -- so the price is inside
440 // the bar and never through the limit. Crediting the limit itself could name a price the bar never
441 // reached; crediting the trigger would assume the book was there.
442 // - **Not marketable at the trigger** (a sell limited above it): it trades only if the price came back
443 // to the limit after the release. The close proves that when it is at or through the limit; then it
444 // is credited the limit. Otherwise the bar's other extreme may have come first, and nothing is.
445 static std::optional<Decimal64> intrabar_release_price(
446 const strategy::PlannedOrder& order,
447 const strategy::BarEvent& next
448 ) {
449 const auto& trigger = *order.trigger_price_;
450 const auto& limit = *order.price_;
451 const bool sell = order.side_ == OrderSide::SELL;
452 if (limit_marketable_at(order, trigger)) {
453 if (sell) {
454 return decimal_less(limit, next.bar_.low_) ? next.bar_.low_ : limit;
455 }
456 return decimal_less(next.bar_.high_, limit) ? next.bar_.high_ : limit;
457 }
458 if (limit_marketable_at(order, next.bar_.close_)) {
459 return limit;
460 }
461 return std::nullopt;
462 }
463
464 // Returns the price before slippage, or nothing when the order does not trade against this bar.
465 //
466 // A market order takes the open. A limit trades only if the bar's range reached it — a buy needs the
467 // low at or below the limit, a sell needs the high at or above it — because a bar that never touched
468 // the price is a bar in which the order sat unfilled, however close it came.
469 //
470 // Sets `at_open` when the price is the bar's stated open -- a market order, or a limit under
471 // `AT_OPEN_IF_BETTER` whose open was better -- which is what lets `decide` date the charge from the
472 // open's day. False for a bar with no open, whose close stands in for it.
473 std::optional<Decimal64> reference_price(
474 const strategy::PlannedOrder& order,
475 const strategy::BarEvent& next,
476 bool& at_open
477 ) const {
478 at_open = false;
479 // A bar with no open prices a market order at its close, which is not the open's moment.
480 const bool has_open = next.bar_.open_.has_value();
481 const auto open = has_open ? *next.bar_.open_ : next.bar_.close_;
482
483 // A released stop is priced as the limit it has become.
484 const bool limit_basis = limit_priced(order.type_);
485 if (!limit_basis || !order.price_) {
486 at_open = has_open;
487 return open;
488 }
489 const auto& limit = *order.price_;
490 bool open_if_better = false;
491 switch (limit_price_) {
492 case LimitFillPrice::AT_OPEN_IF_BETTER:
493 open_if_better = true;
494 break;
495 case LimitFillPrice::AT_LIMIT:
496 break;
497 }
498
499 if (order.side_ == OrderSide::BUY) {
500 if (decimal_less(limit, next.bar_.low_)) {
501 return std::nullopt;
502 }
503 if (open_if_better && decimal_less(open, limit)) {
504 at_open = has_open;
505 return open;
506 }
507 return limit;
508 }
509
510 if (decimal_less(next.bar_.high_, limit)) {
511 return std::nullopt;
512 }
513 if (open_if_better && decimal_less(limit, open)) {
514 at_open = has_open;
515 return open;
516 }
517 return limit;
518 }
