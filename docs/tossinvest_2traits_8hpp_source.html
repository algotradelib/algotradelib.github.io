203 static Result<tossinvest::TossAmendment> to_replacement(
204 const Ref& ref,
205 const AmendInstruction& instruction
206 ) {
207 // Toss's `OrderModifyRequest` carries `orderType`, `quantity`, `price` and
208 // the high-value flag, and **no validity field at all**: a modification
209 // cannot change the condition or the session, and Toss cannot refuse a
210 // request to. So this refuses on Toss's behalf, and the question is not
211 // whether the requested value is one Toss supports -- it can place a
212 // closing-auction order -- but whether it *differs* from what the order
213 // already is.
214 //
215 // Which means the original has to be known. An unset `resting_` is refused
216 // rather than assumed to match: a hand-built instruction naming DAY/REGULAR
217 // against a resting CLS order compared equal to a defaulted original and was
218 // sent, and the order stayed CLS while the caller was told it had moved.
219 if (!instruction.resting_) {
220 BrokerError error;
221 error.category_ = ErrorCategory::INVALID_ARGUMENT;
222 error.reached_broker_ = false;
223 error.broker_code_ = "unknown_resting_shape";
224 error.message_ =
225 "Toss cannot change an order's validity condition or session by amending it, and this "
226 "instruction does not say what the order's are, so neither can be verified; build the "
227 "instruction with reprice() or replace_with() from the working order";
228 return Result<tossinvest::TossAmendment>(std::move(error));
229 }
230 // An unset field on the shape asks for no change, which is always safe here.
231 if (instruction.shape_.time_in_force_ &&
232 *instruction.shape_.time_in_force_ != instruction.resting_->time_in_force_) {
233 BrokerError error;
234 error.category_ = ErrorCategory::INVALID_ARGUMENT;
235 error.reached_broker_ = false;
236 error.broker_code_ = "unsupported_time_in_force";
237 error.message_ =
238 "Toss cannot change an order's validity condition by amending it; cancel the order and "
239 "place a new one instead";
240 return Result<tossinvest::TossAmendment>(std::move(error));
241 }
242 if (instruction.shape_.session_ &&
243 *instruction.shape_.session_ != instruction.resting_->session_) {
244 BrokerError error;
245 error.category_ = ErrorCategory::INVALID_ARGUMENT;
246 error.reached_broker_ = false;
247 error.broker_code_ = "unsupported_session";
248 error.message_ =
249 "Toss cannot move an order to another trading session by amending it; cancel the order "
250 "and place a new one instead";
251 return Result<tossinvest::TossAmendment>(std::move(error));
252 }
253
254 tossinvest::TossAmendment amendment;
255 amendment.order_id_ = ref.primary_;
256 amendment.type_ = instruction.shape_.type_;
257 amendment.quantity_ = instruction.quantity_;
258 amendment.price_ = instruction.shape_.price_;
259 amendment.confirm_high_value_ = instruction.confirm_high_value_;
260 return Result<tossinvest::TossAmendment>(std::move(amendment));
261 }
