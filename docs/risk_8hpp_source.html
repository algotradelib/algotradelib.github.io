114inline constexpr bool halt_blocks(HaltReason reason, InstructionClass klass) noexcept {
115 switch (klass) {
116 case InstructionClass::OPENING:
117 return true;
118 case InstructionClass::WITHDRAWAL:
119 break;
120 }
121 switch (reason) {
122 case HaltReason::OPERATOR:
123 case HaltReason::LOCAL_FAILURE:
124 case HaltReason::STOPPING:
125 return true;
126 // Guards against an opening order duplicating a predecessor's; a named withdrawal duplicates nothing.
127 case HaltReason::ORDER_LIST_UNREAD:
128 // None of these says anything about a row the caller already named.
129 case HaltReason::PORTFOLIO_ABSENT:
130 case HaltReason::PORTFOLIO_STALE:
131 case HaltReason::BUYING_POWER_UNREAD:
132 // A known resting order is still safe to withdraw when polling is abandoned.
133 case HaltReason::POLL_FAILING:
134 return false;
135 }
136 return true;
137}

431template <typename Broker>
432std::optional<Decimal64> projected_gross_exposure(
433 const Portfolio<Broker>& portfolio,
434 const std::vector<InstrumentBasis>& basis,
435 const Currency& currency,
436 const PlannedOrder* candidate,
437 const std::optional<Decimal64>& candidate_price,
438 CandidateContribution contribution
439) {
440 const auto zero = Decimal64::parse("0");
441
442 // Every instrument the answer depends on: held, working, or about to be ordered. Collected first
443 // so each one is valued exactly once however many rows mention it.
444 std::vector<InstrumentId> instruments;
445 const auto note = [&instruments](const InstrumentId& id) {
446 for (const auto& seen : instruments) {
447 if (seen == id) {
448 return;
449 }
450 }
451 instruments.push_back(id);
452 };
453 for (const auto& position : portfolio.positions_) {
454 // A flat row (a closed position a broker or the replay ledger still lists) is not exposure and
455 // needs no basis; if it also has a working order, the loop below collects it.
456 if (position.quantity_.coefficient() == 0) {
457 continue;
458 }
459 note(position.instrument_);
460 }
461 for (const auto& working : portfolio.working_) {
462 if (order_is_open(working.order_.status_)) {
463 note(working.order_.instrument_);
464 }
465 }
466 if (candidate != nullptr) {
467 note(candidate->instrument_);
468 }
469
470 auto total = zero;
471 for (const auto& id : instruments) {
472 const InstrumentBasis* resolved = nullptr;
473 for (const auto& entry : basis) {
474 if (!(entry.instrument_ == id)) {
475 continue;
476 }
477 if (resolved != nullptr) {
478 // Two answers for one instrument. Picking either would be picking one of two currencies or
479 // two marks, and the caller resolved it twice for a reason this cannot see.
480 return std::nullopt;
481 }
482 resolved = &entry;
483 }
484 if (resolved == nullptr) {
485 return std::nullopt;
486 }
487 if (!(resolved->currency_ == currency)) {
488 continue;
489 }
490 // A mark must be positive: zero would omit the position, and a negative one (from bad market
491 // data) would offset real holdings.
492 if (!decimal_less(zero, resolved->price_)) {
493 return std::nullopt;
494 }
495
496 // Summed over every row: a holding can be split across lots.
497 auto held = zero;
498 for (const auto& position : portfolio.positions_) {
499 if (!(position.instrument_ == id)) {
500 continue;
501 }
502 const auto sum = decimal_add(held, position.quantity_);
503 if (!sum) {
504 return std::nullopt;
505 }
506 held = *sum;
507 }
508 auto buys = zero;
509 auto sells = zero;
510 auto price = resolved->price_;
511 for (const auto& working : portfolio.working_) {
512 if (!(working.order_.instrument_ == id) || !order_is_open(working.order_.status_)) {
513 continue;
514 }
515 const auto remaining = decimal_subtract(
516 working.order_.quantity_, working.order_.execution_.filled_quantity_
517 );
518 if (!remaining) {
519 return std::nullopt;
520 }
521 // A row filled beyond its quantity refuses (`working_quantity` skips it): a limit cannot infer
522 // from an inconsistent snapshot that nothing is outstanding.
523 if (decimal_less(*remaining, zero)) {
524 return std::nullopt;
525 }
526 const auto sum = decimal_add(
527 working.order_.side_ == OrderSide::BUY ? buys : sells, *remaining
528 );
529 if (!sum) {
530 return std::nullopt;
531 }
532 (working.order_.side_ == OrderSide::BUY ? buys : sells) = *sum;
533 if (working.order_.price_) {
534 // A non-positive price would silently lose to the mark in `max`; refuse instead.
535 if (!decimal_less(zero, *working.order_.price_)) {
536 return std::nullopt;
537 }
538 price = risk_detail::decimal_max(price, *working.order_.price_);
539 }
540 }
541
542 if (candidate != nullptr && candidate->instrument_ == id) {
543 // The candidate's price applies under both contributions, its quantity only under one, so the
544 // two projections differ only by the quantity.
545 const auto stated = candidate->price_ ? candidate->price_ : candidate_price;
546 if (stated) {
547 // Validated, not just offered to `max`: a negative price would vanish into the mark here and
548 // pass `max_order_notional_` as a negative value. Nothing upstream rules it out.
549 if (!decimal_less(zero, *stated)) {
550 return std::nullopt;
551 }
552 price = risk_detail::decimal_max(price, *stated);
553 }
554 bool adds_quantity = false;
555 switch (contribution) {
556 case CandidateContribution::QUANTITY_AND_PRICE:
557 adds_quantity = true;
558 break;
559 case CandidateContribution::PRICE_ONLY:
560 break;
561 }
562 if (adds_quantity) {
563 const auto sum = decimal_add(
564 candidate->side_ == OrderSide::BUY ? buys : sells, candidate->quantity_
565 );
566 if (!sum) {
567 return std::nullopt;
568 }
569 (candidate->side_ == OrderSide::BUY ? buys : sells) = *sum;
570 }
571 }
572
573 const auto if_bought = decimal_add(held, buys);
574 const auto if_sold = decimal_subtract(held, sells);
575 if (!if_bought || !if_sold) {
576 return std::nullopt;
577 }
578 const auto long_side = risk_detail::decimal_magnitude(*if_bought);
579 const auto short_side = risk_detail::decimal_magnitude(*if_sold);
580 if (!long_side || !short_side) {
581 return std::nullopt;
582 }
583 const auto projected = risk_detail::decimal_max(*long_side, *short_side);
584
585 const auto value = decimal_multiply(projected, price);
586 if (!value) {
587 return std::nullopt;
588 }
589 const auto running = decimal_add(total, *value);
590 if (!running) {
591 return std::nullopt;
592 }
593 total = *running;
594 }
595 return total;
596}

618template <typename Broker>
619std::optional<bool> closes_toward_flat(
620 const Portfolio<Broker>& portfolio, const PlannedOrder& order
621) {
622 const auto zero = Decimal64::parse("0");
623
624 // Net across every row; a split holding must not understate `held`.
625 const auto net = portfolio.net_quantity_of(order.instrument_);
626 if (!net) {
627 return std::nullopt;
628 }
629 const auto held = *net;
630 // No row and a row of zero both mean nothing to close.
631 if (held.coefficient() == 0) {
632 return false;
633 }
634
635 // A non-positive quantity closes nothing; a negative one would otherwise satisfy the bound below.
636 if (!decimal_less(zero, order.quantity_)) {
637 return false;
638 }
639
640 const auto closing = decimal_less(zero, held) ? OrderSide::SELL : OrderSide::BUY;
641 if (order.side_ != closing) {
642 return false;
643 }
644
645 auto resting = zero;
646 for (const auto& working : portfolio.working_) {
647 if (!(working.order_.instrument_ == order.instrument_) || working.order_.side_ != closing) {
648 continue;
649 }
650 if (!order_is_open(working.order_.status_)) {
651 continue;
652 }
653 const auto remaining = decimal_subtract(
654 working.order_.quantity_, working.order_.execution_.filled_quantity_
655 );
656 // A row filled beyond its quantity is a contradiction; refuse rather than count zero.
657 if (!remaining || decimal_less(*remaining, zero)) {
658 return std::nullopt;
659 }
660 const auto sum = decimal_add(resting, *remaining);
661 if (!sum) {
662 return std::nullopt;
663 }
664 resting = *sum;
665 }
666
667 const auto committed = decimal_add(resting, order.quantity_);
668 if (!committed) {
669 return std::nullopt;
670 }
671 const auto magnitude = risk_detail::decimal_magnitude(held);
672 if (!magnitude) {
673 return std::nullopt;
674 }
675 // `<=`: an order that reaches flat exactly is a close (the usual liquidation case).
676 return !decimal_less(*magnitude, *committed);
677}

926 template <typename Broker>
927 GuardDecision evaluate(
928 const PlannedOrder& order,
929 const Portfolio<Broker>& portfolio,
930 const GuardInputs& inputs
931 ) const {
932 const auto refuse = [&order](GuardReason reason) {
933 GuardRejection rejection;
934 rejection.reason_ = reason;
935 rejection.instrument_ = order.instrument_;
936 rejection.tag_ = order.tag_;
937 return GuardDecision{std::optional<GuardRejection>(rejection), false};
938 };
939 const auto refuse_with =
940 [&order](GuardReason reason, Decimal64 limit, Decimal64 attempted) {
941 GuardRejection rejection;
942 rejection.reason_ = reason;
943 rejection.instrument_ = order.instrument_;
944 rejection.tag_ = order.tag_;
945 rejection.limit_ = std::move(limit);
946 rejection.attempted_ = std::move(attempted);
947 return GuardDecision{std::optional<GuardRejection>(rejection), false};
948 };
949 // Set by the phase check and reported only if the order survives every later limit, since
950 // `refuse` and `refuse_with` both build a decision with it false. An order the exception
951 // admitted and a ceiling then refused was not admitted.
952 bool admitted_outside_phase = false;
953
954 if (const auto holding = blocking_reason()) {
955 auto decision = refuse(GuardReason::TRADING_HALTED);
956 decision.rejection_->halt_reason_ = *holding;
957 return decision;
958 }
959
960 if (limits_.permitted_universe_) {
961 bool permitted = false;
962 for (const auto& allowed : *limits_.permitted_universe_) {
963 if (allowed == order.instrument_) {
964 permitted = true;
965 break;
966 }
967 }
968 if (!permitted) {
969 return refuse(GuardReason::INSTRUMENT_NOT_PERMITTED);
970 }
971 }
972
973 if (limits_.permitted_phases_) {
974 bool permitted = false;
975 for (const auto phase : *limits_.permitted_phases_) {
976 if (phase == inputs.phase_) {
977 permitted = true;
978 break;
979 }
980 }
981 if (!permitted) {
982 // The phase restricts what an order opens, not what it closes: an exit must stay expressible
983 // after the session. `closes_toward_flat` is narrower than the exposure rule, which would
984 // admit a sell that opens a short against a resting buy.
985 const auto closing = closes_toward_flat(portfolio, order);
986 // Unknown refuses: this branch only relaxes a restriction.
987 if (!closing || !*closing) {
988 return refuse(GuardReason::PHASE_NOT_PERMITTED);
989 }
990 admitted_outside_phase = true;
991 }
992 }
993
994 if (limits_.max_orders_per_window_ &&
995 recent_orders(inputs.now_) >= *limits_.max_orders_per_window_) {
996 return refuse_with(
997 GuardReason::ORDER_RATE_EXCEEDED,
998 Decimal64(static_cast<std::int64_t>(*limits_.max_orders_per_window_), 0),
999 Decimal64(static_cast<std::int64_t>(recent_orders(inputs.now_) + 1), 0)
1000 );
1001 }
1002
1003 // A market order carries no price, so its value has to come from the runner's last quote.
1004 // Without one, a notional or exposure limit cannot be evaluated, and a limit that cannot be
1005 // evaluated must refuse rather than wave the order through.
1006 const auto price = order.price_ ? order.price_ : inputs.reference_price_;
1007 const bool needs_value =
1008 limits_.max_order_notional_.has_value() || limits_.max_gross_exposure_.has_value() ||
1009 (limits_.enforce_buying_power_ && order.side_ == OrderSide::BUY);
1010 if (needs_value && !price) {
1011 return refuse(GuardReason::VALUE_NOT_KNOWN);
1012 }
1013
1014 std::optional<Decimal64> value;
1015 if (price) {
1016 value = decimal_multiply(order.quantity_, *price);
1017 if (needs_value && !value) {
1018 return refuse(GuardReason::VALUE_NOT_KNOWN);
1019 }
1020 }
1021
1022 if (limits_.max_order_notional_ && value &&
1023 decimal_less(*limits_.max_order_notional_, *value)) {
1024 return refuse_with(
1025 GuardReason::ORDER_NOTIONAL_TOO_LARGE, *limits_.max_order_notional_, *value
1026 );
1027 }
1028
1029 if (limits_.enforce_sellable_quantity_ && order.side_ == OrderSide::SELL) {
1030 // One read, so the figure and its basis come from the same row.
1031 const auto reading = portfolio.sellable_reading_for(order.instrument_);
1032 const auto& sellable = reading.quantity_;
1033 if (!sellable) {
1034 // Absent is unknown, and unknown refuses; the held quantity is not substituted.
1035 return refuse(GuardReason::QUANTITY_NOT_SELLABLE);
1036 }
1037 // A figure whose basis is not established refuses even though present; this fails closed for
1038 // an adapter the library has never seen.
1039 if (!reading.basis_.is_enforceable()) {
1040 return refuse(GuardReason::SELLABLE_BASIS_UNESTABLISHED);
1041 }
1042 // Every open sell (the broker's rows and this library's reservations) counts against the
1043 // figure. That over-refuses where the broker already deducted them and never under-refuses;
1044 // see `SellableBasis::DEDUCT`.
1045 const auto resting = portfolio.working_quantity(order.instrument_, OrderSide::SELL);
1046 const auto committed = decimal_add(resting, order.quantity_);
1047 if (!committed) {
1048 return refuse(GuardReason::QUANTITY_NOT_SELLABLE);
1049 }
1050 if (decimal_less(*sellable, *committed)) {
1051 return refuse_with(GuardReason::QUANTITY_NOT_SELLABLE, *sellable, *committed);
1052 }
1053 }
1054
1055 if (limits_.enforce_buying_power_ && order.side_ == OrderSide::BUY) {
1056 if (!value) {
1057 // A price is present (`needs_value`), so the multiplication itself failed.
1058 return refuse(GuardReason::VALUE_NOT_KNOWN);
1059 }
1060 if (!inputs.order_currency_) {
1061 return refuse(GuardReason::INSUFFICIENT_BUYING_POWER);
1062 }
1063 const auto available = portfolio.buying_power_in(*inputs.order_currency_);
1064 if (!available) {
1065 // Absent is unknown, and unknown refuses. (`cash_in` is empty live, so no fallback.)
1066 return refuse(GuardReason::INSUFFICIENT_BUYING_POWER);
1067 }
1068 if (decimal_less(*available, *value)) {
1069 return refuse_with(GuardReason::INSUFFICIENT_BUYING_POWER, *available, *value);
1070 }
1071 }
1072
1073 if (limits_.max_position_quantity_) {
1074 const auto zero = Decimal64::parse("0");
1075 // Net across every position row, as reconciliation sizes: a split holding read from its first
1076 // row would refuse an exit to flat and pass a buy that both lots together put over the ceiling.
1077 const auto net = portfolio.net_quantity_of(order.instrument_);
1078 if (!net) {
1079 return refuse(GuardReason::POSITION_LIMIT_EXCEEDED);
1080 }
1081 const auto held = *net;
1082 // Orders already working on this side count, so two buys that each fit cannot together exceed
1083 // the ceiling. A resting opposite-side order is not counted as relief.
1084 const auto pending = portfolio.working_quantity(order.instrument_, order.side_);
1085 const auto committed = decimal_add(pending, order.quantity_);
1086 if (!committed) {
1087 return refuse(GuardReason::POSITION_LIMIT_EXCEEDED);
1088 }
1089 const auto signed_quantity = order.side_ == OrderSide::BUY ?
1090 std::optional<Decimal64>(*committed) :
1091 decimal_subtract(zero, *committed);
1092 if (!signed_quantity) {
1093 return refuse(GuardReason::POSITION_LIMIT_EXCEEDED);
1094 }
1095 const auto resulting = decimal_add(held, *signed_quantity);
1096 if (!resulting) {
1097 return refuse(GuardReason::POSITION_LIMIT_EXCEEDED);
1098 }
1099 // The ceiling is on magnitude, so a short of the same size is limited as a long is.
1100 const auto magnitude = decimal_less(*resulting, zero) ?
1101 decimal_subtract(zero, *resulting) :
1102 std::optional<Decimal64>(*resulting);
1103 if (!magnitude) {
1104 return refuse(GuardReason::POSITION_LIMIT_EXCEEDED);
1105 }
1106 if (decimal_less(*limits_.max_position_quantity_, *magnitude)) {
1107 return refuse_with(
1108 GuardReason::POSITION_LIMIT_EXCEEDED, *limits_.max_position_quantity_, *magnitude
1109 );
1110 }
1111 }
1112
1113 if (limits_.max_gross_exposure_) {
1114 // Per currency: an unresolved order currency refuses.
1115 if (!inputs.order_currency_) {
1116 return refuse(GuardReason::VALUE_NOT_KNOWN);
1117 }
1118 const auto resulting = projected_gross_exposure(
1119 portfolio, inputs.exposure_basis_, *inputs.order_currency_, &order, price,
1120 CandidateContribution::QUANTITY_AND_PRICE
1121 );
1122 if (!resulting) {
1123 // The basis does not describe the portfolio, or the arithmetic did not fit.
1124 return refuse(GuardReason::VALUE_NOT_KNOWN);
1125 }
1126 if (decimal_less(*limits_.max_gross_exposure_, *resulting)) {
1127 // Over the ceiling, refuse only an order that raises the projection, so an account already
1128 // over it can still sell out. Compare against `PRICE_ONLY` (same prices, no quantity), not
1129 // `nullptr`, so the difference is the order's quantity and not its limit price.
1130 const auto without = projected_gross_exposure(
1131 portfolio, inputs.exposure_basis_, *inputs.order_currency_, &order, price,
1132 CandidateContribution::PRICE_ONLY
1133 );
1134 if (!without) {
1135 return refuse(GuardReason::VALUE_NOT_KNOWN);
1136 }
1137 if (decimal_less(*without, *resulting)) {
1138 return refuse_with(
1139 GuardReason::GROSS_EXPOSURE_EXCEEDED, *limits_.max_gross_exposure_, *resulting
1140 );
1141 }
1142 }
1143 }
1144
1145 return GuardDecision{std::nullopt, admitted_outside_phase};
1146 }
1147
1148private:
1149 // `reason`'s slot in the hold arrays.
1150 static constexpr std::size_t index_of(HaltReason reason) noexcept {
1151 return static_cast<std::size_t>(reason);
1152 }
1153
1154 RiskLimits limits_;
1155 // Which holds are in force, indexed by `HaltReason`; declaration order is priority order.
1156 std::array<bool, halt_reason_count> held_{};
1157 // Each hold's detail, empty where none or not held.
1158 std::array<std::string, halt_reason_count> details_;
1159 // Timestamps of orders sent, oldest first; a sliding window, not a counter reset on a boundary.
1160 std::deque<std::chrono::system_clock::time_point> sent_;
1161 std::size_t total_sent_{0};
1162};
