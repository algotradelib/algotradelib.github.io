102inline constexpr bool halt_blocks(HaltReason reason, InstructionClass klass) noexcept {
103 if (klass == InstructionClass::OPENING) {
104 return true;
105 }
106 switch (reason) {
107 case HaltReason::OPERATOR:
108 case HaltReason::LOCAL_FAILURE:
109 case HaltReason::STOPPING:
110 return true;
111 // Guards against an opening order duplicating a predecessor's; a named withdrawal duplicates nothing.
112 case HaltReason::ORDER_LIST_UNREAD:
113 // None of these says anything about a row the caller already named.
114 case HaltReason::PORTFOLIO_ABSENT:
115 case HaltReason::PORTFOLIO_STALE:
116 case HaltReason::BUYING_POWER_UNREAD:
117 // A known resting order is still safe to withdraw when polling is abandoned.
118 case HaltReason::POLL_FAILING:
119 return false;
120 }
121 return true;
122}

411template <typename Broker>
412std::optional<Decimal64> projected_gross_exposure(
413 const Portfolio<Broker>& portfolio,
414 const std::vector<InstrumentBasis>& basis,
415 const Currency& currency,
416 const PlannedOrder* candidate,
417 const std::optional<Decimal64>& candidate_price,
418 CandidateContribution contribution
419) {
420 const auto zero = Decimal64::parse("0");
421
422 // Every instrument the answer depends on: held, working, or about to be ordered. Collected first
423 // so each one is valued exactly once however many rows mention it.
424 std::vector<InstrumentId> instruments;
425 const auto note = [&instruments](const InstrumentId& id) {
426 for (const auto& seen : instruments) {
427 if (seen == id) {
428 return;
429 }
430 }
431 instruments.push_back(id);
432 };
433 for (const auto& position : portfolio.positions_) {
434 // A flat row (a closed position a broker or the replay ledger still lists) is not exposure and
435 // needs no basis; if it also has a working order, the loop below collects it.
436 if (position.quantity_.coefficient() == 0) {
437 continue;
438 }
439 note(position.instrument_);
440 }
441 for (const auto& working : portfolio.working_) {
442 if (order_is_open(working.order_.status_)) {
443 note(working.order_.instrument_);
444 }
445 }
446 if (candidate != nullptr) {
447 note(candidate->instrument_);
448 }
449
450 auto total = zero;
451 for (const auto& id : instruments) {
452 const InstrumentBasis* resolved = nullptr;
453 for (const auto& entry : basis) {
454 if (!(entry.instrument_ == id)) {
455 continue;
456 }
457 if (resolved != nullptr) {
458 // Two answers for one instrument. Picking either would be picking one of two currencies or
459 // two marks, and the caller resolved it twice for a reason this cannot see.
460 return std::nullopt;
461 }
462 resolved = &entry;
463 }
464 if (resolved == nullptr) {
465 return std::nullopt;
466 }
467 if (!(resolved->currency_ == currency)) {
468 continue;
469 }
470 // A mark must be positive: zero would omit the position, and a negative one (from bad market
471 // data) would offset real holdings.
472 if (!decimal_less(zero, resolved->price_)) {
473 return std::nullopt;
474 }
475
476 // Summed over every row: a holding can be split across lots.
477 auto held = zero;
478 for (const auto& position : portfolio.positions_) {
479 if (!(position.instrument_ == id)) {
480 continue;
481 }
482 const auto sum = decimal_add(held, position.quantity_);
483 if (!sum) {
484 return std::nullopt;
485 }
486 held = *sum;
487 }
488 auto buys = zero;
489 auto sells = zero;
490 auto price = resolved->price_;
491 for (const auto& working : portfolio.working_) {
492 if (!(working.order_.instrument_ == id) || !order_is_open(working.order_.status_)) {
493 continue;
494 }
495 const auto remaining = decimal_subtract(
496 working.order_.quantity_, working.order_.execution_.filled_quantity_
497 );
498 if (!remaining) {
499 return std::nullopt;
500 }
501 // A row filled beyond its quantity refuses (`working_quantity` skips it): a limit cannot infer
502 // from an inconsistent snapshot that nothing is outstanding.
503 if (decimal_less(*remaining, zero)) {
504 return std::nullopt;
505 }
506 const auto sum = decimal_add(
507 working.order_.side_ == OrderSide::BUY ? buys : sells, *remaining
508 );
509 if (!sum) {
510 return std::nullopt;
511 }
512 (working.order_.side_ == OrderSide::BUY ? buys : sells) = *sum;
513 if (working.order_.price_) {
514 // A non-positive price would silently lose to the mark in `max`; refuse instead.
515 if (!decimal_less(zero, *working.order_.price_)) {
516 return std::nullopt;
517 }
518 price = risk_detail::decimal_max(price, *working.order_.price_);
519 }
520 }
521
522 if (candidate != nullptr && candidate->instrument_ == id) {
523 // The candidate's price applies under both contributions, its quantity only under one, so the
524 // two projections differ only by the quantity.
525 const auto stated = candidate->price_ ? candidate->price_ : candidate_price;
526 if (stated) {
527 // Validated, not just offered to `max`: a negative price would vanish into the mark here and
528 // pass `max_order_notional_` as a negative value. Nothing upstream rules it out.
529 if (!decimal_less(zero, *stated)) {
530 return std::nullopt;
531 }
532 price = risk_detail::decimal_max(price, *stated);
533 }
534 if (contribution == CandidateContribution::QUANTITY_AND_PRICE) {
535 const auto sum = decimal_add(
536 candidate->side_ == OrderSide::BUY ? buys : sells, candidate->quantity_
537 );
538 if (!sum) {
539 return std::nullopt;
540 }
541 (candidate->side_ == OrderSide::BUY ? buys : sells) = *sum;
542 }
543 }
544
545 const auto if_bought = decimal_add(held, buys);
546 const auto if_sold = decimal_subtract(held, sells);
547 if (!if_bought || !if_sold) {
548 return std::nullopt;
549 }
550 const auto long_side = risk_detail::decimal_magnitude(*if_bought);
551 const auto short_side = risk_detail::decimal_magnitude(*if_sold);
552 if (!long_side || !short_side) {
553 return std::nullopt;
554 }
555 const auto projected = risk_detail::decimal_max(*long_side, *short_side);
556
557 const auto value = decimal_multiply(projected, price);
558 if (!value) {
559 return std::nullopt;
560 }
561 const auto running = decimal_add(total, *value);
562 if (!running) {
563 return std::nullopt;
564 }
565 total = *running;
566 }
567 return total;
568}

590template <typename Broker>
591std::optional<bool> closes_toward_flat(
592 const Portfolio<Broker>& portfolio, const PlannedOrder& order
593) {
594 const auto zero = Decimal64::parse("0");
595
596 // Net across every row; a split holding must not understate `held`.
597 const auto net = portfolio.net_quantity_of(order.instrument_);
598 if (!net) {
599 return std::nullopt;
600 }
601 const auto held = *net;
602 // No row and a row of zero both mean nothing to close.
603 if (held.coefficient() == 0) {
604 return false;
605 }
606
607 // A non-positive quantity closes nothing; a negative one would otherwise satisfy the bound below.
608 if (!decimal_less(zero, order.quantity_)) {
609 return false;
610 }
611
612 const auto closing = decimal_less(zero, held) ? OrderSide::SELL : OrderSide::BUY;
613 if (order.side_ != closing) {
614 return false;
615 }
616
617 auto resting = zero;
618 for (const auto& working : portfolio.working_) {
619 if (!(working.order_.instrument_ == order.instrument_) || working.order_.side_ != closing) {
620 continue;
621 }
622 if (!order_is_open(working.order_.status_)) {
623 continue;
624 }
625 const auto remaining = decimal_subtract(
626 working.order_.quantity_, working.order_.execution_.filled_quantity_
627 );
628 // A row filled beyond its quantity is a contradiction; refuse rather than count zero.
629 if (!remaining || decimal_less(*remaining, zero)) {
630 return std::nullopt;
631 }
632 const auto sum = decimal_add(resting, *remaining);
633 if (!sum) {
634 return std::nullopt;
635 }
636 resting = *sum;
637 }
638
639 const auto committed = decimal_add(resting, order.quantity_);
640 if (!committed) {
641 return std::nullopt;
642 }
643 const auto magnitude = risk_detail::decimal_magnitude(held);
644 if (!magnitude) {
645 return std::nullopt;
646 }
647 // `<=`: an order that reaches flat exactly is a close (the usual liquidation case).
648 return !decimal_less(*magnitude, *committed);
649}

898 template <typename Broker>
899 GuardDecision evaluate(
900 const PlannedOrder& order,
901 const Portfolio<Broker>& portfolio,
902 const GuardInputs& inputs
903 ) const {
904 const auto refuse = [&order](GuardReason reason) {
905 GuardRejection rejection;
906 rejection.reason_ = reason;
907 rejection.instrument_ = order.instrument_;
908 rejection.tag_ = order.tag_;
909 return GuardDecision{std::optional<GuardRejection>(rejection), false};
910 };
911 const auto refuse_with =
912 [&order](GuardReason reason, Decimal64 limit, Decimal64 attempted) {
913 GuardRejection rejection;
914 rejection.reason_ = reason;
915 rejection.instrument_ = order.instrument_;
916 rejection.tag_ = order.tag_;
917 rejection.limit_ = std::move(limit);
918 rejection.attempted_ = std::move(attempted);
919 return GuardDecision{std::optional<GuardRejection>(rejection), false};
920 };
921 // Set by the phase check and reported only if the order survives every later limit, since
922 // `refuse` and `refuse_with` both build a decision with it false. An order the exception
923 // admitted and a ceiling then refused was not admitted.
924 bool admitted_outside_phase = false;
925
926 if (const auto holding = blocking_reason()) {
927 auto decision = refuse(GuardReason::TRADING_HALTED);
928 decision.rejection_->halt_reason_ = *holding;
929 return decision;
930 }
931
932 if (limits_.permitted_universe_) {
933 bool permitted = false;
934 for (const auto& allowed : *limits_.permitted_universe_) {
935 if (allowed == order.instrument_) {
936 permitted = true;
937 break;
938 }
939 }
940 if (!permitted) {
941 return refuse(GuardReason::INSTRUMENT_NOT_PERMITTED);
942 }
943 }
944
945 if (limits_.permitted_phases_) {
946 bool permitted = false;
947 for (const auto phase : *limits_.permitted_phases_) {
948 if (phase == inputs.phase_) {
949 permitted = true;
950 break;
951 }
952 }
953 if (!permitted) {
954 // The phase restricts what an order opens, not what it closes: an exit must stay expressible
955 // after the session. `closes_toward_flat` is narrower than the exposure rule, which would
956 // admit a sell that opens a short against a resting buy.
957 const auto closing = closes_toward_flat(portfolio, order);
958 // Unknown refuses: this branch only relaxes a restriction.
959 if (!closing || !*closing) {
960 return refuse(GuardReason::PHASE_NOT_PERMITTED);
961 }
962 admitted_outside_phase = true;
963 }
964 }
965
966 if (limits_.max_orders_per_window_ &&
967 recent_orders(inputs.now_) >= *limits_.max_orders_per_window_) {
968 return refuse_with(
969 GuardReason::ORDER_RATE_EXCEEDED,
970 Decimal64(static_cast<std::int64_t>(*limits_.max_orders_per_window_), 0),
971 Decimal64(static_cast<std::int64_t>(recent_orders(inputs.now_) + 1), 0)
972 );
973 }
974
975 // A market order carries no price, so its value has to come from the runner's last quote.
976 // Without one, a notional or exposure limit cannot be evaluated, and a limit that cannot be
977 // evaluated must refuse rather than wave the order through.
978 const auto price = order.price_ ? order.price_ : inputs.reference_price_;
979 const bool needs_value =
980 limits_.max_order_notional_.has_value() || limits_.max_gross_exposure_.has_value() ||
981 (limits_.enforce_buying_power_ && order.side_ == OrderSide::BUY);
982 if (needs_value && !price) {
983 return refuse(GuardReason::VALUE_NOT_KNOWN);
984 }
985
986 std::optional<Decimal64> value;
987 if (price) {
988 value = decimal_multiply(order.quantity_, *price);
989 if (needs_value && !value) {
990 return refuse(GuardReason::VALUE_NOT_KNOWN);
991 }
992 }
993
994 if (limits_.max_order_notional_ && value &&
995 decimal_less(*limits_.max_order_notional_, *value)) {
996 return refuse_with(
997 GuardReason::ORDER_NOTIONAL_TOO_LARGE, *limits_.max_order_notional_, *value
998 );
999 }
1000
1001 if (limits_.enforce_sellable_quantity_ && order.side_ == OrderSide::SELL) {
1002 // One read, so the figure and its basis come from the same row.
1003 const auto reading = portfolio.sellable_reading_for(order.instrument_);
1004 const auto& sellable = reading.quantity_;
1005 if (!sellable) {
1006 // Absent is unknown, and unknown refuses; the held quantity is not substituted.
1007 return refuse(GuardReason::QUANTITY_NOT_SELLABLE);
1008 }
1009 // A figure whose basis is not established refuses even though present; this fails closed for
1010 // an adapter the library has never seen.
1011 if (!reading.basis_.is_enforceable()) {
1012 return refuse(GuardReason::SELLABLE_BASIS_UNESTABLISHED);
1013 }
1014 // Every open sell (the broker's rows and this library's reservations) counts against the
1015 // figure. That over-refuses where the broker already deducted them and never under-refuses;
1016 // see `SellableBasis::DEDUCT`.
1017 const auto resting = portfolio.working_quantity(order.instrument_, OrderSide::SELL);
1018 const auto committed = decimal_add(resting, order.quantity_);
1019 if (!committed) {
1020 return refuse(GuardReason::QUANTITY_NOT_SELLABLE);
1021 }
1022 if (decimal_less(*sellable, *committed)) {
1023 return refuse_with(GuardReason::QUANTITY_NOT_SELLABLE, *sellable, *committed);
1024 }
1025 }
1026
1027 if (limits_.enforce_buying_power_ && order.side_ == OrderSide::BUY) {
1028 if (!value) {
1029 // A price is present (`needs_value`), so the multiplication itself failed.
1030 return refuse(GuardReason::VALUE_NOT_KNOWN);
1031 }
1032 if (!inputs.order_currency_) {
1033 return refuse(GuardReason::INSUFFICIENT_BUYING_POWER);
1034 }
1035 const auto available = portfolio.buying_power_in(*inputs.order_currency_);
1036 if (!available) {
1037 // Absent is unknown, and unknown refuses. (`cash_in` is empty live, so no fallback.)
1038 return refuse(GuardReason::INSUFFICIENT_BUYING_POWER);
1039 }
1040 if (decimal_less(*available, *value)) {
1041 return refuse_with(GuardReason::INSUFFICIENT_BUYING_POWER, *available, *value);
1042 }
1043 }
1044
1045 if (limits_.max_position_quantity_) {
1046 const auto zero = Decimal64::parse("0");
1047 // Net across every position row, as reconciliation sizes: a split holding read from its first
1048 // row would refuse an exit to flat and pass a buy that both lots together put over the ceiling.
1049 const auto net = portfolio.net_quantity_of(order.instrument_);
1050 if (!net) {
1051 return refuse(GuardReason::POSITION_LIMIT_EXCEEDED);
1052 }
1053 const auto held = *net;
1054 // Orders already working on this side count, so two buys that each fit cannot together exceed
1055 // the ceiling. A resting opposite-side order is not counted as relief.
1056 const auto pending = portfolio.working_quantity(order.instrument_, order.side_);
1057 const auto committed = decimal_add(pending, order.quantity_);
1058 if (!committed) {
1059 return refuse(GuardReason::POSITION_LIMIT_EXCEEDED);
1060 }
1061 const auto signed_quantity = order.side_ == OrderSide::BUY ?
1062 std::optional<Decimal64>(*committed) :
1063 decimal_subtract(zero, *committed);
1064 if (!signed_quantity) {
1065 return refuse(GuardReason::POSITION_LIMIT_EXCEEDED);
1066 }
1067 const auto resulting = decimal_add(held, *signed_quantity);
1068 if (!resulting) {
1069 return refuse(GuardReason::POSITION_LIMIT_EXCEEDED);
1070 }
1071 // The ceiling is on magnitude, so a short of the same size is limited as a long is.
1072 const auto magnitude = decimal_less(*resulting, zero) ?
1073 decimal_subtract(zero, *resulting) :
1074 std::optional<Decimal64>(*resulting);
1075 if (!magnitude) {
1076 return refuse(GuardReason::POSITION_LIMIT_EXCEEDED);
1077 }
1078 if (decimal_less(*limits_.max_position_quantity_, *magnitude)) {
1079 return refuse_with(
1080 GuardReason::POSITION_LIMIT_EXCEEDED, *limits_.max_position_quantity_, *magnitude
1081 );
1082 }
1083 }
1084
1085 if (limits_.max_gross_exposure_) {
1086 // Per currency: an unresolved order currency refuses.
1087 if (!inputs.order_currency_) {
1088 return refuse(GuardReason::VALUE_NOT_KNOWN);
1089 }
1090 const auto resulting = projected_gross_exposure(
1091 portfolio, inputs.exposure_basis_, *inputs.order_currency_, &order, price,
1092 CandidateContribution::QUANTITY_AND_PRICE
1093 );
1094 if (!resulting) {
1095 // The basis does not describe the portfolio, or the arithmetic did not fit.
1096 return refuse(GuardReason::VALUE_NOT_KNOWN);
1097 }
1098 if (decimal_less(*limits_.max_gross_exposure_, *resulting)) {
1099 // Over the ceiling, refuse only an order that raises the projection, so an account already
1100 // over it can still sell out. Compare against `PRICE_ONLY` (same prices, no quantity), not
1101 // `nullptr`, so the difference is the order's quantity and not its limit price.
1102 const auto without = projected_gross_exposure(
1103 portfolio, inputs.exposure_basis_, *inputs.order_currency_, &order, price,
1104 CandidateContribution::PRICE_ONLY
1105 );
1106 if (!without) {
1107 return refuse(GuardReason::VALUE_NOT_KNOWN);
1108 }
1109 if (decimal_less(*without, *resulting)) {
1110 return refuse_with(
1111 GuardReason::GROSS_EXPOSURE_EXCEEDED, *limits_.max_gross_exposure_, *resulting
1112 );
1113 }
1114 }
1115 }
1116
1117 return GuardDecision{std::nullopt, admitted_outside_phase};
1118 }
