95 self->write_queue_.push_back(std::move(message));
96 if (self->connected_ && !self->closing_ && !self->writing_) {

106 void close() {
107 net::post(strand_, [self = shared_derived()] {
108 self->stop_requested_ = true;
109 self->reconnect_timer_.cancel();
110 if (self->closing_) {
111 return;
112 }
113 self->closing_ = true;
114 if (self->connecting_ && !self->connected_) {
115 self->resolver_.cancel();
116 if (self->ws_) {
117 beast::error_code ignored;
118 beast::get_lowest_layer(*self->ws_).socket().cancel(ignored);
119 }
120 return;
121 }
122 if (self->connected_ && !self->writing_) {
123 self->do_close();
124 } else if (!self->connected_) {
125 self->closing_ = false;
126 self->on_closed();
127 }
128 });
129 }

336 boost::ignore_unused(bytes_transferred);
337 writing_ = false;

342 write_queue_.pop_front();

356 void on_read(beast::error_code error, std::size_t bytes_transferred) {
357 boost::ignore_unused(bytes_transferred);
358 if (error == websocket::error::closed) {
359 connected_ = false;
360 connecting_ = false;
361 closing_ = false;
362 on_closed();
363 schedule_reconnect();
364 return;
365 }
366 if (error) {
367 fail("read", error);
368 return;
369 }
370 const auto message = beast::buffers_to_string(buffer_.data());
371 buffer_.consume(buffer_.size());
372 on_message(message);
373 do_read();
374 }
375
376 void do_close() {
377 ws_->async_close(
378 websocket::close_code::normal,
379 beast::bind_front_handler(&WebSocketSessionBase::on_close, shared_derived())
380 );
381 }
382
383 void on_close(beast::error_code error) {
384 connected_ = false;
385 connecting_ = false;
386 closing_ = false;
387 if (error) {
388 on_error("close", error);
389 }
390 on_closed();
391 if (!stop_requested_) {
392 schedule_reconnect();
393 }
394 }
395
396 void schedule_reconnect() {
397 if (stop_requested_ || !options_.automatic_reconnect_ || closing_) {
398 return;
399 }
400 const auto shift = std::min(reconnect_attempt_, 16U);
401 const auto multiplier = static_cast<std::int64_t>(1ULL << shift);
402 const auto delay = std::min(options_.reconnect_initial_ * multiplier, options_.reconnect_max_);
403 ++reconnect_attempt_;
404 const auto base_delay = std::chrono::duration_cast<std::chrono::milliseconds>(delay);
405 const auto jitter_window = base_delay / 5;
406 const auto jitter = jitter_window.count() <= 0 ?
407 std::chrono::milliseconds(0) :
408 std::chrono::milliseconds(
409 std::uniform_int_distribution<std::int64_t>(0, jitter_window.count())(reconnect_rng_)
410 );
411 reconnect_timer_.expires_after(base_delay + jitter);
412 reconnect_timer_.async_wait([self = shared_derived()](beast::error_code error) {
413 if (!error) {
414 self->do_connect();
415 }
416 });
417 }
