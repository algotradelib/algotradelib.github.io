46int main(int argc, char* argv[]) {
47 try {
48 if (argc > 6) {
49 throw std::invalid_argument("too many arguments");
50 }
51
52 atl::OrderRequest order;
53 order.instrument_ = atl::InstrumentId{
54 atl::Country::KR,
55 atl::Venue::COMBINED,
56 argc > 1 ? argv[1] : "005930",
57 };
58 order.quantity_ = atl::Decimal64::parse(argc > 2 ? argv[2] : "1");
59 order.side_ = parse_order_side(argc > 3 ? argv[3] : "BUY");
60 order.type_ = parse_order_type(argc > 4 ? argv[4] : "LIMIT");
61 if (order.type_ == atl::OrderType::LIMIT) {
62 order.price_ = atl::Decimal64::parse(argc > 5 ? argv[5] : "70000");
63 } else if (argc > 5) {
64 throw std::invalid_argument("a MARKET dry-run must not include a limit price");
65 }
66
67 std::cout << "dry-run: this example never sends an order\n";
68 std::cout << (order.side_ == atl::OrderSide::BUY ? "BUY " : "SELL ")
69 << order.quantity_.str() << ' ' << order.instrument_.symbol_;
70 if (order.price_) {
71 std::cout << " @ " << order.price_->str();
72 } else {
73 std::cout << " at market";
74 }
75 std::cout << '\n';
76 return 0;
77 } catch (const std::exception& exception) {
78 std::cerr << "example failed: " << exception.what() << '\n';
79 print_usage(argv[0]);
80 return 1;
81 }
