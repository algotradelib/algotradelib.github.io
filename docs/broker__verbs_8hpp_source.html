78template <typename Client, typename Token>
79void call_claimed_async_verbs(
80 BrokerSession<Client>& session, const typename BrokerSession<Client>::Ref& ref
81) {
82 using Caps = BrokerCaps<Client>;
83 session.async_quote(probe_instrument(), Token{});
84 session.async_holdings(Token{});
85 session.async_bars(BarQuery{}, Token{});
86 session.async_place(OrderRequest{}, Token{});
87 session.async_replace(ref, AmendInstruction{}, Token{});
88 session.async_cancel(ref, AmendInstruction{}, Token{});
89 session.async_order_history(OrderHistoryQuery{}, Token{});
90 if constexpr (Caps::has_buying_power) {
91 session.async_buying_power(Currency{}, Token{});
92 }
93 if constexpr (Caps::sellable_source == SellableSource::INQUIRY) {
94 session.async_sellable_quantity(probe_instrument(), Token{});
95 }
96 if constexpr (Caps::has_commission_schedule) {
97 session.async_commissions(Token{});
98 }
99 if constexpr (Caps::has_working_orders) {
100 session.async_working_orders(Token{});
101 }
102 if constexpr (Caps::has_reference_data) {
103 session.async_instrument(probe_instrument(), Token{});
104 }
105 if constexpr (Caps::has_calendar) {
106 session.async_calendar(Country::KR, std::nullopt, Token{});
107 }
108 if constexpr (Caps::has_price_limits) {
109 session.async_price_limits(probe_instrument(), Token{});
110 }
111 if constexpr (Caps::has_trade_prints) {
112 session.async_trades(probe_instrument(), 1, Token{});
113 }
114 if constexpr (Caps::has_batch_prices) {
115 session.async_prices(std::vector<InstrumentId>{probe_instrument()}, Token{});
116 }
117 if constexpr (Caps::has_depth) {
118 session.async_depth(probe_instrument(), Token{});
119 }
120 if constexpr (Caps::has_order_by_id) {
121 session.async_order(ref, Token{});
122 }
123 if constexpr (Caps::has_account_discovery) {
124 async_accounts(session.client(), Token{});
125 }
126}

190template <typename Client>
191constexpr bool claimed_verbs_compile() {
192 if constexpr (BrokerTraits<Client>::blocking) {
193 static_cast<void>(&call_claimed_blocking_verbs<Client>);
194 } else {
195 static_cast<void>(&call_claimed_async_verbs<Client, PlainCallback>);
196 static_cast<void>(&call_claimed_async_verbs<Client, MoveOnlyCallback>);
197 static_cast<void>(&call_claimed_async_verbs<Client, boost::asio::use_future_t<>>);
198 }
199 return true;
200}
