198 return std::nullopt;

212inline void order_depth(DepthSnapshot& book) {
213 const auto by_price = [](const DepthLevel& lhs, const DepthLevel& rhs) {
214 return decimal_less(lhs.price_, rhs.price_);

216 std::sort(book.asks_.begin(), book.asks_.end(), by_price);
217 std::sort(book.bids_.begin(), book.bids_.end(), [&by_price](const auto& l, const auto& r) {
