241 std::optional<Decimal64> spread() const {
242 const auto bid = best_bid();
243 const auto ask = best_ask();
244 if (!bid || !ask) {
245 return std::nullopt;
246 }
247 return decimal_subtract(ask->price_, bid->price_);
248 }
249
257 std::optional<Decimal64> mid() const {
258 const auto bid = best_bid();
259 const auto ask = best_ask();
260 if (!bid || !ask) {
261 return std::nullopt;
262 }
263 const auto sum = decimal_add(bid->price_, ask->price_);
264 if (!sum) {
265 return std::nullopt;
266 }
267 return decimal_halve(*sum);
268 }

275inline void order_depth(DepthSnapshot& book) {
276 const auto by_price = [](const DepthLevel& lhs, const DepthLevel& rhs) {
277 return decimal_less(lhs.price_, rhs.price_);
278 };
279 std::sort(book.asks_.begin(), book.asks_.end(), by_price);
280 std::sort(book.bids_.begin(), book.bids_.end(), [&by_price](const auto& l, const auto& r) {
281 return by_price(r, l);
282 });
283}
