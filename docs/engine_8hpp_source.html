217 static bool can_model(const strategy::PlannedOrder& order) noexcept {
218 if (order.session_ != OrderSession::REGULAR) {
219 return false;
220 }
221 switch (order.type_) {
222 case OrderType::STOP_LIMIT:
223 // A stop the model can release: both prices, and a day order (the only stop KIS and Kiwoom offer).
224 return order.trigger_price_.has_value() && order.price_.has_value() &&
225 order.time_in_force_ == TimeInForce::DAY;
226 case OrderType::LIMIT:
227 case OrderType::MARKET:
228 // A trigger on a limit or market order would be ignored; refuse it.
229 return models_time_in_force(order.time_in_force_) && !order.trigger_price_;
230 case OrderType::UNKNOWN:
231 case OrderType::CONDITIONAL_LIMIT:
232 case OrderType::BEST_LIMIT:
233 case OrderType::PRIMARY_LIMIT:
234 case OrderType::MID_PRICE:
235 case OrderType::CLOSING_PRICE:
236 return false;
237 }
238 return false;
239 }

441 bool withdraw(const OrderRef<BrokerTag>& ref) {
442 const auto at = live_slot_of(ref);
443 if (!at) {
444 return false;
445 }
446 // **Marked removed where it is, not erased.** Erasing moved every later order (each with three strings)
447 // one place down, for every fill; the storage is compacted instead once removed orders outnumber the
448 // rest, which keeps placement order and costs each removal a constant on average. Finding the order in
449 // its instrument's list is linear in that instrument's outstanding orders, usually one.
450 auto& slot = slots_[*at];
451 slot.live_ = false;
452 --live_;
453 auto& serials = by_instrument_.find(slot.entry_.order_.instrument_)->second;
454 serials.erase(std::find(serials.begin(), serials.end(), slot.serial_));
455 ++changes_;
456 if (slots_.size() - live_ > live_) {
457 compact();
458 }
459 return true;
460 }

515 // Where the order with `serial` is in `slots_`. Serials rise in placement order, which compaction keeps.
516 std::optional<std::size_t> slot_of(std::uint64_t serial) const noexcept {
517 const auto found = std::lower_bound(
518 slots_.begin(), slots_.end(), serial,
519 [](const Slot& slot, std::uint64_t wanted) { return slot.serial_ < wanted; }
520 );
521 if (found == slots_.end() || found->serial_ != serial) {
522 return std::nullopt;
523 }
524 return static_cast<std::size_t>(found - slots_.begin());
525 }
526
527 // Where the outstanding order `ref` names is, by the serial its primary part states: a reference this
528 // executor did not issue, or one already filled or withdrawn, is found nowhere, as by a scan comparing whole
529 // references.
530 std::optional<std::size_t> live_slot_of(const OrderRef<BrokerTag>& ref) const noexcept {
531 const auto& text = ref.primary_;
532 std::uint64_t serial = 0;
533 const auto parsed = std::from_chars(text.data(), text.data() + text.size(), serial);
534 if (parsed.ec != std::errc{} || parsed.ptr != text.data() + text.size()) {
535 return std::nullopt;
536 }
537 const auto at = slot_of(serial);
538 if (!at || !slots_[*at].live_ || !(slots_[*at].entry_.ref_ == ref)) {
539 return std::nullopt;
540 }
541 return at;
542 }

842 void feed(const strategy::BarEvent& event, std::optional<bool> trading_day = true) {
843 // **A halted replay consumes nothing further.** `feed` keeps returning `void` so no caller has to
844 // change, and a caller that ignores `ReplayResult` sees a run that ends early rather than one
845 // that carries on costing nothing: no bar counted, no fill, no mark, no equity point, and no
846 // callback into the strategy. That is API compatibility, which is not the same as invisibility,
847 // and it is why the halt is stated on the result and first in the report.
848 if (result_.halted_) {
849 pending_.clear();
850 return;
851 }
852 // **One lookup of the instrument per bar**: its slot says where its lot, its last price and its series are,
853 // and whether it already has a bar in the period being collected.
854 auto& slot = slot_for(event.instrument_);
855 if constexpr (Runner::takes_bar_slices) {
856 // **Checked before anything is counted, settled, marked or cached**, so a refused bar leaves the
857 // replay as it was. The pending period is delivered here, before this bar's settlement, exactly
858 // where `on_bar` would have run for that period's last bar.
859 check_slice_order(event, slot);
860 if (!pending_.empty() && pending_.period_end() < event.period_end_) {
861 deliver_pending();
862 if (result_.halted_) {
863 return;
864 }
865 }
866 }
867 ++result_.bars_;
868 if (previous_ && !ReplayClock::contiguous(previous_->first, previous_->second, event)) {
869 // A gap is counted rather than interpolated across: nothing happened in it that a strategy
870 // should be told about, and inventing a bar would be inventing data.
871 ++result_.gaps_;
872 }
873
874 // Opened before anything fills against this bar, so that an order placed during it carries a
875 // later stamp than the ones already resting and cannot fill against the bar it was decided on.
876 runner_.executor().open_bar();
877 // And dated, for the half of that rule a counter cannot express -- see `QueuedOrder::decided_at_`.
878 // A bar's decisions are all made at its period end, whichever handler makes them: `on_fill`
879 // during settlement, `on_phase`, `on_bar`, and a handler answering an order update afterwards.
880 //
881 // **Whether that instant is exact depends on the stamp**, and getting this wrong deferred real
882 // fills by a month. A close-stamped bar's period end is its own timestamp, stated by the broker.
883 // An open-stamped one is `bar_period_end`'s arithmetic, which is exact for an intraday bar (late
884 // only for a session's last bar cut short by the close) and deliberately **nominal** for a day, a
885 // week, a month and a year -- and every shipped adapter stamps every daily and longer bar `OPEN`.
886 // February's nominal end is five days into March, so an order decided on February's bar looked as
887 // though it had been decided inside March's period and was deferred to April.
888 runner_.executor().deciding_at(event.period_end_, period_end_is_exact(event));
889
890 // A period's point is recorded once its last bar is in, which is when a bar of another period arrives:
891 // before that bar fills anything, so the point is the equity the period ended with. Any other period end,
892 // later or (bars of mixed intervals) earlier, ends the run, so no period is lost.
893 if (period_point_ && *period_point_ != event.period_end_) {
894 record_point(*period_point_);
895 }
896 settle_queued_against(event, slot);
897 // **The halt stops this bar too, not merely the next one.** Returning only from settlement left
898 // `feed` to mark the book, record an equity point, advance the clock and call `on_phase` and
899 // `on_bar` on the very bar whose fill could not be costed -- so the strategy decided once more
900 // on a portfolio missing that trade, and could queue orders that then survived in the executor
901 // because every later `feed` is a no-op and `finish` returns early. Nothing runs after the
902 // refusal.
903 if (result_.halted_) {
904 return;
905 }
906 // **A mark that will not fit halts the replay**, like a fill it cannot cost or record. The lot
907 // stays at its previous price, while `Runner::feed` is about to hand the strategy this bar's
908 // close as the last price -- so `on_bar` would decide, and `Runner::settle` would size and check
909 // an order, against a position valued one bar ago beside a price from this one. An order placed
910 // from that view fills on a later bar and changes every figure after it. This used to skip only
911 // the equity point and carry on, which reported nothing and let that order through.
912 if (slot.lot_ && !ledger_.mark_lot(*slot.lot_, event.bar_.close_)) {
913 BrokerError error;
914 error.category_ = ErrorCategory::INVALID_ARGUMENT;
915 error.reached_broker_ = false;
916 error.broker_code_ = "mark_not_representable";
917 error.message_ = "a position in " + event.instrument_.symbol_ + " could not be valued at the "
918 "bar's close without leaving an exact decimal, so the replay stopped rather than let a "
919 "strategy decide on a position one bar behind its price";
920 result_.halted_ = std::move(error);
921 abort_queue();
922 return;
923 }
924 count_bar(event);
925
926 // **The price moves with the portfolio, not after it.** `Runner::settle` values a market order
927 // at `context_.last_price`, so a view whose positions are marked to this bar while the price is
928 // still the previous bar's is not a fresher picture -- it is an incoherent one, and it is worse
929 // than a uniformly stale picture because it looks current. After a close moves from 10 to 100, a
930 // market buy from `on_phase` would be checked against buying power and exposure at 10 while the
931 // positions beside it say 100, which can admit an order those limits exist to refuse.
932 set_price(slot, event);
933
934 if (clock_.advance(event, trading_day)) {
935 runner_.context().set_phase(clock_.phase());
936 // **The clock moves with the price and the portfolio, for the same reason.**
937 // `Runner::feed(MarketPhase)` does not advance the context's time -- only `feed(BarEvent)`
938 // does, and that runs later -- so `on_phase` would decide with this bar's marks and close
939 // beside the *previous* bar's instant. `RiskGuard` records a send at `GuardInputs::now_` and
940 // counts sends within `rate_window_` of it, so across a gap longer than that window a phase
941 // order is filed under the old instant and the bar order that follows does not see it: two
942 // orders through a one-order window at what the replay calls the same moment.
943 runner_.context().set_now(event.period_end_);
944 // **Installed before the phase change is acted on, not after.** A strategy placing an order
945 // from `on_phase` used to decide on a portfolio predating this bar's fills and this bar's
946 // mark, while the engine already held a fresher answer and had simply not installed it yet.
947 install_portfolio();
948 drain_reports(runner_.feed(clock_.phase()));
949 } else {
950 runner_.context().set_phase(clock_.phase());
951 }
952
953 if constexpr (Runner::takes_bar_slices) {
954 // Cached and clocked now (priced above, so not again); the strategy sees it with the rest of its period.
955 auto& by_interval = slot.interval(event.interval_);
956#if defined(ATL_HAS_EIGEN)
957 using Access = runner::context_detail::PortfolioAccess;
958 forget_pointers_if_replaced();
959 if constexpr (Runner::records_bar_history) {
960 if (by_interval.series_ == nullptr) {
961 by_interval.series_ = &Access::series_for(runner_.context(), event);
962 }
963 by_interval.series_->push(event.bar_);
964 } else {
965 if (by_interval.bar_time_ == nullptr) {
966 by_interval.bar_time_ = &Access::bar_time_for(runner_.context(), event);
967 }
968 Access::advance_bar_time(*by_interval.bar_time_, event);
969 }
970#endif
971 runner_.context().set_now(event.period_end_);
972 pending_.add_checked(event);
973 by_interval.pending_ = event.period_end_;
974 } else {
975 install_portfolio();
976 drain_reports(runner_.feed(event));
977 }
978 // **After the dispatch, not before it**, so an order this bar's `on_bar` or `on_phase` placed is
979 // reported. Before it would report only what settlement had already changed.
980 publish_order_updates();
981 previous_ = std::make_pair(event.interval_, event.period_end_);
982 }

1001 void feed(const strategy::QuoteEvent& event, std::optional<bool> trading_day = true) {
1002 // The same rule `feed(const BarEvent&)` opens with: a halted replay consumes nothing further,
1003 // because a strategy told about anything after the halt decides on a run that has stopped being
1004 // able to cost what it asks for.
1005 if (result_.halted_) {
1006 pending_.clear();
1007 return;
1008 }
1009 if constexpr (Runner::takes_bar_slices) {
1010 // A quote at or after the pending period comes after it: the period is delivered while its own
1011 // price, clock and phase are still installed, before this quote changes any of them.
1012 const auto at = event.quote_.received_at_;
1013 if (!pending_.empty() && at < pending_.period_end()) {
1014 throw std::invalid_argument(
1015 "a quote earlier than bars already fed: with on_bars, feed quotes and bars in time order"
1016 );
1017 }
1018 if (pending_.empty() && delivered_through_ && at < *delivered_through_) {
1019 throw std::invalid_argument(
1020 "a quote earlier than a period already delivered: with on_bars, feed quotes and bars in time order"
1021 );
1022 }
1023 if (!pending_.empty()) {
1024 deliver_pending();
1025 if (result_.halted_) {
1026 return;
1027 }
1028 }
1029 }
1030 ++result_.quotes_;
1031 // Dated before anything can place an order, so a decision made here is not credited a price from
1032 // earlier in the bar it will fill against. See `QueuedOrder::decided_at_`.
1033 runner_.executor().deciding_at(event.quote_.received_at_);
1034
1035 // **The phase is re-derived at the quote's own instant, before the handler runs.** Without this
1036 // a quote at 10:30 was shown the phase as it stood at the previous close, and the default
1037 // `RiskLimits::permitted_phases_` refuses an opening order in every phase but `OPEN` -- so a
1038 // strategy that lives correctly refused in a replay. `PollingLoop` installs the phase before
1039 // each due poll, this is the same rule, and a change is announced first for the reason
1040 // `feed(const BarEvent&)` announces it first: a handler placing an order from `on_phase` must
1041 // decide against the phase this instant established, not the one before it.
1042 const bool phase_changed = clock_.advance_to(event.quote_.received_at_, trading_day);

1046 // both from the quote, but it runs *after* this, so announcing first left `on_phase` deciding
1047 // on the previous bar's close and stamping its send at the previous instant -- a market order
1048 // sized and guarded at the wrong value. This is the order
1049 // `feed(const BarEvent&)` already uses: mark, price, clock, announce, dispatch.
1050 runner_.context().set_last_price(event.quote_.instrument_, event.quote_.last_price_);

1053 drain_reports(runner_.feed(clock_.phase()));

1064 // working-order poll and announced. Publishing here is the closer parity, and it also

1230 // Returns whether `event`'s period end is enforced as the decision instant (`true`), or treated as a
1231 // nominal upper bound whose fills are allowed and counted as unchecked (`false`).
1232 //
1233 // A close-stamped bar carries its period end as its timestamp, so the broker stated it. An
1234 // open-stamped bar's end is derived by adding the interval's nominal length, which is right for an
1235 // intraday interval save a session's last bar when the session is not a whole number of them: KRX's
1236 // 15:00 hourly bar is given 16:00 for its 15:30 close. That end is late, not exact, and it is still
1237 // enforced strictly, because a late bound only defers a fill (to the next session's bars) and never
1238 // credits a price from before the decision; treating it as nominal would let such fills through
1239 // unchecked. For every calendar interval the end is an approximation -- 24 hours is not a trading day, 31 days
1240 // is not a month, and 366 is not a year. The approximation always runs **long**, which is what
1241 // makes it usable: an upper bound on the decision instant still proves causality whenever a bar's
1242 // start is at or after it.
1243 static bool period_end_is_exact(const strategy::BarEvent& event) noexcept {
1244 return event.stamp_ == BarStamp::CLOSE || bar_interval_length(event.interval_).has_value();

1247 // Returns when `event`'s period began, where that is answerable **from the event itself**.
1248 //
1249 // Three cases, in the order they are decided; the order matters:
1250 //
1251 // 1. **An open-stamped bar states it.** `BarStamp::OPEN` means `bar_.timestamp_` *is* the first
1252 // instant of the period, for every interval. Nothing has to be derived.
1253 // 2. **An intraday interval has a fixed length**, so a close-stamped one began that long before it
1254 // ended. A session's last bar cut short by the close began later than that (KRX's hourly bar ending
1255 // 15:30 began at 15:00, not 14:30): the derived start is early, which refuses fills the check could
1256 // have allowed and never allows one it should refuse.
1257 // 3. **A close-stamped daily bar** began at the session open on its own local date, which is the
1258 // date its period end falls on and where its open price comes from. **That assumes the stamp
1259 // falls on the session's own date**, which is true of a bar stamped at the close and false of
1260 // one stamped at the following local midnight; such a series gets a start a day late.
1261 //
1262 // **Everything else is unanswerable and says so.** A close-stamped weekly bar ends on Friday and
1263 // began on Monday; taking the session open on the *period end's* date would have called Friday
1264 // 09:00 its start, which is four days too late -- so a Tuesday quote would have passed the check
1265 // and filled at Monday's open, a price from two days before the decision. An earlier version of
1266 // this function did exactly that and called it conservative; it was the opposite. The same was
1267 // true of an open-stamped daily bar, whose period end is 24 hours after its start.
1268 //
1269 // What an empty answer means to the caller is in `may_fill`: not "no constraint", and not a bound
1270 // to be inferred from somewhere else either, but "this check cannot run -- allow the fill and
1271 // record that it was not checked".
1272 std::optional<std::chrono::system_clock::time_point> period_start_of(
1273 const strategy::BarEvent& event
1274 ) const {
1275 if (event.stamp_ == BarStamp::OPEN) {
1276 return event.bar_.timestamp_;
1277 }
1278 if (const auto duration = interval_duration(event.interval_)) {
1279 return event.period_end_ - *duration;
1280 }
1281 if (event.interval_ == BarInterval::ONE_DAY) {
1282 return clock_.session().opens_at(event.period_end_);
1283 }
1284 return std::nullopt;
1285 }

1287 // Returns whether `entry` may fill against `event`, and records where the question *could this be
1288 // credited a price from before it was decided* could not be answered at all.
1289 //
1290 // Three answers:
1291 //
1292 // 1. **An undated order is eligible.** That is every order placed by a caller driving
1293 // `QueueingExecutor` directly, and this rule exists to defer fills rather than to invent them.
1294 // **No test can reach this branch through `Backtester`**, because the engine stamps the instant
1295 // before every dispatch that can place -- so a mutation removing it is not caught, and that is
1296 // a property of the branch rather than a weakness in the tests. It is kept because
1297 // `QueuedOrder::decided_at_` is an optional on a public type, and an undated one is a state the
1298 // type admits; nothing outside this class calls this function, so no caller exercises it today.
1299 // 2. **Where `period_start_of` answers, the comparison is made**: the bar's period must have begun
1300 // **at or after** the decision. A period that began before it contains the decision, and its
1301 // open price -- which is what `NextBarFillModel` credits a market order with -- is from before
1302 // the strategy decided anything. This is every bar shape this library produces: intraday,
1303 // open-stamped, and close-stamped daily. A derived start can be early (a close-stamped intraday
1304 // bar cut short by the close), which defers a fill and never credits an earlier price.
1305 // 3. **Where either side of that comparison is a calendar boundary this engine cannot establish,
1306 // it makes no inference at all.** Two ways in, and the second was found last: the bar's start is
1307 // underivable (a close-stamped weekly, monthly or yearly bar), or the decision instant is only
1308 // an upper bound (an open-stamped bar of any calendar interval, which is what every shipped
1309 // adapter produces) and the bar's start falls before it. Either way the fill is allowed, on the
1310 // bar-counter rule that governed every fill before this check existed, and **counted** in
1311 // `ReplayResult::unchecked_fill_causality_` so the report can say the check did not happen.
1312 //
1313 // **Three successive versions of this function tried to infer that period instead, and all three
1314 // were wrong.** The previous bar of the instrument: wrong, because two intervals of one
1315 // instrument overlap. The previous bar of the same instrument *and* interval: wrong, because
1316 // nothing in `feed`'s contract requires one stamp convention or anchor for a series, so two
1317 // same-interval weekly bars can overlap too. Each inference rested on an invariant that is usually
1318 // true and is not guaranteed -- and the repository's own rule covers exactly this: *a conservative
1319 // refusal beats a narrow test you cannot justify.* There is no refusal available here either,
1320 // because deferring an unprovable fill would stop a close-stamped weekly replay filling **anything**
1321 // -- every order in one is decided at a boundary this engine equally cannot place. That is a
1322 // statement about the shape rather than about every interval in it: a sound lower bound of *period
1323 // end minus seven days* does exist for a week and would let a full-week series fill on time, but it
1324 // defers a week after every holiday-shortened one, and the 31-day and 366-day bounds defer nearly
1325 // every monthly and yearly order. So the honest third answer is to check nothing, claim nothing,
1326 // and report the gap.
1327 bool may_fill(
1328 const QueuedOrder<Tag>& entry,
1329 const strategy::BarEvent& event
1330 ) {
1331 if (!entry.decided_at_) {
1332 return true;
1333 }
1334 if (const auto start = period_start_of(event)) {
1335 if (!(*start < *entry.decided_at_)) {
1336 // Proven: the period began at or after the decision. A *nominal* decision instant is an
1337 // upper bound on the true one, so this proves it for those too.
1338 return true;
1339 }
1340 if (entry.decision_is_exact_) {
1341 return false;
1342 }
1343 // The start is known and the decision is only bounded, so this is the unprovable case again
1344 // from the other side: the order may have been decided before this period after all. Allowed
1345 // and counted, on the same terms as an underivable start.
1346 ++result_.unchecked_fill_causality_;
1347 return true;
1348 }
1349 ++result_.unchecked_fill_causality_;
1350 return true;
1351 }

1353 // Fills orders resting from an earlier bar against `event`, and tells a strategy that declares `on_fill`.
1354 //
1355 // **Walked by reference rather than by iterator or index.** Applying a fill calls `on_fill` when the
1356 // strategy declares it, which may place or withdraw orders, so any position into the queue is invalid the moment the strategy
1357 // is told anything. The eligible references are collected first and each is looked up again when
1358 // its turn comes; one the callback withdrew is simply no longer there.
1359 //
1360 // The queue is never emptied, so an order this loop has not reached yet is outstanding, visible in
1361 // every portfolio installed along the way, and addressable by a cancellation from any callback.
1362 void settle_queued_against(const strategy::BarEvent& event, InstrumentSlot& slot) {
1363 // With nothing queued nothing can expire, fill or be assigned a session; only the publication at the end
1364 // has work left (it releases deferred targets).
1365 if (runner_.executor().queued().empty()) {
1366 publish_order_updates();
1367 return;
1368 }
1369 const auto bar_date = clock_.session().local_date(event.bar_.timestamp_);
1370 // A session the replay had no closing bar for -- a missing last minute -- still ended: an order
1371 // of an earlier date is withdrawn before this bar can fill it.
1372 expire_day_orders(event, bar_date, false);
1373 std::vector<OrderRef<Tag>> eligible;
1374 std::vector<OrderRef<Tag>> first_session;
1375 // Only this instrument's orders, oldest first: a different instrument's order waits for that
1376 // instrument's next bar, which is what makes an interleaved multi-instrument replay behave. An order
1377 // accepted during this bar waits for the next one, which is the no-lookahead rule.
1378 runner_.executor().for_each_queued_in(event.instrument_, [&](const QueuedOrder<Tag>& entry) {
1379 if (entry.accepted_on_ >= runner_.executor().bar()) {
1380 return;
1381 }
1382 // **And, where the event says enough to check it, it cannot fill against a bar whose period
1383 // had already begun when it was decided.** The counter above is enough while only bars place
1384 // orders and no two periods overlap; a quote arrives inside a period without opening one, so
1385 // without this an order decided at 10:30 fills at that day's 09:00 open -- a price from before
1386 // the decision. Where the event does *not* say enough, `may_fill` allows the fill and records
1387 // that nothing was checked. See it for all three answers.
1388 if (!may_fill(entry, event)) {
1389 return;
1390 }
1391 eligible.push_back(entry.ref_);
1392 // **Which session a DAY order lives in**, fixed at the first bar it is eligible against: that
1393 // bar's session is the one the order trades in. A decision the bar cannot fill without looking
1394 // ahead -- a quote at 10:00 against the day's bar, or the day's close -- trades in the next
1395 // session, as a market order decided there fills at the next open: one session, shifted, never
1396 // the several a resting order had before. Recorded here, beside the one `may_fill` call, so
1397 // `unchecked_fill_causality_` still counts each order once per bar.
1398 if (!entry.session_ && time_in_force_expires_at_close(entry.order_.time_in_force_)) {
1399 first_session.push_back(entry.ref_);
1400 }
1401 });
1402 for (const auto& ref : first_session) {
1403 runner_.executor().assign_session(ref, bar_date);
1404 }
1405
1406 for (const auto& ref : eligible) {
1407 const auto* entry = runner_.executor().find(ref);
1408 if (entry == nullptr) {
1409 // Withdrawn by a callback earlier in this loop. Nothing to fill.
1410 continue;
1411 }
1412 const auto order = entry->order_;
1413 const auto& rule = rule_for(order.instrument_);
1414 // An `OPEN`-stamped bar states its start, so a fill at its open has a known day; the period
1415 // end's day is nominal there -- a day late for a daily bar, days into the next month for a
1416 // monthly one. See `NextBarFillModel::decide`'s `open_date`.
1417 const auto open_date = event.stamp_ == BarStamp::OPEN
1418 ? std::optional<Date>(clock_.session().local_date(event.bar_.timestamp_))
1419 : std::nullopt;
1420 auto decision = fills_.decide(
1421 order, event, rule, clock_.session().local_date(event.period_end_), open_date,
1422 entry->released_
1423 );
1424 if (decision.released_) {
1425 // Recorded before anything below can withdraw the order, and whether or not it filled: a stop
1426 // that released and still rests is a limit from the next bar on.
1427 runner_.executor().release(ref);
1428 ++result_.stop_releases_;
1429 }
1430 if (decision.costing_error_) {
1431 // **Checked before the generic no-execution branch**, because that branch would leave a
1432 // resting day limit queued to be retried on the next bar, and a refusal is not a bar that
1433 // failed to reach a price. The order goes whatever its type, exactly once.
1434 runner_.executor().withdraw(ref);
1435 result_.halted_ = std::move(*decision.costing_error_);
1436 // **The halt governs the engine, not just this order.** Returning here stops the other
1437 // orders eligible on this same bar from filling behind a cost the model could not compute,
1438 // and `feed` stops consuming bars from its next call onward.
1439 abort_queue();
1440 return;
1441 }
1442 if (!decision.execution_) {
1443 if (!decision.still_working_) {
1444 // A market order the bar could not price, or a limit whose arithmetic did not fit. It does
1445 // not rest and it did not trade, so it stops being outstanding.
1446 runner_.executor().withdraw(ref);
1447 }
1448 continue;
1449 }
1450 // Removed before the ledger moves and before anything is shown a portfolio, so no view ever
1451 // carries a filled order as still working.
1452 runner_.executor().withdraw(ref);
1453 // Stamped from the same reference data the ledger is given, so the two can never disagree
1454 // about what currency one fill was in.
1455 decision.execution_->currency_ = currency_for(order.instrument_);
1456 if (!ledger_.apply(*decision.execution_, currency_for(order.instrument_))) {
1457 // **A modelled fill the ledger cannot record halts the replay**, for the same reason a cost
1458 // it cannot compute does. `apply` is atomic, so nothing is half-written -- but the fill
1459 // happened in the model and did not happen in the books, and continuing would carry on from
1460 // a portfolio missing a trade the strategy was never told about. This used to `continue`,
1461 // which recreated at the ledger boundary exactly the silently-missing-trade path the
1462 // costing refusal above exists to remove.
1463 BrokerError error;
1464 error.category_ = ErrorCategory::INVALID_ARGUMENT;
1465 error.reached_broker_ = false;
1466 error.broker_code_ = "execution_not_representable";
1467 error.message_ = "an execution's cash arithmetic does not fit an exact decimal, so the "
1468 "ledger could not record a fill the model produced";
1469 result_.halted_ = std::move(error);
1470 abort_queue();
1471 return;
1472 }
1473 // **Where the lot is, learnt as soon as it exists.** A fill can only open a lot in this bar's own
1474 // instrument, and it is recorded here, before anything below (recording, `on_fill`) can throw, so a
1475 // handler's exception caught by the caller cannot leave a held lot that is never marked again.
1476 if (!slot.lot_) {
1477 slot.lot_ = ledger_.lot_of(order.instrument_);
1478 }
1479 note_overdraft(*decision.execution_);
1480 if (decision.intrabar_release_fill_) {
1481 ++result_.stop_fills_intrabar_;
1482 }
1483 if (decision.bounded_) {
1484 ++result_.limit_fills_bounded_;
1485 }
1486 // Moved, not copied: the result is the only record of a replay's executions. Nothing below adds to
1487 // `result_.executions_` before `on_fill` returns, so the reference stays valid.
1488 result_.executions_.push_back(std::move(*decision.execution_));
1489 if constexpr (strategy::strategy_detail::declares_on_fill<Strategy>) {
1490 const auto& recorded = result_.executions_.back();
1491 // **No handler, nothing to show.** For a strategy without `on_fill` the install would be overwritten
1492 // unread (every later dispatch, and `settle_deferred`, installs first) and the pass would settle an
1493 // empty instruction list. Skipped at compile time: per fill, the install alone costs a row per queued
1494 // order and a copy of every position.
1495 install_portfolio();
1496 drain_reports(runner_.feed(recorded.fill_));
1497 }
1498 }

1501 // an `on_fill` handler placed or cancelled in response to one.
1502 publish_order_updates();

1599 // Records a point, and the return since the previous point.
1600 void record_point(std::chrono::system_clock::time_point at, const Decimal64& equity) {
1601 EquityPoint point;
1602 point.at_ = at;
1603 point.equity_ = equity;
1604 point.period_return_ = math::missing();
1605 // **No return across an omitted point.** The one below would measure two or more periods and be
1606 // handed to the Sharpe ratio and the annualization as one period's return, so the first point
1607 // after a gap carries none -- as the first point of the run does.
1608 const bool after_gap = equity_gap_;
1609 equity_gap_ = false;
1610 if (!result_.equity_.empty() && !after_gap) {
1611 const auto& previous = result_.equity_.back().equity_;
1612 // Only from positive equity. From a negative total the ratio's sign is reversed: a further loss
1613 // reads as a gain, and a run that lost more than its capital reported a positive Sharpe ratio.
1614 if (previous.coefficient() > 0) {
1615 const auto moved = decimal_subtract(equity, previous);
1616 if (moved) {

1618 if (ratio) {
1619 point.period_return_ = analytics_to_scalar(*ratio);
1620 }
1621 }
1622 }
1623 }

1627 // Converts an exact ratio to the scalar the metrics compute in.
1628 //
1629 // The one crossing of the decimal boundary in this engine, and it goes outward: a return is an input
1630 // to `sharpe` and the rest, which are defined on `Scalar`. Nothing converts back, because no figure
1631 // a report produces is ever sent to a broker. The same conversion as `analytics::to_scalar` (one
1632 // division by an exact power of ten, so the nearest double within its bounds), written out because
1633 // this header does not include the analytics layer; keep the two alike.
1634 static math::Scalar analytics_to_scalar(const Decimal64& value) {
1635 math::Scalar divisor{1};
1636 for (std::uint8_t at = 0; at < value.scale(); ++at) {
1637 divisor *= math::Scalar{10};
1638 }
1639 return static_cast<math::Scalar>(value.coefficient()) / divisor;

1642 // Returns the grid for `instrument`, unknown when no reference data was supplied.
1643 // Returns the currency `instrument` trades in, from the reference data the replay was given.
1644 //
1645 // **Every fill used to be booked in `config_.currency_`**, which is documented as the currency the
1646 // *starting cash* is in. A replay of a dollar-quoted instrument therefore banked dollars as won at
1647 // a rate of one -- an implicit currency conversion, which this library never makes silently.
1648 //
1649 // The fallback is defensive rather than a policy: `reconcile` refuses an instrument with no
1650 // reference data before a `PlannedOrder` exists (`strategy/reconcile.hpp`), so nothing can reach
1651 // the fill model unregistered. It is written the conservative way round anyway -- an unregistered
1652 // instrument books in the currency the ledger was opened in, which is the behaviour every existing
1653 // replay already had.
1654 // Installs the portfolio a strategy and its guards decide on.
1655 //
1656 // The single place the replay answers "what does this account look like", so that every callback
1657 // boundary is answered the same way.
1658 //
1659 // **It has to be a complete answer, because it overwrites one.**
1660 // `Runner::persist_reservation` writes an accepted order into the context portfolio and relies on
1661 // a wholesale replacement carrying what the reservation was standing in for
1662 // (`Runner::persist_reservation`'s comment). Live that replacement is a broker's listing. Here it is this
1663 // function, and a partial answer is therefore not a smaller improvement -- it is a *loss*, because
1664 // the reservation it erased was carrying the part it left out.
1665 //
1666 // The reservation carried two things, so this carries both: the outstanding orders themselves,
1667 // and the money an outstanding buy has committed.
1668 // Reports every queued order whose state has changed since the last observation.
1669 //
1670 // ## Where this is called from, and why not simply beside `install_portfolio`
1671 //
1672 // At the three points where a *batch* of changes to the queue has finished: the end of
1673 // settlement, the end of `feed`, and the end of `finish`. The queue changes in exactly three
1674 // ways -- `place` and `cancel` from a strategy dispatch, and `withdraw` from settlement -- and
1675 // each of those three boundaries closes one batch of them.
1676 //
1677 // The obvious alternative, publishing from `install_portfolio`, is wrong and quietly so: the
1678 // last install of a bar runs *before* `feed(BarEvent)` dispatches `on_bar`, so an order placed
1679 // there enters the queue after it, and the next bar's settlement withdraws the order before that
1680 // bar's first install. An order placed on one bar and filled on the next -- which is every market
1681 // order -- would be present at no install at all and would be reported nowhere.
1682 //
1683 // ## The portfolio is installed before each dispatch, not once for the batch
1684 //
1685 // A handler told about an order has to be able to find it. With every optional limit disabled
1686 // `Runner::persist_reservation` writes no working row, so nothing else would put the order in
1687 // the view, and a strategy would be told about an order it could neither see nor withdraw.
1688 // Installing once for the batch is not enough either, because a handler answering the first
1689 // update may place or cancel, which invalidates the picture the second one reads.
1690 //
1691 // ## What a handler's own order does
1692 //
1693 // It enters the queue behind this observation and is reported at the *next* boundary, never
1694 // within this one, so a strategy cannot drive itself round a loop here. An order placed by
1695 // `on_order_update` during `finish` has no later boundary and is not reported at all.
1696 void publish_order_updates() {
1697 if (result_.halted_) {
1698 // Nothing runs after a refusal, for the reason `feed` documents: a handler called here would
1699 // decide on a replay that has stopped being able to cost what it asks for.
1700 return;
1701 }
1702 if constexpr (!strategy::strategy_detail::declares_on_order_update<Strategy>) {
1703 // **No handler, no updates.** An update would reach no code: dispatching it is a no-op and the
1704 // settlement after it has no instruction to settle. The rows, the comparison and the installs
1705 // around it are skipped; the release of deferred targets is not, because it answers the passes
1706 // before this point, not the updates.
1707 settle_deferred();
1708 return;
1709 } else {
1710 publish_to_handler();
1711 }
1712 }

1714 // The part of `publish_order_updates` a strategy declaring `on_order_update` needs.
1715 void publish_to_handler() {
1716 // Nothing queued and nothing tracked: no order can have changed.
1717 if (runner_.executor().queued().empty() && updates_.tracked_orders() == 0) {
1718 settle_deferred();
1719 return;
1720 }
1721 std::vector<WorkingOrder<Tag>> rows;
1722 rows.reserve(runner_.executor().queued().size());
1723 for (const auto& entry : runner_.executor().queued()) {
1724 rows.push_back(working_row(entry));
1725 }
1726 for (const auto& update : updates_.changed(rows)) {
1727 // **Re-checked against the queue immediately before it is dispatched, not only when the batch
1728 // was observed.** A handler answering the first update of a batch can cancel a *different*
1729 // order in the same batch, and the event captured for that order is then a statement about an
1730 // order that no longer exists -- the portfolio installed below would correctly omit it while
1731 // the event announced it, which is exactly the contradiction the install exists to prevent.
1732 //
1733 // Checked on the state as well as the reference, because a handler can also replace an order
1734 // with another carrying the same one.
1735 const auto* still_queued = find_queued(update);
1736 if (still_queued == nullptr) {
1737 continue;
1738 }
1739 install_portfolio();
1740 drain_reports(runner_.feed(update));
1741 if (result_.halted_) {
1742 return;
1743 }
1744 }
1745 // **Once, after the whole batch.** Every pass that can withdraw a stop -- the bar's, the phase's,
1746 // the fills' and these updates' -- ends here, so the strategy has answered every update before a
1747 // fallback is released; releasing after each pass let a later update's instruction arrive after
1748 // the fallback had already been queued. The live path waits for a whole listing the same way.
1749 settle_deferred();
1750 }

1806 void install_portfolio() {
1807 // The value a weight is sized against: the ledger's equity in the replay's currency at the
1808 // latest marks, which is what the equity curve records. Cleared when that total does not fit,
1809 // so a weight is refused on such a bar rather than sized against the previous one.
1810 if (const auto equity = ledger_.equity()) {
1811 runner_.set_portfolio_value(Money{*equity, config_.currency_});
1812 } else {
1813 runner_.clear_portfolio_value();
1814 }
1815 using Access = runner::context_detail::PortfolioAccess;
1816 auto& context = runner_.context();
1817 const auto& queued = runner_.executor().queued();
1818 // **Built whole once, then kept up to date in place**, so a callback costs what changed rather than a
1819 // copy of every position. Afterwards the portfolio is what a fresh snapshot with a row per queued order
1820 // would be. Built whole on the first install and after anything replaced the portfolio outright
1821 // (`Context::set_portfolio`); otherwise the ledger rewrites the lots it changed and the cash, and the
1822 // working rows are rebuilt when the queue gained or lost an order, a runner edited them in place
1823 // (`Runner::persist_reservation`), or reference data (a row's currency) changed. What was seen is
1824 // acknowledged only once everything is in, so the next install finishes what one that threw left undone.
1825 if (!installed_ || Access::replacements(context) != replacements_seen_) {
1826 auto portfolio = ledger_.snapshot<Tag>(clock_.now());
1827 portfolio.working_.reserve(queued.size());
1828 for (const auto& entry : queued) {
1829 portfolio.working_.push_back(working_row(entry));
1830 commit_open_buy(portfolio, entry);
1831 }
1832 context.set_portfolio(std::move(portfolio));
1833 ledger_.clear_changed();
1834 } else {
1835 // A runner's reservation follows an order the queue accepted, so today the queue's count moves with it;
1836 // the runner's own count is checked too, so a reservation written without a new order is not kept.
1837 const bool rows_stale = runner_.executor().changes() != queue_seen_ ||
1838 Access::runner_edits(context) != edits_seen_ ||
1839 Access::reference_changes(context) != references_seen_;
1840 auto& portfolio = Access::for_replay_sync(context);
1841 ledger_.update(portfolio, clock_.now());
1842 if (rows_stale) {
1843 portfolio.working_.clear();
1844 for (const auto& entry : queued) {
1845 portfolio.working_.push_back(working_row(entry));
1846 }
1847 }
1848 // `update` set the buying power back to the cash, so every open buy is taken out again; a market
1849 // buy is valued at the last price, which moves.
1850 if (config_.limits_.enforce_buying_power_) {
1851 for (const auto& entry : queued) {
1852 commit_open_buy(portfolio, entry);
1853 }
1854 }
1855 }
1856 installed_ = true;
1857 replacements_seen_ = Access::replacements(context);
1858 edits_seen_ = Access::runner_edits(context);
1859 references_seen_ = Access::reference_changes(context);
1860 queue_seen_ = runner_.executor().changes();
1861#if defined(ATL_REPLAY_CHECK_PORTFOLIO)
1862 // A test build's check (the tests define the macro as their comparison): the portfolio as kept against
1863 // the one the replay used to build from scratch, and the running equity against the full sum.
1864 {
1865 auto fresh = ledger_.snapshot<Tag>(clock_.now());
1866 for (const auto& entry : queued) {
1867 fresh.working_.push_back(working_row(entry));
1868 commit_open_buy(fresh, entry);
1869 }
1870 ATL_REPLAY_CHECK_PORTFOLIO(context.portfolio(), fresh, ledger_.equity(), ledger_.equity_in(config_.currency_));
1871 }
1872#endif
1873 }

1875 // Returns the working-order row for an outstanding replay order.
1876 //
1877 // `reserved_` stays false, unlike a runner's own reservation. The flag distinguishes this
1878 // library's arithmetic from what a broker reported, and in a replay the engine *is* the broker:
1879 // this row is the listing. The one consequence worth stating is that
1880 // `strategy::SellableConventionProbe`, which reads unreserved rows as evidence about a broker's
1881 // convention, will read these -- and a replay's sellable figure is a model that never moves, so
1882 // the probe reports what the simulator does rather than what a broker does.

1889 // stable for as long as the order is outstanding. Nothing in the library matched a replay

1892 row.order_.instrument_ = entry.order_.instrument_;
1893 row.order_.side_ = entry.order_.side_;
1894 row.order_.type_ = entry.order_.type_;
1895 row.order_.time_in_force_ = entry.order_.time_in_force_;
1896 row.order_.session_ = entry.order_.session_;
1897 row.order_.venue_ = entry.order_.venue_;
1898 row.order_.price_ = entry.order_.price_;

1912 // Takes what an outstanding buy has committed out of the buying power the replay models.
1913 //
1914 // **Rows alone do not fix `enforce_buying_power_`**, which reads the scalar and not the orders.
1915 // `ReplayLedger::snapshot` builds buying power from ledger cash, and that cash has not paid for
1916 // anything still resting -- so with a hundred of cash, a resting buy costing a hundred, and a
1917 // second buy costing a hundred in another instrument, the second passes and both can still fill.
1918 //
1919 // Valued at the order's own price, else the last close, which is the same estimate the guard
1920 // checked the order against — and it is an **estimate**, not a bound. A limit buy cannot fill
1921 // above its limit, so for those the commitment is at least the eventual spend before costs. A
1922 // market order has no stated price at all: it fills where the next bar opens, and slippage and
1923 // commission are both outside this figure, so the real spend can exceed what was committed. An
1924 // order that can be valued by neither route is left uncommitted rather than guessed at.
1925 //
1926 // **Scoped to the limit, exactly as `Runner::reserve` is.**
1927 // `buying_power_in()` is a public figure a strategy is free to size its next order from, so
1928 // lowering it for a caller who never enabled the check would change how their strategy sizes --
1929 // a behaviour change well past enforcing a ceiling, and the reason the runner passes `spends`
1930 // rather than always debiting. It also keeps `ReportCaveats::buying_power_is_ledger_cash` true
1931 // for every replay that does not ask for the check.
1932 void commit_open_buy(strategy::Portfolio<Tag>& portfolio, const QueuedOrder<Tag>& entry) const {
1933 if (!config_.limits_.enforce_buying_power_ || entry.order_.side_ != OrderSide::BUY) {
1934 return;
1935 }
1936 const auto price = entry.order_.price_ ?
1937 entry.order_.price_ :
1938 runner_.context().last_price(entry.order_.instrument_);
1939 if (!price) {
1940 return;
1941 }
1942 const auto spent = decimal_multiply(entry.order_.quantity_, *price);
1943 if (!spent) {
1944 return;
1945 }
1946 const auto currency = currency_for(entry.order_.instrument_);
1947 for (auto& reading : portfolio.buying_power_) {
1948 if (!(reading.cash_.currency_ == currency)) {
1949 continue;
1950 }
1951 if (const auto left = decimal_subtract(reading.cash_.amount_, *spent)) {
1952 reading.cash_.amount_ = *left;
1953 }
1954 return;
1955 }
1956 }
