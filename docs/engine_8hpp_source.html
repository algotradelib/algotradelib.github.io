474 void feed(const strategy::BarEvent& event, std::optional<bool> trading_day = true) {
475 // **A halted replay consumes nothing further.** `feed` keeps returning `void` so no caller has to
476 // change, and a caller that ignores `ReplayResult` sees a run that ends early rather than one
477 // that carries on costing nothing: no bar counted, no fill, no mark, no equity point, and no
478 // callback into the strategy. That is API compatibility, which is not the same as invisibility,
479 // and it is why the halt is stated on the result and first in the report.
480 if (result_.halted_) {
481 return;
482 }
483 ++result_.bars_;
484 if (previous_ && !ReplayClock::contiguous(*previous_, event)) {
485 // A gap is counted rather than interpolated across: nothing happened in it that a strategy
486 // should be told about, and inventing a bar would be inventing data.
487 ++result_.gaps_;
488 }
489
490 // Opened before anything fills against this bar, so that an order placed during it carries a
491 // later stamp than the ones already resting and cannot fill against the bar it was decided on.
492 runner_.executor().open_bar();
493 // And dated, for the half of that rule a counter cannot express -- see `QueuedOrder::decided_at_`.
494 // A bar's decisions are all made at its period end, whichever handler makes them: `on_fill`
495 // during settlement, `on_phase`, `on_bar`, and a handler answering an order update afterwards.
496 //
497 // **Whether that instant is exact depends on the stamp**, and getting this wrong deferred real
498 // fills by a month. A close-stamped bar's period end is its own timestamp, stated by the broker.
499 // An open-stamped one is `bar_period_end`'s arithmetic, which is exact for a minute and
500 // deliberately **nominal** for a day, a week, a month and a year -- and every shipped adapter
501 // stamps every non-minute bar `OPEN`. February's nominal end is five days into March, so an
502 // order decided on February's bar looked as though it had been decided inside March's period and
503 // was deferred to April.
504 runner_.executor().deciding_at(event.period_end_, period_end_is_exact(event));
505
506 settle_queued_against(event);
507 // **The halt stops this bar too, not merely the next one.** Returning only from settlement left
508 // `feed` to mark the book, record an equity point, advance the clock and call `on_phase` and
509 // `on_bar` on the very bar whose fill could not be costed -- so the strategy decided once more
510 // on a portfolio missing that trade, and could queue orders that then survived in the executor
511 // because every later `feed` is a no-op and `finish` returns early. Nothing runs after the
512 // refusal.
513 if (result_.halted_) {
514 return;
515 }
516 // **A mark that will not fit halts the replay**, like a fill it cannot cost or record. The lot
517 // stays at its previous price, while `Runner::feed` is about to hand the strategy this bar's
518 // close as the last price -- so `on_bar` would decide, and `Runner::settle` would size and check
519 // an order, against a position valued one bar ago beside a price from this one. An order placed
520 // from that view fills on a later bar and changes every figure after it. This used to skip only
521 // the equity point and carry on, which reported nothing and let that order through.
522 if (!ledger_.mark(event.instrument_, event.bar_.close_)) {
523 BrokerError error;
524 error.category_ = ErrorCategory::INVALID_ARGUMENT;
525 error.reached_broker_ = false;
526 error.broker_code_ = "mark_not_representable";
527 error.message_ = "a position in " + event.instrument_.symbol_ + " could not be valued at the "
528 "bar's close without leaving an exact decimal, so the replay stopped rather than let a "
529 "strategy decide on a position one bar behind its price";
530 result_.halted_ = std::move(error);
531 abort_queue();
532 return;
533 }
534 record_equity(event);
535
536 // **The price moves with the portfolio, not after it.** `Runner::settle` values a market order
537 // at `context_.last_price`, so a view whose positions are marked to this bar while the price is
538 // still the previous bar's is not a fresher picture -- it is an incoherent one, and it is worse
539 // than a uniformly stale picture because it looks current. After a close moves from 10 to 100, a
540 // market buy from `on_phase` would be checked against buying power and exposure at 10 while the
541 // positions beside it say 100, which can admit an order those limits exist to refuse.
542 runner_.context().set_last_price(event.instrument_, event.bar_.close_);
543
544 if (clock_.advance(event, trading_day)) {
545 runner_.context().set_phase(clock_.phase());
546 // **The clock moves with the price and the portfolio, for the same reason.**
547 // `Runner::feed(MarketPhase)` does not advance the context's time -- only `feed(BarEvent)`
548 // does, and that runs later -- so `on_phase` would decide with this bar's marks and close
549 // beside the *previous* bar's instant. `RiskGuard` records a send at `GuardInputs::now_` and
550 // counts sends within `rate_window_` of it, so across a gap longer than that window a phase
551 // order is filed under the old instant and the bar order that follows does not see it: two
552 // orders through a one-order window at what the replay calls the same moment.
553 runner_.context().set_now(event.period_end_);
554 // **Installed before the phase change is acted on, not after.** A strategy placing an order
555 // from `on_phase` used to decide on a portfolio predating this bar's fills and this bar's
556 // mark, while the engine already held a fresher answer and had simply not installed it yet.
557 install_portfolio();
558 drain_reports(runner_.feed(clock_.phase()));
559 } else {
560 runner_.context().set_phase(clock_.phase());
561 }
562
563 install_portfolio();
564 drain_reports(runner_.feed(event));
565 // **After the dispatch, not before it**, so an order this bar's `on_bar` or `on_phase` placed is
566 // reported. Before it would report only what settlement had already changed.
567 publish_order_updates();
568 previous_ = event;
569 }

585 void feed(const strategy::QuoteEvent& event, std::optional<bool> trading_day = true) {
586 // The same rule `feed(const BarEvent&)` opens with: a halted replay consumes nothing further,
587 // because a strategy told about anything after the halt decides on a run that has stopped being
588 // able to cost what it asks for.
589 if (result_.halted_) {
590 return;
591 }
592 ++result_.quotes_;
593 // Dated before anything can place an order, so a decision made here is not credited a price from
594 // earlier in the bar it will fill against. See `QueuedOrder::decided_at_`.
595 runner_.executor().deciding_at(event.quote_.received_at_);
596
597 // **The phase is re-derived at the quote's own instant, before the handler runs.** Without this
598 // a quote at 10:30 was shown the phase as it stood at the previous close, and the default
599 // `RiskLimits::permitted_phases_` refuses an opening order in every phase but `OPEN` -- so a
600 // strategy that lives correctly refused in a replay. `PollingLoop` installs the phase before
601 // each due poll, this is the same rule, and a change is announced first for the reason
602 // `feed(const BarEvent&)` announces it first: a handler placing an order from `on_phase` must
603 // decide against the phase this instant established, not the one before it.
604 const bool phase_changed = clock_.advance_to(event.quote_.received_at_, trading_day);
605 runner_.context().set_phase(clock_.phase());
606 if (phase_changed) {
607 // **The price and the clock move first, then the change is announced.** `Runner::feed` sets
608 // both from the quote, but it runs *after* this, so announcing first left `on_phase` deciding
609 // on the previous bar's close and stamping its send at the previous instant -- a market order
610 // sized and guarded at the wrong value. This is the order
611 // `feed(const BarEvent&)` already uses: mark, price, clock, announce, dispatch.
612 runner_.context().set_last_price(event.quote_.instrument_, event.quote_.last_price_);
613 runner_.context().set_now(event.quote_.received_at_);
614 install_portfolio();
615 drain_reports(runner_.feed(clock_.phase()));
616 }
617
618 // Installed before the dispatch for the reason every other dispatch site here installs: a
619 // handler reads `ctx.portfolio()`, and the engine holds a fresher answer than the last one it
620 // pushed -- an order placed by the *previous* quote has had no settlement pass behind it.
621 install_portfolio();
622 drain_reports(runner_.feed(event));
623 // **After the dispatch, exactly as the bar path does.** An earlier version left
624 // it out, arguing that live a quote poll publishes no order update -- which is true of that one
625 // call and false of the loop it sits in: an order placed from `on_quote` can be listed by a later
626 // working-order poll and announced. Publishing here is the closer parity, and it also
627 // announces an order that would otherwise fill and be gone before any later observation point.
628 publish_order_updates();
629 }

637 void finish() {
638 if (result_.halted_) {
639 // **The strategy is not stopped normally, and that is deliberate.** `Runner::stop` calls
640 // `on_stop`, which may place or flatten -- and an order placed into a replay that has already
641 // stopped being able to cost one would either fill uncosted or be silently dropped. Neither is
642 // a shutdown a reader could interpret, so nothing runs: the queue was cleared at the halt and
643 // `aborted_` says how much was in it. The report states that the strategy never saw a stop.
644 return;
645 }
646 install_portfolio();
647 drain_reports(runner_.stop());
648 // The only boundary that can observe what `on_stop` placed or withdrew. Before `unfilled_` is
649 // counted, so the count describes the queue the run actually ended with.
650 publish_order_updates();
651 result_.unfilled_ = runner_.executor().queued().size();
652 }

673 // Returns whether `event`'s period end is the instant itself rather than an upper bound on it.
674 //
675 // A close-stamped bar carries its period end as its timestamp, so the broker stated it. An
676 // open-stamped bar's end is derived by adding the interval's nominal length, which is right for a
677 // minute and an approximation for every calendar interval -- 24 hours is not a trading day, 31 days
678 // is not a month, and 366 is not a year. The approximation always runs **long**, which is what
679 // makes it usable: an upper bound on the decision instant still proves causality whenever a bar's
680 // start is at or after it.
681 static bool period_end_is_exact(const strategy::BarEvent& event) noexcept {
682 return event.stamp_ == BarStamp::CLOSE || event.interval_ == BarInterval::ONE_MINUTE;
683 }

685 // Returns when `event`'s period began, where that is answerable **from the event itself**.
686 //
687 // Three cases, in the order they are decided; the order matters:
688 //
689 // 1. **An open-stamped bar states it.** `BarStamp::OPEN` means `bar_.timestamp_` *is* the first
690 // instant of the period, for every interval. Nothing has to be derived.
691 // 2. **An intraday interval has a fixed length**, so a close-stamped one began that long before it
692 // ended.
693 // 3. **A close-stamped daily bar** began at the session open on its own local date, which is the
694 // date its period end falls on and where its open price comes from. **That assumes the stamp
695 // falls on the session's own date**, which is true of a bar stamped at the close and false of
696 // one stamped at the following local midnight; such a series gets a start a day late.
697 //
698 // **Everything else is unanswerable and says so.** A close-stamped weekly bar ends on Friday and
699 // began on Monday; taking the session open on the *period end's* date would have called Friday
700 // 09:00 its start, which is four days too late -- so a Tuesday quote would have passed the check
701 // and filled at Monday's open, a price from two days before the decision. An earlier version of
702 // this function did exactly that and called it conservative; it was the opposite. The same was
703 // true of an open-stamped daily bar, whose period end is 24 hours after its start.
704 //
705 // What an empty answer means to the caller is in `may_fill`: not "no constraint", and not a bound
706 // to be inferred from somewhere else either, but "this check cannot run -- allow the fill and
707 // record that it was not checked".
708 std::optional<std::chrono::system_clock::time_point> period_start_of(
709 const strategy::BarEvent& event
710 ) const {
711 if (event.stamp_ == BarStamp::OPEN) {
712 return event.bar_.timestamp_;
713 }
714 if (const auto duration = interval_duration(event.interval_)) {
715 return event.period_end_ - *duration;
716 }
717 if (event.interval_ == BarInterval::ONE_DAY) {
718 return clock_.session().opens_at(event.period_end_);
719 }
720 return std::nullopt;
721 }

723 // Returns whether `entry` may fill against `event`, and records where the question *could this be
724 // credited a price from before it was decided* could not be answered at all.
725 //
726 // Three answers:
727 //
728 // 1. **An undated order is eligible.** That is every order placed by a caller driving
729 // `QueueingExecutor` directly, and this rule exists to defer fills rather than to invent them.
730 // **No test can reach this branch through `Backtester`**, because the engine stamps the instant
731 // before every dispatch that can place -- so a mutation removing it is not caught, and that is
732 // a property of the branch rather than a weakness in the tests. It is kept because
733 // `QueuedOrder::decided_at_` is an optional on a public type, and an undated one is a state the
734 // type admits; nothing outside this class calls this function, so no caller exercises it today.
735 // 2. **Where `period_start_of` answers, the comparison is exact**: the bar's period must have begun
736 // **at or after** the decision. A period that began before it contains the decision, and its
737 // open price -- which is what `NextBarFillModel` credits a market order with -- is from before
738 // the strategy decided anything. This is every bar shape this library produces: intraday,
739 // open-stamped, and close-stamped daily.
740 // 3. **Where either side of that comparison is a calendar boundary this engine cannot establish,
741 // it makes no inference at all.** Two ways in, and the second was found last: the bar's start is
742 // underivable (a close-stamped weekly, monthly or yearly bar), or the decision instant is only
743 // an upper bound (an open-stamped bar of any calendar interval, which is what every shipped
744 // adapter produces) and the bar's start falls before it. Either way the fill is allowed, on the
745 // bar-counter rule that governed every fill before this check existed, and **counted** in
746 // `ReplayResult::unchecked_fill_causality_` so the report can say the check did not happen.
747 //
748 // **Three successive versions of this function tried to infer that period instead, and all three
749 // were wrong.** The previous bar of the instrument: wrong, because two intervals of one
750 // instrument overlap. The previous bar of the same instrument *and* interval: wrong, because
751 // nothing in `feed`'s contract requires one stamp convention or anchor for a series, so two
752 // same-interval weekly bars can overlap too. Each inference rested on an invariant that is usually
753 // true and is not guaranteed -- and the repository's own rule covers exactly this: *a conservative
754 // refusal beats a narrow test you cannot justify.* There is no refusal available here either,
755 // because deferring an unprovable fill would stop a close-stamped weekly replay filling **anything**
756 // -- every order in one is decided at a boundary this engine equally cannot place. That is a
757 // statement about the shape rather than about every interval in it: a sound lower bound of *period
758 // end minus seven days* does exist for a week and would let a full-week series fill on time, but it
759 // defers a week after every holiday-shortened one, and the 31-day and 366-day bounds defer nearly
760 // every monthly and yearly order. So the honest third answer is to check nothing, claim nothing,
761 // and report the gap.
762 bool may_fill(
763 const QueuedOrder<Tag>& entry,
764 const strategy::BarEvent& event
765 ) {
766 if (!entry.decided_at_) {
767 return true;
768 }
769 if (const auto start = period_start_of(event)) {
770 if (!(*start < *entry.decided_at_)) {
771 // Proven: the period began at or after the decision. A *nominal* decision instant is an
772 // upper bound on the true one, so this proves it for those too.
773 return true;
774 }
775 if (entry.decision_is_exact_) {
776 return false;
777 }
778 // The start is known and the decision is only bounded, so this is the unprovable case again
779 // from the other side: the order may have been decided before this period after all. Allowed
780 // and counted, on the same terms as an underivable start.
781 ++result_.unchecked_fill_causality_;
782 return true;
783 }
784 ++result_.unchecked_fill_causality_;
785 return true;
786 }

788 // Fills orders resting from an earlier bar against `event`, and tells the strategy.
789 //
790 // **Walked by reference rather than by iterator or index.** Applying a fill calls `on_fill`, which
791 // may place or withdraw orders, so any position into the queue is invalid the moment the strategy
792 // is told anything. The eligible references are collected first and each is looked up again when
793 // its turn comes; one the callback withdrew is simply no longer there.
794 //
795 // The queue is never emptied, so an order this loop has not reached yet is outstanding, visible in
796 // every portfolio installed along the way, and addressable by a cancellation from any callback.
797 void settle_queued_against(const strategy::BarEvent& event) {
798 const auto bar_date = clock_.session().local_date(event.bar_.timestamp_);
799 // A session the replay had no closing bar for -- a missing last minute -- still ended: an order
800 // of an earlier date is withdrawn before this bar can fill it.
801 expire_day_orders(event, bar_date, false);
802 std::vector<OrderRef<Tag>> eligible;
803 std::vector<OrderRef<Tag>> first_session;
804 for (const auto& entry : runner_.executor().queued()) {
805 // A different instrument's order waits for that instrument's next bar, which is what makes an
806 // interleaved multi-instrument replay behave. An order accepted during this bar waits for the
807 // next one, which is the no-lookahead rule.
808 if (!(entry.order_.instrument_ == event.instrument_)) {
809 continue;
810 }
811 if (entry.accepted_on_ >= runner_.executor().bar()) {
812 continue;
813 }
814 // **And, where the event says enough to check it, it cannot fill against a bar whose period
815 // had already begun when it was decided.** The counter above is enough while only bars place
816 // orders and no two periods overlap; a quote arrives inside a period without opening one, so
817 // without this an order decided at 10:30 fills at that day's 09:00 open -- a price from before
818 // the decision. Where the event does *not* say enough, `may_fill` allows the fill and records
819 // that nothing was checked. See it for all three answers.
820 if (!may_fill(entry, event)) {
821 continue;
822 }
823 eligible.push_back(entry.ref_);
824 // **Which session a DAY order lives in**, fixed at the first bar it is eligible against: that
825 // bar's session is the one the order trades in. A decision the bar cannot fill without looking
826 // ahead -- a quote at 10:00 against the day's bar, or the day's close -- trades in the next
827 // session, as a market order decided there fills at the next open: one session, shifted, never
828 // the several a resting order had before. Recorded here, beside the one `may_fill` call, so
829 // `unchecked_fill_causality_` still counts each order once per bar.

847 // end's day is nominal there -- a day late for a daily bar, days into the next month for a

864 // resting day limit queued to be retried on the next bar, and a refusal is not a bar that
865 // failed to reach a price. The order goes whatever its type, exactly once.
866 runner_.executor().withdraw(ref);

869 // orders eligible on this same bar from filling behind a cost the model could not compute,
870 // and `feed` stops consuming bars from its next call onward.
871 abort_queue();
872 return;
873 }
874 if (!decision.execution_) {
875 if (!decision.still_working_) {
876 // A market order the bar could not price, or a limit whose arithmetic did not fit. It does
877 // not rest and it did not trade, so it stops being outstanding.
878 runner_.executor().withdraw(ref);
879 }
880 continue;
881 }
882 // Removed before the ledger moves and before anything is shown a portfolio, so no view ever
883 // carries a filled order as still working.
884 runner_.executor().withdraw(ref);
885 // Stamped from the same reference data the ledger is given, so the two can never disagree
886 // about what currency one fill was in.
887 decision.execution_->currency_ = currency_for(order.instrument_);
888 if (!ledger_.apply(*decision.execution_, currency_for(order.instrument_))) {
889 // **A modelled fill the ledger cannot record halts the replay**, for the same reason a cost
890 // it cannot compute does. `apply` is atomic, so nothing is half-written -- but the fill
891 // happened in the model and did not happen in the books, and continuing would carry on from
892 // a portfolio missing a trade the strategy was never told about. This used to `continue`,
893 // which recreated at the ledger boundary exactly the silently-missing-trade path the
894 // costing refusal above exists to remove.
895 BrokerError error;
896 error.category_ = ErrorCategory::INVALID_ARGUMENT;
897 error.reached_broker_ = false;
898 error.broker_code_ = "execution_not_representable";
899 error.message_ = "an execution's cash arithmetic does not fit an exact decimal, so the "
900 "ledger could not record a fill the model produced";
901 result_.halted_ = std::move(error);
902 abort_queue();
903 return;
904 }
905 note_overdraft(*decision.execution_);
906 if (decision.intrabar_release_fill_) {
907 ++result_.stop_fills_intrabar_;
908 }
909 if (decision.bounded_) {
910 ++result_.limit_fills_bounded_;
911 }
912 result_.executions_.push_back(*decision.execution_);
913 install_portfolio();
914 drain_reports(runner_.feed(decision.execution_->fill_));

970 // Records equity at this bar, and the return since the previous point.
971 void record_equity(const strategy::BarEvent& event) {
972 if (const auto cash = ledger_.cash_in(config_.currency_);
973 cash && decimal_less(*cash, Decimal64::parse("0"))) {
974 ++result_.overdrawn_bars_;
975 if (!result_.first_overdrawn_) {
976 result_.first_overdrawn_ = event.period_end_;
977 }

979 const auto equity = ledger_.equity_in(config_.currency_);
980 if (!equity) {
981 // Every lot was marked, and the total of cash and lots still does not fit an exact decimal.
982 // No point is recorded -- a point at an approximate total would be a figure nobody computed
983 // -- and the omission is counted, because a gap in the curve has other causes too.
984 ++result_.unvalued_bars_;
985 if (!result_.first_unvalued_) {
986 result_.first_unvalued_ = event.period_end_;
987 }
988 equity_gap_ = true;
989 return;
990 }
991 EquityPoint point;
992 point.at_ = event.period_end_;
993 point.equity_ = *equity;
994 point.period_return_ = math::missing();
995 // **No return across an omitted point.** The one below would measure two or more bars and be
996 // handed to the Sharpe ratio and the annualization as one bar's return, so the first point
997 // after a gap carries none -- as the first point of the run does.
998 const bool after_gap = equity_gap_;
999 equity_gap_ = false;
1000 if (!result_.equity_.empty() && !after_gap) {
1001 const auto& previous = result_.equity_.back().equity_;
1002 // Only from positive equity. From a negative total the ratio's sign is reversed: a further loss
1003 // reads as a gain, and a run that lost more than its capital reported a positive Sharpe ratio.
1004 if (previous.coefficient() > 0) {
1005 const auto moved = decimal_subtract(*equity, previous);
1006 if (moved) {
1007 const auto ratio = decimal_divide(*moved, previous, 8);
1008 if (ratio) {
1009 point.period_return_ = analytics_to_scalar(*ratio);
1010 }
1011 }
1012 }
1013 }
1014 result_.equity_.push_back(point);
1015 }

1017 // Converts an exact ratio to the scalar the metrics compute in.
1018 //
1019 // The one crossing of the decimal boundary in this engine, and it goes outward: a return is an input
1020 // to `sharpe` and the rest, which are defined on `Scalar`. Nothing converts back, because no figure
1021 // a report produces is ever sent to a broker. The same conversion as `analytics::to_scalar` (one
1022 // division by an exact power of ten, so the nearest double within its bounds), written out because
1023 // this header does not include the analytics layer; keep the two alike.
1024 static math::Scalar analytics_to_scalar(const Decimal64& value) {
1025 math::Scalar divisor{1};
1026 for (std::uint8_t at = 0; at < value.scale(); ++at) {
1027 divisor *= math::Scalar{10};
1028 }
1029 return static_cast<math::Scalar>(value.coefficient()) / divisor;
1030 }

1032 // Returns the grid for `instrument`, unknown when no reference data was supplied.
1033 // Returns the currency `instrument` trades in, from the reference data the replay was given.
1034 //
1035 // **Every fill used to be booked in `config_.currency_`**, which is documented as the currency the
1036 // *starting cash* is in. A replay of a dollar-quoted instrument therefore banked dollars as won at
1037 // a rate of one -- an implicit currency conversion, which this library never makes silently.
1038 //
1039 // The fallback is defensive rather than a policy: `reconcile` refuses an instrument with no
1040 // reference data before a `PlannedOrder` exists (`strategy/reconcile.hpp`), so nothing can reach
1041 // the fill model unregistered. It is written the conservative way round anyway -- an unregistered
1042 // instrument books in the currency the ledger was opened in, which is the behaviour every existing
1043 // replay already had.
1044 // Installs the portfolio a strategy and its guards decide on.
1045 //
1046 // The single place the replay answers "what does this account look like", so that every callback
1047 // boundary is answered the same way.
1048 //
1049 // **It has to be a complete answer, because it overwrites one.**
1050 // `Runner::persist_reservation` writes an accepted order into the context portfolio and relies on
1051 // a wholesale replacement carrying what the reservation was standing in for
1052 // (`Runner::persist_reservation`'s comment). Live that replacement is a broker's listing. Here it is this
1053 // function, and a partial answer is therefore not a smaller improvement -- it is a *loss*, because
1054 // the reservation it erased was carrying the part it left out.
1055 //
1056 // The reservation carried two things, so this carries both: the outstanding orders themselves,
1057 // and the money an outstanding buy has committed.
1058 // Reports every queued order whose state has changed since the last observation.
1059 //
1060 // ## Where this is called from, and why not simply beside `install_portfolio`
1061 //
1062 // At the three points where a *batch* of changes to the queue has finished: the end of
1063 // settlement, the end of `feed`, and the end of `finish`. The queue changes in exactly three
1064 // ways -- `place` and `cancel` from a strategy dispatch, and `withdraw` from settlement -- and
1065 // each of those three boundaries closes one batch of them.
1066 //
1067 // The obvious alternative, publishing from `install_portfolio`, is wrong and quietly so: the
1068 // last install of a bar runs *before* `feed(BarEvent)` dispatches `on_bar`, so an order placed
1069 // there enters the queue after it, and the next bar's settlement withdraws the order before that
1070 // bar's first install. An order placed on one bar and filled on the next -- which is every market
1071 // order -- would be present at no install at all and would be reported nowhere.
1072 //
1073 // ## The portfolio is installed before each dispatch, not once for the batch
1074 //
1075 // A handler told about an order has to be able to find it. With every optional limit disabled
1076 // `Runner::persist_reservation` writes no working row, so nothing else would put the order in
1077 // the view, and a strategy would be told about an order it could neither see nor withdraw.
1078 // Installing once for the batch is not enough either, because a handler answering the first
1079 // update may place or cancel, which invalidates the picture the second one reads.
1080 //
1081 // ## What a handler's own order does
1082 //
1083 // It enters the queue behind this observation and is reported at the *next* boundary, never
1084 // within this one, so a strategy cannot drive itself round a loop here. An order placed by
1085 // `on_order_update` during `finish` has no later boundary and is not reported at all.
1086 void publish_order_updates() {
1087 if (result_.halted_) {
1088 // Nothing runs after a refusal, for the reason `feed` documents: a handler called here would
1089 // decide on a replay that has stopped being able to cost what it asks for.
1090 return;
1091 }
1092 std::vector<WorkingOrder<Tag>> rows;
1093 rows.reserve(runner_.executor().queued().size());
1094 for (const auto& entry : runner_.executor().queued()) {
1095 rows.push_back(working_row(entry));
1096 }
1097 for (const auto& update : updates_.changed(rows)) {
1098 // **Re-checked against the queue immediately before it is dispatched, not only when the batch
1099 // was observed.** A handler answering the first update of a batch can cancel a *different*
1100 // order in the same batch, and the event captured for that order is then a statement about an
1101 // order that no longer exists -- the portfolio installed below would correctly omit it while
1102 // the event announced it, which is exactly the contradiction the install exists to prevent.
1103 //
1104 // Checked on the state as well as the reference, because a handler can also replace an order
1105 // with another carrying the same one.
1106 const auto* still_queued = find_queued(update);
1107 if (still_queued == nullptr) {
1108 continue;
1109 }
1110 install_portfolio();
1111 drain_reports(runner_.feed(update));
1112 if (result_.halted_) {
1113 return;
1114 }
1115 }
1116 // **Once, after the whole batch.** Every pass that can withdraw a stop -- the bar's, the phase's,
1117 // the fills' and these updates' -- ends here, so the strategy has answered every update before a
1118 // fallback is released; releasing after each pass let a later update's instruction arrive after
1119 // the fallback had already been queued. The live path waits for a whole listing the same way.
1120 settle_deferred();
1121 }

1126 // since the batch was observed is not announced as it stood.
1127 const QueuedOrder<Tag>* find_queued(const OrderSnapshot& update) const {

1194 // Takes what an outstanding buy has committed out of the buying power the replay models.
1195 //
1196 // **Rows alone do not fix `enforce_buying_power_`**, which reads the scalar and not the orders.
1197 // `ReplayLedger::snapshot` builds buying power from ledger cash, and that cash has not paid for
1198 // anything still resting -- so with a hundred of cash, a resting buy costing a hundred, and a
1199 // second buy costing a hundred in another instrument, the second passes and both can still fill.
1200 //
1201 // Valued at the order's own price, else the last close, which is the same estimate the guard
1202 // checked the order against — and it is an **estimate**, not a bound. A limit buy cannot fill
1203 // above its limit, so for those the commitment is at least the eventual spend before costs. A
1204 // market order has no stated price at all: it fills where the next bar opens, and slippage and
1205 // commission are both outside this figure, so the real spend can exceed what was committed. An
1206 // order that can be valued by neither route is left uncommitted rather than guessed at.
1207 //
1208 // **Scoped to the limit, exactly as `Runner::reserve` is.**
1209 // `buying_power_in()` is a public figure a strategy is free to size its next order from, so
1210 // lowering it for a caller who never enabled the check would change how their strategy sizes --
1211 // a behaviour change well past enforcing a ceiling, and the reason the runner passes `spends`
1212 // rather than always debiting. It also keeps `ReportCaveats::buying_power_is_ledger_cash` true
1213 // for every replay that does not ask for the check.
1214 void commit_open_buy(strategy::Portfolio<Tag>& portfolio, const QueuedOrder<Tag>& entry) const {
1215 if (!config_.limits_.enforce_buying_power_ || entry.order_.side_ != OrderSide::BUY) {
1216 return;
1217 }
1218 const auto price = entry.order_.price_ ?
1219 entry.order_.price_ :
1220 runner_.context().last_price(entry.order_.instrument_);
1221 if (!price) {
1222 return;
1223 }
1224 const auto spent = decimal_multiply(entry.order_.quantity_, *price);
1225 if (!spent) {
1226 return;
1227 }
1228 const auto currency = currency_for(entry.order_.instrument_);
1229 for (auto& reading : portfolio.buying_power_) {
1230 if (!(reading.cash_.currency_ == currency)) {
1231 continue;
1232 }
1233 if (const auto left = decimal_subtract(reading.cash_.amount_, *spent)) {
1234 reading.cash_.amount_ = *left;
1235 }
1236 return;
1237 }
