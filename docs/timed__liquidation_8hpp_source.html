67 // Targets zero in every instrument held, once per instrument (a holding may span several rows,
68 // and duplicate targets would each plan a sell). Working orders are left to reconciliation.
69 template <class Ctx>
70 void flatten(Ctx& ctx) {
71 std::vector<InstrumentId> targeted;
72 for (const auto& position : ctx.portfolio().positions_) {
73 if (position.quantity_.coefficient() == 0) {
74 continue;
75 }
76 bool seen = false;
77 for (const auto& already : targeted) {
78 if (already == position.instrument_) {
79 seen = true;
80 break;
81 }
82 }
83 if (seen) {
84 continue;
85 }
86 targeted.push_back(position.instrument_);
87
88 TargetPosition target;
89 target.instrument_ = position.instrument_;
90 target.target_quantity_ = Decimal64::parse("0");
91 target.tag_ = params_.tag_;
92 ctx.submit(target);
93 }
