150struct PriceRule {
152 enum class Kind {
154 STATED,
156 UNCONSTRAINED,
161 UNKNOWN,
162 };

167 std::vector<TickBand> bands_;
169 std::string source_;
170
177 std::optional<Decimal64> tick_at(const Decimal64& price) const {
178 std::optional<Decimal64> tick;
179 for (const auto& band : bands_) {
180 if (decimal_less(price, band.from_)) {
181 break;
182 }
183 tick = band.tick_;
184 }
185 return tick;
186 }
187
195 std::optional<bool> on_tick(const Decimal64& price) const {
196 if (kind_ == Kind::UNKNOWN) {
197 return std::nullopt;
198 }
199 const auto tick = tick_at(price);
200 if (!tick) {
201 return true;
202 }
203 const auto rest = remainder(price, *tick);
204 if (!rest) {
205 return std::nullopt;
206 }
207 return rest->coefficient() == 0;
208 }
209
222 std::optional<Decimal64> round_to_tick(const Decimal64& price, OrderSide side) const {
223 if (kind_ == Kind::UNKNOWN) {
224 return std::nullopt;
225 }
226 const auto tick = tick_at(price);
227 if (!tick) {
228 return price;
229 }
230 const auto rest = remainder(price, *tick);
231 if (!rest) {
232 return std::nullopt;
233 }
234 if (rest->coefficient() == 0) {
235 return decimal_normalized(price);
236 }
237 const auto floor_price = decimal_subtract(price, *rest);
238 if (!floor_price) {
239 return std::nullopt;
240 }
241 if (side == OrderSide::BUY) {
242 return floor_price;
243 }
244 return decimal_add(*floor_price, *tick);
245 }

291 // `price` modulo `tick`, exactly.
292 static std::optional<Decimal64> remainder(const Decimal64& price, const Decimal64& tick) {
293 std::int64_t left = 0;
294 std::int64_t right = 0;
295 std::uint8_t scale = 0;
296 if (!decimal_detail::align(price, tick, left, right, scale) || right == 0) {
297 return std::nullopt;
298 }
299 auto rest = left % right;
300 if (rest < 0) {
301 rest += right < 0 ? -right : right;
302 }
303 return Decimal64(rest, scale);
304 }

318inline PriceRule price_rule_for(Country country, SecurityType type) {
319 if (country != Country::KR) {
320 return PriceRule::unconstrained();
321 }
322 switch (type) {
323 case SecurityType::STOCK:
324 case SecurityType::FOREIGN_STOCK:
325 case SecurityType::DEPOSITARY_RECEIPT:
326 case SecurityType::INFRASTRUCTURE_FUND:
327 case SecurityType::REIT:
328 return PriceRule::krx_equity();
329 case SecurityType::ETF:
330 case SecurityType::FOREIGN_ETF:
331 case SecurityType::ETN:
332 return PriceRule::unknown(
333 "KRX prices funds and notes on a ladder this library does not carry"
334 );
335 case SecurityType::STOCK_WARRANT:
336 case SecurityType::UNKNOWN:
337 break;
338 }
339 return PriceRule::unknown("no price grid is known for this security type");
340}
