174 constexpr bool grid_known() const noexcept {

179 case Kind::UNKNOWN:

191 std::optional<Decimal64> tick_at(const Decimal64& price) const {
192 std::optional<Decimal64> tick;
193 for (const auto& band : bands_) {
194 if (decimal_less(price, band.from_)) {
195 break;
196 }
197 tick = band.tick_;
198 }
199 return tick;
200 }

209 std::optional<bool> on_tick(const Decimal64& price) const {
210 if (!grid_known()) {
211 return std::nullopt;
212 }
213 const auto tick = tick_at(price);
214 if (!tick) {
215 return true;
216 }
217 const auto rest = remainder(price, *tick);
218 if (!rest) {
219 return std::nullopt;
220 }
221 return rest->coefficient() == 0;
222 }
223
235 std::optional<Decimal64> round_to_tick(const Decimal64& price, OrderSide side) const {
236 if (!grid_known()) {
237 return std::nullopt;
238 }
239 const auto tick = tick_at(price);
240 if (!tick) {
241 return price;
242 }
243 const auto rest = remainder(price, *tick);
244 if (!rest) {
245 return std::nullopt;
246 }
247 if (rest->coefficient() == 0) {
248 return decimal_normalized(price);
249 }
250 const auto floor_price = decimal_subtract(price, *rest);
251 if (!floor_price) {
252 return std::nullopt;
253 }
254 if (side == OrderSide::BUY) {
255 return floor_price;
256 }
257 return decimal_add(*floor_price, *tick);
258 }

264 static PriceRule unconstrained() {
265 PriceRule rule;
266 rule.kind_ = Kind::UNCONSTRAINED;
267 rule.source_ = "no exchange price grid";
268 return rule;
269 }

278 rule.kind_ = Kind::UNKNOWN;
279 rule.source_ = std::move(reason);
280 return rule;
281 }

283private:
284 // `price` modulo `tick`, exactly.
285 static std::optional<Decimal64> remainder(const Decimal64& price, const Decimal64& tick) {
286 std::int64_t left = 0;
287 std::int64_t right = 0;
288 std::uint8_t scale = 0;
289 if (!decimal_detail::align(price, tick, left, right, scale) || right == 0) {
290 return std::nullopt;
291 }
292 auto rest = left % right;
293 if (rest < 0) {
294 rest += right < 0 ? -right : right;
295 }
296 return Decimal64(rest, scale);
297 }

479 switch (state) {

481 return true;
482 case StatedDate::State::NOT_STATED:
483 case StatedDate::State::STATED:
