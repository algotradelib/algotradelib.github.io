62inline ErrorCategory kiwoom_error_category(std::string_view code) noexcept {
63 // Header-level authentication faults: the authorization field is missing or malformed.
64 if (code == "1512" || code == "1513" || code == "1514" || code == "1515" || code == "1516") {
65 return ErrorCategory::AUTHENTICATION;
66 }

97inline std::optional<BrokerError> kiwoom_declared_error(const boost::json::object& document) {
98 const auto code = document.if_contains("return_code");
99 if (!code) {
100 return std::nullopt;
101 }
102 std::string text;
103 if (code->is_int64()) {
104 text = std::to_string(code->as_int64());
105 } else if (code->is_uint64()) {
106 text = std::to_string(code->as_uint64());
107 } else if (code->is_string()) {
108 text = std::string(code->as_string());
109 } else {
110 BrokerError error;
111 error.category_ = ErrorCategory::PROTOCOL;
112 error.broker_code_ = "invalid_return_code";
113 error.message_ = "return_code is neither a number nor a string";
114 return error;
115 }
116 if (text == "0") {
117 return std::nullopt;
118 }
119
120 BrokerError error;
121 error.broker_code_ = text;
122 error.reached_broker_ = true;
123 if (const auto message = document.if_contains("return_msg"); message && message->is_string()) {
124 error.message_ = std::string(message->as_string());
125 } else {
126 error.message_ = "Kiwoom declared return_code " + text;
127 }
128 error.category_ = kiwoom_error_category(text);
129 return error;
130}

160inline std::string_view kiwoom_refusal_code(std::string_view message) noexcept {
161 constexpr std::string_view prefix = "[2000](";
162 if (message.size() < prefix.size() || message.substr(0, prefix.size()) != prefix || message.back() != ')') {
163 return {};
164 }
165 const auto inner = message.substr(prefix.size(), message.size() - prefix.size() - 1);
166 const auto colon = inner.find(':');
167 if (colon == std::string_view::npos || colon == 0 || colon + 1 == inner.size()) {
168 return {};
169 }
170 const auto code = inner.substr(0, colon);
171 for (const char character : code) {
172 if (!((character >= 'A' && character <= 'Z') || (character >= '0' && character <= '9'))) {
173 return {};
174 }
175 }
176 return code;
177}

204inline std::optional<BrokerError> kiwoom_recognized_refusal(
205 int http_status,
206 const boost::json::object& document,
207 KiwoomOrderOperation operation,
208 bool demo
209) {
210 if (http_status != 200) {
211 return std::nullopt;
212 }
213 const auto code = document.if_contains("return_code");
214 const bool generic = code && (
215 (code->is_int64() && code->as_int64() == 20) || (code->is_uint64() && code->as_uint64() == 20) ||
216 (code->is_string() && code->as_string() == "20")
217 );
218 const auto message = document.if_contains("return_msg");
219 if (!generic || !message || !message->is_string()) {
220 return std::nullopt;
221 }
222 const std::string stated(kiwoom_refusal_code(std::string_view(message->as_string())));
223 const char* name = "";
224 const bool placement = operation == KiwoomOrderOperation::PLACE &&
225 (demo ? (stated == "800033" || stated == "RC4057" || stated == "RC4058") : stated == "571445");
226 const bool cancellation = operation == KiwoomOrderOperation::CANCEL &&
227 (demo ? stated == "RC4033" : stated == "506550");
228 if (placement) {
229 name = "placement";
230 } else if (cancellation) {
231 name = "cancellation";
232 } else {
233 return std::nullopt;
234 }
235 if (const auto number = document.if_contains("ord_no");
236 number && !(number->is_string() && number->as_string().empty())) {
237 return std::nullopt;
238 }
239 for (const char* field : {"cncl_qty", "mdfy_qty"}) {
240 const auto quantity = document.if_contains(field);
241 if (!quantity) {
242 continue;
243 }
244 if (!quantity->is_string() ||
245 std::string_view(quantity->as_string()).find_first_not_of('0') != std::string_view::npos) {
246 return std::nullopt;
247 }
248 }
249 BrokerError error;
250 error.category_ = ErrorCategory::INVALID_ARGUMENT;
251 error.reached_broker_ = true;
252 error.http_status_ = http_status;
253 error.broker_code_ = stated;
254 error.message_ = std::string(message->as_string()) + " (return_code 20; recognized refusal: Kiwoom " +
255 (demo ? "demo" : "real") + ", cash " + name + ", " + stated + ")";
256 return error;
257}
