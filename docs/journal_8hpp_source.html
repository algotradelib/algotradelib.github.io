547inline std::vector<OpenInstruction> unresolved_instructions(const JournalReplay& replay) {
548 std::vector<OpenInstruction> open;
549 for (const auto& entry : replay.entries_) {
550 if (!journal_outcome_is_terminal(entry.outcome_)) {
551 OpenInstruction instruction;
552 instruction.entry_ = entry;
553 instruction.opened_at_ = entry.recorded_at_;
554 instruction.detail_ = entry.detail_;
555 open.push_back(std::move(instruction));
556 continue;
557 }
558
559 bool paired = false;
560 for (auto at = open.begin(); at != open.end(); ++at) {
561 if (at->entry_.id_ != entry.id_) {
562 continue;
563 }
564 paired = true;
565 if (journal_outcome_leaves_open(entry.outcome_)) {
566 at->answered_ = true;
567 at->closed_at_ = entry.recorded_at_;
568 at->detail_ = entry.detail_;
569 // The broker issued no reference, so the placement keeps the one its pending line had.
570 at->entry_.order_id_ = entry.order_id_.empty() ? at->entry_.order_id_ : entry.order_id_;
571 } else {
572 open.erase(at);
573 }
574 break;
575 }
576 if (!paired && journal_outcome_leaves_open(entry.outcome_)) {
577 OpenInstruction instruction;
578 instruction.entry_ = entry;
579 instruction.opened_at_ = entry.recorded_at_;
580 instruction.closed_at_ = entry.recorded_at_;
581 instruction.answered_ = true;
582 instruction.detail_ = entry.detail_;
583 open.push_back(std::move(instruction));
584 }
585 }
586 return open;
587}

602inline std::vector<OpenInstruction> placements_of(const JournalReplay& replay, const std::vector<std::string>& ids) {
603 std::vector<OpenInstruction> found;
604 for (const auto& entry : replay.entries_) {
605 if (
606 entry.action_ != JournalAction::PLACE || entry.requires_operator_ ||
607 !journal_outcome_leaves_open(entry.outcome_) || std::find(ids.begin(), ids.end(), entry.id_) == ids.end()
608 ) {
609 continue;
610 }
611 auto known = std::find_if(found.begin(), found.end(), [&](const OpenInstruction& instruction) {
612 return instruction.entry_.id_ == entry.id_;
613 });
614 if (known == found.end()) {
615 OpenInstruction instruction;
616 instruction.entry_ = entry;
617 instruction.opened_at_ = entry.recorded_at_;
618 instruction.detail_ = entry.detail_;
619 found.push_back(std::move(instruction));
620 known = std::prev(found.end());
621 }
622 if (journal_outcome_is_terminal(entry.outcome_)) {
623 known->answered_ = true;
624 known->closed_at_ = entry.recorded_at_;
625 known->detail_ = entry.detail_;
626 }
627 }
628 return found;
629}

793 SendOutcome place(const strategy::PlannedOrder& order, std::optional<Decimal64> baseline) {
794 JournalEntry entry;
795 entry.id_ = next_id();
796 entry.action_ = JournalAction::PLACE;
797 entry.outcome_ = JournalOutcome::PENDING;
798 entry.recorded_at_ = clock_.now();
799 entry.instrument_ = order.instrument_;
800 entry.side_ = order.side_;
801 entry.type_ = order.type_;
802 entry.time_in_force_ = order.time_in_force_;
803 entry.session_ = order.session_;
804 // Where the order will actually go (an unrouted KIS order reaches KRX), asked of the binding;
805 // a pure function of the order, so the line is still written before any I/O.
806 entry.venue_ = effective_route_of(order);
807 entry.quantity_ = order.quantity_;
808 entry.price_ = order.price_;
809 entry.trigger_price_ = order.trigger_price_;
810 entry.baseline_ = std::move(baseline);
811 if (!journal_->append(entry)) {
812 return refuse_unjournaled();
813 }
814
815 const auto outcome = inner_.place(order);
816
817 entry.outcome_ = journal_outcome_of(outcome);
818 entry.recorded_at_ = clock_.now();
819 entry.order_id_ = accepted_order_id(outcome);
820 entry.detail_ = failure_detail(outcome);
821 last_entry_ = entry;
822 if (!journal_->append(entry)) {
823 // The send happened; only its answer is lost, so recovery will see it unresolved. Recorded so
824 // the runner stops.
825 note_failure("the order journal could not record an instruction's outcome");
826 }
827 return outcome;
828 }
