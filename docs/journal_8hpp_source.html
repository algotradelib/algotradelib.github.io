510struct OpenInstruction {

521 bool answered_{false};

537inline std::vector<OpenInstruction> unresolved_instructions(const JournalReplay& replay) {
538 std::vector<OpenInstruction> open;
539 for (const auto& entry : replay.entries_) {
540 if (!journal_outcome_is_terminal(entry.outcome_)) {
541 OpenInstruction instruction;
542 instruction.entry_ = entry;
543 instruction.opened_at_ = entry.recorded_at_;
544 instruction.detail_ = entry.detail_;
545 open.push_back(std::move(instruction));
546 continue;
547 }
548
549 bool paired = false;
550 for (auto at = open.begin(); at != open.end(); ++at) {
551 if (at->entry_.id_ != entry.id_) {
552 continue;
553 }
554 paired = true;
555 if (journal_outcome_leaves_open(entry.outcome_)) {
556 at->answered_ = true;
557 at->closed_at_ = entry.recorded_at_;
558 at->detail_ = entry.detail_;
559 // The broker issued no reference, so the placement keeps the one its pending line had.
560 at->entry_.order_id_ = entry.order_id_.empty() ? at->entry_.order_id_ : entry.order_id_;
561 } else {
562 open.erase(at);
563 }
564 break;
565 }
566 if (!paired && journal_outcome_leaves_open(entry.outcome_)) {
567 OpenInstruction instruction;
568 instruction.entry_ = entry;
569 instruction.opened_at_ = entry.recorded_at_;
570 instruction.closed_at_ = entry.recorded_at_;
571 instruction.answered_ = true;
572 instruction.detail_ = entry.detail_;
573 open.push_back(std::move(instruction));
574 }
575 }
576 return open;
577}

592inline std::vector<OpenInstruction> placements_of(const JournalReplay& replay, const std::vector<std::string>& ids) {
593 std::vector<OpenInstruction> found;
594 for (const auto& entry : replay.entries_) {
595 if (entry.action_ != JournalAction::PLACE || entry.requires_operator_ ||
596 !journal_outcome_leaves_open(entry.outcome_) ||
597 std::find(ids.begin(), ids.end(), entry.id_) == ids.end()) {
598 continue;
599 }
600 auto known = std::find_if(found.begin(), found.end(), [&](const OpenInstruction& instruction) {
601 return instruction.entry_.id_ == entry.id_;
602 });
603 if (known == found.end()) {
604 OpenInstruction instruction;
605 instruction.entry_ = entry;
606 instruction.opened_at_ = entry.recorded_at_;
607 instruction.detail_ = entry.detail_;
608 found.push_back(std::move(instruction));
609 known = std::prev(found.end());
610 }
611 if (journal_outcome_is_terminal(entry.outcome_)) {
612 known->answered_ = true;
613 known->closed_at_ = entry.recorded_at_;
614 known->detail_ = entry.detail_;
615 }
616 }
617 return found;
618}

770 SendOutcome place(const strategy::PlannedOrder& order, std::optional<Decimal64> baseline) {
771 JournalEntry entry;
772 entry.id_ = next_id();
773 entry.action_ = JournalAction::PLACE;
774 entry.outcome_ = JournalOutcome::PENDING;
775 entry.recorded_at_ = clock_.now();
776 entry.instrument_ = order.instrument_;
777 entry.side_ = order.side_;
778 entry.type_ = order.type_;
779 entry.time_in_force_ = order.time_in_force_;
780 entry.session_ = order.session_;
781 // Where the order will actually go (an unrouted KIS order reaches KRX), asked of the binding;

783 entry.venue_ = effective_route_of(order);
784 entry.quantity_ = order.quantity_;
785 entry.price_ = order.price_;
786 entry.trigger_price_ = order.trigger_price_;
787 entry.baseline_ = std::move(baseline);

789 return refuse_unjournaled();
790 }
791
792 const auto outcome = inner_.place(order);
793
794 entry.outcome_ = journal_outcome_of(outcome);
795 entry.recorded_at_ = clock_.now();
796 entry.order_id_ = accepted_order_id(outcome);
797 entry.detail_ = failure_detail(outcome);
798 last_entry_ = entry;

800 // The send happened; only its answer is lost, so recovery will see it unresolved. Recorded so
801 // the runner stops.
802 note_failure("the order journal could not record an instruction's outcome");
803 }
