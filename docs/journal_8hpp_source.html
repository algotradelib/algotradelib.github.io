362 bool answered_{false};

378inline std::vector<OpenInstruction> unresolved_instructions(const JournalReplay& replay) {
379 std::vector<OpenInstruction> open;
380 for (const auto& entry : replay.entries_) {
381 if (!journal_outcome_is_terminal(entry.outcome_)) {
382 OpenInstruction instruction;
383 instruction.entry_ = entry;
384 instruction.opened_at_ = entry.recorded_at_;
385 instruction.detail_ = entry.detail_;
386 open.push_back(std::move(instruction));
387 continue;
388 }
389
390 bool paired = false;
391 for (auto at = open.begin(); at != open.end(); ++at) {
392 if (at->entry_.id_ != entry.id_) {
393 continue;
394 }
395 paired = true;
396 if (entry.outcome_ == JournalOutcome::INDETERMINATE) {
397 at->answered_ = true;
398 at->closed_at_ = entry.recorded_at_;
399 at->detail_ = entry.detail_;
400 // The broker issued no reference, so the placement keeps the one its pending line had.
401 at->entry_.order_id_ = entry.order_id_.empty() ? at->entry_.order_id_ : entry.order_id_;
402 } else {
403 open.erase(at);
404 }
405 break;
406 }
407 if (!paired && entry.outcome_ == JournalOutcome::INDETERMINATE) {
408 OpenInstruction instruction;
409 instruction.entry_ = entry;
410 instruction.opened_at_ = entry.recorded_at_;
411 instruction.closed_at_ = entry.recorded_at_;
412 instruction.answered_ = true;
413 instruction.detail_ = entry.detail_;
414 open.push_back(std::move(instruction));
415 }
416 }
417 return open;
418}

433inline std::vector<OpenInstruction> placements_of(const JournalReplay& replay, const std::vector<std::string>& ids) {
434 std::vector<OpenInstruction> found;
435 for (const auto& entry : replay.entries_) {
436 if (entry.action_ != JournalAction::PLACE || entry.requires_operator_ ||
437 (entry.outcome_ != JournalOutcome::PENDING && entry.outcome_ != JournalOutcome::INDETERMINATE) ||
438 std::find(ids.begin(), ids.end(), entry.id_) == ids.end()) {
439 continue;
440 }
441 auto known = std::find_if(found.begin(), found.end(), [&](const OpenInstruction& instruction) {
442 return instruction.entry_.id_ == entry.id_;
443 });
444 if (known == found.end()) {
445 OpenInstruction instruction;
446 instruction.entry_ = entry;
447 instruction.opened_at_ = entry.recorded_at_;
448 instruction.detail_ = entry.detail_;
449 found.push_back(std::move(instruction));
450 known = std::prev(found.end());
451 }
452 if (entry.outcome_ == JournalOutcome::INDETERMINATE) {
453 known->answered_ = true;
454 known->closed_at_ = entry.recorded_at_;
455 known->detail_ = entry.detail_;
456 }
457 }
458 return found;
459}

611 SendOutcome place(const strategy::PlannedOrder& order, std::optional<Decimal64> baseline) {
612 JournalEntry entry;
613 entry.id_ = next_id();
614 entry.action_ = JournalAction::PLACE;
615 entry.outcome_ = JournalOutcome::PENDING;
616 entry.recorded_at_ = clock_.now();
617 entry.instrument_ = order.instrument_;
618 entry.side_ = order.side_;
619 entry.type_ = order.type_;
620 entry.time_in_force_ = order.time_in_force_;
621 entry.session_ = order.session_;
622 // Where the order will actually go (an unrouted KIS order reaches KRX), asked of the binding;
623 // a pure function of the order, so the line is still written before any I/O.
624 entry.venue_ = effective_route_of(order);
625 entry.quantity_ = order.quantity_;
626 entry.price_ = order.price_;
627 entry.trigger_price_ = order.trigger_price_;
628 entry.baseline_ = std::move(baseline);
629 if (!journal_->append(entry)) {
630 return refuse_unjournaled();
631 }
632
633 const auto outcome = inner_.place(order);
634
635 entry.outcome_ = journal_outcome_of(outcome);
636 entry.recorded_at_ = clock_.now();
637 entry.order_id_ = accepted_order_id(outcome);
638 entry.detail_ = failure_detail(outcome);
639 last_entry_ = entry;
640 if (!journal_->append(entry)) {
641 // The send happened; only its answer is lost, so recovery will see it unresolved. Recorded so
642 // the runner stops.
643 note_failure("the order journal could not record an instruction's outcome");
644 }
645 return outcome;
646 }
