673inline std::vector<OpenInstruction> unresolved_instructions(const JournalReplay& replay) {
674 std::vector<OpenInstruction> open;
675 for (const auto& entry : replay.entries_) {
676 // A ledger record answers no instruction, whatever its id.
677 if (entry.action_ == JournalAction::LEDGER) {
678 continue;
679 }
680 if (!journal_outcome_is_terminal(entry.outcome_)) {
681 OpenInstruction instruction;
682 instruction.entry_ = entry;
683 instruction.opened_at_ = entry.recorded_at_;
684 instruction.detail_ = entry.detail_;
685 open.push_back(std::move(instruction));
686 continue;
687 }
688
689 bool paired = false;
690 for (auto at = open.begin(); at != open.end(); ++at) {
691 if (at->entry_.id_ != entry.id_) {
692 continue;
693 }
694 paired = true;
695 if (journal_outcome_leaves_open(entry.outcome_)) {
696 at->answered_ = true;
697 at->closed_at_ = entry.recorded_at_;
698 at->detail_ = entry.detail_;
699 // The broker issued no reference, so the placement keeps the one its pending line had.
700 at->entry_.order_id_ = entry.order_id_.empty() ? at->entry_.order_id_ : entry.order_id_;
701 } else {
702 open.erase(at);
703 }
704 break;
705 }
706 if (!paired && journal_outcome_leaves_open(entry.outcome_)) {
707 OpenInstruction instruction;
708 instruction.entry_ = entry;
709 instruction.opened_at_ = entry.recorded_at_;
710 instruction.closed_at_ = entry.recorded_at_;
711 instruction.answered_ = true;
712 instruction.detail_ = entry.detail_;
713 open.push_back(std::move(instruction));
714 }
715 }
716 return open;
717}

732inline std::vector<OpenInstruction> placements_of(const JournalReplay& replay, const std::vector<std::string>& ids) {
733 std::vector<OpenInstruction> found;
734 for (const auto& entry : replay.entries_) {
735 if (
736 entry.action_ != JournalAction::PLACE || entry.requires_operator_ ||
737 !journal_outcome_leaves_open(entry.outcome_) || std::find(ids.begin(), ids.end(), entry.id_) == ids.end()
738 ) {
739 continue;
740 }
741 auto known = std::find_if(found.begin(), found.end(), [&](const OpenInstruction& instruction) {
742 return instruction.entry_.id_ == entry.id_;
743 });
744 if (known == found.end()) {
745 OpenInstruction instruction;
746 instruction.entry_ = entry;
747 instruction.opened_at_ = entry.recorded_at_;
748 instruction.detail_ = entry.detail_;
749 found.push_back(std::move(instruction));
750 known = std::prev(found.end());
751 }
752 if (journal_outcome_is_terminal(entry.outcome_)) {
753 known->answered_ = true;
754 known->closed_at_ = entry.recorded_at_;
755 known->detail_ = entry.detail_;
756 }
757 }
758 return found;
759}
