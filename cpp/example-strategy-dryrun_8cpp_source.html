136struct BreakoutThenFlatten {
138 Decimal64 quantity_;
140 std::vector<Decimal64> closes_;
142 bool flattened_{false};
143
144 static constexpr std::size_t lookback = 5;
145
146 template <class Ctx>
147 void on_bar(Ctx& ctx, const strategy::BarEvent& bar) {
148 closes_.push_back(bar.bar_.close_);
149 if (flattened_ || closes_.size() <= lookback) {
150 return;
151 }
152 // Exact decimals, so this needs no linear algebra backend.
153 Decimal64 highest = closes_[closes_.size() - 1 - lookback];
154 for (std::size_t at = closes_.size() - lookback; at + 1 < closes_.size(); ++at) {
155 if (algotradelib::decimal_less(highest, closes_[at])) {
156 highest = closes_[at];
157 }
158 }
159 if (!algotradelib::decimal_less(highest, closes_.back())) {
160 return;
161 }
162 strategy::TargetPosition target;
163 target.instrument_ = bar.instrument_;
164 target.target_quantity_ = quantity_;
165 target.tag_ = "breakout";
166 ctx.submit(target);
167 }
168
169 template <class Ctx>
170 void on_phase(Ctx& ctx, strategy::MarketPhase phase) {
171 if (phase == strategy::MarketPhase::OPEN || phase == strategy::MarketPhase::PRE_OPEN) {
172 return;
173 }
174 std::vector<InstrumentId> targeted;
175 for (const auto& position : ctx.portfolio().positions_) {
176 bool seen = false;
177 for (const auto& already : targeted) {
178 seen = seen || already == position.instrument_;
179 }
180 if (seen || position.quantity_.coefficient() == 0) {
181 continue;
182 }
183 targeted.push_back(position.instrument_);
184 strategy::TargetPosition target;
185 target.instrument_ = position.instrument_;
186 target.target_quantity_ = Decimal64::parse("0");
187 target.tag_ = "session-end";
188 ctx.submit(target);
189 }
190 flattened_ = true;
191 }
192};

198strategy::BarEvent bar(std::size_t index) {
199 Bar value;
200 value.timestamp_ =
201 std::chrono::system_clock::time_point{} + std::chrono::hours(24 * (20000 + static_cast<int>(index)));
202 value.open_ = Decimal64::parse(closes()[index]);
203 value.high_ = Decimal64::parse(closes()[index]);
204 value.low_ = Decimal64::parse(closes()[index]);
205 value.close_ = Decimal64::parse(closes()[index]);
206 value.volume_ = Decimal64::parse("1200000");
207 return strategy::make_bar_event(samsung(), BarInterval::ONE_DAY, BarStamp::CLOSE, value);
208}

211template <typename Report>
212void print(std::size_t day, const Report& report) {
213 for (const auto& order : report.would_place_) {
214 std::cout
215 << " day " << std::setw(2) << day << " " << (order.side_ == OrderSide::BUY ? "BUY " : "SELL") << ' '
216 << std::setw(4) << order.quantity_.str() << " shares";
217 if (order.price_) {
218 std::cout << " at " << order.price_->str();
219 } else {
220 std::cout << " at market";
221 }
222 std::cout << " [" << order.tag_ << "]\n";
223 }
224 for (const auto& rejection : report.rejected_) {
225 std::cout
226 << " day " << std::setw(2) << day << " rejected: " << strategy::reject_reason_name(rejection.reason_) << '\n';
227 }
228}

237using RefusalTally = std::map<std::string, std::pair<std::size_t, std::size_t>>;

239template <typename Report>
240void tally(std::size_t day, const Report& report, RefusalTally& counts) {
241 for (const auto& refusal : report.refused_) {
242 const std::string reason(strategy::guard_reason_name(refusal.reason_));
243 auto found = counts.find(reason);
244 if (found == counts.end()) {
245 counts.emplace(reason, std::pair<std::size_t, std::size_t>{day, 1});
246 } else {
247 ++found->second.second;
248 }
249 }
250}

253strategy::Portfolio<runner::DryTag> portfolio_of(const Decimal64& held) {
254 strategy::Portfolio<runner::DryTag> portfolio;
255 HoldingPosition position;
256 position.instrument_ = samsung();
257 position.quantity_ = held;
258 // What the account may sell, which a broker's holdings state and the guard checks a sell against by
259 // default: here the whole holding. Without it a dry run refuses every sell (`quantity_not_sellable`).
260 position.sellable_quantity_ = held;
261 portfolio.positions_.push_back(position);
262 portfolio.sellable_basis_ = algotradelib::SellableBasis{
263 algotradelib::SellableBasis::Coverage::WHOLE_HOLDING,
264 algotradelib::SellableBasis::RestingTreatment::DEDUCT,
265 false,
266 };
267 return portfolio;
268}

271template <typename Report>
272void apply(const Report& report, Decimal64& held) {
273 for (const auto& order : report.would_place_) {
274 const auto signed_quantity = order.side_ == OrderSide::BUY ?
275 algotradelib::decimal_add(held, order.quantity_) :
276 algotradelib::decimal_subtract(held, order.quantity_);
277 if (signed_quantity) {
278 held = *signed_quantity;
279 }
280 }

290template <typename Strategy>
291void run(const char* title, Strategy instance, const strategy::RiskLimits& limits = {}, bool end_session = false) {
292 std::cout << title << '\n';
293
294 runner::DryRunner<Strategy> dry(std::move(instance), limits);
295 dry.context().set_instrument(samsung_reference());
296 dry.context().set_phase(strategy::MarketPhase::OPEN);
297 dry.set_portfolio_value(Decimal64::parse("100000000"));
298 // A runner driven by hand starts its strategy before the first feed.
299 dry.start();
300
301 Decimal64 held = Decimal64::parse("0");
302 RefusalTally refusals;
303 for (std::size_t day = 0; day < closes().size(); ++day) {
304 dry.context().set_last_price(samsung(), Decimal64::parse(closes()[day]));
305
306 dry.context().set_portfolio(portfolio_of(held));
307
308 const auto report = dry.feed(bar(day));
309 print(day, report);
310 tally(day, report, refusals);
311 apply(report, held);
312 }
313 if (end_session) {
314 // The session ends after the last bar. The runner sees the holding the last bar left, then the phase.
315 std::cout << " holding " << held.str() << " shares when the session closes\n";
316 dry.context().set_portfolio(portfolio_of(held));
317 const auto report = dry.feed(strategy::MarketPhase::POST_CLOSE);
318 print(closes().size(), report);
319 tally(closes().size(), report, refusals);
320 apply(report, held);
321 }
322 for (const auto& entry : refusals) {
323 std::cout
324 << " day " << std::setw(2) << entry.second.first << " refused: " << entry.first << " (and on "
325 << entry.second.second - 1 << " later bars)\n";
326 }
327 std::cout << " ending position: " << held.str() << " shares\n\n";
328}

335 run("HoldsShares: one target, then nothing, and it remembers nothing to do it", HoldsShares{Decimal64::parse("100")});
336
337 run(
338 "BreakoutThenFlatten: in on a breakout, out when the session closes",
339 BreakoutThenFlatten{Decimal64::parse("100"), {}, false}, {}, true
340 );
341
342 // A guard refusing an order is part of the pipeline, not an error, so it belongs in the output.
343 strategy::RiskLimits tight;
344 tight.max_order_notional_ = Decimal64::parse("1000000");
345 run(
346 "HoldsShares with a 1,000,000 KRW per-order ceiling: refused, never resized", HoldsShares{Decimal64::parse("100")},
347 tight
348 );
349
350 return 0;
