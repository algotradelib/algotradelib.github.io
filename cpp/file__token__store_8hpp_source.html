59 Result<std::unique_ptr<Lock>> lock(const std::string& identity) override {
60 Descriptor folder;
61 if (auto error = open_folder(identity, folder)) {
62 return *error;
63 }
64 const auto name = identity + ".lock";
65 Descriptor file(::openat(folder.get(), name.c_str(), FILE_FLAGS | O_RDWR | O_CREAT, 0600));
66 if (file.get() < 0 && errno == ENOENT) {
67 // A concurrent create that answers no-such-file is retried once.
68 file.reset(::openat(folder.get(), name.c_str(), FILE_FLAGS | O_RDWR | O_CREAT, 0600));
69 }
70 if (auto error = check_file(file)) {
71 return *error;
72 }
73 const auto deadline = std::chrono::steady_clock::now() + lock_wait_;
74 while (::flock(file.get(), LOCK_EX | LOCK_NB) != 0) {
75 if (errno != EWOULDBLOCK && errno != EINTR) {
76 return token_store_error("token store: the lock file could not be locked");
77 }
78 if (std::chrono::steady_clock::now() >= deadline) {
79 return token_store_error("token store: lock not obtained in time");
80 }
81 std::this_thread::sleep_for(std::chrono::milliseconds(10));
82 }
83 std::unique_ptr<Lock> held = std::make_unique<FileLock>(std::move(file));

210 // Tests whether an identity can name a file: letters, digits, `-` and `_`, at most 200.
211 static bool file_safe(const std::string& identity) noexcept {
212 if (identity.empty() || identity.size() > 200) {
213 return false;
214 }
215 for (const char ch : identity) {
216 const bool allowed =
217 (ch >= 'a' && ch <= 'z') || (ch >= 'A' && ch <= 'Z') || (ch >= '0' && ch <= '9') || ch == '-' || ch == '_';
218 if (!allowed) {
219 return false;
220 }
221 }
222 return true;
223 }

225 // Tests what the store requires of everything it opens: the type, the user's own, nothing for group or others.
226 static bool owned_privately(int descriptor, mode_t type) noexcept {
227 struct stat status{};
228 return ::fstat(descriptor, &status) == 0 && (status.st_mode & S_IFMT) == type && status.st_uid == ::geteuid() &&
229 (status.st_mode & 077) == 0;
230 }

232 // Creates the folder if it does not exist and opens it, checked.
233 std::optional<BrokerError> open_folder(const std::string& identity, Descriptor& folder) const {
234 if (!file_safe(identity)) {
235 return token_store_error("token store: the identity cannot name a file");
236 }
237 if (::mkdir(folder_.c_str(), 0700) != 0 && errno != EEXIST) {
238 return token_store_error("token store: the folder could not be created");
239 }
240 folder.reset(::open(folder_.c_str(), O_RDONLY | O_DIRECTORY | O_NOFOLLOW | O_CLOEXEC));
241 if (folder.get() < 0) {
242 struct stat status{};
243 const bool link = ::lstat(folder_.c_str(), &status) == 0 && S_ISLNK(status.st_mode);
244 return token_store_error(
245 link ? "token store: the folder is a symbolic link" : "token store: the folder could not be opened"
246 );
247 }
248 if (!owned_privately(folder.get(), S_IFDIR)) {
249 return token_store_error("token store: the folder is not private (owner or permissions)");
250 }
251 return std::nullopt;
252 }

254 // Checks a file just opened.
255 static std::optional<BrokerError> check_file(const Descriptor& file) {
256 if (file.get() < 0) {
257 return token_store_error(
258 errno == ELOOP ? "token store: a file is a symbolic link" : "token store: a file could not be opened"

262 return token_store_error("token store: a file is not private (owner, permissions or type)");
263 }
264 return std::nullopt;

267 // Reads the token file into `text`; leaves it empty when there is none.
268 static std::optional<BrokerError> read_token(
269 const Descriptor& folder, const std::string& identity, std::optional<std::string>& text
270 ) {
271 Descriptor file(::openat(folder.get(), (identity + ".token").c_str(), FILE_FLAGS | O_RDONLY));
272 if (file.get() < 0 && errno == ENOENT) {
273 return std::nullopt;
274 }
275 if (auto error = check_file(file)) {
276 return error;
277 }
278 std::string contents;
279 char buffer[4096];
280 while (contents.size() <= MAXIMUM_TOKEN_FILE) {
281 const auto count = ::read(file.get(), buffer, sizeof(buffer));
282 if (count == 0) {
283 text = std::move(contents);
284 return std::nullopt;
285 }
286 if (count < 0 && errno != EINTR) {
287 return token_store_error("token store: a file could not be read");
288 }
289 if (count > 0) {
290 contents.append(buffer, static_cast<std::size_t>(count));
291 }
292 }
293 text = std::string();
294 return std::nullopt;
295 }

297 // Reads a token file's text; nothing unless it is a JSON object with a nonempty token and an expiry.
298 static std::optional<StoredToken> parse(const std::string& text) {
299 boost::system::error_code ec;
300 const auto document = boost::json::parse(text, ec);
301 if (ec || !document.is_object()) {
302 return std::nullopt;
303 }
304 const auto* token = document.as_object().if_contains("access_token");
305 const auto* expiry = document.as_object().if_contains("expires_at_ns");
306 if (!token || !token->is_string() || token->as_string().empty() || !expiry || !expiry->is_int64()) {
307 return std::nullopt;
308 }
309 const auto since_epoch = std::chrono::nanoseconds(expiry->as_int64());
310 return StoredToken{
311 std::string(token->as_string()),
312 std::chrono::system_clock::time_point(
313 std::chrono::duration_cast<std::chrono::system_clock::duration>(since_epoch)
314 ),
315 };
316 }

318 // Writes the token to the temporary file, flushes it and renames it over the token file.
319 static std::optional<BrokerError> write_token(
320 const Descriptor& folder, const std::string& identity, const StoredToken& token
321 ) {
322 boost::json::object document;
323 document["access_token"] = token.access_token_;
324 document["expires_at_ns"] =
325 std::chrono::duration_cast<std::chrono::nanoseconds>(token.expires_at_.time_since_epoch()).count();
326 const auto text = boost::json::serialize(document);
327
328 const auto temporary = identity + ".tmp";
329 // A writer killed before its rename left this file; the lock is held, so nobody else writes it.
330 ::unlinkat(folder.get(), temporary.c_str(), 0);
331 Descriptor file(::openat(folder.get(), temporary.c_str(), FILE_FLAGS | O_WRONLY | O_CREAT | O_EXCL, 0600));
332 bool written = file.get() >= 0;
333 for (std::size_t offset = 0; written && offset < text.size();) {
334 const auto count = ::write(file.get(), text.data() + offset, text.size() - offset);
335 if (count > 0) {
336 offset += static_cast<std::size_t>(count);
337 } else if (count == 0 || errno != EINTR) {
338 written = false;
339 }
340 }
341 written = written && ::fsync(file.get()) == 0 &&
342 ::renameat(folder.get(), temporary.c_str(), folder.get(), (identity + ".token").c_str()) == 0;
343 if (!written) {
344 if (file.get() >= 0) {
345 ::unlinkat(folder.get(), temporary.c_str(), 0);
346 }
347 return token_store_error("token store: the token file could not be written");
348 }
349 return std::nullopt;
350 }
