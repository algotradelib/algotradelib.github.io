31// The checks before the identity draw: two prices or more (`few`), the indices, and without `any_sign` every price
32// above zero (`sign`). Returns whether `indices` is the identity draw.
33inline bool identity_draw(
34 std::size_t count, const std::vector<std::size_t>& indices, const std::vector<Decimal64>& prices, bool any_sign,
35 const char* who, const char* few, const char* sign
36) {
37 if (count < 2 || indices.size() > count - 1) {
38 throw refusal(who, few);
39 }
40 bool identity = indices.size() == count - 1;
41 for (std::size_t k = 0; k < indices.size(); ++k) {
42 identity = identity && indices[k] == k;
43 if (indices[k] >= count - 1) {
44 throw refusal(who, "an index names no return");
45 }
46 }
47 for (const auto& price : prices) {
48 if (!any_sign && price.coefficient() <= 0) {
49 throw refusal(who, sign);
50 }
51 }
52 return identity;
53}

55// A price grid in whole units of `10^-scale_`, checked aligned, and the rounding of a path's level onto it.
56class Grid {
57public:
58 // A value in grid units is refused at or beyond 2^53, where a double would no longer hold every integer.
59 static constexpr std::int64_t LIMIT = std::int64_t{1} << 53;
60
61 // The bands at their own scale; `rule` must outlive the grid.
62 Grid(const PriceRule& rule, bool any_sign, const char* who) : rule_(rule), any_sign_(any_sign), who_(who) {
63 if (!rule.grid_known()) {
64 throw refusal(who_, "the price grid is unknown");
65 }
66 build();
67 }
68
69 // The bands again, at a scale covering the values'.
70 void widen(const std::vector<Decimal64>& values) {
71 for (const auto& value : values) {
72 scale_ = std::max(scale_, value.scale());
73 }
74 build();
75 }
76
77 std::int64_t units(const Decimal64& value) const {
78 std::int64_t out = value.coefficient();
79 for (auto step = value.scale();; ++step) {
80 if (out >= LIMIT || out <= -LIMIT) {
81 throw refusal(who_, "a value at or beyond 2^53 grid units");
82 }
83 if (step >= scale_) {
84 return out;
85 }
86 out *= 10;
87 }
88 }
89
90 // The nearest grid price, a tie going up; below the first band only unconstrained with `any_sign`.
91 Decimal64 round(double level, const char* zero) const {
92 constexpr auto limit = static_cast<double>(LIMIT);
93 std::size_t band = 0;
94 while (band + 1 < grid_.size() && static_cast<double>(grid_[band + 1].first) <= level) {
95 ++band;
96 }
97 const bool floored = !any_sign_ || rule_.kind_ != PriceRule::Kind::UNCONSTRAINED;
98 if (
99 !(level < limit) || !(level > -limit) || grid_.empty() ||
100 (floored && static_cast<double>(grid_[band].first) > level)
101 ) {
102 throw refusal(who_, "a level at or beyond 2^53 grid units, or below the first band");
103 }
104 const auto tick = grid_[band].second;
105 auto lower = static_cast<std::int64_t>(std::floor(level / static_cast<double>(tick)));
106 for (;;) {
107 const std::int64_t lo = lower * tick;
108 const std::int64_t hi = lo + tick;
109 if (hi >= LIMIT || lo <= -LIMIT) {
110 throw refusal(who_, "a grid neighbour at or beyond 2^53 grid units");
111 }
112 if (static_cast<double>(lo) > level) {
113 --lower;
114 } else if (static_cast<double>(hi) <= level) {
115 ++lower;
116 } else {
117 const auto nearest = level - static_cast<double>(lo) < static_cast<double>(hi) - level ? lo : hi;
118 if (nearest == 0 && !any_sign_) {
119 throw refusal(who_, zero);
120 }
121 return Decimal64(nearest, scale_);
122 }
123 }
124 }
125
126private:
127 // The bands in units, checked aligned; one unit unconstrained.
128 void build() {
129 grid_.clear();
130 if (rule_.kind_ == PriceRule::Kind::UNCONSTRAINED) {
131 grid_.emplace_back(0, 1);
132 return;
133 }
134 for (const auto& band : rule_.bands_) {
135 scale_ = std::max({scale_, band.from_.scale(), band.tick_.scale()});
136 }
137 for (const auto& band : rule_.bands_) {
138 const auto start = units(band.from_);
139 const auto tick = units(band.tick_);
140 if (tick <= 0 || start % tick != 0 || (!grid_.empty() && start % grid_.back().second != 0)) {
141 throw refusal(who_, "a band's start is not a multiple of its tick and of the previous band's");
142 }
143 grid_.emplace_back(start, tick);
144 }
145 }
146
147 const PriceRule& rule_;
148 bool any_sign_;
149 const char* who_;
150 std::uint8_t scale_{0};
151 std::vector<std::pair<std::int64_t, std::int64_t>> grid_;
152};

268template <typename Join = Returns>
269std::vector<Decimal64> resample_closes(
270 const std::vector<Decimal64>& closes, const PriceRule& rule, const std::vector<std::size_t>& indices,
271 const Join& join = Join{}
272) {
273 static_assert(
274 is_price_join<Join>::value,
275 "a bootstrap join needs the member double join(double level, double from, double value) const"
276 );
277 constexpr bool any_sign = bootstrap_detail::has_any_sign<Join>::value;
278 const bool identity = bootstrap_detail::identity_draw(
279 closes.size(), indices, closes, any_sign, "resample_closes",
280 "needs two closes or more, and at most one index per return", "a close at or below zero"
281 );
282 // The bands checked aligned at their own scale, then at the closes'.
283 bootstrap_detail::Grid grid(rule, any_sign, "resample_closes");
284 if (identity) {
285 return closes;
286 }
287 grid.widen(closes);
288 std::vector<double> in;
289 for (const auto& close : closes) {
290 in.push_back(static_cast<double>(grid.units(close)));
291 }
292 std::vector<Decimal64> out;
293 double level = in.front();
294 for (std::size_t k = 0; k <= indices.size(); ++k) {
295 if (k > 0) {
296 level = join.join(level, in[indices[k - 1]], in[indices[k - 1] + 1]);
297 }
298 out.push_back(grid.round(level, "a close rounded to zero"));
299 }
300 return out;
301}

331std::vector<Bar> resample_bars(
332 const std::vector<Bar>& bars, const PriceRule& rule, const std::vector<std::size_t>& indices,
333 const Join& join = Join{}
334) {
335 static_assert(

338 );
339 constexpr bool any_sign = bootstrap_detail::has_any_sign<Join>::value;
340 std::vector<Decimal64> prices;
341 for (const auto& bar : bars) {
342 if (bar.open_) {
343 prices.push_back(*bar.open_);
344 }
345 prices.insert(prices.end(), {bar.high_, bar.low_, bar.close_});
346 }
347 const bool identity = bootstrap_detail::identity_draw(
348 bars.size(), indices, prices, any_sign, "resample_bars", "needs two bars or more, and at most one index per return",
349 "a price at or below zero"
350 );
351 bootstrap_detail::Grid grid(rule, any_sign, "resample_bars");
352 if (identity) {
353 return bars;
354 }
355 grid.widen(prices);
356 // Every price in bounds, drawn or not, as `resample_closes` reads every close.
357 for (const auto& price : prices) {
358 grid.units(price);
359 }
360 const auto units = [&grid](const Decimal64& value) {
361 return static_cast<double>(grid.units(value));
362 };
363 std::vector<Bar> out;
364 double level = units(bars.front().close_);
365 for (std::size_t k = 0; k <= indices.size(); ++k) {
366 // Bar 0 is the first bar on the grid; bar `k` is day `j + 1` joined from close `j`.
367 const auto& drawn = k == 0 ? bars.front() : bars[indices[k - 1] + 1];
368 const double from = k == 0 ? 0.0 : units(bars[indices[k - 1]].close_);
369 const auto price = [&](const Decimal64& value) {
370 return grid.round(k == 0 ? units(value) : join.join(level, from, units(value)), "a price rounded to zero");
371 };
372 Bar bar = drawn;
373 bar.timestamp_ = bars[k].timestamp_;
374 if (drawn.open_) {
375 bar.open_ = price(*drawn.open_);
376 }
377 bar.high_ = price(drawn.high_);
378 bar.low_ = price(drawn.low_);
379 bar.close_ = price(drawn.close_);
380 if (k > 0) {
381 level = join.join(level, from, units(drawn.close_));
382 bar.traded_value_.reset();
383 }
384 out.push_back(bar);
385 }
386 return out;
387}
