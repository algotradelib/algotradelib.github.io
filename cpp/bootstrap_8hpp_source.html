45 const auto refuse = [](const char* why) {

48 if (closes.size() < 2 || indices.size() > closes.size() - 1) {
49 throw refuse("needs two closes or more, and at most one index per return");
50 }
51 bool identity = indices.size() == closes.size() - 1;
52 for (std::size_t k = 0; k < indices.size(); ++k) {
53 identity = identity && indices[k] == k;
54 if (indices[k] >= closes.size() - 1) {
55 throw refuse("an index names no return");
56 }
57 }
58 for (const auto& close : closes) {
59 if (close.coefficient() <= 0) {
60 throw refuse("a close at or below zero");
61 }
62 }
63 if (!rule.grid_known()) {
64 throw refuse("the price grid is unknown");
65 }
66 std::uint8_t scale = 0;
67 // A value in grid units, refused at or beyond 2^53, where a double would no longer hold every integer.
68 constexpr std::int64_t limit = std::int64_t{1} << 53;
69 const auto units = [&](const Decimal64& value) {
70 std::int64_t out = value.coefficient();
71 for (auto step = value.scale();; ++step) {
72 if (out >= limit || out <= -limit) {
73 throw refuse("a value at or beyond 2^53 grid units");
74 }
75 if (step >= scale) {
76 return out;
77 }
78 out *= 10;
79 }
80 };
81 // The bands in units, checked aligned, at the bands' own scale and again at the closes'; one unit unconstrained.
82 std::vector<std::pair<std::int64_t, std::int64_t>> grid;
83 const auto grid_in_units = [&]() {
84 grid.clear();
85 if (rule.kind_ == PriceRule::Kind::UNCONSTRAINED) {
86 grid.emplace_back(0, 1);
87 return;
88 }
89 for (const auto& band : rule.bands_) {
90 scale = std::max({scale, band.from_.scale(), band.tick_.scale()});
91 }
92 for (const auto& band : rule.bands_) {

94 const auto tick = units(band.tick_);
95 if (tick <= 0 || start % tick != 0 || (!grid.empty() && start % grid.back().second != 0)) {
96 throw refuse("a band's start is not a multiple of its tick and of the previous band's");
97 }
98 grid.emplace_back(start, tick);
99 }
100 };
101 grid_in_units();
102 if (identity) {
103 return closes;
104 }
105 for (const auto& close : closes) {

129 for (;;) {

132 if (hi >= limit) {

134 }
135 if (static_cast<double>(lo) > level) {
