226 void advance_to(std::chrono::system_clock::time_point to) {
227 if (to < now_) {
228 throw std::invalid_argument("the mock market's clock does not move backwards");
229 }
230 if (halted_) {
231 throw std::runtime_error(*halted_);
232 }
233 const auto failed = now_ < to ? record(Change{nullptr, nullptr, to}) : std::nullopt;
234 if (failed) {
235 throw MarketFileError(*failed);
236 }
237 for (;;) {
238 Listing* earliest = nullptr;
239 for (auto& listing : listings_) {
240 if (listing.next_ >= listing.series_.bars_.size()) {
241 continue;
242 }
243 if (earliest == nullptr || period_end(listing, listing.next_) < period_end(*earliest, earliest->next_)) {
244 earliest = &listing;
245 }
246 }
247 if (earliest == nullptr || to < period_end(*earliest, earliest->next_)) {
248 break;
249 }
250 now_ = period_end(*earliest, earliest->next_);
251 apply_bar(*earliest);
252 if (halted_) {
253 throw std::runtime_error(*halted_);
254 }
255 ++earliest->next_;
256 }
257 now_ = to;
258 }

266 Result<BarSeries> bars(const BarQuery& query) const {
267 const auto* listing = find_listing(query.instrument_);
268 if (listing == nullptr) {
269 return refused<BarSeries>("unknown_instrument", "the mock market has no bars of this instrument");
270 }
271 if (query.interval_ != listing->series_.interval_) {
272 return refused<BarSeries>("unsupported_interval", "the mock market has no bars of this interval");
273 }
274 if (query.count_ < 1) {
275 return refused<BarSeries>("bars_count", "a bar request asks for at least one bar");
276 }
277 BarSeries answer;
278 answer.instrument_ = listing->series_.instrument_;
279 answer.interval_ = listing->series_.interval_;
280 answer.stamp_ = listing->series_.stamp_;
281 answer.currency_ = listing->series_.currency_;
282 answer.adjusted_ = listing->series_.adjusted_;
283 answer.received_at_ = now_;
284 const auto& all = listing->series_.bars_;
285 // The bars ended by now and stamped at or before `before_`: a prefix of the series, found by a binary search. A
286 // prefix because the bars are strictly ascending and a bar's period end is its stamp plus a fixed length; a period
287 // end taken from the session would need this search revisited.
288 const auto readable = [this, listing, &query, &all](std::size_t k) {
289 return !(now_ < period_end(*listing, k)) && (!query.before_ || !(*query.before_ < all[k].timestamp_));
290 };
291 std::size_t end = 0;
292 std::size_t high = all.size();
293 while (end < high) {
294 const std::size_t middle = end + (high - end) / 2;
295 if (readable(middle)) {
296 end = middle + 1;
297 } else {
298 high = middle;
299 }
300 }
301 const auto count = static_cast<std::size_t>(query.count_);
302 const std::size_t begin = end > count ? end - count : 0;
303 answer.bars_.assign(
304 all.begin() + static_cast<std::ptrdiff_t>(begin), all.begin() + static_cast<std::ptrdiff_t>(end)
305 );
306 return Result<BarSeries>(std::move(answer));
307 }

459 Result<PlacedOrder<Broker>> place(const OrderRequest& order) {
460 if (halted_) {
461 return refused<PlacedOrder<Broker>>("market_halted", "the mock market halted and takes no orders");
462 }
463 const auto* listing = find_listing(order.instrument_);
464 if (listing == nullptr) {
465 return refused<PlacedOrder<Broker>>("unknown_instrument", "the mock market has no bars of this instrument");
466 }
467 if (order.type_ != OrderType::MARKET && order.type_ != OrderType::LIMIT) {
468 return refused<PlacedOrder<Broker>>(
469 "unsupported_order_type", "the mock market takes market and limit orders only"
470 );
471 }
472 switch (order.time_in_force_) {
473 case TimeInForce::DAY:
474 case TimeInForce::IMMEDIATE_OR_CANCEL:
475 case TimeInForce::FILL_OR_KILL:
476 break;
477 case TimeInForce::UNKNOWN:
478 return refused<PlacedOrder<Broker>>("unsupported_time_in_force", "the mock market needs a time in force");
479 }
480 const auto quantity = decimal_normalized(order.quantity_);
481 if (quantity.coefficient() <= 0 || quantity.scale() != 0) {
482 return refused<PlacedOrder<Broker>>("invalid_quantity", "an order's quantity is a positive whole number");
483 }
484 const bool limit = order.type_ == OrderType::LIMIT;
485 if (limit != order.price_.has_value() || (order.price_ && order.price_->coefficient() <= 0)) {
486 return refused<PlacedOrder<Broker>>(
487 "invalid_price", "a limit order has a positive price and a market order none"
488 );
489 }
490 const auto currency = listing->series_.currency_;
491 std::optional<Decimal64> reserve_price = order.price_;
492 if (!reserve_price) {
493 if (const auto last = last_closed(*listing)) {
494 reserve_price = listing->series_.bars_[*last].close_;
495 }
496 }
497 if (order.side_ == OrderSide::BUY) {
498 if (!reserve_price) {
499 return refused<PlacedOrder<Broker>>("no_price", "no bar of this instrument has ended to price a market buy at");
500 }
501 const auto cash = ledger_.cash_in(currency);
502 const auto spendable = cash ? spendable_in(currency, *cash) : std::nullopt;
503 const auto cost = decimal_multiply(quantity, *reserve_price);
504 if (!spendable || !cost || decimal_less(*spendable, *cost)) {
505 return refused<PlacedOrder<Broker>>("insufficient_cash", "the buy costs more than the buying power");
506 }
507 } else {
508 const auto held = ledger_.quantity_of(order.instrument_).value_or(Decimal64(0, 0));
509 const auto free = decimal_subtract(held, open_sells_of(order.instrument_));
510 if (!free || decimal_less(*free, quantity)) {
511 return refused<PlacedOrder<Broker>>(
512 "insufficient_holdings", "the sell is more than the shares held and not already offered"
513 );
514 }
515 }
516 if (const auto failed = record(Change{&order, nullptr, std::nullopt})) {
517 return not_recorded<PlacedOrder<Broker>>(*failed);
518 }
519 MarketOrder accepted;
520 accepted.snapshot_.order_id_ = std::to_string(++last_id_);
521 accepted.snapshot_.instrument_ = order.instrument_;
522 accepted.snapshot_.side_ = order.side_;
523 accepted.snapshot_.type_ = order.type_;
524 accepted.snapshot_.time_in_force_ = order.time_in_force_;
525 accepted.snapshot_.session_ = order.session_;
526 accepted.snapshot_.venue_ = order.venue_;
527 accepted.snapshot_.status_ = OrderStatus::PENDING;
528 accepted.snapshot_.price_ = order.price_;
529 accepted.snapshot_.quantity_ = quantity;
530 accepted.snapshot_.currency_ = currency;
531 accepted.snapshot_.ordered_at_ = now_;
532 accepted.snapshot_.execution_.filled_quantity_ = Decimal64(0, 0);
533 accepted.reserve_price_ = reserve_price;
534 if (const auto last = last_closed(*listing)) {
535 accepted.decision_price_ = listing->series_.bars_[*last].close_;
536 accepted.decision_price_at_ = period_end(*listing, *last);
537 }
538 orders_.push_back(std::move(accepted));
539 PlacedOrder<Broker> placed;
540 placed.ack_.order_id_ = orders_.back().snapshot_.order_id_;
541 placed.ack_.accepted_at_ = now_;
542 placed.ref_.primary_ = placed.ack_.order_id_;
543 return Result<PlacedOrder<Broker>>(std::move(placed));
544 }

571 }
572 return refused<PlacedOrder<Broker>>("nothing_to_cancel", "no open order has this id");

717 TradingCalendar calendar_of_bars() const {
718 using Days = std::chrono::duration<std::int64_t, std::ratio<86400>>;
719 const auto& session = settings_.session_;
720 const auto local_day = [&](std::chrono::system_clock::time_point at) {
721 return std::chrono::floor<Days>(at + session.utc_offset_).time_since_epoch().count();
722 };
723 TradingCalendar calendar;
724 std::set<std::int64_t> open;
725 for (const auto& listing : listings_) {
726 if (listing.series_.bars_.empty()) {
727 continue;
728 }
729 if (open.empty()) {
730 calendar.country_ = listing.series_.instrument_.country_;
731 }
732 for (const auto& bar : listing.series_.bars_) {
733 open.insert(local_day(bar.timestamp_));
734 }
735 }
736 if (open.empty()) {
737 return calendar;
738 }
739 for (auto day = *open.begin(); day <= *open.rbegin(); ++day) {
740 calendar.days_.push_back(TradingDay{date_from_epoch_days(day), open.count(day) > 0});
741 }
742 return calendar;
743 }

796 std::optional<Decimal64> spendable_in(const Currency& currency, const Decimal64& cash) const {
797 std::optional<Decimal64> left = cash;
798 for (const auto& order : orders_) {
799 if (!left) {
800 break;
801 }
802 if (
803 order.snapshot_.status_ != OrderStatus::PENDING || order.snapshot_.side_ != OrderSide::BUY ||
804 !(order.snapshot_.currency_ == currency) || !order.reserve_price_
805 ) {
806 continue;

809 left = reserved ? decimal_subtract(*left, *reserved) : std::nullopt;
810 }
811 return left;

815 static bool add_to_totals(
816 std::vector<HoldingsTotals>& totals, const Decimal64& market_value, const Decimal64& cost, const Decimal64& gain,
817 const Currency& currency
818 ) {
819 auto found = std::find_if(totals.begin(), totals.end(), [&](const HoldingsTotals& t) {
820 return t.currency_ == currency;
821 });
822 const auto zero = Decimal64(0, 0);
823 const auto purchase = decimal_add(found == totals.end() ? zero : found->purchase_value_, cost);
824 const auto market = decimal_add(found == totals.end() ? zero : found->market_value_, market_value);
825 const auto unrealized = decimal_add(found == totals.end() ? zero : found->unrealized_profit_loss_, gain);
826 if (!purchase || !market || !unrealized) {
827 return false;
828 }
829 if (found == totals.end()) {
830 HoldingsTotals fresh;
831 fresh.currency_ = currency;
832 totals.push_back(fresh);
833 found = totals.end() - 1;
834 }
835 found->purchase_value_ = *purchase;
836 found->market_value_ = *market;
837 found->unrealized_profit_loss_ = *unrealized;
838 return true;
839 }

866 void apply_bar(Listing& listing) {
867 const auto at = listing.next_;
868 const auto& series = listing.series_;
869 const auto& session = settings_.session_;
870 strategy::BarEvent event;
871 event.instrument_ = series.instrument_;
872 event.interval_ = series.interval_;
873 event.stamp_ = series.stamp_;
874 event.bar_ = series.bars_[at];
875 event.period_end_ = period_end(listing, at);
876 const auto start = period_start(listing, at);
877 const auto date = session.local_date(event.bar_.timestamp_);
878 const auto fill_date = session.local_date(event.period_end_);
879 const auto open_date = series.stamp_ == BarStamp::OPEN ? std::optional<Date>(date) : std::nullopt;
880 const auto closes = session.opens_at(event.bar_.timestamp_) - session.open_ + session.close_;
881 const bool ends_session = !bar_interval_length(series.interval_) || !(event.period_end_ < closes);
882 expire_day_orders(listing, date, false, event.period_end_);
883 for (auto& order : orders_) {
884 auto& snapshot = order.snapshot_;
885 if (
886 snapshot.status_ != OrderStatus::PENDING || !(snapshot.instrument_ == series.instrument_) ||
887 start < *snapshot.ordered_at_
888 ) {
889 continue;
890 }
891 if (!order.session_) {
892 order.session_ = date;
893 }
894 strategy::PlannedOrder planned;
895 planned.instrument_ = snapshot.instrument_;
896 planned.side_ = snapshot.side_;
897 planned.type_ = snapshot.type_;
898 planned.time_in_force_ = snapshot.time_in_force_;
899 planned.quantity_ = snapshot.quantity_;
900 planned.price_ = snapshot.price_;
901 const auto decision = settings_.fills_.decide(
902 planned, event, listing.reference_.price_rule_, fill_date, open_date, false, order.decision_price_,
903 order.decision_price_at_, snapshot.ordered_at_
904 );
905 if (decision.costing_error_) {
906 halted_ = "the mock market halted: order " + snapshot.order_id_ + "'s fill could not be costed (" +
907 decision.costing_error_->message_ + ")";
908 return;
909 }
910 if (decision.execution_) {
911 auto execution = *decision.execution_;
912 execution.fill_.order_id_ = snapshot.order_id_;
913 execution.currency_ = snapshot.currency_;
914 if (!ledger_.apply(execution, snapshot.currency_)) {
915 halted_ = "the mock market halted: order " + snapshot.order_id_ + "'s fill does not fit an exact decimal";
916 return;
917 }
918 record_fill(snapshot, execution, event.period_end_);
919 continue;
920 }
921 if (!decision.still_working_) {
922 cancel_at(order, event.period_end_);
923 }
924 }
925 expire_day_orders(listing, date, ends_session, event.period_end_);
926 if (!ledger_.mark(series.instrument_, event.bar_.close_)) {
927 halted_ = "the mock market halted: a position's value at the close of " + series.instrument_.symbol_ +
928 " does not fit an exact decimal";
929 }
930 }

933 static void record_fill(
934 OrderSnapshot& snapshot, const backtest::Execution& execution, std::chrono::system_clock::time_point at
935 ) {
936 const auto& fill = execution.fill_;
937 snapshot.status_ = OrderStatus::FILLED;
938 auto& done = snapshot.execution_;
939 done.filled_quantity_ = fill.quantity_;
940 done.average_filled_price_ = fill.price_;
941 if (const auto amount = decimal_multiply(fill.quantity_, fill.price_)) {
942 done.filled_amount_ = Money{*amount, snapshot.currency_};
943 }
944 auto commission = execution.commission_;
945 if (execution.institution_fee_) {
946 commission = decimal_add(commission, *execution.institution_fee_).value_or(commission);
947 }
948 done.commission_ = Money{commission, snapshot.currency_};
949 done.tax_ = Money{execution.tax_, snapshot.currency_};
950 done.filled_at_ = at;
951 }
