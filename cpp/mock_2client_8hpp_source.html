293 template <typename T, typename Ask>
294 Result<T> read(Call call, std::optional<T>& last, Ask ask) {
295 auto fault = take(call);
296 if (fault && fault->effect_ == Effect::FAIL) {
297 return Result<T>(std::move(fault->error_));
298 }
299 if (fault && last) {
300 auto repeated = *last;
301 repeated.received_at_ = market_->now();
302 return Result<T>(std::move(repeated));
303 }
304 auto answer = ask();
305 if (answer) {
306 last = answer.value();
307 }
308 return answer;
309 }
310
312 std::optional<Fault> take(Call call) {
313 for (auto it = faults_.begin(); it != faults_.end(); ++it) {
314 if (it->call_ == call) {
315 auto fault = std::move(*it);
316 faults_.erase(it);
317 return fault;
318 }
319 }
320 return std::nullopt;
321 }
