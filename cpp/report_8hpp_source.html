331// `summarize`, also handing back the curve's levels and returns as the columns the metrics take (both empty
332// for an empty curve), so `measure_input` keeps them without a second pass.
333inline Report summarize(
334 const ReplayResult& result, const NextBarFillModel& fills, std::optional<math::Scalar> periods_per_year,
335 math::Scalar risk_free, math::Vector& levels, math::Vector& returns
336) {
337 Report report;
338 report.bars_ = result.bars_;
339 report.executions_ = result.executions_.size();
340 report.rejected_ = result.rejected_.size();
341 report.refused_ = result.refused_.size();
342
343 report.caveats_.limit_fill_ = fills.limit_price();
344 report.halted_ = result.halted_;
345 report.aborted_ = result.aborted_;
346 if (fills.commission().is_scheduled()) {
347 report.caveats_.asserted_venue_ = fills.commission().asserted_venue();
348 }
349 report.caveats_.commission_tariff_ = std::string(fills.commission().tariff());
350 report.caveats_.slippage_basis_points_ = fills.slippage().basis_points_;
351 report.caveats_.slippage_ticks_ = fills.slippage().ticks_;
352 report.caveats_.gaps_ = result.gaps_;
353 report.caveats_.unfilled_ = result.unfilled_;
354 report.caveats_.stop_releases_ = result.stop_releases_;
355 report.caveats_.stop_fills_intrabar_ = result.stop_fills_intrabar_;
356 report.caveats_.limit_fills_bounded_ = result.limit_fills_bounded_;
357 report.caveats_.market_price_rule_ = result.market_price_rule_;
358 report.caveats_.market_fills_by_rule_ = result.market_fills_by_rule_;
359 report.caveats_.quotes_ = result.quotes_;
360 report.caveats_.unchecked_fill_causality_ = result.unchecked_fill_causality_;
361 report.caveats_.unvalued_bars_ = result.unvalued_bars_;
362 report.caveats_.first_unvalued_ = result.first_unvalued_;
363 report.caveats_.overdrawn_bars_ = result.overdrawn_bars_;
364 report.caveats_.first_overdrawn_ = result.first_overdrawn_;
365 report.caveats_.lowest_cash_ = result.lowest_cash_;
366 report.caveats_.lowest_cash_at_ = result.lowest_cash_at_;
367 report.caveats_.expired_day_orders_ = result.expired_day_orders_;
368
369 // Costs summed exactly, because they are money -- and **per currency**, because a replay whose
370 // instruments span markets would otherwise have won added to dollars to make one scalar.
371 // Checked, and a refusal is recorded rather than swallowed: leaving the previous total in place
372 // and carrying on produced a plausible figure that was short by an unknown amount, which is the
373 // one outcome every exact-decimal path here exists to prevent. Two individually representable
374 // commissions whose sum does not fit is enough to reach it, and a probe did.
375 bool costs_fit = true;
376 const auto accumulate = [&costs_fit](std::vector<Money>& totals, const Money& amount) {
377 for (auto& held : totals) {
378 if (held.currency_ == amount.currency_) {
379 const auto sum = decimal_add(held.amount_, amount.amount_);
380 if (!sum) {
381 costs_fit = false;
382 return;
383 }
384 held.amount_ = *sum;
385 return;
386 }
387 }
388 totals.push_back(amount);
389 };
390
391 std::vector<InstrumentId> seen;
392 for (const auto& execution : result.executions_) {
393 accumulate(report.commission_by_currency_, Money{execution.commission_, execution.currency_});
394 if (execution.institution_fee_) {
395 accumulate(report.institution_by_currency_, Money{*execution.institution_fee_, execution.currency_});
396 } else {
397 // The rate stated one figure, so this execution's commission is inclusive and the split total
398 // below cannot be completed. Counted rather than approximated.
399 ++report.caveats_.inclusive_commission_executions_;
400 }
401 // The rate that charged it, collapsed so an unchanged rate is one row with a count rather than
402 // one row per fill. Accumulated here, from the executions the ledger accepted, rather than from
403 // the schedule: a row the replay never used is not a rate this run applied.
404 bool counted = false;
405 for (auto& applied : report.caveats_.applied_rates_) {
406 if (applied == execution.applied_) {
407 ++applied.executions_;
408 counted = true;
409 break;
410 }
411 }
412 if (!counted) {
413 auto entry = execution.applied_;
414 entry.executions_ = 1;
415 report.caveats_.applied_rates_.push_back(std::move(entry));
416 }
417 // The tax, totalled per currency, over the executions in the market it was stated for: another
418 // market's fills pay none by construction, and counting them would add a zero row in their
419 // currency and cost a single-currency total its scalar.
420 if (fills.sell_tax() && execution.fill_.instrument_.country_ == fills.sell_tax()->country_) {
421 accumulate(report.tax_by_currency_, Money{execution.tax_, execution.currency_});
422 }
423 // Slippage is a per-share price difference, so its cost is that times the quantity -- and it
424 // is totalled per currency like every other cost, because a replay can hold fills in more
425 // than one and adding them would produce a figure denominated in nothing.
426 if (const auto cost = decimal_multiply(execution.slippage_, execution.fill_.quantity_)) {
427 accumulate(report.slippage_by_currency_, Money{*cost, execution.currency_});
428 } else {
429 costs_fit = false;
430 }
431 bool known = false;
432 for (const auto& id : seen) {
433 if (id == execution.fill_.instrument_) {
434 known = true;
435 break;
436 }
437 }
438 if (!known) {
439 seen.push_back(execution.fill_.instrument_);
440 }
441 }
442 report.caveats_.cost_totals_incomplete_ = !costs_fit;
443 // A scalar is stated only where one can be true: one currency, something executed, and every
444 // addition representable. On a mixed-currency replay the per-currency vectors are the whole
445 // answer, and on an arithmetic refusal nothing is stated at all rather than a figure short by an
446 // amount nobody can name.
447 if (costs_fit && report.commission_by_currency_.size() == 1) {
448 report.commission_paid_ = report.commission_by_currency_.front().amount_;
449 }
450 // Present only when every charged rate split the two components, so the total is complete rather
451 // than a sum over the subset that happened to state one.
452 if (
453 costs_fit && report.caveats_.inclusive_commission_executions_ == 0 && report.institution_by_currency_.size() == 1
454 ) {
455 report.institution_fees_paid_ = report.institution_by_currency_.front().amount_;
456 }
457 if (costs_fit && report.slippage_by_currency_.size() == 1) {
458 report.slippage_paid_ = report.slippage_by_currency_.front().amount_;
459 }
460 report.caveats_.sell_tax_ = fills.sell_tax();
461 if (fills.sell_tax() && costs_fit && report.tax_by_currency_.size() == 1) {
462 report.tax_paid_ = report.tax_by_currency_.front().amount_;
463 }
464 report.caveats_.multiple_instruments_ = seen.size() > 1;
465 for (const auto& execution : result.executions_) {
466 if (!(execution.currency_ == result.currency_)) {
467 ++report.caveats_.other_currency_executions_;
468 }
469 }
470
471 if (result.equity_.empty()) {
472 // Nothing to measure. Every figure stays unobserved rather than becoming zero, which would read as
473 // a flat result rather than as an absent one.
474 report.total_return_ = math::missing();
475 report.sharpe_ = math::missing();
476 report.annualized_sharpe_ = math::missing();
477 report.sortino_ = math::missing();
478 report.max_drawdown_ = math::missing();
479 report.total_turnover_ = math::missing();
480 return report;
481 }
482
483 report.opening_equity_ = result.equity_.front().equity_;
484 report.closing_equity_ = result.equity_.back().equity_;
485
486 // The returns and the equity curve, as the columns the metrics take.
487 returns.resize(static_cast<math::Index>(result.equity_.size()));
488 levels.resize(static_cast<math::Index>(result.equity_.size()));
489 for (std::size_t at = 0; at < result.equity_.size(); ++at) {
490 returns[static_cast<math::Index>(at)] = result.equity_[at].period_return_;
491 levels[static_cast<math::Index>(at)] = analytics::to_scalar(result.equity_[at].equity_);
492 }
493
494 // From a negative opening the ratio's sign reverses (a replay may start with negative cash), so a
495 // non-positive opening has no return and no turnover, as a zero one has none.
496 const auto opening = analytics::to_scalar(report.opening_equity_);
497 report.total_return_ = !(opening > math::Scalar{0}) ?
498 math::undefined() :
499 analytics::to_scalar(report.closing_equity_) / opening - math::Scalar{1};
500
501 report.sharpe_ = analytics::sharpe(returns, risk_free);
502 report.sortino_ = analytics::sortino(returns, risk_free);
503 // Equity at or below zero before the end leaves the periods after it without a return
504 // (`EquityPoint::period_return_`), so no ratio describes the run.
505 for (std::size_t at = 0; at + 1 < result.equity_.size(); ++at) {
506 if (result.equity_[at].equity_.coefficient() <= 0) {
507 report.sharpe_ = math::undefined();
508 report.sortino_ = math::undefined();
509 break;
510 }
511 }
512 report.max_drawdown_ = analytics::max_drawdown(levels).magnitude_;
513
514 report.annualized_sharpe_ =
515 periods_per_year ? analytics::annualize_ratio(report.sharpe_, *periods_per_year) : math::missing();
516
517 // Every trade outside the curve: it holds only the opening cash, and its figures describe no trading.
518 const auto outside = report.caveats_.other_currency_executions_;
519 if (outside > 0 && outside == result.executions_.size()) {
520 report.total_return_ = math::missing();
521 report.sharpe_ = math::missing();
522 report.annualized_sharpe_ = math::missing();
523 report.sortino_ = math::missing();
524 report.max_drawdown_ = math::missing();
525 }
526
527 // Turnover from the executions rather than from a weight matrix, because a replay knows what it
528 // actually traded. One-way: the fraction of equity that changed hands.
529 // Same money on both sides, or no figure at all.
530 report.total_turnover_ = !(opening > math::Scalar{0}) ? math::undefined() : math::Scalar{0};
531 if (outside > 0) {
532 report.total_turnover_ = math::missing();
533 }
534 if (math::observed(report.total_turnover_)) {
535 for (const auto& execution : result.executions_) {
536 const auto traded = decimal_multiply(execution.fill_.quantity_, execution.fill_.price_);
537 if (traded) {
538 report.total_turnover_ += analytics::to_scalar(*traded) / opening;
539 }
540 }
541 }
542
543 return report;
544}

650inline ReplayResult chain_results(const std::vector<const ReplayResult*>& results) {
651 if (results.empty()) {
652 throw std::invalid_argument("chain_results needs a result");
653 }
654 ReplayResult out;
655 for (std::size_t f = 0; f < results.size(); ++f) {
656 const ReplayResult* fold = results[f];
657 const auto refused = "chain_results: fold " + std::to_string(f) + " ";
658 if (fold == nullptr || fold->halted_ || fold->equity_.empty()) {
659 throw std::invalid_argument(refused + "is null, halted, or has no curve point");
660 }
661 if (f == 0) {
662 out.currency_ = fold->currency_;
663 out.equity_ = fold->equity_;
664 } else {
665 const auto close = out.equity_.back();
666 const auto& first = fold->equity_.front();
667 const auto from = analytics::to_scalar(first.equity_);
668 const auto to = analytics::to_scalar(close.equity_);
669 const bool capital = first.equity_.coefficient() > 0 && close.equity_.coefficient() > 0;
670 if (!(fold->currency_ == out.currency_) || !(close.at_ < first.at_) || !capital) {
671 throw std::invalid_argument(refused + "differs in currency, overlaps the previous one, or follows no capital");
672 }
673 for (std::size_t at = 0; at < fold->equity_.size(); ++at) {
674 auto point = fold->equity_[at];
675 const auto value = analytics::to_scalar(point.equity_) * (to / from);
676 // As many places as fifteen significant digits need (`to_decimal` multiplies by ten as often), up to 255.
677 std::uint8_t places = 0;
678 for (auto magnitude = std::abs(value); magnitude != 0 && magnitude < 1e15 && places < 255; ++places) {
679 magnitude *= 10;
680 }
681 const auto scaled = at == 0 ? std::optional<Decimal64>(close.equity_) : analytics::to_decimal(value, places);
682 // Nonzero is read from the decimal, not `value`: the scale is positive, and the product can underflow to 0.
683 if (!scaled || (point.equity_.coefficient() != 0 && scaled->coefficient() == 0)) {
684 throw std::invalid_argument(refused + "has an equity that does not fit a decimal once scaled");
685 }
686 point.equity_ = *scaled;
687 point.period_return_ = at == 0 ? math::missing() : point.period_return_;
688 out.equity_.push_back(point);
689 }
690 }
691 out.executions_.insert(out.executions_.end(), fold->executions_.begin(), fold->executions_.end());
692 out.rejected_.insert(out.rejected_.end(), fold->rejected_.begin(), fold->rejected_.end());
693 out.refused_.insert(out.refused_.end(), fold->refused_.begin(), fold->refused_.end());
694 for (
695 auto [total, count] : std::initializer_list<std::pair<std::size_t*, std::size_t>>{
696 {&out.gaps_, fold->gaps_}, {&out.bars_, fold->bars_}, {&out.unfilled_, fold->unfilled_},
697 {&out.aborted_, fold->aborted_}, {&out.quotes_, fold->quotes_}, {&out.records_, fold->records_},
698 {&out.unchecked_fill_causality_, fold->unchecked_fill_causality_}, {&out.unvalued_bars_, fold->unvalued_bars_},
699 {&out.stop_releases_, fold->stop_releases_}, {&out.stop_fills_intrabar_, fold->stop_fills_intrabar_},
700 {&out.limit_fills_bounded_, fold->limit_fills_bounded_},
701 {&out.market_fills_by_rule_, fold->market_fills_by_rule_}, {&out.overdrawn_bars_, fold->overdrawn_bars_},
702 {&out.expired_day_orders_, fold->expired_day_orders_}, {&out.history_bars_, fold->history_bars_}
703 }
704 ) {
705 *total += count;
706 }
707 out.first_unvalued_ = out.first_unvalued_ ? out.first_unvalued_ : fold->first_unvalued_;
708 out.first_overdrawn_ = out.first_overdrawn_ ? out.first_overdrawn_ : fold->first_overdrawn_;
709 out.market_price_rule_ = out.market_price_rule_.empty() ? fold->market_price_rule_ : out.market_price_rule_;
710 if (
711 fold->lowest_cash_ && (!out.lowest_cash_ || decimal_less(fold->lowest_cash_->amount_, out.lowest_cash_->amount_))
712 ) {
713 out.lowest_cash_ = fold->lowest_cash_;
714 out.lowest_cash_at_ = fold->lowest_cash_at_;
715 }
716 }
717 return out;
718}
