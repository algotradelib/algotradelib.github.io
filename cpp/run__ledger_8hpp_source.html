181 bool book(
182 const InstrumentId& instrument, OrderSide side, const Decimal64& filled, const Money& amount, const Decimal64& fee
183 ) {
184 const auto next = booked(instrument, side, filled, amount, fee);
185 if (!next) {
186 return false;
187 }
188 const auto at = find(instrument);
189 if (at) {
190 shares_[*at].quantity_ = next->second;
191 } else {
192 add(instrument, next->second);
193 }
194 cash_.amount_ = next->first;
195 return true;
196 }

257 // Applies a deposit or a withdrawal; `Runner` refuses a withdrawal past the cash before it is written.
258 bool transfer(const RunTransfer& transfer) {
259 if (
260 transfer.id_.empty() || transfer_of(transfer.id_) != nullptr ||
261 !(transfer.amount_.currency_ == cash_.currency_) || !decimal_less(Decimal64::parse("0"), transfer.amount_.amount_)
262 ) {
263 return false;
264 }
265 const auto cash = transfer.withdrawal_ ?
266 decimal_subtract(cash_.amount_, transfer.amount_.amount_) :
267 decimal_add(cash_.amount_, transfer.amount_.amount_);
268 if (!cash) {
269 return false;
270 }
271 transfers_.push_back(transfer);
272 cash_.amount_ = *cash;

318 // Books a placement released without a booking, as a `RunRecordKind::FILL` record does.
319 bool book_released(const std::string& id, const Decimal64& filled, const Money& amount, const Decimal64& fee) {
320 if (!can_book_released(id, filled, amount, fee)) {
321 return false;

328 // The cash and the instrument's shares after a fill, or nothing if `book` refuses it.
329 std::optional<std::pair<Decimal64, Decimal64>> booked(
330 const InstrumentId& instrument, OrderSide side, const Decimal64& filled, const Money& amount, const Decimal64& fee
331 ) const {
332 const auto zero = Decimal64::parse("0");
333 if (
334 !(amount.currency_ == cash_.currency_) || decimal_less(filled, zero) || decimal_less(amount.amount_, zero) ||
335 decimal_less(fee, zero)
336 ) {
337 return std::nullopt;
338 }
339 const auto at = find(instrument);
340 const Decimal64 held = at ? shares_[*at].quantity_ : zero;
341 std::optional<Decimal64> shares;
342 std::optional<Decimal64> cash;
343 if (side == OrderSide::BUY) {
344 shares = decimal_add(held, filled);
345 const auto spent = decimal_add(amount.amount_, fee);
346 cash = spent ? decimal_subtract(cash_.amount_, *spent) : std::nullopt;
347 } else {
348 if (decimal_less(held, filled)) {
349 return std::nullopt;
350 }
351 shares = decimal_subtract(held, filled);
352 const auto returned = decimal_subtract(amount.amount_, fee);
353 cash = returned ? decimal_add(cash_.amount_, *returned) : std::nullopt;
354 }
355 if (!shares || !cash) {
356 return std::nullopt;
357 }
358 return std::make_pair(*cash, *shares);
359 }

361 // Moves a placement to `state`, recording it if new; refused once it is booked (a released one is booked only by
362 // `book_released`).
363 bool settle_state(const RunPlacement& placement, RunPlacementState state) {
364 const auto at = placement_at_.find(placement.id_);
365 if (at == placement_at_.end()) {
366 auto added = placement;
367 added.state_ = state;
368 placement_at_.emplace(added.id_, placements_.size());
369 placements_.push_back(std::move(added));
370 unsettled_ += state != RunPlacementState::BOOKED ? 1 : 0;
371 return true;
372 }
373 auto& known = placements_[at->second];
374 if (known.state_ == RunPlacementState::BOOKED) {
375 return false;
376 }
377 // Leaving `HELD` or `UNBOOKED` for `BOOKED` settles it.
378 if (state == RunPlacementState::BOOKED) {
379 --unsettled_;
380 }
381 known.state_ = state;
382 return true;
383 }

385 // One instrument's shares.
386 struct Shares {
387 InstrumentId instrument_;
388 Decimal64 quantity_;
389 };

391 // The instrument at a position of `shares_`, for the table.
392 struct IdOf {
393 const std::vector<Shares>* shares_;
394 const InstrumentId& operator()(std::size_t at) const noexcept {
395 return (*shares_)[at].instrument_;
396 }
397 };
398
399 // Where `instrument` sits in `shares_`: a scan up to `id_detail::scan_limit` rows, the table beyond.
400 std::optional<std::size_t> find(const InstrumentId& instrument) const noexcept {
401 if (table_.empty()) {
402 for (std::size_t at = 0; at < shares_.size(); ++at) {
403 if (shares_[at].instrument_ == instrument) {
404 return at;
405 }
406 }
407 return std::nullopt;
408 }
409 return table_.find(instrument, IdOf{&shares_});
410 }
411
412 // Appends a row; if the table cannot take it, the row is removed again and the exception passes on.
413 void add(const InstrumentId& instrument, const Decimal64& quantity) {
414 shares_.push_back(Shares{instrument, quantity});
415 try {
416 if (!table_.empty()) {
417 table_.insert(instrument, shares_.size() - 1, IdOf{&shares_});
418 } else if (shares_.size() > id_detail::scan_limit) {
419 id_detail::PositionTable built;
420 for (std::size_t at = 0; at < shares_.size(); ++at) {
421 built.insert(shares_[at].instrument_, at, IdOf{&shares_});
422 }
423 table_ = std::move(built);
424 }
425 } catch (...) {
426 shares_.pop_back();
427 throw;
428 }
429 }

454inline run_detail::Fold RunLedger::fold(const JournalReplay& replay) {
455 run_detail::Fold folded;
456 std::optional<std::size_t> opened;
457 for (std::size_t at = 0; at < replay.entries_.size(); ++at) {
458 const auto& entry = replay.entries_[at];
459 if (entry.action_ == JournalAction::LEDGER && entry.run_ && entry.run_->kind_ == RunRecordKind::OPEN) {
460 opened = at;
461 }
462 }
463 if (!opened) {
464 return folded;
465 }
466 const auto& open = *replay.entries_[*opened].run_;
467 if (!open.amount_ || open.run_ != replay.entries_[*opened].id_) {
468 folded.failure_ = "the run's opening line " + replay.entries_[*opened].id_ + " is incomplete";
469 return folded;
470 }
471 // A line of the run before its latest opening: the opening repeated, which would restart the run at its budget.
472 for (std::size_t at = 0; at < *opened; ++at) {
473 if (replay.entries_[at].run_ && replay.entries_[at].run_->run_ == open.run_) {
474 folded.failure_ = "the run's opening line " + open.run_ + " is repeated";
475 return folded;
476 }
477 }
478 RunLedger run(open.run_, *open.amount_);
479 const auto fail = [&folded](const JournalEntry& entry, const char* why) {
480 folded.failure_ = "the run's journal line " + entry.id_ + " " + why;
481 return std::move(folded);
482 };
483 for (std::size_t at = *opened + 1; at < replay.entries_.size(); ++at) {
484 const auto& entry = replay.entries_[at];
485 if (!entry.run_ || entry.run_->run_ != run.id_) {
486 continue;
487 }
488 const auto& record = *entry.run_;
489 const RunPlacement placement{entry.id_, entry.instrument_, entry.side_, entry.quantity_, RunPlacementState::HELD};
490 // What the file reader guarantees, checked again for a journal held in memory.
491 const bool whole = entry.action_ == JournalAction::LEDGER ?
492 record.kind_ && record.amount_ && (*record.kind_ != RunRecordKind::FILL || (record.filled_ && record.fee_)) :
493 !record.filled_ || (record.amount_ && record.fee_);
494 if (!whole) {
495 return fail(entry, "is incomplete");
496 }
497 if (entry.action_ == JournalAction::LEDGER) {
498 if (*record.kind_ == RunRecordKind::OPEN) {
499 return fail(entry, "opens its run a second time");
500 }
501 if (*record.kind_ == RunRecordKind::FILL) {
502 if (!run.book_released(record.of_, *record.filled_, *record.amount_, *record.fee_)) {
503 return fail(entry, "books a placement that was not released unbooked, or a fill it cannot take");
504 }
505 } else if (
506 !run.transfer(RunTransfer{record.given_, *record.amount_, *record.kind_ == RunRecordKind::WITHDRAWAL})
507 ) {
508 return fail(entry, "repeats a transfer id, or is in another currency, or does not fit");
509 }
510 continue;
511 }
512 if (entry.action_ != JournalAction::PLACE) {
513 continue;
514 }
515 if (entry.outcome_ == JournalOutcome::HELD) {
516 if (!run.hold(placement)) {
517 return fail(entry, "holds a placement already booked");
518 }
519 } else if (entry.outcome_ == JournalOutcome::CONFIRMED) {
520 const bool booked = record.filled_.has_value();
521 if (
522 booked ?
523 !run.book_placement(placement, *record.filled_, *record.amount_, *record.fee_) :
524 !run.release(placement)
525 ) {
526 return fail(entry, "accounts for a placement twice, or books a fill the run cannot take");
527 }
528 }
529 }
530 folded.run_ = std::move(run);
531 return folded;
532}
