261 static Result<KiwoomAmendment> to_amendment(const Ref& ref, const AmendInstruction& instruction, bool cancel) {
262 BrokerError error;
263 error.category_ = ErrorCategory::INVALID_ARGUMENT;
264 error.reached_broker_ = false;
265
266 if (!instruction.all_remaining_ && !instruction.quantity_) {
267 error.broker_code_ = "quantity_required";
268 error.message_ =
269 "a Kiwoom modification or cancellation affects a stated quantity; set all_remaining_ to "
270 "act on everything still resting, or build the instruction with cancel_all()";
271 return Result<KiwoomAmendment>(std::move(error));
272 }
273 if (auto refusal = refuse_shape_change(instruction, cancel)) {
274 return Result<KiwoomAmendment>(std::move(*refusal));
275 }
276
277 KiwoomAmendment amendment;
278 amendment.order_ = from_ref(ref);
279 amendment.venue_ = instruction.venue_;
280 // Kiwoom's own "everything still resting", which it evaluates when it processes the
281 // instruction rather than when the caller read the listing.
282 amendment.quantity_ = instruction.all_remaining_ ? Decimal64() : *instruction.quantity_;
283 if (!cancel) {
284 // The basis is known here and nowhere after: `kt10002` carries no `trde_tp`, so this is the
285 // last point that can tell a trigger the order wants from one it would silently lose.
286 const auto trigger = amended_trigger_price(instruction);
287 if (auto invalid = kiwoom::invalid_trigger(instruction.shape_.type_, trigger)) {
288 return Result<KiwoomAmendment>(std::move(*invalid));
289 }
290 amendment.price_ = instruction.shape_.price_;
291 amendment.condition_price_ = trigger;
292 }
293 // **Never on a cancellation.** `kt10003` has no trigger field and the client refuses a
294 // cancellation carrying one; `cancel_all` restates the resting trigger, so copying it here would
295 // make every stop order impossible to withdraw through the port.
296 return Result<KiwoomAmendment>(std::move(amendment));
297 }

311 static std::optional<BrokerError> refuse_shape_change(const AmendInstruction& instruction, bool cancel) {
312 BrokerError error;
313 error.category_ = ErrorCategory::INVALID_ARGUMENT;
314 error.reached_broker_ = false;
315 error.broker_code_ = "shape_change_unsupported";
316
317 const auto refuse = [&error, cancel](const char* what) {
318 error.message_ = cancel ?
319 std::string(
320 "Kiwoom's cancellation carries a quantity and nothing else, so it cannot "
321 "change an order's "
322 ) +
323 what + " on the way out" :
324 std::string(
325 "Kiwoom's amendment endpoints carry a quantity and a price only, so they "
326 "cannot change an order's "
327 ) +
328 what + "; cancel the order and place a new one instead";
329 return error;
330 };
331
332 // An unset field means "leave it as it is", so it is never a change. A set field is a change
333 // only when it differs from what the order already is -- and a set field with no resting shape
334 // to compare against cannot be shown to be a restatement, so it is refused rather than assumed
335 // harmless.
336 const auto differs = [](auto stated, auto current) {
337 return stated.has_value() && (!current.has_value() || *stated != *current);
338 };
339
340 if (cancel) {
341 // **A cancellation transmits a quantity and nothing else**, so every other stated field is
342 // one Kiwoom cannot act on. The price and the trigger are refused because acting on them is
343 // what a caller would expect; these three are refused for the same reason and it is not a
344 // lesser one -- reporting acceptance for a stated change that never reached the broker is
345 // the defect, whatever the field.
346 //
347 // `cancel_all()` and `cancel_quantity()` copy the resting order's whole shape into both
348 // `shape_` and `resting_`, so every comparison below sees a restatement and the ordinary
349 // withdrawal path is unaffected.
350 const bool restates_price = instruction.resting_ && instruction.resting_->price_ == instruction.shape_.price_;
351 if (instruction.shape_.price_ && !restates_price) {
352 error.broker_code_ = "price_on_cancellation";
353 error.message_ =
354 "Kiwoom's cancellation carries a quantity and no price, so it cannot reprice an order on "
355 "the way out; cancel it and place a new one instead";
356 return error;
357 }
358 // The same rule as the price: a restatement of the resting trigger -- which is what
359 // `cancel_all` writes -- is the ordinary withdrawal, and only a stated *change* is refused.
360 const bool restates_trigger =
361 instruction.resting_ && instruction.resting_->trigger_price_ == instruction.shape_.trigger_price_;
362 if (instruction.shape_.trigger_price_ && !restates_trigger) {
363 error.broker_code_ = "trigger_on_cancellation";
364 error.message_ =
365 "Kiwoom's cancellation carries a quantity and no trigger price, so it cannot move an "
366 "order's stop on the way out; cancel it and place a new one instead";
367 return error;
368 }
369 } else if (!instruction.resting_) {
370 // **A modification with no resting shape is refused, and that is the conservative
371 // direction.** `OrderShape::type_` is a plain value defaulting to `LIMIT`, so this adapter
372 // cannot tell a caller who *means* `LIMIT` from one who left it alone -- and if the resting
373 // order is a best-limit order, accepting the instruction would send a price and leave the
374 // basis where it was, reporting success for a change that did not happen. KIS refuses the
375 // same case for the same reason, as `incomplete_shape`.
376 error.broker_code_ = "incomplete_shape";
377 error.message_ =
378 "Kiwoom's modification cannot change an order's price basis, validity condition or session, "
379 "and without the order's current shape this adapter cannot tell a restatement from a "
380 "change; build the instruction with reprice() or replace_with(), which take it from the "
381 "resting order";
382 return error;
383 }
384
385 const auto resting_condition =
386 instruction.resting_ ? instruction.resting_->time_in_force_ : std::optional<TimeInForce>();
387 const auto resting_session = instruction.resting_ ? instruction.resting_->session_ : std::optional<OrderSession>();
388 if (differs(instruction.shape_.time_in_force_, resting_condition)) {
389 return refuse("validity condition");
390 }
391 if (differs(instruction.shape_.session_, resting_session)) {
392 return refuse("session");
393 }
394 // `OrderShape::type_` is a plain value rather than an optional, so "left alone" and "stated
395 // `LIMIT`" are the same bytes and the comparison has to be made in two steps.
396 if (instruction.resting_) {
397 if (instruction.shape_.type_ != instruction.resting_->type_) {
398 return refuse("price basis");
399 }
400 } else if (instruction.shape_.type_ != OrderShape().type_) {
401 // **With no resting shape, a value that differs from the default is still a stated change.**
402 // Only the default is ambiguous; `MARKET` on a cancellation is a caller saying something, and
403 // `kt10003` transmits no basis, so accepting it would report acceptance for a change the
404 // broker never saw. Compared against `OrderShape`'s own default rather than a literal, so the
405 // two cannot drift apart.
406 return refuse("price basis");
407 }
408 return std::nullopt;
409 }

441 static Result<KiwoomHistoryQuery> to_history_query(const KiwoomAccount&, const OrderHistoryQuery& query) {
442 BrokerError error;
443 error.category_ = ErrorCategory::INVALID_ARGUMENT;
444 error.reached_broker_ = false;
445 if (!query.from_) {
446 error.broker_code_ = "date_required";
447 error.message_ = "Kiwoom's order-history inquiry lists one calendar date, which must be given";
448 return Result<KiwoomHistoryQuery>(std::move(error));
449 }
450 if (
451 query.to_ &&
452 !(query.to_->year_ == query.from_->year_ && query.to_->month_ == query.from_->month_ &&
453 query.to_->day_ == query.from_->day_)
454 ) {
455 error.broker_code_ = "single_date_only";
456 error.message_ =
457 "Kiwoom's order-history inquiry lists one calendar date; ask for each date separately "
458 "rather than for a range";
459 return Result<KiwoomHistoryQuery>(std::move(error));
460 }
461 if (query.cursor_) {
462 error.broker_code_ = "cursor_unsupported";
463 error.message_ = "Kiwoom order history pages internally and exposes no cursor";
464 return Result<KiwoomHistoryQuery>(std::move(error));
465 }
466 if (query.limit_) {
467 error.broker_code_ = "limit_unsupported";
468 error.message_ = "Kiwoom order history documents no row limit and cannot honor one";
469 return Result<KiwoomHistoryQuery>(std::move(error));
470 }
471 KiwoomHistoryQuery converted;
472 converted.date_ = *query.from_;
473 converted.symbol_ = query.symbol_ ? *query.symbol_ : std::string();
474 return Result<KiwoomHistoryQuery>(std::move(converted));
475 }
