265 static Result<KiwoomAmendment> to_amendment(const Ref& ref, const AmendInstruction& instruction, bool cancel) {
266 BrokerError error;
267 error.category_ = ErrorCategory::INVALID_ARGUMENT;
268 error.reached_broker_ = false;
269
270 if (!instruction.all_remaining_ && !instruction.quantity_) {
271 error.broker_code_ = "quantity_required";
272 error.message_ =
273 "a Kiwoom modification or cancellation affects a stated quantity; set all_remaining_ to "
274 "act on everything still resting, or build the instruction with cancel_all()";
275 return Result<KiwoomAmendment>(std::move(error));
276 }
277 if (auto refusal = refuse_shape_change(instruction, cancel)) {
278 return Result<KiwoomAmendment>(std::move(*refusal));
279 }
280
281 KiwoomAmendment amendment;
282 amendment.order_ = from_ref(ref);
283 amendment.venue_ = instruction.venue_;
284 // Kiwoom's own "everything still resting", which it evaluates when it processes the
285 // instruction rather than when the caller read the listing.
286 amendment.quantity_ = instruction.all_remaining_ ? Decimal64() : *instruction.quantity_;
287 if (!cancel) {
288 // The basis is known here and nowhere after: `kt10002` carries no `trde_tp`, so this is the
289 // last point that can tell a trigger the order wants from one it would silently lose.
290 const auto trigger = amended_trigger_price(instruction);
291 if (auto invalid = kiwoom::invalid_trigger(instruction.shape_.type_, trigger)) {
292 return Result<KiwoomAmendment>(std::move(*invalid));
293 }
294 amendment.price_ = instruction.shape_.price_;
295 amendment.condition_price_ = trigger;
296 }
297 // **Never on a cancellation.** `kt10003` has no trigger field and the client refuses a
298 // cancellation carrying one; `cancel_all` restates the resting trigger, so copying it here would
299 // make every stop order impossible to withdraw through the port.
300 return Result<KiwoomAmendment>(std::move(amendment));
301 }

315 static std::optional<BrokerError> refuse_shape_change(const AmendInstruction& instruction, bool cancel) {
316 BrokerError error;
317 error.category_ = ErrorCategory::INVALID_ARGUMENT;
318 error.reached_broker_ = false;
319 error.broker_code_ = "shape_change_unsupported";
320
321 const auto refuse = [&error, cancel](const char* what) {
322 error.message_ = cancel ?
323 std::string(
324 "Kiwoom's cancellation carries a quantity and nothing else, so it cannot "
325 "change an order's "
326 ) +
327 what + " on the way out" :
328 std::string(
329 "Kiwoom's amendment endpoints carry a quantity and a price only, so they "
330 "cannot change an order's "
331 ) +
332 what + "; cancel the order and place a new one instead";
333 return error;
334 };
335
336 // An unset field means "leave it as it is", so it is never a change. A set field is a change
337 // only when it differs from what the order already is -- and a set field with no resting shape
338 // to compare against cannot be shown to be a restatement, so it is refused rather than assumed
339 // harmless.
340 const auto differs = [](auto stated, auto current) {
341 return stated.has_value() && (!current.has_value() || *stated != *current);
342 };
343
344 if (cancel) {
345 // **A cancellation transmits a quantity and nothing else**, so every other stated field is
346 // one Kiwoom cannot act on. The price and the trigger are refused because acting on them is
347 // what a caller would expect; these three are refused for the same reason and it is not a
348 // lesser one -- reporting acceptance for a stated change that never reached the broker is
349 // the defect, whatever the field.
350 //
351 // `cancel_all()` and `cancel_quantity()` copy the resting order's whole shape into both
352 // `shape_` and `resting_`, so every comparison below sees a restatement and the ordinary
353 // withdrawal path is unaffected.
354 const bool restates_price = instruction.resting_ && instruction.resting_->price_ == instruction.shape_.price_;
355 if (instruction.shape_.price_ && !restates_price) {
356 error.broker_code_ = "price_on_cancellation";
357 error.message_ =
358 "Kiwoom's cancellation carries a quantity and no price, so it cannot reprice an order on "
359 "the way out; cancel it and place a new one instead";
360 return error;
361 }
362 // The same rule as the price: a restatement of the resting trigger -- which is what
363 // `cancel_all` writes -- is the ordinary withdrawal, and only a stated *change* is refused.
364 const bool restates_trigger =
365 instruction.resting_ && instruction.resting_->trigger_price_ == instruction.shape_.trigger_price_;
366 if (instruction.shape_.trigger_price_ && !restates_trigger) {
367 error.broker_code_ = "trigger_on_cancellation";
368 error.message_ =
369 "Kiwoom's cancellation carries a quantity and no trigger price, so it cannot move an "
370 "order's stop on the way out; cancel it and place a new one instead";
371 return error;
372 }
373 } else if (!instruction.resting_) {
374 // **A modification with no resting shape is refused, and that is the conservative
375 // direction.** `OrderShape::type_` is a plain value defaulting to `LIMIT`, so this adapter
376 // cannot tell a caller who *means* `LIMIT` from one who left it alone -- and if the resting
377 // order is a best-limit order, accepting the instruction would send a price and leave the
378 // basis where it was, reporting success for a change that did not happen. KIS refuses the
379 // same case for the same reason, as `incomplete_shape`.
380 error.broker_code_ = "incomplete_shape";
381 error.message_ =
382 "Kiwoom's modification cannot change an order's price basis, validity condition or session, "
383 "and without the order's current shape this adapter cannot tell a restatement from a "
384 "change; build the instruction with reprice() or replace_with(), which take it from the "
385 "resting order";
386 return error;
387 }
388
389 const auto resting_condition =
390 instruction.resting_ ? instruction.resting_->time_in_force_ : std::optional<TimeInForce>();
391 const auto resting_session = instruction.resting_ ? instruction.resting_->session_ : std::optional<OrderSession>();
392 if (differs(instruction.shape_.time_in_force_, resting_condition)) {
393 return refuse("validity condition");
394 }
395 if (differs(instruction.shape_.session_, resting_session)) {
396 return refuse("session");
397 }
398 // `OrderShape::type_` is a plain value rather than an optional, so "left alone" and "stated
399 // `LIMIT`" are the same bytes and the comparison has to be made in two steps.
400 if (instruction.resting_) {
401 if (instruction.shape_.type_ != instruction.resting_->type_) {
402 return refuse("price basis");
403 }
404 } else if (instruction.shape_.type_ != OrderShape().type_) {
405 // **With no resting shape, a value that differs from the default is still a stated change.**
406 // Only the default is ambiguous; `MARKET` on a cancellation is a caller saying something, and
407 // `kt10003` transmits no basis, so accepting it would report acceptance for a change the
408 // broker never saw. Compared against `OrderShape`'s own default rather than a literal, so the
409 // two cannot drift apart.
410 return refuse("price basis");
411 }
412 return std::nullopt;
413 }

445 static Result<KiwoomHistoryQuery> to_history_query(const KiwoomAccount&, const OrderHistoryQuery& query) {
446 BrokerError error;
447 error.category_ = ErrorCategory::INVALID_ARGUMENT;
448 error.reached_broker_ = false;
449 if (!query.from_) {
450 error.broker_code_ = "date_required";
451 error.message_ = "Kiwoom's order-history inquiry lists one calendar date, which must be given";
452 return Result<KiwoomHistoryQuery>(std::move(error));
453 }
454 if (
455 query.to_ &&
456 !(query.to_->year_ == query.from_->year_ && query.to_->month_ == query.from_->month_ &&
457 query.to_->day_ == query.from_->day_)
458 ) {
459 error.broker_code_ = "single_date_only";
460 error.message_ =
461 "Kiwoom's order-history inquiry lists one calendar date; ask for each date separately "
462 "rather than for a range";
463 return Result<KiwoomHistoryQuery>(std::move(error));
464 }
465 if (query.cursor_) {
466 error.broker_code_ = "cursor_unsupported";
467 error.message_ = "Kiwoom order history pages internally and exposes no cursor";
468 return Result<KiwoomHistoryQuery>(std::move(error));
469 }
470 if (query.limit_) {
471 error.broker_code_ = "limit_unsupported";
472 error.message_ = "Kiwoom order history documents no row limit and cannot honor one";
473 return Result<KiwoomHistoryQuery>(std::move(error));
474 }
475 KiwoomHistoryQuery converted;
476 converted.date_ = *query.from_;
477 converted.symbol_ = query.symbol_ ? *query.symbol_ : std::string();
478 return Result<KiwoomHistoryQuery>(std::move(converted));
479 }
