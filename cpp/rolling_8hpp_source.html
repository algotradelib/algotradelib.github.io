39inline Reduction reduce(const math::Observed& parts, math::Index first, math::Index length) {

69inline Dispersion dispersion(const math::Observed& parts, math::Index first, math::Index length, std::size_t needed) {
70 Dispersion result;
71 const auto reduced = reduce(parts, first, length);
72 result.count_ = reduced.count_;
73 if (reduced.count_ < needed || reduced.count_ < 2) {
74 return result;
75 }
76 result.mean_ = reduced.sum_ / static_cast<Scalar>(reduced.count_);
77 const Scalar squared =
78 (parts.values_.segment(first, length) - result.mean_ * parts.present_.segment(first, length)).squaredNorm();
79 result.deviation_ = std::sqrt(squared / static_cast<Scalar>(reduced.count_ - 1));
80 result.reported_ = true;
81 return result;
82}

125inline math::Vector rolling_mean(math::SeriesView values, std::size_t window, std::size_t min_observed) {
126 rolling_detail::check(window, min_observed);
127 const auto parts = math::observed_parts(values);
128 math::Vector result = math::unobserved(static_cast<std::size_t>(values.size()));
129 for (math::Index row = 0; row < values.size(); ++row) {
130 const math::Index first = rolling_detail::window_start(row, window);
131 const auto reduced = rolling_detail::reduce(parts, first, row - first + 1);
132 if (reduced.count_ >= min_observed) {
133 result[row] = reduced.sum_ / static_cast<Scalar>(reduced.count_);
134 }
135 }
136 return result;
137}

199inline math::Vector rolling_min(math::SeriesView values, std::size_t window, std::size_t min_observed) {
200 rolling_detail::check(window, min_observed);
201 const auto parts = math::observed_parts(values);
202 // Absent rows become `+inf` so they lose every comparison without needing a mask inside the
203 // window. A window with no observation at all is rejected by the count, never read from here.
204 const math::Vector ceiling = math::observed_or(values, std::numeric_limits<Scalar>::infinity());
205 math::Vector result = math::unobserved(static_cast<std::size_t>(values.size()));
206 for (math::Index row = 0; row < values.size(); ++row) {
207 const math::Index first = rolling_detail::window_start(row, window);
208 const math::Index length = row - first + 1;
209 if (parts.present_.segment(first, length).sum() >= static_cast<Scalar>(min_observed)) {
210 result[row] = ceiling.segment(first, length).minCoeff();
211 }
212 }
213 return result;
214}

339inline math::Vector ema(math::SeriesView values, Scalar alpha, std::size_t min_observed = 1) {
340 rolling_detail::check_alpha(alpha);
341 if (min_observed == 0) {
342 throw std::invalid_argument("ema must require at least one observation");
343 }
344 math::Vector result = math::unobserved(static_cast<std::size_t>(values.size()));
345 Scalar level{0};
346 std::size_t count = 0;
347 for (math::Index row = 0; row < values.size(); ++row) {
348 const Scalar value = values[row];
349 if (!observed(value)) {
350 continue;
351 }
352 // Seeded by count rather than by an unobserved level, so a level an infinity turned into `NaN`
353 // stays `NaN` instead of being mistaken for "no level yet".
354 level = count == 0 ? value : level + alpha * (value - level);
355 ++count;
356 if (count >= min_observed) {
357 result[row] = level;
358 }
359 }
360 return result;
361}
