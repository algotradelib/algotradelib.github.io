393 FillDecision decide(
394 const strategy::PlannedOrder& order, const strategy::BarEvent& next, const PriceRule& rule, const Date& date,
395 const std::optional<Date>& open_date = std::nullopt, bool released = false,
396 const std::optional<Decimal64>& decision_price = std::nullopt,
397 const std::optional<std::chrono::system_clock::time_point>& decision_price_at = std::nullopt,
398 const std::optional<std::chrono::system_clock::time_point>& decided_at = std::nullopt
399 ) const {
400 FillDecision decision;
401 bool at_open = false;
402 std::optional<Decimal64> reference;
403 if (market_price_ && !limit_priced(order.type_)) {
404 if (next_close_ && longer_than_a_day(next.interval_)) {
405 // The close of a calendar bar has no known date or session to charge and expire it by.
406 BrokerError error;
407 error.category_ = ErrorCategory::INVALID_ARGUMENT;
408 error.reached_broker_ = false;
409 error.broker_code_ = "next_close_bar_too_long";
410 error.message_ =
411 "the market-price rule 'next-close' prices bars of a day or less, not a weekly, monthly or "
412 "yearly bar";
413 decision.costing_error_ = std::move(error);
414 return decision;
415 }
416 reference = market_price_(MarketPriceInput{order, decision_price, decision_price_at, decided_at, next, rule});
417 if (!reference) {
418 // Not this bar: a DAY order rests for the rest of its session; any other is withdrawn.
419 decision.still_working_ = order.time_in_force_ == TimeInForce::DAY;
420 return decision;
421 }
422 if (reference->coefficient() <= 0) {
423 // A defect in the rule, not a price: filling at it would book a trade no market made.
424 BrokerError error;
425 error.category_ = ErrorCategory::INVALID_ARGUMENT;
426 error.reached_broker_ = false;
427 error.broker_code_ = "market_price_not_positive";
428 error.message_ = "the market-price rule '" + market_price_name_ + "' named a price that is not positive";
429 decision.costing_error_ = std::move(error);
430 return decision;
431 }
432 decision.priced_by_rule_ = true;
433 // Charged on the fill bar's own day, as an open's fill is.
434 at_open = true;
435 } else if (carries_trigger_price(order.type_) && !released) {
436 if (!order.trigger_price_ || !order.price_) {
437 // Not a stop this model can release (a caller driving the model directly).
438 return decision;
439 }
440 switch (release_of(order, next)) {
441 case Release::NONE:
442 // Dormant: the bar never reached the trigger.
443 decision.still_working_ = true;
444 return decision;
445 case Release::AT_OPEN:
446 // The whole bar is after the release, so it is priced as the limit it now is.
447 decision.released_ = true;
448 reference = reference_price(order, next, at_open);
449 // Dated from the open whatever the price policy credits: it traded at the open.
450 if (reference && limit_marketable_at(order, *next.bar_.open_)) {
451 at_open = true;
452 }
453 break;
454 case Release::INTRABAR:
455 decision.released_ = true;
456 reference = intrabar_release_price(order, next);
457 if (!reference) {
458 // The bar cannot show the price reached the limit after the release. Rests for the rest of
459 // its session (on daily bars, it expires unfilled).
460 decision.still_working_ = true;
461 return decision;
462 }
463 decision.intrabar_release_fill_ = true;
464 break;
465 }
466 } else {
467 reference = reference_price(order, next, at_open);
468 }
469 if (!reference) {
470 // A limit the bar never reached stays queued for the rest of its session.
471 decision.still_working_ = rests_unfilled(order);
472 return decision;
473 }
474
475 // A limit fill is credited a price its bar traded and never one through the limit: the bound is a
476 // sell's max(low, limit) or a buy's min(high, limit), applied before and after slippage.
477 std::optional<Decimal64> bound;
478 const bool limit_basis = limit_priced(order.type_);
479 if (limit_basis && order.price_) {
480 const auto& limit = *order.price_;
481 if (order.side_ == OrderSide::SELL) {
482 bound = decimal_less(limit, next.bar_.low_) ? next.bar_.low_ : limit;
483 if (decimal_less(*reference, *bound)) {
484 reference = bound;
485 decision.bounded_ = true;
486 }
487 } else {
488 bound = decimal_less(next.bar_.high_, limit) ? next.bar_.high_ : limit;
489 if (decimal_less(*bound, *reference)) {
490 reference = bound;
491 decision.bounded_ = true;
492 }
493 }
494 }
495 auto slipped = slippage_.apply(*reference, order.side_, rule);
496 if (slipped && decision.intrabar_release_fill_) {
497 // No favourable slippage after an intrabar release: the bar's other extreme may have come first.
498 const bool favourable = order.side_ == OrderSide::SELL ?
499 decimal_less(*reference, slipped->first) :
500 decimal_less(slipped->first, *reference);
501 if (favourable) {
502 slipped->first = *reference;
503 slipped->second = Decimal64::parse("0");
504 }
505 }
506 if (slipped && bound) {
507 // Negative slippage cannot carry a fill past the bar's opposite extreme either.
508 const auto& ceiling = order.side_ == OrderSide::SELL ? next.bar_.high_ : next.bar_.low_;
509 const bool past =
510 order.side_ == OrderSide::SELL ? decimal_less(ceiling, slipped->first) : decimal_less(slipped->first, ceiling);
511 if (past) {
512 const auto moved = order.side_ == OrderSide::SELL ?
513 decimal_subtract(*reference, ceiling) :
514 decimal_subtract(ceiling, *reference);
515 if (!moved) {
516 slipped.reset();
517 } else {
518 slipped->first = ceiling;
519 slipped->second = *moved;
520 decision.bounded_ = true;
521 }
522 }
523 }
524 if (slipped && bound) {
525 const bool beyond =
526 order.side_ == OrderSide::SELL ? decimal_less(slipped->first, *bound) : decimal_less(*bound, slipped->first);
527 if (beyond) {
528 const auto moved =
529 order.side_ == OrderSide::SELL ? decimal_subtract(*reference, *bound) : decimal_subtract(*bound, *reference);
530 if (!moved) {
531 slipped.reset();
532 } else {
533 slipped->first = *bound;
534 slipped->second = *moved;
535 decision.bounded_ = true;
536 }
537 }
538 }
539 if (!slipped) {
540 decision.still_working_ = rests_unfilled(order);
541 return decision;
542 }
543 const auto& charged_on = at_open && open_date ? *open_date : date;
544 auto charged = commission_.charge(order.quantity_, slipped->first, charged_on, order.instrument_.country_);
545 if (!charged) {
546 // A cost that cannot be resolved halts the engine; it is neither a non-fill nor silent.
547 decision.costing_error_ = std::move(charged.error());
548 return decision;
549 }
550
551 // The tax on a sale, where a rate was stated: of the sale's value, truncated to whole units.
552 std::optional<Decimal64> tax;
553 if (sell_tax_ && order.side_ == OrderSide::SELL && order.instrument_.country_ == sell_tax_->country_) {
554 const auto value = decimal_multiply(order.quantity_, slipped->first);
555 const auto charged = value ? decimal_multiply(*value, sell_tax_->rate_) : std::nullopt;
556 tax = charged ? decimal_floor_to_multiple(*charged, Decimal64::parse("1")) : std::nullopt;
557 if (!tax) {
558 // Not a tax of zero: the cost is unknown, so the engine halts.
559 BrokerError error;
560 error.category_ = ErrorCategory::INVALID_ARGUMENT;
561 error.reached_broker_ = false;
562 error.broker_code_ = "tax_not_representable";
563 error.message_ = "the tax on that sale does not fit an exact decimal";
564 decision.costing_error_ = std::move(error);
565 return decision;
566 }
567 }
568
569 Execution execution;
570 execution.fill_.instrument_ = order.instrument_;
571 execution.fill_.side_ = order.side_;
572 execution.fill_.quantity_ = order.quantity_;
573 execution.fill_.price_ = slipped->first;
574 execution.fill_.tag_ = order.tag_;
575 execution.fill_.at_ = next.period_end_;
576 execution.commission_ = charged.value().broker_;
577 execution.institution_fee_ = charged.value().institution_;
578 execution.applied_ = std::move(charged.value().applied_);
579 execution.slippage_ = slipped->second;
580 if (tax) {
581 execution.tax_ = *tax;
582 }
583 decision.execution_ = std::move(execution);
584 return decision;
585 }

588 // Where in a bar a dormant stop released, if it did.
589 enum class Release {
590 // The bar never reached the trigger.
591 NONE,
592 // The open was already through the trigger, so the whole bar followed the release.
593 AT_OPEN,
594 // The bar's extreme reached the trigger and its open did not.
595 INTRABAR,
596 };
597
598 // Returns whether, and where, `next` released a dormant stop.
599 //
600 // **A sell stop releases when the price falls to its trigger and a buy stop when it rises to it.**
601 // That is an assumption about the broker -- neither KIS nor Kiwoom documents the direction -- and
602 // it is the universal convention. `ReportCaveats::stop_releases_` counts how often it was used.
603 //
604 // A bar with no open cannot establish a release *at* the open, so a release there is intrabar,
605 // which is the answer that credits less.
606 static Release release_of(const strategy::PlannedOrder& order, const strategy::BarEvent& next) {
607 const auto& trigger = *order.trigger_price_;
608 const bool sell = order.side_ == OrderSide::SELL;
609 const bool reached = sell ? !decimal_less(trigger, next.bar_.low_) : !decimal_less(next.bar_.high_, trigger);
610 if (!reached) {
611 return Release::NONE;
612 }
613 if (next.bar_.open_) {
614 const auto& open = *next.bar_.open_;
615 const bool through = sell ? !decimal_less(trigger, open) : !decimal_less(open, trigger);
616 if (through) {
617 return Release::AT_OPEN;
618 }
619 }
620 return Release::INTRABAR;
621 }

623 // Returns whether the order's limit would trade at `price` -- a sell at or above it, a buy at or below.
624 static bool limit_marketable_at(const strategy::PlannedOrder& order, const Decimal64& price) {
625 const auto& limit = *order.price_;
626 return order.side_ == OrderSide::SELL ? !decimal_less(price, limit) : !decimal_less(limit, price);
627 }
628
629 // Returns what a stop released inside `next` is credited with, or nothing when the bar cannot show
630 // it traded after the release.
631 //
632 // At the release the price *was* the trigger. Two cases, using only prices the bar proves came after
633 // it -- the trigger itself and the close, which is always last:
634 //
635 // - **Marketable at the trigger** (a sell limited at or below it, a buy at or above): it trades. It is
636 // credited the **worst price the bar traded that the limit allows** -- a sell the higher of its
637 // limit and the bar's low, a buy the lower of its limit and the bar's high -- so the price is inside
638 // the bar and never through the limit. Crediting the limit itself could name a price the bar never
639 // reached; crediting the trigger would assume the book was there.
640 // - **Not marketable at the trigger** (a sell limited above it): it trades only if the price came back
641 // to the limit after the release. The close proves that when it is at or through the limit; then it
642 // is credited the limit. Otherwise the bar's other extreme may have come first, and nothing is.
643 static std::optional<Decimal64> intrabar_release_price(
644 const strategy::PlannedOrder& order, const strategy::BarEvent& next
645 ) {
646 const auto& trigger = *order.trigger_price_;
647 const auto& limit = *order.price_;
648 const bool sell = order.side_ == OrderSide::SELL;
649 if (limit_marketable_at(order, trigger)) {
650 if (sell) {
651 return decimal_less(limit, next.bar_.low_) ? next.bar_.low_ : limit;
652 }
653 return decimal_less(next.bar_.high_, limit) ? next.bar_.high_ : limit;
654 }
655 if (limit_marketable_at(order, next.bar_.close_)) {
656 return limit;
657 }
658 return std::nullopt;
659 }

661 // Whether a bar spans more than one trading day (a week, a month, a year).
662 static bool longer_than_a_day(BarInterval interval) noexcept {
663 switch (interval) {
664 case BarInterval::ONE_WEEK:
665 case BarInterval::ONE_MONTH:
666 case BarInterval::ONE_YEAR:
667 return true;
668 case BarInterval::ONE_MINUTE:
669 case BarInterval::FIVE_MINUTES:
670 case BarInterval::FIFTEEN_MINUTES:
671 case BarInterval::THIRTY_MINUTES:
672 case BarInterval::ONE_HOUR:
673 case BarInterval::ONE_DAY:
674 return false;
675 }
676 return true;
677 }

679 // Returns the price before slippage, or nothing when the order does not trade against this bar.
680 //
681 // A market order takes the open. A limit trades only if the bar's range reached it — a buy needs the
682 // low at or below the limit, a sell needs the high at or above it — because a bar that never touched
683 // the price is a bar in which the order sat unfilled, however close it came.
684 //
685 // Sets `at_open` when the price is the bar's stated open -- a market order, or a limit under
686 // `AT_OPEN_IF_BETTER` whose open was better -- which is what lets `decide` date the charge from the
687 // open's day. False for a bar with no open, whose close stands in for it.
688 std::optional<Decimal64> reference_price(
689 const strategy::PlannedOrder& order, const strategy::BarEvent& next, bool& at_open
690 ) const {
691 at_open = false;
692 // A bar with no open prices a market order at its close, which is not the open's moment.
693 const bool has_open = next.bar_.open_.has_value();
694 const auto open = has_open ? *next.bar_.open_ : next.bar_.close_;
695
696 // A released stop is priced as the limit it has become.
697 const bool limit_basis = limit_priced(order.type_);
698 if (!limit_basis || !order.price_) {
699 at_open = has_open;
700 return open;
701 }
702 const auto& limit = *order.price_;
703 bool open_if_better = false;
704 switch (limit_price_) {
705 case LimitFillPrice::AT_OPEN_IF_BETTER:
706 open_if_better = true;
707 break;
708 case LimitFillPrice::AT_LIMIT:
709 break;
710 }
711
712 if (order.side_ == OrderSide::BUY) {
713 if (decimal_less(limit, next.bar_.low_)) {
714 return std::nullopt;
715 }
716 if (open_if_better && decimal_less(open, limit)) {
717 at_open = has_open;
718 return open;
719 }
720 return limit;
721 }
722
723 if (decimal_less(next.bar_.high_, limit)) {
724 return std::nullopt;
725 }
726 if (open_if_better && decimal_less(limit, open)) {
727 at_open = has_open;
728 return open;
729 }
730 return limit;
731 }
