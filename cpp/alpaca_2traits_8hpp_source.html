214 static Result<PlacedOrder<alpaca::Broker>> normalize_place(Result<PlaceAck> raw) {
215 if (!raw) {
216 return Result<PlacedOrder<alpaca::Broker>>(std::move(raw.error()));
217 }
218 PlacedOrder<alpaca::Broker> placed;
219 placed.ack_.order_id_ = raw.value().order_id_;
220 placed.ack_.client_order_id_ = raw.value().client_order_id_;
221 placed.ack_.accepted_at_ = raw.value().received_at_;
222 placed.ref_.primary_ = raw.value().order_id_;
223 return Result<PlacedOrder<alpaca::Broker>>(std::move(placed));
224 }

231 static Result<WorkingOrders<alpaca::Broker>> normalize_working_orders(Result<WorkingOrdersResult> raw) {
232 if (!raw) {
233 return Result<WorkingOrders<alpaca::Broker>>(std::move(raw.error()));
234 }
235 WorkingOrders<alpaca::Broker> orders;
236 orders.received_at_ = raw.value().received_at_;
237 for (auto& open : raw.value().orders_) {
238 WorkingOrder<alpaca::Broker> working;
239 working.ref_.primary_ = open.order_id_;
240 working.amendable_quantity_ = decimal_subtract(open.quantity_, open.execution_.filled_quantity_);
241 working.order_ = std::move(open);
242 orders.orders_.push_back(std::move(working));
243 }
244 return Result<WorkingOrders<alpaca::Broker>>(std::move(orders));
245 }
