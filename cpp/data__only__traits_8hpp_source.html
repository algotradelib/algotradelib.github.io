88template <typename Out, typename Handler>
89void post_refusal(Handler&& handler) {
90 auto executor = net::get_associated_executor(handler, net::system_executor());
91 auto complete = [handler = std::forward<Handler>(handler)]() mutable {
92 std::move(handler)(Result<Out>(refusal()));
93 };
94 if constexpr (detail::postable<decltype(executor)>::value) {
95 net::post(executor, std::move(complete));
96 } else {
97 net::post(net::system_executor(), net::bind_executor(executor, std::move(complete)));
98 }
99}
