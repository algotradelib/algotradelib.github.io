34// Whether no two clients share `BrokerTraits<Client>::name`.
35template <typename... Clients>
36constexpr bool distinct_names() {
37 const std::string_view names[] = {BrokerTraits<Clients>::name...};
38 for (std::size_t i = 0; i < sizeof...(Clients); ++i) {
39 for (std::size_t j = i + 1; j < sizeof...(Clients); ++j) {
40 if (names[i] == names[j]) {
41 return false;
42 }
43 }

48template <typename Visit, typename First, typename... Clients>
49auto visit_named(type_list<First, Clients...>, std::string_view name, Visit& visit) {
50 using R = std::invoke_result_t<Visit&, type_tag<First>>;
51 static_assert(
52 (std::is_same_v<R, std::invoke_result_t<Visit&, type_tag<Clients>>> && ...),
53 "with_broker: visit must return the same type for every client"
54 );
55 static_assert(
56 !std::is_void_v<R> && !std::is_reference_v<R>, "with_broker: visit must return a value, not void or a reference"
57 );
58 static_assert(
59 distinct_names<First, Clients...>(),
60 "with_broker: listed clients need distinct names; a broker's asynchronous and blocking clients share one"
61 );
62 std::optional<R> result;
63 const auto match = [&](auto tag) {
64 return name == BrokerTraits<typename decltype(tag)::type>::name &&
65 (static_cast<void>(result.emplace(visit(tag))), true);
66 };
67 static_cast<void>((match(type_tag<First>{}) || ... || match(type_tag<Clients>{})));
68 return result;
69}
