203 static Result<tossinvest::TossAmendment> to_replacement(const Ref& ref, const AmendInstruction& instruction) {
204 // Toss's `OrderModifyRequest` carries `orderType`, `quantity`, `price` and
205 // the high-value flag, and **no validity field at all**: a modification
206 // cannot change the condition or the session, and Toss cannot refuse a
207 // request to. So this refuses on Toss's behalf, and the question is not
208 // whether the requested value is one Toss supports -- it can place a
209 // closing-auction order -- but whether it *differs* from what the order
210 // already is.
211 //
212 // Which means the original has to be known. An unset `resting_` is refused
213 // rather than assumed to match: a hand-built instruction naming DAY/REGULAR
214 // against a resting CLS order compared equal to a defaulted original and was
215 // sent, and the order stayed CLS while the caller was told it had moved.
216 if (!instruction.resting_) {
217 BrokerError error;
218 error.category_ = ErrorCategory::INVALID_ARGUMENT;
219 error.reached_broker_ = false;
220 error.broker_code_ = "unknown_resting_shape";
221 error.message_ =
222 "Toss cannot change an order's validity condition or session by amending it, and this "
223 "instruction does not say what the order's are, so neither can be verified; build the "
224 "instruction with reprice() or replace_with() from the working order";
225 return Result<tossinvest::TossAmendment>(std::move(error));
226 }
227 // An unset field on the shape asks for no change, which is always safe here.
228 if (
229 instruction.shape_.time_in_force_ && *instruction.shape_.time_in_force_ != instruction.resting_->time_in_force_
230 ) {
231 BrokerError error;
232 error.category_ = ErrorCategory::INVALID_ARGUMENT;
233 error.reached_broker_ = false;
234 error.broker_code_ = "unsupported_time_in_force";
235 error.message_ =
236 "Toss cannot change an order's validity condition by amending it; cancel the order and "
237 "place a new one instead";
238 return Result<tossinvest::TossAmendment>(std::move(error));
239 }
240 if (instruction.shape_.session_ && *instruction.shape_.session_ != instruction.resting_->session_) {
241 BrokerError error;
242 error.category_ = ErrorCategory::INVALID_ARGUMENT;
243 error.reached_broker_ = false;
244 error.broker_code_ = "unsupported_session";
245 error.message_ =
246 "Toss cannot move an order to another trading session by amending it; cancel the order "
247 "and place a new one instead";
248 return Result<tossinvest::TossAmendment>(std::move(error));
249 }
250
251 tossinvest::TossAmendment amendment;
252 amendment.order_id_ = ref.primary_;
253 amendment.type_ = instruction.shape_.type_;
254 amendment.quantity_ = instruction.quantity_;
255 amendment.price_ = instruction.shape_.price_;
256 amendment.confirm_high_value_ = instruction.confirm_high_value_;
257 return Result<tossinvest::TossAmendment>(std::move(amendment));
258 }

269 static Result<tossinvest::TossAmendment> to_cancellation(const Ref& ref, const AmendInstruction& instruction) {

282 return Result<tossinvest::TossAmendment>(std::move(amendment));
283 }
