162 // A 64-bit FNV-1a digest, fed fixed-width integers and length-prefixed text, so two different inputs never feed it the
163 // same bytes.
164 class Digest {
165 public:
166 void add(std::int64_t value) {
167 for (int at = 0; at < 64; at += 8) {
168 state_ = (state_ ^ ((static_cast<std::uint64_t>(value) >> at) & 0xffU)) * 0x100000001b3ULL;
169 }
170 }
171
172 void add(std::string_view text) {
173 add(static_cast<std::int64_t>(text.size()));
174 for (const char ch : text) {
175 state_ = (state_ ^ static_cast<unsigned char>(ch)) * 0x100000001b3ULL;
176 }
177 }

179 void add(const Decimal64& value) {
180 add(value.coefficient());
181 add(static_cast<std::int64_t>(value.scale()));
182 }

185 add(static_cast<std::int64_t>(date.year_));
186 add(static_cast<std::int64_t>(date.month_));
187 add(static_cast<std::int64_t>(date.day_));

190 void add(Clock::time_point at) {

192 }

355 // Opens, locks, checks and replays the file; on any refusal says why in `rejection_`. Only the repair of an
356 // incomplete last line or header changes the file before a refusal.
357 void open_file() {
358 if (!market_.settings_.fills_.library_market_price()) {
359 rejection_ = "a mock market whose market orders a caller's own function prices cannot be kept in a file";
360 return;
361 }
362#if defined(_WIN32)
363 handle_.reset(std::fopen(path_.c_str(), "a+b"));
364#else
365 const int descriptor = ::open(path_.c_str(), O_RDWR | O_APPEND | O_CREAT | O_CLOEXEC, 0666);
366 if (descriptor < 0) {
367 rejection_ = "the mock market's state file could not be opened";
368 return;
369 }
370 if (::flock(descriptor, LOCK_EX | LOCK_NB) != 0) {
371 // Closed before the reason is stored, which may allocate and throw.
372 const bool held_elsewhere = errno == EWOULDBLOCK;
373 ::close(descriptor);
374 rejection_ = held_elsewhere ?
375 "the mock market's state file is open in another market (one at a time)" :
376 "the mock market's state file could not be locked";
377 return;
378 }
379 handle_.reset(::fdopen(descriptor, "a+b"));
380 if (handle_ == nullptr) {
381 ::close(descriptor);
382 }
383#endif
384 if (handle_ == nullptr) {
385 rejection_ = "the mock market's state file could not be opened";
386 return;
387 }
388 // Unbuffered: a failed write leaves nothing behind for a later flush to complete.
389 std::setvbuf(handle_.get(), nullptr, _IONBF, 0);
390 std::string text;
391 char buffer[65536];
392 std::rewind(handle_.get());
393 while (const auto count = std::fread(buffer, 1, sizeof(buffer), handle_.get())) {
394 text.append(buffer, count);
395 }
396 if (std::ferror(handle_.get()) != 0 || std::fseek(handle_.get(), 0, SEEK_END) != 0) {
397 rejection_ = "the mock market's state file could not be read";
398 return;
399 }
400 const auto header = header_line();
401 const auto last = text.rfind('\n');
402 if (last == std::string::npos) {
403 // Empty, or a header cut short, before any change was made.
404 if (header.compare(0, text.size(), text) != 0) {
405 rejection_ = "the file is not a mock market's state file";
406 } else if (!cut(0) || !write_whole(header) || !sync() || !sync_folder()) {
407 rejection_ = "the mock market's state file could not be started";
408 } else {
409 opened_ = true;
410 }
411 return;
412 }
413 const auto first = text.find('\n');
414 if (!check_header(std::string_view(text).substr(0, first + 1), header)) {
415 return;
416 }
417 for (auto begin = first + 1; begin <= last;) {
418 const auto end = text.find('\n', begin);
419 if (!replay(std::string_view(text).substr(begin, end - begin))) {
420 rejection_ = "the mock market's state file does not replay: a line is damaged or out of order";
421 return;
422 }
423 begin = end + 1;
424 }
425 if (last + 1 < text.size() && !cut(last + 1)) {
426 rejection_ = "the mock market's state file ends in a line cut short, which could not be removed";
427 return;
428 }
429 // An earlier process may have stopped before either flush: what was replayed must be on the disk before it stands.
430 if (!sync() || !sync_folder()) {
431 rejection_ = "the mock market's state file could not be flushed to disk";
432 return;
433 }
434 opened_ = true;
435 }

437 // Tests the first line against this market's; says why in `rejection_` when it differs.
438 bool check_header(std::string_view line, const std::string& expected) {
439 if (line == expected) {
440 return true;
441 }
442 boost::system::error_code error;
443 const auto value = boost::json::parse(line, error);
444 const auto* fields = error ? nullptr : value.if_object();
445 const auto format = fields ? integer(*fields, "mock_market") : std::nullopt;
446 if (!format) {
447 rejection_ = "the file is not a mock market's state file";
448 } else if (*format != FORMAT) {
449 rejection_ = "the mock market's state file is of another format";
450 } else if (integer(*fields, "semantics") != SEMANTICS) {
451 rejection_ = "the mock market's state file was written under other market rules";
452 } else {
453 rejection_ =
454 "the mock market's state file belongs to another market: its bars, settings or reference data differ";
455 }
456 return false;
457 }

459 // Shortens the file to `size` bytes and flushes it.
460 bool cut(std::size_t size) {
461#if defined(_WIN32)
462 const bool shortened = ::_chsize_s(::_fileno(handle_.get()), static_cast<long long>(size)) == 0;
463#else
464 const bool shortened = ::ftruncate(::fileno(handle_.get()), static_cast<off_t>(size)) == 0;
465#endif
466 return shortened && std::fseek(handle_.get(), 0, SEEK_END) == 0 && sync();
467 }
468
469 // Writes `line` at the end of the file; `false` unless every byte was.
470 bool write_whole(const std::string& line) {
471 return std::fwrite(line.data(), 1, line.size(), handle_.get()) == line.size() && std::fflush(handle_.get()) == 0;
472 }

474 // Asks the system to put the file's bytes on the disk.
475 bool sync() {
476#if defined(_WIN32)
477 return ::_commit(::_fileno(handle_.get())) == 0;
478#else
479 return ::fsync(::fileno(handle_.get())) == 0;
480#endif
481 }

483 // Puts a new file's name on the disk, so a crash cannot lose the file; Windows has no such call.
484 bool sync_folder() const {
485#if defined(_WIN32)
486 return true;
487#else
488 const auto slash = path_.rfind('/');
489 const auto folder = slash == std::string::npos ? std::string(".") : path_.substr(0, slash == 0 ? 1 : slash);
490 const int descriptor = ::open(folder.c_str(), O_RDONLY | O_CLOEXEC);
491 const bool synced = descriptor >= 0 && ::fsync(descriptor) == 0;
492 if (descriptor >= 0) {
493 ::close(descriptor);
494 }
495 return synced;
496#endif
497 }

499 // Writes and flushes the line for a change before the market makes it; what became of a line that failed.
500 std::optional<MarketFileError::Kind> record(const Market::Change& change) {
501 if (!opened_ || failed_) {
502 return MarketFileError::Kind::NOT_WRITTEN;
503 }
504 boost::json::object entry;
505 entry["at"] = ticks(market_.now());
506 if (change.order_ != nullptr) {
507 const auto& order = *change.order_;
508 boost::json::object fields;
509 fields["symbol"] = order.instrument_.symbol_;
510 fields["country"] = static_cast<std::int64_t>(order.instrument_.country_);
511 fields["venue"] = static_cast<std::int64_t>(order.instrument_.venue_);
512 fields["side"] = static_cast<std::int64_t>(order.side_);
513 fields["type"] = static_cast<std::int64_t>(order.type_);
514 fields["time_in_force"] = static_cast<std::int64_t>(order.time_in_force_);
515 fields["session"] = static_cast<std::int64_t>(order.session_);
516 fields["route"] = static_cast<std::int64_t>(order.venue_);
517 fields["quantity"] = decimal(order.quantity_);
518 if (order.price_) {
519 fields["price"] = decimal(*order.price_);
520 }
521 entry["place"] = std::move(fields);
522 } else if (change.order_id_ != nullptr) {
523 entry["cancel"] = *change.order_id_;
524 } else {
525 entry["advance"] = ticks(*change.to_);
526 }
527 if (!write_whole(boost::json::serialize(entry) + "\n")) {
528 failed_ = true;
529 return MarketFileError::Kind::NOT_WRITTEN;
530 }
531 if (!sync()) {
532 failed_ = true;
533 return MarketFileError::Kind::UNCONFIRMED;
534 }
535 return std::nullopt;
536 }

538 // Makes the change a line records; `false` if the line is damaged or the market does not take it as it did.
539 bool replay(std::string_view line) {
540 boost::system::error_code error;
541 const auto value = boost::json::parse(line, error);
542 const auto* entry = error ? nullptr : value.if_object();
543 if (entry == nullptr || integer(*entry, "at") != ticks(market_.now())) {
544 return false;
545 }
546 if (const auto* place = entry->if_contains("place")) {
547 const auto order = order_from(*place);
548 return order && market_.place(*order).has_value();
549 }
550 if (const auto* cancel = entry->if_contains("cancel")) {
551 return cancel->is_string() && market_.cancel(std::string(cancel->as_string())).has_value();
552 }
553 const auto to = integer(*entry, "advance");
554 if (!to || *to <= ticks(market_.now())) {
555 return false;
556 }
557 try {
558 market_.advance_to(instant(*to));
559 } catch (const std::runtime_error&) {
560 // The market halted on this move, as it did when the move was recorded.
561 return market_.halted().has_value();
562 }
563 return true;
564 }

566 // Reads a placement's fields; nothing if any is missing or out of range.
567 static std::optional<OrderRequest> order_from(const boost::json::value& value) {
568 const auto* fields = value.if_object();
569 const auto* symbol = fields ? fields->if_contains("symbol") : nullptr;
570 if (symbol == nullptr || !symbol->is_string()) {
571 return std::nullopt;
572 }
573 const auto country = enumerated<Country>(*fields, "country", all_countries.size());
574 const auto venue = enumerated<Venue>(*fields, "venue", all_venues.size());
575 const auto side = enumerated<OrderSide>(*fields, "side", all_order_sides.size());
576 const auto type = enumerated<OrderType>(*fields, "type", all_order_types.size());
577 const auto time_in_force = enumerated<TimeInForce>(*fields, "time_in_force", all_times_in_force.size());
578 const auto session = enumerated<OrderSession>(*fields, "session", all_order_sessions.size());
579 const auto route = enumerated<Venue>(*fields, "route", all_venues.size());
580 const auto quantity = decimal_from(fields->if_contains("quantity"));
581 const auto* price = fields->if_contains("price");
582 const auto limit = decimal_from(price);
583 if (
584 !country || !venue || !side || !type || !time_in_force || !session || !route || !quantity || (price && !limit)
585 ) {
586 return std::nullopt;
587 }
588 OrderRequest order;
589 order.instrument_ = InstrumentId{*country, *venue, std::string(symbol->as_string())};
590 order.side_ = *side;
591 order.type_ = *type;
592 order.time_in_force_ = *time_in_force;
593 order.session_ = *session;
594 order.venue_ = *route;
595 order.quantity_ = *quantity;
596 order.price_ = limit;
597 return order;
598 }

619 // Reads a decimal `decimal` wrote; nothing if it is not one.
620 static std::optional<Decimal64> decimal_from(const boost::json::value* value) {
621 const auto* pair = value != nullptr ? value->if_array() : nullptr;
622 if (pair == nullptr || pair->size() != 2 || !(*pair)[0].is_int64() || !(*pair)[1].is_int64()) {
623 return std::nullopt;
624 }
625 const auto scale = (*pair)[1].as_int64();
626 if (scale < 0 || scale > 255) {
627 return std::nullopt;
628 }
629 return Decimal64((*pair)[0].as_int64(), static_cast<std::uint8_t>(scale));
630 }
