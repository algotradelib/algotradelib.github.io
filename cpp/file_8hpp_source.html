166 // A 64-bit FNV-1a digest, fed fixed-width integers and length-prefixed text, so two different inputs never feed it the
167 // same bytes.
168 class Digest {
169 public:
170 void add(std::int64_t value) {
171 for (int at = 0; at < 64; at += 8) {
172 state_ = (state_ ^ ((static_cast<std::uint64_t>(value) >> at) & 0xffU)) * 0x100000001b3ULL;
173 }
174 }
175
176 void add(std::string_view text) {
177 add(static_cast<std::int64_t>(text.size()));
178 for (const char ch : text) {
179 state_ = (state_ ^ static_cast<unsigned char>(ch)) * 0x100000001b3ULL;
180 }
181 }
182
183 void add(const Decimal64& value) {
184 add(value.coefficient());
185 add(static_cast<std::int64_t>(value.scale()));
186 }
187
188 void add(const Date& date) {
189 add(static_cast<std::int64_t>(date.year_));
190 add(static_cast<std::int64_t>(date.month_));
191 add(static_cast<std::int64_t>(date.day_));
192 }
193
194 void add(Clock::time_point at) {
195 add(ticks(at));
196 }
197
198 void add(const InstrumentId& id) {
199 add(static_cast<std::int64_t>(id.country_));
200 add(static_cast<std::int64_t>(id.venue_));
201 add(id.symbol_);
202 }
203
204 void add(const VenueKey& venue) {
205 add(static_cast<std::int64_t>(venue.kind()));
206 add(static_cast<std::int64_t>(venue.venue()));
207 add(venue.raw_code().str());
208 }
209
210 void add(const StatedDate& date) {
211 add(static_cast<std::int64_t>(date.state()));
212 if (date.state() == StatedDate::State::STATED) {
213 add(date.date());
214 }
215 }
216
217 template <typename T>
218 void add(const std::optional<T>& value) {
219 add(static_cast<std::int64_t>(value.has_value()));
220 if (value) {
221 add(*value);
222 }
223 }

225 std::string hex() const {
226 std::string text(16, '0');
227 for (std::size_t at = 0; at < 16; ++at) {
228 text[15 - at] = "0123456789abcdef"[(state_ >> (4 * at)) & 0xfU];
229 }
230 return text;
231 }

354 // Opens, locks, checks and replays the file; on any refusal says why in `rejection_`. Only the repair of an
355 // incomplete last line or header changes the file before a refusal.
356 void open_file() {
357 if (!market_.settings_.fills_.library_market_price()) {
358 rejection_ = "a mock market whose market orders a caller's own function prices cannot be kept in a file";
359 return;
360 }
361#if defined(_WIN32)
362 handle_ = std::fopen(path_.c_str(), "a+b");
363#else
364 const int descriptor = ::open(path_.c_str(), O_RDWR | O_APPEND | O_CREAT | O_CLOEXEC, 0666);
365 if (descriptor < 0) {
366 rejection_ = "the mock market's state file could not be opened";
367 return;
368 }
369 if (::flock(descriptor, LOCK_EX | LOCK_NB) != 0) {
370 rejection_ = errno == EWOULDBLOCK ?
371 "the mock market's state file is open in another market (one at a time)" :
372 "the mock market's state file could not be locked";
373 ::close(descriptor);
374 return;
375 }
376 handle_ = ::fdopen(descriptor, "a+b");
377 if (handle_ == nullptr) {
378 ::close(descriptor);
379 }
380#endif
381 if (handle_ == nullptr) {
382 rejection_ = "the mock market's state file could not be opened";
383 return;
384 }
385 // Unbuffered: a failed write leaves nothing behind for a later flush to complete.
386 std::setvbuf(handle_, nullptr, _IONBF, 0);
387 std::string text;
388 char buffer[65536];
389 std::rewind(handle_);
390 while (const auto count = std::fread(buffer, 1, sizeof(buffer), handle_)) {
391 text.append(buffer, count);
392 }
393 if (std::ferror(handle_) != 0 || std::fseek(handle_, 0, SEEK_END) != 0) {
394 rejection_ = "the mock market's state file could not be read";
395 return;
396 }
397 const auto header = header_line();
398 const auto last = text.rfind('\n');
399 if (last == std::string::npos) {
400 // Empty, or a header cut short, before any change was made.
401 if (header.compare(0, text.size(), text) != 0) {
402 rejection_ = "the file is not a mock market's state file";
403 } else if (!cut(0) || !write_whole(header) || !sync() || !sync_folder()) {
404 rejection_ = "the mock market's state file could not be started";
405 } else {
406 opened_ = true;
407 }
408 return;
409 }
410 const auto first = text.find('\n');
411 if (!check_header(std::string_view(text).substr(0, first + 1), header)) {
412 return;
413 }
414 for (auto begin = first + 1; begin <= last;) {
415 const auto end = text.find('\n', begin);
416 if (!replay(std::string_view(text).substr(begin, end - begin))) {
417 rejection_ = "the mock market's state file does not replay: a line is damaged or out of order";
418 return;
419 }
420 begin = end + 1;
421 }
422 if (last + 1 < text.size() && !cut(last + 1)) {
423 rejection_ = "the mock market's state file ends in a line cut short, which could not be removed";
424 return;
425 }
426 // An earlier process may have stopped before either flush: what was replayed must be on the disk before it stands.
427 if (!sync() || !sync_folder()) {
428 rejection_ = "the mock market's state file could not be flushed to disk";
429 return;
430 }
431 opened_ = true;
432 }

434 // Tests the first line against this market's; says why in `rejection_` when it differs.
435 bool check_header(std::string_view line, const std::string& expected) {
436 if (line == expected) {
437 return true;
438 }
439 boost::system::error_code error;
440 const auto value = boost::json::parse(line, error);
441 const auto* fields = error ? nullptr : value.if_object();
442 const auto format = fields ? integer(*fields, "mock_market") : std::nullopt;
443 if (!format) {
444 rejection_ = "the file is not a mock market's state file";
445 } else if (*format != FORMAT) {
446 rejection_ = "the mock market's state file is of another format";
447 } else if (integer(*fields, "semantics") != SEMANTICS) {
448 rejection_ = "the mock market's state file was written under other market rules";
449 } else {
450 rejection_ =
451 "the mock market's state file belongs to another market: its bars, settings or reference data differ";
452 }
453 return false;
454 }

456 // Shortens the file to `size` bytes and flushes it.
457 bool cut(std::size_t size) {
458#if defined(_WIN32)
459 const bool shortened = ::_chsize_s(::_fileno(handle_), static_cast<long long>(size)) == 0;
460#else
461 const bool shortened = ::ftruncate(::fileno(handle_), static_cast<off_t>(size)) == 0;
462#endif
463 return shortened && std::fseek(handle_, 0, SEEK_END) == 0 && sync();
464 }
465
466 // Writes `line` at the end of the file; `false` unless every byte was.
467 bool write_whole(const std::string& line) {
468 return std::fwrite(line.data(), 1, line.size(), handle_) == line.size() && std::fflush(handle_) == 0;
469 }
470
471 // Asks the system to put the file's bytes on the disk.
472 bool sync() {
473#if defined(_WIN32)
474 return ::_commit(::_fileno(handle_)) == 0;
475#else
476 return ::fsync(::fileno(handle_)) == 0;
477#endif
478 }

480 // Puts a new file's name on the disk, so a crash cannot lose the file; Windows has no such call.
481 bool sync_folder() const {
482#if defined(_WIN32)
483 return true;
484#else
485 const auto slash = path_.rfind('/');
486 const auto folder = slash == std::string::npos ? std::string(".") : path_.substr(0, slash == 0 ? 1 : slash);
487 const int descriptor = ::open(folder.c_str(), O_RDONLY | O_CLOEXEC);
488 const bool synced = descriptor >= 0 && ::fsync(descriptor) == 0;
489 if (descriptor >= 0) {
490 ::close(descriptor);
491 }
492 return synced;
493#endif

496 // Writes and flushes the line for a change before the market makes it; what became of a line that failed.
497 std::optional<MarketFileError::Kind> record(const Market::Change& change) {
498 if (!opened_ || failed_) {
499 return MarketFileError::Kind::NOT_WRITTEN;
500 }
501 boost::json::object entry;
502 entry["at"] = ticks(market_.now());
503 if (change.order_ != nullptr) {
504 const auto& order = *change.order_;
505 boost::json::object fields;
506 fields["symbol"] = order.instrument_.symbol_;
507 fields["country"] = static_cast<std::int64_t>(order.instrument_.country_);
508 fields["venue"] = static_cast<std::int64_t>(order.instrument_.venue_);
509 fields["side"] = static_cast<std::int64_t>(order.side_);
510 fields["type"] = static_cast<std::int64_t>(order.type_);
511 fields["time_in_force"] = static_cast<std::int64_t>(order.time_in_force_);
512 fields["session"] = static_cast<std::int64_t>(order.session_);
513 fields["route"] = static_cast<std::int64_t>(order.venue_);
514 fields["quantity"] = decimal(order.quantity_);
515 if (order.price_) {
516 fields["price"] = decimal(*order.price_);
517 }
518 entry["place"] = std::move(fields);
519 } else if (change.order_id_ != nullptr) {
520 entry["cancel"] = *change.order_id_;
521 } else {
522 entry["advance"] = ticks(*change.to_);
523 }
524 if (!write_whole(boost::json::serialize(entry) + "\n")) {
525 failed_ = true;
526 return MarketFileError::Kind::NOT_WRITTEN;
527 }
528 if (!sync()) {
529 failed_ = true;
530 return MarketFileError::Kind::UNCONFIRMED;
531 }
532 return std::nullopt;
533 }

535 // Makes the change a line records; `false` if the line is damaged or the market does not take it as it did.
536 bool replay(std::string_view line) {
537 boost::system::error_code error;
538 const auto value = boost::json::parse(line, error);
539 const auto* entry = error ? nullptr : value.if_object();
540 if (entry == nullptr || integer(*entry, "at") != ticks(market_.now())) {
541 return false;
542 }
543 if (const auto* place = entry->if_contains("place")) {
544 const auto order = order_from(*place);
545 return order && market_.place(*order).has_value();
546 }
547 if (const auto* cancel = entry->if_contains("cancel")) {
548 return cancel->is_string() && market_.cancel(std::string(cancel->as_string())).has_value();
549 }
550 const auto to = integer(*entry, "advance");
551 if (!to || *to <= ticks(market_.now())) {
552 return false;
553 }
554 try {
555 market_.advance_to(instant(*to));
556 } catch (const std::runtime_error&) {
557 // The market halted on this move, as it did when the move was recorded.
558 return market_.halted().has_value();
559 }
560 return true;
561 }

563 // Reads a placement's fields; nothing if any is missing or out of range.
564 static std::optional<OrderRequest> order_from(const boost::json::value& value) {
565 const auto* fields = value.if_object();
566 const auto* symbol = fields ? fields->if_contains("symbol") : nullptr;
567 if (symbol == nullptr || !symbol->is_string()) {
568 return std::nullopt;
569 }
570 const auto country = enumerated<Country>(*fields, "country", all_countries.size());
571 const auto venue = enumerated<Venue>(*fields, "venue", all_venues.size());
572 const auto side = enumerated<OrderSide>(*fields, "side", all_order_sides.size());
573 const auto type = enumerated<OrderType>(*fields, "type", all_order_types.size());
574 const auto time_in_force = enumerated<TimeInForce>(*fields, "time_in_force", all_times_in_force.size());
575 const auto session = enumerated<OrderSession>(*fields, "session", all_order_sessions.size());
576 const auto route = enumerated<Venue>(*fields, "route", all_venues.size());
577 const auto quantity = decimal_from(fields->if_contains("quantity"));
578 const auto* price = fields->if_contains("price");
579 const auto limit = decimal_from(price);
580 if (
581 !country || !venue || !side || !type || !time_in_force || !session || !route || !quantity || (price && !limit)
582 ) {
583 return std::nullopt;
584 }
585 OrderRequest order;
586 order.instrument_ = InstrumentId{*country, *venue, std::string(symbol->as_string())};
587 order.side_ = *side;
588 order.type_ = *type;
589 order.time_in_force_ = *time_in_force;
590 order.session_ = *session;
591 order.venue_ = *route;
592 order.quantity_ = *quantity;
593 order.price_ = limit;
594 return order;
595 }

616 // Reads a decimal `decimal` wrote; nothing if it is not one.
617 static std::optional<Decimal64> decimal_from(const boost::json::value* value) {
618 const auto* pair = value != nullptr ? value->if_array() : nullptr;
619 if (pair == nullptr || pair->size() != 2 || !(*pair)[0].is_int64() || !(*pair)[1].is_int64()) {
620 return std::nullopt;
621 }
622 const auto scale = (*pair)[1].as_int64();
623 if (scale < 0 || scale > 255) {
624 return std::nullopt;
625 }
626 return Decimal64((*pair)[0].as_int64(), static_cast<std::uint8_t>(scale));
627 }
