114std::vector<Fold> folds(std::size_t count, const FoldLengths& lengths, const Scheme& scheme = Scheme{}) {
115 static_assert(
116 is_window_scheme<Scheme>::value,
117 "a fold scheme needs the member IndexRange train(std::size_t test_begin, const FoldLengths& lengths) const"
118 );
119 if (lengths.train_ == 0 || lengths.step_ == 0) {
120 throw std::invalid_argument("folds need a positive train_ and step_");
121 }
122 const auto& [train, test, step, warmup] = lengths;
123 if (warmup > count || train > count - warmup || test > count - warmup - train) {
124 throw std::invalid_argument(
125 "folds: warmup_ + train_ + test_ (" + std::to_string(warmup) + " + " + std::to_string(train) + " + " +
126 std::to_string(test) + ") exceeds the " + std::to_string(count) + " positions"
127 );
128 }
129 std::vector<Fold> out;
130 for (std::size_t begin = warmup + train;; begin += step) {
131 const IndexRange range = scheme.train(begin, lengths);
132 if (range.begin_ < warmup || range.begin_ >= range.end_ || range.end_ > begin) {
133 throw std::invalid_argument(
134 "folds: the scheme's training range [" + std::to_string(range.begin_) + ", " + std::to_string(range.end_) +
135 ") is empty, starts before warmup_ " + std::to_string(warmup) + ", or ends after the test start " +
136 std::to_string(begin)
137 );
138 }
139 out.push_back(Fold{out.size(), range, IndexRange{begin, begin + test}, range.begin_ - warmup, begin - warmup});
140 if (step > count - begin || test > count - begin - step) {
141 break;
142 }
143 }
144 return out;
145}
