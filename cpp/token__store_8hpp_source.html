123inline std::string token_identity(std::string_view broker, std::string_view environment, std::string_view key) {
124 unsigned char digest[EVP_MAX_MD_SIZE];
125 unsigned int length = 0;
126 if (EVP_Digest(key.data(), key.size(), digest, &length, EVP_sha256(), nullptr) != 1) {
127 return {};
128 }
129 static constexpr char hex[] = "0123456789abcdef";
130 std::string identity = std::string(broker) + '-' + std::string(environment) + '-';
131 for (unsigned int index = 0; index < length; ++index) {
132 identity.push_back(hex[digest[index] >> 4]);
133 identity.push_back(hex[digest[index] & 0x0F]);
134 }
135 return identity;
136}

155template <typename Client, typename Request>
156std::optional<BrokerError> prepare(TokenStore& store, const std::string& identity, Client& client, Request request) {
157 using Token = decltype(client.token());
158 if (client.token().valid()) {
159 return std::nullopt;
160 }
161 auto held = store.lock(identity);
162 if (!held) {
163 return std::move(held.error());
164 }
165 auto loaded = store.load(identity);
166 if (!loaded) {
167 return std::move(loaded.error());
168 }
169 if (loaded.value()) {
170 Token token;
171 token.access_token_ = std::move(loaded.value()->access_token_);
172 token.expires_at_ = loaded.value()->expires_at_;
173 if (client.adopt_token(std::move(token))) {
174 return std::nullopt;
175 }
176 }
177 auto issued = request();
178 if (!issued) {
179 // A token response read only in part can leave a token that looks valid; it was never published.
180 client.forget_token();
181 return std::move(issued.error());
182 }
183 if (auto error = store.publish(identity, StoredToken{issued.value().access_token_, issued.value().expires_at_})) {
184 client.forget_token();
185 return error;
186 }
187 return std::nullopt;
188}
