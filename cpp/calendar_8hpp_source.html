97 std::optional<bool> is_trading_day(const Date& date) const {
98 // `days_` should be kept ascending, so a bisection finds a covered date; a miss is confirmed by the scan below, so a
99 // calendar filled out of order still finds every date it covers. Written by hand rather than with
100 // `std::lower_bound`, whose precondition (a sorted range) an out-of-order calendar would break. A date listed twice
101 // with different answers in an out-of-order calendar may get either.
102 std::size_t low = 0;
103 std::size_t high = days_.size();
104 while (low < high) {
105 const std::size_t middle = low + (high - low) / 2;
106 if (days_[middle].date_ < date) {
107 low = middle + 1;
108 } else {
109 high = middle;
110 }
111 }
112 if (low < days_.size() && days_[low].date_ == date) {
113 return days_[low].open_;
114 }
115 for (const auto& day : days_) {
116 if (day.date_ == date) {
117 return day.open_;
118 }
119 }
120 return std::nullopt;
121 }
122
130 std::optional<Date> next_trading_day(const Date& date) const {
131 for (const auto& day : days_) {
132 if (day.open_ && date < day.date_) {
133 return day.date_;
134 }
135 }
136 return std::nullopt;
137 }
138
146 std::optional<Date> previous_trading_day(const Date& date) const {
147 std::optional<Date> found;
148 for (const auto& day : days_) {
149 if (!(day.date_ < date)) {
150 break;
151 }
152 if (day.open_) {
153 found = day.date_;
154 }
155 }
156 return found;
157 }
158
164 std::optional<std::pair<Date, Date>> covered() const {
165 if (days_.empty()) {
166 return std::nullopt;
167 }
168 return std::make_pair(days_.front().date_, days_.back().date_);
169 }
