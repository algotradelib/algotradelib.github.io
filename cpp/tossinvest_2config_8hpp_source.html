71 void from_json_text_impl(std::string_view text, const std::string& origin) {
72 boost::system::error_code ec;
73 const auto jv = json::parse(text, ec);
74 if (ec || !jv.is_object()) {
75 throw std::runtime_error(
76 "Failed to parse Toss Invest credentials " + origin +
77 (ec ? " — " + ec.message() : std::string(" — not a JSON object"))
78 );
79 }
80 const auto& doc = jv.as_object();
81 const json::value* key = doc.if_contains("client_id");
82 const json::value* secret = doc.if_contains("client_secret");
83 if (!key || !secret || !key->is_string() || !secret->is_string()) {
84 throw std::runtime_error(
85 "Toss Invest credential " + origin + " should contain client_id and client_secret as strings"
86 );
87 }
88 // Both are checked before either is assigned, so a refused document leaves these credentials as they
89 // were.
90 key_ = key->get_string().c_str();
91 secret_ = secret->get_string().c_str();
92 }
