408inline std::optional<std::string> salvage_string(std::string_view line, std::string_view key) {
409 std::string needle;
410 needle.reserve(key.size() + 4);
411 needle += '"';
412 needle.append(key);
413 needle += "\":\"";
414 const auto at = line.find(needle);
415 if (at == std::string_view::npos) {
416 return std::nullopt;
417 }
418 const auto begin = at + needle.size();
419 // **The closing quote is the first one not escaped**, not the first one. A value holding `\"`
420 // was cut short at the escape, and a symbol cut short is another symbol's key.
421 auto end = begin;
422 while (end < line.size() && line[end] != '"') {
423 end += line[end] == '\\' ? 2 : 1;
424 }
425 if (end >= line.size()) {
426 // The value itself was cut off mid-write. Reporting what is there would invent a symbol.
427 return std::nullopt;
428 }
429 // Decoded as the JSON string it is, so an escape reads as the character it stands for. A value
430 // whose escapes do not decode is not salvaged, for the same reason a cut-off one is not.
431 boost::system::error_code error;
432 const auto quoted = json::parse(line.substr(begin - 1, end - begin + 2), error);
433 if (error || !quoted.is_string()) {
434 return std::nullopt;
435 }
436 return std::string(quoted.as_string());
437}

454inline std::string checksum_text(std::uint32_t value) {
455 static constexpr char digits[] = "0123456789abcdef";
456 std::string text(8, '0');
457 for (int i = 7; i >= 0; --i) {
458 text[static_cast<std::size_t>(i)] = digits[value & 0xFu];
459 value >>= 4;
460 }
461 return text;
462}

465inline constexpr std::string_view kChecksumMember = ",\"crc\":\"";
466
468inline constexpr std::size_t kChecksumSuffixSize = kChecksumMember.size() + 8 + 2;
469
471inline bool ends_in_checksum(std::string_view line) {
472 if (line.size() <= kChecksumSuffixSize) {
473 return false;
474 }
475 const auto suffix = line.substr(line.size() - kChecksumSuffixSize);
476 if (suffix.substr(0, kChecksumMember.size()) != kChecksumMember || suffix.substr(kChecksumSuffixSize - 2) != "\"}") {
477 return false;
478 }
479 for (const char digit : suffix.substr(kChecksumMember.size(), 8)) {
480 if (!((digit >= '0' && digit <= '9') || (digit >= 'a' && digit <= 'f'))) {
481 return false;
482 }
483 }
484 return true;
485}

493inline bool checksum_matches(std::string_view line) {
494 std::string body(line.substr(0, line.size() - kChecksumSuffixSize));
495 body += '}';
496 const auto stated = line.substr(line.size() - kChecksumSuffixSize + kChecksumMember.size(), 8);
497 return checksum_text(line_checksum(body)) == stated;
498}

736inline void write_order_keys(json::object& row, const JournalEntry& entry) {
737 row["country"] = std::string(journal_detail::country_name(entry.instrument_.country_));
738 row["venue"] = std::string(journal_detail::venue_name(entry.instrument_.venue_));
739 row["symbol"] = entry.instrument_.symbol_;
740 row["side"] = std::string(journal_detail::side_name(entry.side_));
741 row["type"] = std::string(journal_detail::order_type_name(entry.type_));
742 row["time_in_force"] = std::string(journal_detail::time_in_force_name(entry.time_in_force_));
743 row["session"] = std::string(journal_detail::order_session_name(entry.session_));
744 // Distinct from `venue`, which is the instrument's identity venue. This is
745 // where the order actually goes -- the binding resolves a request that named
746 // no venue into the one the broker's default reaches -- and a line that omits
747 // the key entirely is
748 // one written before the field existed -- which is not the same as one that
749 // asked for no particular venue, and recovery treats the two differently.
750 if (entry.venue_) {
751 row["route"] = std::string(journal_detail::venue_name(*entry.venue_));
752 }
753 row["quantity"] = entry.quantity_.str();
754 if (entry.price_) {
755 row["price"] = entry.price_->str();
756 }
757 // Only when present, so a line that is not a stop carries no `trigger` key and every old journal
758 // reads unchanged. (Every new line ends in a `crc` member, added below; the fields before it are
759 // what an earlier writer produced.)
760 if (entry.trigger_price_) {
761 row["trigger"] = entry.trigger_price_->str();
762 }
763 // Only on a serialized runner's lines, so an overlapping runner's journal reads as before.
764 if (entry.baseline_) {
765 row["baseline"] = entry.baseline_->str();
766 }
767}

803inline std::string journal_entry_to_line(const JournalEntry& entry) {
804 json::object row;
805 // Ordered so that the fields a damaged line most needs to yield come first. A truncated write
806 // keeps its prefix, so putting the instrument ahead of the detail text is what makes salvage
807 // work in practice rather than only in principle.
808 row["id"] = entry.id_;
809 row["action"] = std::string(journal_action_name(entry.action_));
810 row["outcome"] = std::string(journal_outcome_name(entry.outcome_));
811 // A ledger record names no order, so it carries none of an order's keys: torn, it salvages to no instrument.
812 const bool order = entry.action_ != JournalAction::LEDGER;
813 if (order) {
814 journal_detail::write_order_keys(row, entry);
815 }
816 // The authoritative instant, exact on every platform. `at_text` beside it is for the operator
817 // reading the file with `tail`, is only second-resolution, and is never read back: one field is
818 // the record and the other is the caption.
819 row["at"] = journal_detail::micros_of(entry.recorded_at_);
820 row["at_text"] = format_iso8601_utc(entry.recorded_at_);
821 if (order && !entry.order_id_.empty()) {
822 row["order_id"] = entry.order_id_;
823 }
824 if (!entry.detail_.empty()) {
825 row["detail"] = entry.detail_;
826 }
827 if (entry.requires_operator_) {
828 row["requires_operator"] = true;
829 }
830 if (!entry.scope_.empty()) {
831 row["scope"] = entry.scope_;
832 }
833 // Only on a budgeted runner's lines, so a journal without a run budget is byte for byte as before.
834 if (entry.run_) {
835 journal_detail::write_run_keys(row, *entry.run_);
836 }
837 // **Last, and over the line as it reads without it** -- the whole object, closing brace included,
838 // which the member then replaces. See `journal_detail::line_checksum` for what it detects
839 // and `journal_entry_from_line` for what a line that fails it becomes.
840 auto line = json::serialize(row);
841 const auto checksum = journal_detail::checksum_text(journal_detail::line_checksum(line));
842 line.pop_back();
843 line.append(journal_detail::kChecksumMember);
844 line += checksum;
845 line += "\"}";
846 return line;
847}

884inline RunKeys run_keys_from(const json::object& row) {
885 RunKeys keys;
886 const auto text = [&](std::string_view key, std::string& out) {
887 if (!has_field(row, key)) {
888 return false;
889 }
890 keys.present_ = true;
891 const auto value = readable_string(row, key);
892 if (!value) {
893 keys.unreadable_ = true;
894 return false;
895 }
896 out = *value;
897 return true;
898 };
899 const auto number = [&](std::string_view key, std::optional<Decimal64>& out) {
900 std::string value;
901 if (text(key, value)) {
902 out = known_decimal_from(value);
903 keys.unreadable_ = keys.unreadable_ || !out;
904 }
905 };
906 keys.has_run_ = text("run", keys.record_.run_);
907 std::string kind;
908 if (text("ledger", kind)) {
909 keys.record_.kind_ = run_record_kind_from(kind);
910 keys.unreadable_ = keys.unreadable_ || !keys.record_.kind_;
911 }
912 text("given", keys.record_.given_);
913 text("of", keys.record_.of_);
914 std::optional<Decimal64> amount;
915 number("amount", amount);
916 keys.has_amount_ = amount.has_value();
917 std::string code;
918 std::optional<Currency> currency;
919 if (text("currency", code)) {
920 currency = known_currency_from(code);
921 keys.unreadable_ = keys.unreadable_ || !currency;
922 }
923 keys.has_currency_ = currency.has_value();
924 if (amount && currency) {
925 keys.record_.amount_ = Money{*amount, *currency};
926 }
927 number("filled", keys.record_.filled_);
928 number("fee", keys.record_.fee_);
929 if (const auto estimated = row.if_contains("fee_estimated")) {
930 keys.present_ = true;
931 keys.has_fee_estimated_ = true;
932 if (!estimated->is_bool()) {
933 keys.unreadable_ = true;
934 } else {
935 keys.record_.fee_estimated_ = estimated->as_bool();
936 }
937 }
938 return keys;
939}

968 return outcome == JournalOutcome::CONFIRMED && keys.has_run_ && record.filled_ && record.amount_ && record.fee_ &&
969 booking_figures_hold(record);

1037inline std::optional<JournalEntry> journal_entry_from_line(std::string_view line) {
1038 boost::system::error_code error;
1039 const auto value = json::parse(line, error);
1040 if (error || !value.is_object()) {
1041 return std::nullopt;
1042 }
1043 const auto& row = value.as_object();
1044
1045 const auto action = journal_action_from(journal_detail::string_field(row, "action"));
1046 const auto outcome = journal_outcome_from(journal_detail::string_field(row, "outcome"));
1047 const auto operator_only = row.if_contains("requires_operator");
1048 const bool marked = operator_only != nullptr && operator_only->is_bool() && operator_only->as_bool();
1049
1050 const auto at = row.if_contains("at");
1051 // An operator-only line may carry the epoch, below the bound, and deliberately: a salvaged fragment records the epoch precisely *because* it has no timestamp, and `requires_operator_` is what stops that epoch window resolving it.
1052 // Repairing such a line writes it back, so rejecting the epoch here would make every repair unreadable on the next open.
1053 // Any other instant keeps the bound, which also keeps it inside what `system_clock` holds.
1054 const bool epoch = marked && at != nullptr && at->is_int64() && at->as_int64() == 0;
1055 if (
1056 !action || !outcome || at == nullptr || !at->is_int64() ||
1057 (!epoch && !journal_detail::plausible_instant(at->as_int64(), journal_detail::has_field(row, "crc")))
1058 ) {
1059 // A well-formed JSON object that is not a journal line. Treated as damage rather than as an
1060 // empty entry, because an entry with a default instrument would be a claim about an order.
1061 return std::nullopt;
1062 }
1063
1064 // **A present value this reader cannot parse is damage, not a default.**
1065 // Each of these has a documented default for an *absent* key -- a line that
1066 // predates the field -- and applying that default to a token the line does
1067 // carry turns damage into evidence: a damaged `"SELL"` reads as a buy, a
1068 // damaged `"IOC"` as a day order, and a damaged `"COMBINED"` moves the entry
1069 // to an instrument the runner is not trading, so the ledger marks the wrong
1070 // key and the right one is never withheld at all.
1071 //
1072 // An unreadable value therefore produces an entry with **no instrument**,
1073 // which `unattributable_journal_entry` documents and recovery already turns
1074 // into a global halt. Marking a guessed instrument would not protect the real
1075 // one, because they are different keys.
1076 // Each field is asked two questions, because they have different answers.
1077 // *Is the key there?* -- if not, the line predates the field and its
1078 // documented default is a fact about such a line. *Is what it holds
1079 // readable?* -- if not, the line says something this release cannot use, and
1080 // guessing turns damage into evidence. `string_field` answers neither: it
1081 // flattens absent, non-string and empty into the same `""`.
1082 const auto read = [&row](std::string_view key) {
1083 return journal_detail::readable_string(row, key);
1084 };
1085 // Present but unusable, for any reason: not a string, an empty string, or a
1086 // token the reader could not parse. The first version of this asked only
1087 // whether the *string* was readable, which let `"IOC"` damaged to `"IO?"`
1088 // through -- that is a perfectly good string, and a perfectly bad condition.
1089 const auto unusable = [&row](std::string_view key, bool parsed) {
1090 return journal_detail::has_field(row, key) && !parsed;
1091 };
1092
1093 // `std::optional::and_then` is C++23 and this library is C++17.
1094 const auto parsed = [&read](std::string_view key, auto reader) {
1095 using Result = decltype(reader(std::string_view{}));
1096 const auto text = read(key);
1097 return text ? reader(*text) : Result{};
1098 };
1099
1100 // A `crc` member anywhere but last, or not eight lowercase hex digits, is not this writer's; one
1101 // that is must match the line without it -- the object with its own closing brace. See
1102 // `journal_detail::line_checksum`.
1103 const bool checksum_holds = !journal_detail::has_field(row, "crc") ||
1104 (journal_detail::ends_in_checksum(line) && journal_detail::checksum_matches(line));
1105 bool unknown_key = false;
1106 bool run_key = false;
1107 for (const auto& member : row) {
1108 if (!journal_detail::known_entry_key(member.key())) {
1109 if (journal_detail::run_entry_key(member.key())) {
1110 run_key = true;
1111 } else {
1112 unknown_key = true;
1113 }
1114 }
1115 }
1116 // **A line whose bytes changed is not read at all**, rather than read as a damaged entry. A damaged
1117 // entry carries its own outcome and an id suffixed `~damaged`, which is right for a line whose
1118 // *value* no longer parses -- but a damaged *terminal* line with an intact pending partner then
1119 // pairs with nothing, is dropped as an orphan, and the pending line stays open as an ordinary one:
1120 // the change was detected and then discarded before recovery decided anything. An unreadable line
1121 // is counted by `parse_journal_bytes` and halts recovery outright, whatever it was.
1122 if (!checksum_holds || unknown_key) {
1123 return std::nullopt;
1124 }
1125
1126 // **A run budget's keys are read strictly**: one that is unparseable, of the wrong type, or inconsistent
1127 // with the line makes the whole line unreadable, so recovery halts. Never a `~damaged` entry, which a
1128 // damaged `CONFIRMED` would become and then be dropped as an orphan, its booking lost. Read only on a line
1129 // that carries one or is a ledger record, so an ordinary line pays nothing for them.
1130 std::optional<journal_detail::RunKeys> run;
1131 if (run_key || *action == JournalAction::LEDGER || *outcome == JournalOutcome::RECORDED) {
1132 run = journal_detail::run_keys_from(row);
1133 if (run->unreadable_) {
1134 return std::nullopt;
1135 }
1136 if (*action == JournalAction::LEDGER) {
1137 return journal_detail::ledger_entry_from(row, *outcome, marked, at->as_int64(), *run);
1138 }
1139 if (!journal_detail::run_keys_fit_order_line(*action, *outcome, *run)) {
1140 return std::nullopt;
1141 }
1142 }

1144 const auto symbol = read("symbol");
1145 const auto country = parsed("country", journal_detail::known_country_from);
1146 const auto venue = parsed("venue", journal_detail::known_venue_from);
1147 const auto side = parsed("side", journal_detail::known_side_from);
1148 const auto type = parsed("type", journal_detail::known_order_type_from);
1149 const auto condition = parsed("time_in_force", journal_detail::known_time_in_force_from);
1150 const auto session = parsed("session", journal_detail::known_order_session_from);
1151 const auto quantity = parsed("quantity", journal_detail::known_decimal_from);
1152 const auto price = parsed("price", journal_detail::known_decimal_from);
1153 const auto trigger = parsed("trigger", journal_detail::known_decimal_from);
1154 const auto baseline = parsed("baseline", journal_detail::known_decimal_from);
1155
1156 const bool damaged =
1157 // Identity and the fields every complete line carries: required outright.
1158 // A line this reader cannot attribute to an instrument must not be
1159 // attributed to a guess, and `quantity` at a silent zero is a claim about
1160 // an order's size.
1161 !symbol || !country || !venue || !side || !type || !quantity ||
1162 // Legacy-defaulted fields: absent is fine, present and unreadable is not.
1163 unusable("time_in_force", condition.has_value()) || unusable("session", session.has_value()) ||
1164 // A price is absent for a market order and required for a basis that
1165 // carries one, so its presence is judged against the basis rather than
1166 // assumed either way.
1167 unusable("price", price.has_value()) ||
1168 // The same rule for a stop's trigger: absent on every other order and on every line written
1169 // before the field existed, and a present value that does not parse is damage.
1170 unusable("trigger", trigger.has_value()) || unusable("baseline", baseline.has_value()) ||
1171 // Checked above, where a changed line becomes unreadable rather than damaged; restated so the
1172 // condition reads whole.
1173 !checksum_holds || unknown_key;
1174 // **A price missing from a priced basis, and a cancellation with no target
1175 // id, are deliberately *not* damage here.** Both are lines this library's own
1176 // writer emits: `JournalingExecutor` appends the pending line *before* the
1177 // executor runs, so an order the binding then refuses locally -- a limit with
1178 // no price, a cancellation of a reservation whose broker id is not known --
1179 // has already been written by the time anyone could object. Rejecting them
1180 // turned an ordinary local refusal into a global halt on the next restart,
1181 // which is worse than the hazard the rejection was meant to close.
1182 //
1183 // The hazard is real and it lives one step later: such a line must not be
1184 // *resolved* on evidence it does not have. `runner::match_against` refuses to
1185 // conclude anything from one, which leaves it withheld for an operator while
1186 // its own terminal line still closes it when the refusal was ordinary.
1187 if (damaged) {
1188 // A run's line whose order cannot be read: its booking or its run would be lost with it.
1189 if (run && run->present_) {
1190 return std::nullopt;
1191 }
1192 auto unattributable = unattributable_journal_entry(line);
1193 const auto id = journal_detail::string_field(row, "id");
1194 unattributable.id_ = (id.empty() ? std::string("unknown") : id) + "~damaged";
1195 unattributable.action_ = *action;
1196 // Its own outcome, not `PENDING`. Both lines of a pair carry the same id, so
1197 // a damaged *terminal* line still closes its damaged pending partner --
1198 // forcing `PENDING` left both open and halted a runner over an instruction
1199 // that had already resolved.
1200 unattributable.outcome_ = *outcome;
1201 unattributable.recorded_at_ = journal_detail::instant_from(at->as_int64());
1202 unattributable.detail_ =
1203 "a journal line carries a value this release cannot read, so the order it describes cannot "
1204 "be identified; nothing may be traded until an operator resolves it: " +
1205 std::string(line.substr(0, line.size() < 160 ? line.size() : 160));
1206 return unattributable;
1207 }
1208
1209 JournalEntry entry;
1210 entry.id_ = journal_detail::string_field(row, "id");
1211 entry.action_ = *action;
1212 entry.outcome_ = *outcome;
1213 entry.instrument_ = InstrumentId{*country, *venue, *symbol};
1214 entry.side_ = *side;
1215 entry.type_ = *type;
1216 // These two keep their documented defaults for an *absent* key only; a
1217 // present value that failed to parse was caught above.
1218 entry.time_in_force_ = condition ? *condition : TimeInForce::DAY;
1219 entry.session_ = session ? *session : OrderSession::REGULAR;
1220 // Absent key -> absent value, which is "this line predates the field" rather
1221 // than "no route was asked for". Recovery will not attribute a row to an
1222 // instruction whose route nobody recorded. The field defaults to a recorded
1223 // `UNSPECIFIED`, so this clears it rather than leaving the default standing.
1224 // An absent key means the line predates the field. An unrecognized value means
1225 // the line says something this reader cannot use. Both are "unknown", and
1226 // neither may become the `UNSPECIFIED` that recovery treats as a recorded
1227 // request for no venue.
1228 entry.venue_ = journal_detail::known_venue_from(journal_detail::string_field(row, "route"));
1229 // Damaged numbers land in the same place as damaged enumerations, for the
1230 // same reason -- and they used to land somewhere far worse, throwing out of
1231 // recovery entirely. See `known_decimal_from`.
1232 entry.quantity_ = *quantity;
1233 if (price) {
1234 entry.price_ = *price;
1235 }
1236 if (trigger) {
1237 entry.trigger_price_ = *trigger;
1238 }
1239 entry.baseline_ = baseline;
1240 entry.recorded_at_ = journal_detail::instant_from(at->as_int64());
1241 entry.order_id_ = journal_detail::string_field(row, "order_id");
1242 entry.detail_ = journal_detail::string_field(row, "detail");
1243 entry.requires_operator_ = marked;
1244 entry.scope_ = journal_detail::string_field(row, "scope");
1245 if (run && run->present_) {
1246 entry.run_ = std::make_shared<const RunRecord>(run->record_);
1247 }
1248 return entry;
1249}

1281inline std::optional<JournalEntry> salvage_journal_line(std::string_view line) {
1282 const auto symbol = journal_detail::salvage_string(line, "symbol");
1283 if (!symbol || symbol->empty()) {
1284 return std::nullopt;
1285 }
1286 JournalEntry entry;
1287 // The flag that makes the promise in this comment block true. Without it the entry's missing
1288 // timestamp reads as the epoch, every window test passes immediately, no real row falls inside
1289 // the window, and the first sweep resolves it `NOT_PLACED`.
1290 entry.requires_operator_ = true;
1291 const auto id = journal_detail::salvage_string(line, "id");
1292 entry.id_ = (id ? *id : std::string("unknown")) + "~truncated";
1293 entry.action_ =
1294 journal_action_from(journal_detail::salvage_string(line, "action").value_or("place"))
1295 .value_or(JournalAction::PLACE);
1296 entry.outcome_ = JournalOutcome::PENDING;
1297 // **Absent is a default; present and unreadable is damage** -- the same rule
1298 // the strict reader follows, and for the same reason. A truncated line is
1299 // missing its *tail*, so an absent field is expected and its default is a fact
1300 // about what the line could have said. A field that is there and cannot be
1301 // read is corruption in the middle, and guessing it hands this entry a
1302 // different instrument than the order had: `requires_operator_` would then
1303 // withhold a key the runner is not trading under, while the real one stays
1304 // open. Such a line becomes unattributable instead, carrying no instrument at
1305 // all, which recovery turns into a global halt.
1306 // A key the truncation removed takes its default; a key that is *there* and
1307 // unreadable is damage, and `salvage_string` cannot tell those apart on its
1308 // own -- it recognizes `"key":"value"` and nothing else, so `null` and `7`
1309 // both read as absent.
1310 const auto salvaged = [&line](std::string_view key, const char* fallback) {
1311 const auto text = journal_detail::salvage_string(line, key);
1312 if (text) {
1313 return std::optional<std::string>(*text);
1314 }
1315 return salvage_damaged(line, key) ? std::optional<std::string>() : std::optional<std::string>(fallback);
1316 };
1317 const auto parsed = [&salvaged](std::string_view key, const char* fallback, auto reader) {
1318 using Result = decltype(reader(std::string_view{}));
1319 const auto text = salvaged(key, fallback);
1320 return text ? reader(*text) : Result{};
1321 };

1323 const auto country = parsed("country", "KR", journal_detail::known_country_from);
1324 const auto venue = parsed("venue", "UNSPECIFIED", journal_detail::known_venue_from);
1325 const auto side = parsed("side", "BUY", journal_detail::known_side_from);
1326 const auto type = parsed("type", "UNKNOWN", journal_detail::known_order_type_from);
1327 const auto condition = parsed("time_in_force", "DAY", journal_detail::known_time_in_force_from);
1328 const auto session = parsed("session", "REGULAR", journal_detail::known_order_session_from);
1329 if (!country || !venue || !side || !type || !condition || !session) {
1330 auto unattributable = unattributable_journal_entry(line);
1331 unattributable.id_ = entry.id_;
1332 unattributable.detail_ =
1333 "a truncated journal line carries a value this release cannot read, so the order it "
1334 "describes cannot be identified; nothing may be traded until an operator resolves it: " +
1335 std::string(line.substr(0, line.size() < 160 ? line.size() : 160));
1336 return unattributable;
1337 }
1338 entry.instrument_ = InstrumentId{*country, *venue, *symbol};
1339 entry.side_ = *side;
1340 entry.type_ = *type;
1341 entry.time_in_force_ = *condition;
1342 entry.session_ = *session;
1343 entry.venue_ = journal_detail::known_venue_from(journal_detail::salvage_string(line, "route").value_or(""));
1344 if (const auto quantity = journal_detail::salvage_string(line, "quantity")) {
1345 // A truncated line's last field is routinely half-written, so an unparseable
1346 // number here is expected rather than exceptional -- and it must not throw
1347 // out of a salvage path that exists to survive exactly this.
1348 if (const auto value = journal_detail::known_decimal_from(*quantity)) {
1349 entry.quantity_ = *value;
1350 }
1351 }
1352 if (const auto price = journal_detail::salvage_string(line, "price")) {
1353 if (const auto value = journal_detail::known_decimal_from(*price)) {
1354 entry.price_ = *value;
1355 }
1356 }
1357 if (const auto trigger = journal_detail::salvage_string(line, "trigger")) {
1358 if (const auto value = journal_detail::known_decimal_from(*trigger)) {
1359 entry.trigger_price_ = *value;
1360 }
1361 }
1362 entry.detail_ = "recovered from a truncated journal line; only an operator can resolve it";
1363 return entry;
1364}

1372inline JournalBytes read_journal_bytes(const std::string& path) {
1373 JournalBytes bytes;
1374 errno = 0;
1375 std::unique_ptr<std::FILE, int (*)(std::FILE*)> handle(std::fopen(path.c_str(), "rb"), [](std::FILE* file) {
1376 return file ? std::fclose(file) : 0;
1377 });
1378 if (handle == nullptr) {
1379 if (errno == ENOENT) {
1380 bytes.status_ = JournalReadStatus::MISSING;
1381 return bytes;
1382 }
1383 // Every other reason -- a permission the process does not have, a directory where a file was
1384 // expected, a device error -- means the file exists and its contents are unknown. Reporting
1385 // that as "no journal" is how a process claims a path holding its predecessor's in-flight
1386 // orders and then trades on an empty ledger.
1387 bytes.status_ = JournalReadStatus::UNREADABLE;
1388 bytes.detail_ = "the order journal at " + path + " exists and could not be opened for reading";
1389 return bytes;
1390 }
1391
1392 char buffer[4096];
1393 while (const auto read = std::fread(buffer, 1, sizeof(buffer), handle.get())) {
1394 bytes.text_.append(buffer, read);
1395 }
1396 if (std::ferror(handle.get()) != 0) {
1397 // A read that stopped partway looks exactly like an ordinary end of file to the loop above, so
1398 // the truncated tail it produces would be a fiction about where the journal ends.
1399 bytes.status_ = JournalReadStatus::UNREADABLE;
1400 bytes.detail_ = "the order journal at " + path + " could not be read to the end";
1401 bytes.text_.clear();
1402 return bytes;
1403 }
1404 bytes.status_ = JournalReadStatus::READ;
1405 return bytes;
1406}

1427inline std::size_t parse_complete_lines(std::string_view text, JournalReplay& replay, ParseCarry& carry) {
1428 std::size_t begin = 0;
1429 while (begin < text.size()) {
1430 const auto end = text.find('\n', begin);
1431 if (end == std::string_view::npos) {
1432 break;
1433 }
1434 const std::string_view line(text.data() + begin, end - begin);
1435 begin = end + 1;
1436 // **An empty line, or a header anywhere but first, is damage.** This writer produces neither: every
1437 // append is one non-empty line and its newline, and the header is written once, before anything
1438 // else. Skipping them was how an entry overwritten by a bare newline, or by a copy of the header,
1439 // vanished from recovery with nothing counted -- a changed line removed rather than detected.
1440 const auto unreadable = [&replay](const char* why) {
1441 ++replay.unreadable_lines_;
1442 if (replay.detail_.empty()) {
1443 replay.detail_ = why;
1444 }
1445 };
1446 if (line.empty()) {
1447 unreadable("the journal contains an empty line, which this library never writes");
1448 continue;
1449 }
1450 if (auto scope = journal_header_scope(line)) {
1451 if (replay.has_header_ || !replay.entries_.empty() || replay.unreadable_lines_ > 0) {
1452 unreadable(
1453 "the journal contains a header after its first line, which this library never "
1454 "writes"
1455 );
1456 continue;
1457 }
1458 // Not an instruction and not damage. Recorded so an opener can check ownership before it
1459 // writes anything at all.
1460 replay.has_header_ = true;
1461 replay.scope_ = std::move(*scope);
1462 // A header that declares checksums protects every entry after it, the first one included.
1463 carry.seen_checksum_ = carry.seen_checksum_ || journal_header_declares_checksums(line);
1464 continue;
1465 }
1466 const bool checksummed = ends_in_checksum(line);
1467 if (carry.seen_checksum_ && !checksummed) {
1468 // **This writer never puts a checksum-less line after a checksummed one, or after a header that
1469 // declares checksums**, so this is a protected line that lost its checksum -- damage that removed
1470 // the whole member, which the line's own reader cannot tell from a line written before checksums
1471 // existed. Only the first checksummed line of a file begun by an earlier release is left without
1472 // this protection; see `journal_header_line`.
1473 ++replay.unreadable_lines_;
1474 if (replay.detail_.empty()) {
1475 replay.detail_ =
1476 "a journal line without a checksum follows a checksummed one, so it was changed after it "
1477 "was written";
1478 }
1479 continue;
1480 }
1481 carry.seen_checksum_ = carry.seen_checksum_ || checksummed;
1482 if (auto entry = journal_entry_from_line(line)) {
1483 replay.entries_.push_back(std::move(*entry));
1484 } else {
1485 // A complete line that is not a journal entry. Not expected from this writer, so it means
1486 // the file was edited or written by something else, or a repair was interrupted; counted
1487 // rather than ignored, and recovery halts on it.
1488 ++replay.unreadable_lines_;
1489 if (replay.detail_.empty()) {
1490 replay.detail_ = "the journal contains a line this library did not write";
1491 }
1492 }
1493 }
1494 return begin;
1495}

1503inline void parse_torn_tail(std::string_view tail, JournalReplay& replay) {
1504 // Everything after the last newline is a write that never finished. This is the whole reason
1505 // the format is line-oriented, and dropping it silently would defeat the feature.
1506 replay.truncated_tail_ = true;
1507 if (auto salvaged = salvage_journal_line(tail)) {

1511 } else {
1512 replay.detail_ = "the final journal line was truncated and named no instrument; something was in flight";
1513 // The same entry a writer's repair appends for this tail, so recovery halts on it whether or
1514 // not the file was repaired, and whatever else is open beside it.

1883 // Returns whether this journal may write to the file, deciding ownership from the header alone.
1884 //
1885 // **The header, not the records, and that distinction is the fix.** Reading ownership from the
1886 // records leaves a hole exactly where it matters most: the scope is one field among many on a
1887 // record, so a process killed partway through writing the *first* instruction leaves a file whose
1888 // only line is torn and unscoped. Another account could then open the path, repair that line into
1889 // its own journal, and reconcile the first account's instruction against its own rows.
1890 //
1891 // A header is written before any instruction is contemplated, so it is complete or the file is
1892 // empty. Three cases, and the third is the strict one:

1895 // - **Header present.** The scopes must agree.
1896 // - **Non-empty with no header.** Refused. Something else wrote this file, or it predates the
