107inline std::int64_t days_from_civil(const Date& date) {
108 std::int64_t year = date.year_;
109 const unsigned month = date.month_;
110 year -= month <= 2 ? 1 : 0;
111 const std::int64_t era = (year >= 0 ? year : year - 399) / 400;
112 const auto year_of_era = static_cast<unsigned>(year - era * 400);
113 const unsigned day_of_year = (153 * (month + (month > 2 ? -3 : 9)) + 2) / 5 + date.day_ - 1;
114 const unsigned day_of_era = year_of_era * 365 + year_of_era / 4 - year_of_era / 100 + day_of_year;
115 return era * 146097 + static_cast<std::int64_t>(day_of_era) - 719468;
116}

138inline std::chrono::hours eastern_offset(std::chrono::system_clock::time_point instant) {
139 const auto seconds = std::chrono::duration_cast<std::chrono::seconds>(instant.time_since_epoch()).count();
140 const auto year = utc_date(instant - std::chrono::hours(5)).year_;
141 // The n-th Sunday of a month: 1970-01-01 was a Thursday (day 4 counting Sunday as 0).
142 const auto nth_sunday = [year](unsigned month, int nth) {
143 const auto first = days_from_civil(Date{year, month, 1});
144 const auto weekday = ((first % 7) + 7 + 4) % 7;
145 return first + (7 - weekday) % 7 + 7 * (nth - 1);
146 };
147 const auto starts = nth_sunday(3, 2) * 86400 + 2 * 3600 + 5 * 3600;
148 const auto ends = nth_sunday(11, 1) * 86400 + 2 * 3600 + 4 * 3600;
149 return seconds >= starts && seconds < ends ? std::chrono::hours(-4) : std::chrono::hours(-5);
150}

157inline std::chrono::system_clock::time_point eastern_midnight(const Date& date) {
158 const auto utc_midnight =
159 std::chrono::system_clock::time_point{} + std::chrono::seconds(days_from_civil(date) * 86400);
160 // Local midnight is 04:00Z or 05:00Z, before either change (07:00Z in March, 06:00Z in November), so the offset at
161 // 04:00Z of the date is the one at midnight, change days included.
162 return utc_midnight - eastern_offset(utc_midnight + std::chrono::hours(4));
163}

181inline std::optional<Date> parse_date(std::string_view text) {
182 if (text.size() != 10 || text[4] != '-' || text[7] != '-') {
183 return std::nullopt;
184 }
185 const auto field = [text](std::size_t from, std::size_t count, int& out) {
186 const auto parsed = std::from_chars(text.data() + from, text.data() + from + count, out);
187 return parsed.ec == std::errc{} && parsed.ptr == text.data() + from + count;
188 };
189 int year = 0;
190 int month = 0;
191 int day = 0;
192 if (!field(0, 4, year) || !field(5, 2, month) || !field(8, 2, day) || month < 1 || day < 1) {
193 return std::nullopt;
194 }
195 const Date date{year, static_cast<unsigned>(month), static_cast<unsigned>(day)};
196 if (!is_valid_date(date)) {
197 return std::nullopt;
198 }
199 return date;
200}

213inline std::optional<Decimal64> number(const json::object& object, const char* key) {
214 const auto* value = object.if_contains(key);
215 if (value == nullptr || value->is_null()) {
216 return std::nullopt;
217 }
218 try {
219 if (value->is_string()) {
220 return Decimal64::parse(std::string_view(value->as_string()));
221 }
222 if (value->is_int64()) {
223 return Decimal64::parse(std::to_string(value->as_int64()));
224 }
225 if (value->is_uint64()) {
226 return Decimal64::parse(std::to_string(value->as_uint64()));
227 }
228 if (value->is_double()) {
229 // The fewest decimal places that give the same double back, found by exact division: no locale, and no
230 // floating-point `std::to_chars`, which older standard libraries and macOS targets lack.
231 const double number = value->as_double();
232 double power = 1;
233 for (std::uint8_t scale = 0; std::isfinite(number) && scale <= 15; ++scale, power *= 10) {
234 const double scaled = std::round(number * power);
235 if (std::fabs(scaled) < 9007199254740992.0) {
236 const auto coefficient = static_cast<std::int64_t>(scaled);
237 if (static_cast<double>(coefficient) / power == number) {
238 return Decimal64(coefficient, scale);
239 }
240 }
241 }
242 }
243 } catch (const std::exception&) {
244 }
245 throw std::runtime_error(std::string("a field is not a number: ") + key);
246}

439inline std::optional<BrokerError> validate_order(const OrderRequest& order) {
440 if (auto refused = unsupported_instrument(order.instrument_)) {
441 return refused;
442 }
443 if (order.venue_ != Venue::UNSPECIFIED) {
444 return refusal("venue_unsupported", "Alpaca routes orders itself; leave the venue unspecified");
445 }
446 if (
447 order.client_order_id_ &&
448 (order.client_order_id_->empty() || order.client_order_id_->size() > client_order_id_limit)
449 ) {
450 return refusal("invalid_client_order_id", "Alpaca takes a client_order_id of 1 to 128 characters");
451 }
452 if (order.time_in_force_ != TimeInForce::DAY) {
453 return refusal("time_in_force_unsupported", "the Alpaca adapter sends day orders only");
454 }
455 if (order.session_ != OrderSession::REGULAR) {
456 return refusal("session_unsupported", "the Alpaca adapter sends regular-session orders only, no extended hours");
457 }
458 if (order.trigger_price_) {
459 return refusal("order_type_unsupported", "the Alpaca adapter sends no stop orders, so no trigger price");
460 }
461 switch (order.type_) {
462 case OrderType::LIMIT: {
463 if (!order.price_ || !positive(*order.price_)) {
464 return refusal("invalid_price", "a limit order needs a positive price");
465 }
466 const auto on_grid = us::equity_price_rule().on_tick(*order.price_);
467 if (!on_grid || !*on_grid) {
468 return refusal("off_tick", "a US limit price has at most 2 decimals from $1 and 4 below (Rule 612)");
469 }
470 break;
471 }
472 case OrderType::MARKET:
473 if (order.price_) {
474 return refusal("invalid_price", "a market order carries no price");
475 }
476 break;
477 case OrderType::UNKNOWN:
478 case OrderType::CONDITIONAL_LIMIT:
479 case OrderType::BEST_LIMIT:
480 case OrderType::PRIMARY_LIMIT:
481 case OrderType::MID_PRICE:
482 case OrderType::STOP_LIMIT:
483 case OrderType::CLOSING_PRICE:
484 return refusal("order_type_unsupported", "the Alpaca adapter sends limit and market orders only");
485 }
486 const auto quantity = decimal_normalized(order.quantity_);
487 if (quantity.coefficient() <= 0 || quantity.scale() != 0) {
488 return refusal("invalid_quantity", "the Alpaca adapter sends whole, positive share quantities only");
489 }
490 return std::nullopt;
491}

518inline std::optional<BrokerError> unlisted_asset(const json::object& asset, const std::string& symbol) {
519 const auto* tradable = asset.if_contains("tradable");
520 const auto asset_class = text(asset, "class");
521 const auto status = text(asset, "status");
522 const auto exchange = text(asset, "exchange");
523 if (text(asset, "symbol") != symbol) {
524 auto error = refusal("instrument_mismatch", "Alpaca's asset reply names another symbol than " + symbol);
525 error.category_ = ErrorCategory::PROTOCOL;
526 return error;
527 }
528 if (tradable == nullptr || !tradable->is_bool() || asset_class.empty() || status.empty() || exchange.empty()) {
529 auto error =
530 refusal("instrument_unreadable", "Alpaca's asset reply for " + symbol + " leaves out a field it needs");
531 error.category_ = ErrorCategory::PROTOCOL;
532 return error;
533 }
534 if (asset_class != "us_equity" || !listed_exchange(exchange)) {
535 return refusal(
536 "instrument_not_listed",
537 symbol + " is a " + asset_class + " asset on " + exchange +
538 ": the Alpaca adapter trades US equities listed on a national exchange only, not OTC"
539 );
540 }
541 if (status != "active" || !tradable->as_bool()) {
542 return refusal("instrument_not_tradable", symbol + " is not active and tradable at Alpaca");
543 }
544 return std::nullopt;
545}

675 Result<BarSeries> bars(BarQuery query) {
676 if (auto refused = detail::validate_bar_query(query)) {
677 return Result<BarSeries>(std::move(*refused));
678 }
679 return attempt<BarSeries>([&] {
680 const auto now = clock_();
681 const auto end = query.before_ ? *query.before_ : now;
682 const auto length =
683 query.interval_ == BarInterval::ONE_DAY ? std::chrono::hours(24) : *bar_interval_length(query.interval_);
684 const auto reply = get_object(
685 *data_,
686 "/v2/stocks/" + broker_detail::query_encode(query.instrument_.symbol_) + "/bars?timeframe=" +
687 timeframe(query.interval_) + "&feed=iex&adjustment=" + (query.adjusted_ ? "all" : "raw") +
688 "&sort=desc&start=2015-01-01T00:00:00Z&end=" + format_iso8601_utc(end + std::chrono::seconds(1)) +
689 "&limit=" + std::to_string(query.count_ + 1),
690 network::RateLimitGroup::MARKET_DATA
691 );
692 BarSeries series;
693 series.instrument_ = query.instrument_;
694 series.interval_ = query.interval_;
695 series.stamp_ = BarStamp::OPEN;
696 series.currency_ = Currency::usd();
697 series.adjusted_ = query.adjusted_;
698 const auto* rows = reply.if_contains("bars");
699 if (rows != nullptr && !rows->is_null() && !rows->is_array()) {
700 throw std::runtime_error("the bars are not a list");
701 }
702 if (rows != nullptr && rows->is_array()) {
703 for (const auto& value : rows->as_array()) {
704 if (static_cast<int>(series.bars_.size()) == query.count_) {
705 break;
706 }
707 if (!value.is_object()) {
708 throw std::runtime_error("a bar is not an object");
709 }
710 const auto& row = value.as_object();
711 const auto stamp = detail::instant(row, "t");
712 if (!stamp) {
713 throw std::runtime_error("a bar has no time");
714 }
715 // A daily bar ends 24 hours after its New York midnight: New York's clock changes on a Sunday.
716 if (*stamp + length > now || end < *stamp) {
717 continue;
718 }
719 Bar bar;
720 bar.timestamp_ = *stamp;
721 bar.open_ = detail::number(row, "o");
722 bar.high_ = detail::required(row, "h");
723 bar.low_ = detail::required(row, "l");
724 bar.close_ = detail::required(row, "c");
725 bar.volume_ = detail::required(row, "v");
726 series.bars_.push_back(std::move(bar));
727 }
728 }
729 std::reverse(series.bars_.begin(), series.bars_.end());
730 if (!series.ascending()) {
731 throw std::runtime_error("the bars are out of order or repeated");
732 }
733 series.received_at_ = clock_();
734 return series;
735 });
736 }

777 }

780 position.quantity_ = detail::required(row, "qty");
781 position.sellable_quantity_ = detail::number(row, "qty_available");

852 Result<TradingCalendar> calendar(Country country, std::optional<Date> from = std::nullopt) {
853 if (country != Country::US) {
854 return Result<TradingCalendar>(detail::refusal("market_unsupported", "Alpaca's calendar is the US market's"));

876 calendar.country_ = Country::US;

1140 // The reads made before an order is sent: the asset, the position for a sell, and the clock last.
1141 std::optional<BrokerError> refuse_before_order(const OrderRequest& order) {
1142 const auto& symbol = order.instrument_.symbol_;
1143 const auto unsent = [](BrokerError error, const char* what) {
1144 error.reached_broker_ = false;
1145 error.message_ = std::string(what) + " could not be read before the order, so it was not sent: " + error.message_;
1146 return error;
1147 };
1148 auto asset = send(
1149 *trading_, http::verb::get, "/v2/assets/" + broker_detail::query_encode(symbol), {},
1150 network::RateLimitGroup::ACCOUNT
1151 );
1152 if (asset && asset.value().result_int() == 404) {
1153 return detail::refusal("instrument_not_listed", "Alpaca lists no asset " + symbol);
1154 }
1155 auto asset_reply = read_object(std::move(asset));
1156 if (!asset_reply) {
1157 return unsent(std::move(asset_reply.error()), "the asset");
1158 }
1159 if (auto refused = detail::unlisted_asset(asset_reply.value(), symbol)) {
1160 return refused;
1161 }
1162 if (order.side_ == OrderSide::SELL) {
1163 auto position = send(
1164 *trading_, http::verb::get, "/v2/positions/" + broker_detail::query_encode(symbol), {},
1165 network::RateLimitGroup::ACCOUNT
1166 );
1167 if (position && position.value().result_int() == 404) {
1168 return detail::refusal("no_position", "no position in " + symbol + ": a sell would open a short");
1169 }
1170 auto position_reply = read_object(std::move(position));
1171 if (!position_reply) {
1172 return unsent(std::move(position_reply.error()), "the position");
1173 }
1174 try {
1175 const auto& row = position_reply.value();
1176 if (detail::text(row, "symbol") != symbol || detail::text(row, "asset_class") != "us_equity") {
1177 BrokerError error;
1178 error.category_ = ErrorCategory::PROTOCOL;
1179 error.broker_code_ = "position_mismatch";
1180 error.message_ = "Alpaca's position reply is not the US equity position in " + symbol;
1181 return unsent(std::move(error), "the position");
1182 }
1183 const auto available = detail::number(row, "qty_available");
1184 const auto held = detail::required(row, "qty");
1185 if (
1186 detail::text(row, "side") != "long" || !available || !detail::positive(held) ||
1187 decimal_less(*available, order.quantity_) || decimal_less(held, order.quantity_)
1188 ) {
1189 return detail::refusal(
1190 "sell_exceeds_position", "the sell exceeds the position available in " + symbol + ": it would open a short"
1191 );
1192 }
1193 } catch (const std::exception& exception) {
1194 BrokerError error;
1195 error.category_ = ErrorCategory::PROTOCOL;
1196 error.broker_code_ = "invalid_position_response";
1197 error.message_ = exception.what();

1199 }
1200 }
1201 return refuse_unless_core();
1202 }

1204 // Reads Alpaca's clock for the New York Stock Exchange and refuses unless its phase is `core`.
1205 std::optional<BrokerError> refuse_unless_core() {
1206 auto reply =
1207 read_object(send(*trading_, http::verb::get, "/v3/clock?markets=XNYS", {}, network::RateLimitGroup::ACCOUNT));
1208 BrokerError error;
1209 error.reached_broker_ = false;
1210 if (!reply) {
1211 error = std::move(reply.error());
1212 error.reached_broker_ = false;
1213 error.message_ = "Alpaca's clock could not be read before the order, so it was not sent: " + error.message_;
1214 return error;
1215 }
1216 const json::object* clock = nullptr;
1217 int matches = 0;
1218 if (const auto* clocks = reply.value().if_contains("clocks"); clocks != nullptr && clocks->is_array()) {

1226 }
1227 }

1285 // The error a reply with a failure status carries.
1286 BrokerError reply_error(const network::HttpResponse& response, Call call) const {
1287 BrokerError error;
1288 const auto status = static_cast<int>(response.result_int());
1289 error.http_status_ = status;
1290 boost::system::error_code ec;
1291 const auto value = json::parse(response.body(), ec);
1292 const auto* object = ec ? nullptr : value.if_object();
1293 if (object != nullptr) {
1294 error.message_ = "Alpaca: " + detail::text(*object, "message");
1295 // Read without a throwing conversion: a code that is not a whole number is left out.
1296 if (const auto* code = object->if_contains("code"); code != nullptr && code->is_int64()) {
1297 error.broker_code_ = std::to_string(code->as_int64());
1298 } else if (code != nullptr && code->is_uint64()) {
1299 error.broker_code_ = std::to_string(code->as_uint64());
1300 }
1301 } else {
1302 error.message_ = "Alpaca answered HTTP " + std::to_string(status) + " with a body that is not JSON";
1303 }
1304 if (status == 401) {
1305 // The gateway's refusal of the keys, before any route (seen on every host).
1306 error.category_ = ErrorCategory::AUTHENTICATION;
1307 } else if (status == 429 && call == Call::READ) {
1308 error.category_ = ErrorCategory::RATE_LIMIT;
1309 if (const auto reset = broker_detail::integer_header(response, "X-RateLimit-Reset")) {
1310 const auto now = std::chrono::duration_cast<std::chrono::seconds>(clock_().time_since_epoch()).count();
1311 error.retry_after_ = std::chrono::seconds(std::max<std::int64_t>(1, *reset - now));
1312 }
1313 } else if (call == Call::READ || status >= 500) {
1314 error.category_ = ErrorCategory::HTTP;
1315 } else if (
1316 (call == Call::PLACE &&
1317 ((status == 422 && error.broker_code_ == "42210000") || (status == 403 && error.broker_code_ == "40310000"))) ||
1318 (call == Call::CANCEL && status == 422 && error.broker_code_ == "42210000")
1319 ) {
1320 // Refusals seen on the paper host: the order was not placed, or not cancelled.
1321 error.category_ = ErrorCategory::INVALID_ARGUMENT;
1322 } else {
1323 // A code not seen on an order call says nothing of the order (40010001 is also a duplicate client_order_id,
1324 // which says the order exists): read it back.
1325 error.category_ = ErrorCategory::PROTOCOL;
1326 }
1327 return error;
1328 }

1471 if (order.order_id_.empty()) {

1473 }
