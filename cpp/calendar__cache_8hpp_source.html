108 Result<std::optional<TradingCalendar>> load(const std::string& name) override {
109 Descriptor folder;
110 if (auto error = open_folder(name, folder)) {
111 return *error;
112 }
113 Descriptor file;
114 if (auto error = open_file(folder, name, file)) {
115 return *error;
116 }
117 if (file.get() < 0) {
118 return std::optional<TradingCalendar>();
119 }
120 std::string text;
121 char buffer[4096];
122 while (text.size() <= MAXIMUM_FILE) {
123 const auto count = ::read(file.get(), buffer, sizeof(buffer));
124 if (count == 0) {
125 return parse(text);
126 }
127 if (count < 0 && errno != EINTR) {
128 return calendar_cache_error("calendar cache: the file could not be read");
129 }
130 if (count > 0) {
131 text.append(buffer, static_cast<std::size_t>(count));
132 }
133 }
134 return std::optional<TradingCalendar>();
135 }

143 std::optional<BrokerError> publish(const std::string& name, const TradingCalendar& calendar) override {
144 const auto latest = std::chrono::duration_cast<std::chrono::system_clock::duration>(LATEST_RECEIPT);
145 const auto received = calendar.received_at_.time_since_epoch();
146 if (received < -latest || latest < received) {
147 return calendar_cache_error("calendar cache: the receipt time is outside the range the file holds");
148 }
149 Descriptor folder;
150 if (auto error = open_folder(name, folder)) {
151 return error;
152 }
153 Descriptor existing;
154 if (auto error = open_file(folder, name, existing)) {
155 return error;
156 }
157 auto temporary = write_temporary(folder, name, serialize(calendar));
158 if (!temporary) {
159 return std::move(temporary.error());
160 }
161 return rename_temporary(folder, name, temporary.value());
162 }

203 // Tests what the cache requires of everything it opens: the type, the user's own, nothing for group or others.
204 static bool owned_privately(int descriptor, mode_t type) noexcept {
205 struct stat status{};
206 return ::fstat(descriptor, &status) == 0 && (status.st_mode & S_IFMT) == type && status.st_uid == ::geteuid() &&
207 (status.st_mode & 077) == 0;
208 }

210 // Checks the name can name a file (letters, digits, `-` and `_`, at most 200), then creates the folder if it does not exist and opens it, checked.
211 std::optional<BrokerError> open_folder(const std::string& name, Descriptor& folder) const {
212 bool safe = !name.empty() && name.size() <= 200;
213 for (const char ch : name) {
214 safe = safe &&
215 ((ch >= 'a' && ch <= 'z') || (ch >= 'A' && ch <= 'Z') || (ch >= '0' && ch <= '9') || ch == '-' || ch == '_');
216 }
217 if (!safe) {
218 return calendar_cache_error("calendar cache: the name cannot name a file");
219 }
220 if (::mkdir(folder_.c_str(), 0700) != 0 && errno != EEXIST) {
221 return calendar_cache_error("calendar cache: the folder could not be created");
222 }
223 folder.reset(::open(folder_.c_str(), O_RDONLY | O_DIRECTORY | O_NOFOLLOW | O_CLOEXEC));
224 if (folder.get() < 0) {
225 struct stat status{};
226 const bool link = ::lstat(folder_.c_str(), &status) == 0 && S_ISLNK(status.st_mode);
227 return calendar_cache_error(
228 link ? "calendar cache: the folder is a symbolic link" : "calendar cache: the folder could not be opened"
229 );
230 }
231 if (!owned_privately(folder.get(), S_IFDIR)) {
232 return calendar_cache_error("calendar cache: the folder is not private (owner or permissions)");
233 }
234 return std::nullopt;
235 }

237 // Opens the name's calendar file, checked; leaves `file` closed when there is none.
238 static std::optional<BrokerError> open_file(const Descriptor& folder, const std::string& name, Descriptor& file) {
239 file.reset(::openat(folder.get(), (name + ".calendar").c_str(), FILE_FLAGS | O_RDONLY));
240 if (file.get() < 0 && errno == ENOENT) {
241 return std::nullopt;
242 }
243 if (file.get() < 0) {
244 return calendar_cache_error(
245 errno == ELOOP ? "calendar cache: the file is a symbolic link" : "calendar cache: the file could not be opened"
246 );
247 }
248 if (!owned_privately(file.get(), S_IFREG)) {
249 return calendar_cache_error("calendar cache: the file is not private (owner, permissions or type)");
250 }
251 return std::nullopt;
252 }

265 // Writes the calendar's file text.
266 static std::string serialize(const TradingCalendar& calendar) {
267 boost::json::array days;
268 days.reserve(calendar.days_.size());
269 for (const auto& day : calendar.days_) {
270 const std::int64_t date =
271 std::int64_t{day.date_.year_} * 10000 + std::int64_t{day.date_.month_} * 100 + std::int64_t{day.date_.day_};
272 days.push_back(boost::json::array{date, day.open_});
273 }
274 boost::json::object document;
275 document["format"] = 1;
276 document["country"] = country_code(calendar.country_);
277 document["received_at_ns"] =
278 std::chrono::duration_cast<std::chrono::nanoseconds>(calendar.received_at_.time_since_epoch()).count();
279 document["unreadable_days"] = calendar.unreadable_days_;
280 document["days"] = std::move(days);
281 return boost::json::serialize(document);
282 }

284 // Reads a calendar file's text; nothing unless every field is present, of its type and in its range.
285 static std::optional<TradingCalendar> parse(const std::string& text) {
286 boost::system::error_code ec;
287 const auto document = boost::json::parse(text, ec);
288 if (ec || !document.is_object()) {
289 return std::nullopt;
290 }
291 const auto& object = document.as_object();
292 const auto* format = object.if_contains("format");
293 const auto* country = object.if_contains("country");
294 const auto* received = object.if_contains("received_at_ns");
295 const auto* unreadable = object.if_contains("unreadable_days");
296 const auto* days = object.if_contains("days");
297 if (
298 !format || !format->is_int64() || format->as_int64() != 1 || !country || !country->is_string() || !received ||
299 !received->is_int64() || !unreadable || !unreadable->is_int64() || !days || !days->is_array()
300 ) {
301 return std::nullopt;
302 }
303 TradingCalendar calendar;
304 const auto named = std::find_if(all_countries.begin(), all_countries.end(), [country](Country candidate) {
305 return country->as_string() == country_code(candidate);
306 });
307 if (named == all_countries.end()) {
308 return std::nullopt;
309 }
310 calendar.country_ = *named;
311 const auto since_epoch = std::chrono::nanoseconds(received->as_int64());
312 if (since_epoch < -LATEST_RECEIPT || LATEST_RECEIPT < since_epoch) {
313 return std::nullopt;
314 }
315 calendar.received_at_ = std::chrono::system_clock::time_point(
316 std::chrono::duration_cast<std::chrono::system_clock::duration>(since_epoch)
317 );
318 if (unreadable->as_int64() < 0 || static_cast<std::uint64_t>(unreadable->as_int64()) > SIZE_MAX) {
319 return std::nullopt;
320 }
321 calendar.unreadable_days_ = static_cast<std::size_t>(unreadable->as_int64());
322 calendar.days_.reserve(days->as_array().size());
323 for (const auto& row : days->as_array()) {
324 if (
325 !row.is_array() || row.as_array().size() != 2 || !row.as_array()[0].is_int64() || !row.as_array()[1].is_bool()
326 ) {
327 return std::nullopt;
328 }
329 const auto number = row.as_array()[0].as_int64();
330 if (number < 10101 || number > 99991231) {
331 return std::nullopt;
332 }
333 const Date date{
334 static_cast<int>(number / 10000),
335 static_cast<unsigned>(number / 100 % 100),
336 static_cast<unsigned>(number % 100),
337 };
338 if (!is_valid_date(date)) {
339 return std::nullopt;
340 }
341 calendar.days_.push_back(TradingDay{date, row.as_array()[1].as_bool()});
342 }
343 return calendar;
344 }

346 // Creates a temporary file of the caller's own (a name taken by another is not opened), writes `text` to it and flushes it; returns its name.
347 static Result<std::string> write_temporary(
348 const Descriptor& folder, const std::string& name, const std::string& text
349 ) {
350 static std::atomic<unsigned long> counter{0};
351 std::string temporary;
352 Descriptor file;
353 for (int attempt = 0; attempt < 100 && file.get() < 0; ++attempt) {
354 temporary = name + ".calendar." + std::to_string(::getpid()) + '-' + std::to_string(counter++) + ".tmp";
355 file.reset(::openat(folder.get(), temporary.c_str(), FILE_FLAGS | O_WRONLY | O_CREAT | O_EXCL, 0600));
356 if (file.get() < 0 && errno != EEXIST) {
357 break;
358 }
359 }
360 if (file.get() < 0) {
361 return calendar_cache_error("calendar cache: a temporary file could not be created");
362 }
363 bool written = true;
364 for (std::size_t offset = 0; written && offset < text.size();) {
365 const auto count = ::write(file.get(), text.data() + offset, text.size() - offset);
366 if (count > 0) {
367 offset += static_cast<std::size_t>(count);
368 } else if (count == 0 || errno != EINTR) {
369 written = false;
370 }
371 }
372 if (!written || ::fsync(file.get()) != 0) {
373 ::unlinkat(folder.get(), temporary.c_str(), 0);
374 return calendar_cache_error("calendar cache: the calendar file could not be written");
375 }
376 return temporary;
377 }

379 // Renames the caller's own temporary file over the calendar file; removes it if the rename fails.
380 static std::optional<BrokerError> rename_temporary(

384 ::unlinkat(folder.get(), temporary.c_str(), 0);

386 }
387 return std::nullopt;
