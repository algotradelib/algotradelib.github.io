149 MarketPhase phase(std::chrono::system_clock::time_point now, std::optional<bool> trading_day) const {
150 if (!trading_day) {
151 return MarketPhase::UNKNOWN;
152 }
153 if (!*trading_day) {
154 return MarketPhase::CLOSED;
155 }
156 const auto local = now + utc_offset_;
157 const auto since_midnight = std::chrono::duration_cast<std::chrono::minutes>(local - local_midnight(now));
158 if (since_midnight < open_) {
159 return MarketPhase::PRE_OPEN;
160 }
161 if (close_ <= since_midnight) {
162 return MarketPhase::POST_CLOSE;
163 }
164 return MarketPhase::OPEN;
165 }

178 return date_from_epoch_days(std::chrono::duration_cast<Days>(local_midnight(now).time_since_epoch()).count());

238inline MarketPhase stated_session_phase(const TradingCalendar& calendar, std::chrono::system_clock::time_point now) {
239 const auto has_hours = [](const TradingDay& day) {
240 return day.open_ && day.regular_ && day.regular_->opens_at_ < day.regular_->closes_at_;
241 };
242 const auto span_of = [](const TradingDay& day) {
243 const auto& regular = *day.regular_;
244 if (
245 day.extended_ && day.extended_->opens_at_ <= regular.opens_at_ && regular.closes_at_ <= day.extended_->closes_at_
246 ) {
247 return *day.extended_;
248 }
249 return regular;
250 };
251 const auto& days = calendar.days_;
252 for (std::size_t i = 0; i < days.size(); ++i) {
253 if (!has_hours(days[i])) {
254 continue;
255 }
256 const auto first = span_of(days[i]);
257 for (std::size_t j = i + 1; j < days.size(); ++j) {
258 if (!has_hours(days[j])) {
259 continue;
260 }
261 const auto second = span_of(days[j]);
262 if (first.opens_at_ < second.closes_at_ && second.opens_at_ < first.closes_at_) {
263 return MarketPhase::UNKNOWN;
264 }
265 }
266 }
267 const TradingDay* previous = nullptr;
268 const TradingDay* next = nullptr;
269 for (const auto& day : days) {
270 if (!has_hours(day)) {
271 continue;
272 }
273 const auto& regular = *day.regular_;
274 const auto span = span_of(day);
275 if (regular.opens_at_ <= now && now < regular.closes_at_) {
276 return MarketPhase::OPEN;
277 }
278 if (span.opens_at_ <= now && now < span.closes_at_) {
279 return now < regular.opens_at_ ? MarketPhase::PRE_OPEN : MarketPhase::POST_CLOSE;
280 }
281 if (span.closes_at_ <= now && (previous == nullptr || span_of(*previous).closes_at_ < span.closes_at_)) {
282 previous = &day;
283 }
284 if (now < span.opens_at_ && (next == nullptr || span.opens_at_ < span_of(*next).opens_at_)) {
285 next = &day;
286 }
287 }
288 if (previous == nullptr || next == nullptr || !(previous->date_ < next->date_)) {
289 return MarketPhase::UNKNOWN;
290 }
291 bool closed_between = false;
292 for (const auto& day : days) {
293 if (previous->date_ < day.date_ && day.date_ < next->date_) {
294 if (day.open_) {
295 return MarketPhase::UNKNOWN;
296 }
297 closed_between = true;
298 }
299 }
300 if (calendar.unreadable_days_ > 0) {
301 return MarketPhase::UNKNOWN;
302 }
303 return closed_between ? MarketPhase::CLOSED : MarketPhase::POST_CLOSE;
304}

343inline std::chrono::system_clock::time_point bar_period_end(const Bar& bar, BarInterval interval, BarStamp stamp) {
344 if (stamp == BarStamp::CLOSE) {
345 return bar.timestamp_;
346 }
347 switch (interval) {
348 case BarInterval::ONE_MINUTE:
349 return bar.timestamp_ + std::chrono::minutes(1);
350 case BarInterval::FIVE_MINUTES:
351 return bar.timestamp_ + std::chrono::minutes(5);
352 case BarInterval::FIFTEEN_MINUTES:
353 return bar.timestamp_ + std::chrono::minutes(15);
354 case BarInterval::THIRTY_MINUTES:
355 return bar.timestamp_ + std::chrono::minutes(30);
356 case BarInterval::ONE_HOUR:
357 return bar.timestamp_ + std::chrono::hours(1);
358 case BarInterval::ONE_DAY:

360 case BarInterval::ONE_WEEK:
361 return bar.timestamp_ + std::chrono::hours(24 * 7);
362 case BarInterval::ONE_MONTH:

450class SessionEdges {
451public:
458 SessionEdge next(const BarEvent& event, const SessionWindow& session) {
459 SessionEdge edge;
460 const auto end = event.period_end_;
461 const auto start = bar_period_start(event, session).value_or(end);
462 const auto date = session.local_date(start);
463 const auto open = session.opens_at(start);
464 const auto close = session.closes_at(start);
465 edge.point_ = start == end;
466 const bool reaches = start < close && (open < end || (edge.point_ && end == open));
467 const bool holds_close = start <= close && close <= end;
468 if (!reaches && !holds_close) {
469 // Before the open or after the close: pre-market or after-hours data, which still enters a strategy's series.
470 ++outside_count_;
471 }
472 if (reaches && !(opened_ && *opened_ == date)) {
473 edge.opens_ = true;
474 opened_ = date;
475 ++opened_count_;
476 }
477 if (holds_close && !(closed_ && *closed_ == date)) {
478 edge.closes_ = true;
479 closed_ = date;
480 ++closed_count_;
481 }
482 return edge;
483 }

680 void add_checked(const BarEvent& event) {
681 if (slices_.empty()) {
682 period_end_ = event.period_end_;
683 }
684 for (auto& slice : slices_) {
685 if (slice.interval_ == event.interval_) {
686 slice.bars_.push_back(event);
687 return;
688 }
689 }
690 BarSlice slice;

692 slice.interval_ = event.interval_;
693 slice.bars_.push_back(event);
694 slices_.push_back(std::move(slice));

698 void add(const BarEvent& event) {
699 if (slices_.empty()) {
700 period_end_ = event.period_end_;
701 }
702 for (auto& slice : slices_) {
703 if (slice.interval_ == event.interval_) {
704 slice.bars_.push_back(event);
705 // If the table cannot grow, the bar comes out again, so the two never disagree.
706 try {
707 if (slice.bars_.size() == id_detail::scan_limit + 1) {
708 widen(slice);
709 } else if (slice.bars_.size() > id_detail::scan_limit) {
710 if (auto* entry = wide(slice.interval_)) {
711 entry->table_.insert(event.instrument_, slice.bars_.size() - 1, ids_of(slice));
712 entry->count_ = slice.bars_.size();
713 }
714 }
715 } catch (...) {
716 slice.bars_.pop_back();
717 throw;
718 }
719 return;
720 }
721 }
722 BarSlice slice;
723 slice.period_end_ = event.period_end_;
724 slice.interval_ = event.interval_;
725 slice.bars_.push_back(event);
726 slices_.push_back(std::move(slice));
727 }

753private:
754 // An interval's table, which describes that interval's slice only while the slice holds more than
755 // `id_detail::scan_limit` bars: `widen` refills it when a slice passes that size and `add` extends it after,
756 // and nothing reads it while the slice is narrower. It outlives the slice, so the next wide slice of the
757 // interval reuses its storage; a bucket that never holds a wide slice has none.
758 struct Wide {
759 BarInterval interval_;
760 id_detail::PositionTable table_;
761 // How many of the slice's bars the table holds: all of them after `add` or `widen`, fewer after
762 // `add_checked`.
763 std::size_t count_{0};
764 };
765
766 const Wide* wide(BarInterval interval) const noexcept {
767 for (const auto& entry : wide_) {
768 if (entry.interval_ == interval) {
769 return &entry;
770 }
771 }
772 return nullptr;
773 }
774
775 Wide* wide(BarInterval interval) noexcept {
776 for (auto& entry : wide_) {
777 if (entry.interval_ == interval) {
778 return &entry;
779 }
780 }
781 return nullptr;
782 }
783
784 // Moves the table of a wide slice into it, with the bars it describes, so `BarSlice::find` can use it after
785 // the slice is delivered. Moving a slice keeps its bars' storage, so the table stays valid for it; the bucket's
786 // own table is rebuilt (`widen`) when a later period's slice grows wide again.
787 void hand_index_to(BarSlice& slice) noexcept {
788 if (slice.bars_.size() <= id_detail::scan_limit) {
789 return;
790 }
791 // Built now if `add_checked` left it unbuilt (or behind); a failed allocation only leaves the slice scanned.
792 const auto* built = wide(slice.interval_);
793 if (built == nullptr || built->count_ != slice.bars_.size() || built->table_.empty()) {
794 try {
795 widen(slice);
796 } catch (...) {
797 return;
798 }
799 }
800 if (auto* entry = wide(slice.interval_)) {
801 slice.index_.table_ = std::move(entry->table_);
802 entry->table_.reset();
803 entry->count_ = 0;
804 slice.index_.built_over_ = slice.bars_.data();
805 slice.index_.count_ = slice.bars_.size();
806 }
807 }
808
809 // Refills the interval's table from its slice, which has just passed `id_detail::scan_limit` bars (or is being
810 // delivered). On a failure the table is left empty, and the caller takes the bar back out or scans, so nothing
811 // reads it.
812 void widen(const BarSlice& slice) {
813 Wide* entry = wide(slice.interval_);
814 if (entry == nullptr) {
815 wide_.push_back(Wide{slice.interval_, {}});
816 entry = &wide_.back();
817 }
818 entry->table_.reset();
819 entry->count_ = 0;
820 try {
821 for (std::size_t at = 0; at < slice.bars_.size(); ++at) {
822 entry->table_.insert(slice.bars_[at].instrument_, at, ids_of(slice));
823 }
824 } catch (...) {
825 entry->table_.reset();
826 throw;
827 }
828 entry->count_ = slice.bars_.size();
829 }
830
831 std::chrono::system_clock::time_point period_end_{};
832 std::vector<BarSlice> slices_;
833 // At most one entry per interval that has had a wide slice.
834 std::vector<Wide> wide_;
835};
