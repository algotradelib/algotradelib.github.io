59static inline std::chrono::system_clock::time_point parse_expiry(const char* s) {
60 std::tm tm{};
61 if (
62 std::sscanf(s, "%d-%d-%d %d:%d:%d", &tm.tm_year, &tm.tm_mon, &tm.tm_mday, &tm.tm_hour, &tm.tm_min, &tm.tm_sec) != 6
63 ) {
64 return {};
65 }
66 tm.tm_year -= 1900;
67 tm.tm_mon -= 1;
68 tm.tm_isdst = 0;
69#ifdef _WIN32
70 const std::time_t utc = _mkgmtime(&tm) - 9 * 3600;
71#else
72 const std::time_t utc = timegm(&tm) - 9 * 3600;
73#endif
74 return std::chrono::system_clock::from_time_t(utc);
75}

150 void from_json_text_impl(std::string_view text, const std::string& origin) {
151 boost::system::error_code ec;
152 const auto jv = json::parse(text, ec);
153 if (ec) {
154 throw std::runtime_error("Failed to parse KIS credentials " + origin + " — " + ec.message());
155 }
156 if (!jv.is_object()) {
157 throw std::runtime_error("Failed to parse KIS credentials " + origin + " — not a JSON object");
158 }
159 const auto& doc = jv.as_object();
160 const json::value* key = doc.if_contains("app_key");
161 const json::value* secret = doc.if_contains("app_secret");
162 if (!key || !secret || !key->is_string() || !secret->is_string()) {
163 throw std::runtime_error("KIS credential " + origin + " should contain app_key and app_secret as strings");
164 }
165 // Everything is read and checked before anything is assigned, so a refused document leaves this
166 // config -- and the endpoint it already selected -- exactly as it was.
167 const bool demo = demo_from_json(doc, origin);
168 key_ = key->get_string().c_str();
169 secret_ = secret->get_string().c_str();
170 is_demo_ = demo;
171 }

178 static bool demo_from_env() {
179 const char* demo = std::getenv("KIS_DEMO");
180 if (!demo || std::string_view(demo) == "1") {
181 return true;
182 }
183 if (std::string_view(demo) == "0") {
184 return false;
185 }
186 throw std::runtime_error(
187 std::string("KIS_DEMO is \"") + demo +
188 "\"; set it to 1 for the demo endpoints or 0 to trade "
189 "live, or leave it unset for demo."
190 );
191 }

200 static bool demo_from_json(const json::object& doc, const std::string& origin) {
