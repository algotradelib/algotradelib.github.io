163class StopSource {
164public:
172 void request(StopIntent intent) noexcept {
173 const int desired = static_cast<int>(intent);
174 int current = requested_.load(std::memory_order_relaxed);
175 // A failed exchange reloads `current`, so the held intent's strength is read again each time.
176 while (current < 0 || stop_intent_strength(static_cast<StopIntent>(current)) < stop_intent_strength(intent)) {
177 if (requested_.compare_exchange_weak(current, desired, std::memory_order_release, std::memory_order_relaxed)) {
178 return;
179 }
180 }
181 }
182
187 bool requested() const noexcept {
188 return requested_.load(std::memory_order_acquire) >= 0;
189 }
190
195 std::optional<StopIntent> intent() const noexcept {
196 const int held = requested_.load(std::memory_order_acquire);
197 if (held < 0) {
198 return std::nullopt;
199 }
200 return static_cast<StopIntent>(held);
201 }
202
204 void clear() noexcept {
205 requested_.store(-1, std::memory_order_release);
206 }
207
208private:
209 // Negative means no request; otherwise a `StopIntent`.
210 std::atomic<int> requested_{-1};
211
212 // Asserted rather than assumed, because the signal-safety argument above rests entirely on it. If
213 // this ever fails on some platform, `request()` is no longer callable from a handler and the
214 // signal installer is unsound -- which is worth a compile error rather than a subtle deadlock in a
215 // handler that fires once a month.
216 static_assert(
217 std::atomic<int>::is_always_lock_free, "StopSource requires a lock-free atomic<int> to stay signal-safe"
218 );
219};

265 unrecognized_.clear();
266 if (path_.empty()) {

269 std::FILE* handle = std::fopen(path_.c_str(), "rb");
270 if (handle == nullptr) {
271 return std::nullopt;
272 }
273 char buffer[64] = {};
274 const std::size_t read = std::fread(buffer, 1, sizeof(buffer) - 1, handle);
275 std::fclose(handle);
276
277 std::string_view text(buffer, read);
278 const auto begin = text.find_first_not_of(" \t\r\n");
279 if (begin == std::string_view::npos) {
280 return std::nullopt;
281 }
282 text.remove_prefix(begin);
283 const auto end = text.find_first_of(" \t\r\n");
284 if (end != std::string_view::npos) {
285 text = text.substr(0, end);
286 }
287
288 if (auto intent = stop_intent_from(text)) {
289 return intent;
290 }
291 unrecognized_.assign(text);
292 return std::nullopt;
293 }

300 return unrecognized_;
301 }
